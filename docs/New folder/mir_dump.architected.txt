/**
 * @enum ABIConvention
 * @brief Target calling convention/ABI specification.
 */
enum class ABIConvention : uint8_t {
    SystemV_x64 = 0,    ///< System V AMD64 ABI (Linux/BSD/macOS standard)
    Microsoft_x64 = 1,  ///< Microsoft x64 calling convention (Windows)
    ARM64_EABI = 2,     ///< ARM64 EABI (AArch64 / Apple Silicon)
    WebAssembly = 3,    ///< WebAssembly (wasm32/wasm64)
    RISCV64_LP64D = 4,  ///< RISC-V 64 LP64D ABI
    ARM32_AAPCS = 5     ///< ARM32 AAPCS / EABI
};

/**
 * @enum ABIValueClass
 * @brief Categorizes the data type to route to the correct hardware register bank.
 */
enum class ABIValueClass : uint8_t {
    IntegerOrPointer = 0, ///< Uses GPRs (e.g., RAX, X0)
    FloatingPoint = 1,    ///< Uses FPRs (e.g., XMM0, V0)
    Vector = 2,           ///< Uses SIMD registers (e.g., YMM, ZMM)
    Memory = 3            ///< Forced to stack (large structs)
};

/**
 * @struct ABIParameter
 * @brief Describes parameter location and properties after lowering.
 */
struct ABIParameter {
    uint32_t register_id = 0;    ///< Hardware register ID (if in_register=true)
    uint32_t stack_offset = 0;   ///< Byte offset from frame pointer (if in_register=false)
    uint32_t size = 0;           ///< Parameter size in bytes
    bool in_register = false;    ///< true = bound to register, false = spilled to stack

    [[nodiscard]] constexpr bool is_in_register() const noexcept { return in_register; }
    [[nodiscard]] constexpr bool is_on_stack() const noexcept { return !in_register; }
    [[nodiscard]] constexpr uint32_t get_register_id() const noexcept { return register_id; }
    [[nodiscard]] constexpr uint32_t get_stack_offset() const noexcept { return stack_offset; }
};

struct ABICallLayout {
    std::vector<ABIParameter> parameters;
    uint32_t return_register = 0;
    uint32_t argument_stack_size = 0;
    bool uses_variadic_policy = false;
};

/**
 * @class ABI_Lowering
 * @brief Converts high-level parameters to ABI-conformant register/stack layout.
 * Implements the rigorous calling conventions for HFT/System-level targets,
 * handling shadow spaces, split register banks, and struct alignment.
 */
class ABI_Lowering {
public:
    /**
     * @brief Create lowering engine for target convention.
     * @param convention Target ABI (defaults to System V x64)
     */
    explicit ABI_Lowering(ABIConvention convention = ABIConvention::SystemV_x64) noexcept
        : convention_(convention) {
        reset();
    }

    ~ABI_Lowering() = default;

    ABI_Lowering(const ABI_Lowering&) = delete;
    ABI_Lowering& operator=(const ABI_Lowering&) = delete;

    /**
     * @brief Lower a parameter to its ABI location.
     * @param val_class Type class (Int/Ptr vs Float vs Vector)
     * @param size Parameter size in bytes
     * @param alignment Memory alignment requirement (e.g., 8 for double, 16 for vectors)
     * @return ABIParameter describing exact register or stack layout
     */
    [[nodiscard]] ABIParameter lower_parameter(ABIValueClass val_class, 
                                               uint32_t size, 
                                               uint32_t alignment = 8) noexcept;

    /**
     * @brief Lower a variadic argument with target-specific safety rules.
     * Variadic values are conservatively placed to preserve ABI correctness.
     */
    [[nodiscard]] ABIParameter lower_variadic_parameter(ABIValueClass val_class,
                                                        uint32_t size,
                                                        uint32_t alignment = 8) noexcept;

    /**
     * @brief Plan a full call ABI layout for deterministic argument assignment.
     */
    [[nodiscard]] ABICallLayout plan_call_layout(const std::vector<ABIValueClass>& value_classes,
                                                 const std::vector<uint32_t>& sizes,
                                                 ABIValueClass return_class,
                                                 bool is_variadic = false,
                                                 size_t fixed_parameter_count = 0) noexcept;

    /**
     * @brief Get the primary register ID for scalar return values.
     * @param val_class The return type class
     */
    [[nodiscard]] uint32_t get_return_register(ABIValueClass val_class) const noexcept;

    /**
     * @brief Get the hardware register ID for the stack pointer.
     */
    [[nodiscard]] constexpr uint32_t get_stack_pointer_register() const noexcept {
        switch (convention_) {
            case ABIConvention::ARM64_EABI:    return 31; /* SP (X31) */
            case ABIConvention::RISCV64_LP64D: return 2;  /* sp (x2) */
            case ABIConvention::ARM32_AAPCS:   return 13; /* sp (r13) */
            default:                           return 4;  /* RSP / pseudo-SP */
        }
    }

    /**
     * @brief Get the hardware register ID for the frame pointer.
     */
    [[nodiscard]] constexpr uint32_t get_frame_pointer_register() const noexcept {
        switch (convention_) {
            case ABIConvention::ARM64_EABI:    return 29; /* X29 */
            case ABIConvention::RISCV64_LP64D: return 8;  /* s0/fp (x8) */
            case ABIConvention::ARM32_AAPCS:   return 11; /* r11/fp */
            default:                           return 5;  /* RBP / pseudo-FP */
        }
    }

    /**
     * @brief Reset state for a new function compilation pass.
     * Re-initializes register counters and allocates MS x64 shadow space if needed.
     */
    void reset() noexcept {
        int_regs_used_ = 0;
        float_regs_used_ = 0;
        unified_slots_used_ = 0;
        
        // Windows x64 requires a mandatory 32-byte shadow space (home space) 
        // for the first 4 arguments, even if they are passed in registers.
        stack_offset_ = (convention_ == ABIConvention::Microsoft_x64) ? 32 : 0;
    }

    /**
     * @brief Get current target convention.
     */
    [[nodiscard]] constexpr ABIConvention get_convention() const noexcept { 
        return convention_; 
    }

    /**
     * @brief Get total stack size required for arguments.
     */
    [[nodiscard]] constexpr uint32_t get_argument_stack_size() const noexcept {
        return stack_offset_;
    }

    [[nodiscard]] uint32_t finalize_argument_stack_size(uint32_t call_alignment = 16) const noexcept {
        if (call_alignment == 0 || (call_alignment & (call_alignment - 1u)) != 0u) {
            call_alignment = 16;
        }
        return (stack_offset_ + call_alignment - 1u) & ~(call_alignment - 1u);
    }

private:
    ABIConvention convention_;
    
    // --- State Trackers for the current function being lowered ---
    uint8_t int_regs_used_ = 0;
    uint8_t float_regs_used_ = 0;
    uint8_t unified_slots_used_ = 0; ///< Used by Windows ABI (shares Int/Float slots)
    uint32_t stack_offset_ = 0;

    /**
     * @brief Aligns the current stack offset to the required boundary.
     */
    void align_stack(uint32_t alignment) noexcept {
        if (alignment > 1) [[likely]] {
            stack_offset_ = (stack_offset_ + alignment - 1) & ~(alignment - 1);
        }
    }

    // =========================================================================
    // HARDWARE REGISTER MAPPINGS
    // =========================================================================

    // System V x64: 6 Integer (RDI, RSI, RDX, RCX, R8, R9)
    static constexpr std::array<uint32_t, 6> SYSTEMV_INT_REGS = {
        0, 1, 2, 3, 4, 5 // SysV abstract IDs
    };

    // System V x64: 8 Float (XMM0-XMM7)
    static constexpr std::array<uint32_t, 8> SYSTEMV_FLOAT_REGS = {
        16, 17, 18, 19, 20, 21, 22, 23 // XMM mapped to IDs 16-23
    };

    // Microsoft x64: 4 Integer (RCX, RDX, R8, R9)
    static constexpr std::array<uint32_t, 4> MICROSOFT_INT_REGS = {
        3, 2, 4, 5 // Windows abstract IDs
    };

    // Microsoft x64: 4 Float (XMM0-XMM3) mapped to unified slots
    static constexpr std::array<uint32_t, 4> MICROSOFT_FLOAT_REGS = {
        16, 17, 18, 19
    };

    // ARM64: 8 Integer (X0-X7)
    static constexpr std::array<uint32_t, 8> ARM64_INT_REGS = {
        0, 1, 2, 3, 4, 5, 6, 7
    };

    // ARM64: 8 Float (V0-V7)
    static constexpr std::array<uint32_t, 8> ARM64_FLOAT_REGS = {
        32, 33, 34, 35, 36, 37, 38, 39 // Vector register abstract IDs
    };

    // RISC-V 64: 8 Integer (a0-a7)
    static constexpr std::array<uint32_t, 8> RISCV64_INT_REGS = {
        10, 11, 12, 13, 14, 15, 16, 17
    };

    // RISC-V 64: 8 Float (fa0-fa7)
    static constexpr std::array<uint32_t, 8> RISCV64_FLOAT_REGS = {
        42, 43, 44, 45, 46, 47, 48, 49
    };

    // ARM32: 4 Integer (r0-r3)
    static constexpr std::array<uint32_t, 4> ARM32_INT_REGS = {
        0, 1, 2, 3
    };

    // ARM32: 8 Float/Double (s0-s15 / d0-d7)
    static constexpr std::array<uint32_t, 8> ARM32_FLOAT_REGS = {
        16, 17, 18, 19, 20, 21, 22, 23
    };
};

inline ABIParameter ABI_Lowering::lower_parameter(ABIValueClass val_class,
                                                  uint32_t size,
                                                  uint32_t alignment) noexcept {
    if (size == 0) [[unlikely]] {
        size = 8;
    }

    if (alignment == 0 || (alignment & (alignment - 1u)) != 0u) [[unlikely]] {
        alignment = 8;
    }

    const auto spill_to_stack = [&]() noexcept {
        align_stack(alignment);
        const uint32_t offset = stack_offset_;
        const uint32_t rounded = (size + 7u) & ~7u;
        stack_offset_ += rounded;
        return ABIParameter{0xFFFFFFFFu, offset, size, false};
    };

    // Large aggregates and explicit memory-class values always spill.
    if (val_class == ABIValueClass::Memory || size > 16u) [[unlikely]] {
        return spill_to_stack();
    }

    const bool fp_bank = (val_class == ABIValueClass::FloatingPoint ||
                          val_class == ABIValueClass::Vector);

    switch (convention_) {
        case ABIConvention::SystemV_x64: {
            if (fp_bank) {
                if (float_regs_used_ < SYSTEMV_FLOAT_REGS.size()) [[likely]] {
                    const uint32_t reg = SYSTEMV_FLOAT_REGS[float_regs_used_++];
                    return ABIParameter{reg, 0u, size, true};
                }
            } else {
                if (int_regs_used_ < SYSTEMV_INT_REGS.size()) [[likely]] {
                    const uint32_t reg = SYSTEMV_INT_REGS[int_regs_used_++];
                    return ABIParameter{reg, 0u, size, true};
                }
            }
            return spill_to_stack();
        }

        case ABIConvention::Microsoft_x64: {
            // Microsoft x64 uses a shared 4-slot argument window for integer + FP.
            if (unified_slots_used_ < 4u) [[likely]] {
                const uint8_t slot = unified_slots_used_++;
                if (fp_bank) {
                    return ABIParameter{MICROSOFT_FLOAT_REGS[slot], 0u, size, true};
                }
                return ABIParameter{MICROSOFT_INT_REGS[slot], 0u, size, true};
            }
            return spill_to_stack();
        }

        case ABIConvention::ARM64_EABI: {
            if (fp_bank) {
                if (float_regs_used_ < ARM64_FLOAT_REGS.size()) [[likely]] {
                    const uint32_t reg = ARM64_FLOAT_REGS[float_regs_used_++];
                    return ABIParameter{reg, 0u, size, true};
                }
            } else {
                if (int_regs_used_ < ARM64_INT_REGS.size()) [[likely]] {
                    const uint32_t reg = ARM64_INT_REGS[int_regs_used_++];
                    return ABIParameter{reg, 0u, size, true};
                }
            }
            return spill_to_stack();
        }

        case ABIConvention::WebAssembly:
            // WASM uses virtual locals/operand stack; keep deterministic pseudo-register IDs
            // so call layouts can still model ordered argument lanes.
            if (val_class == ABIValueClass::Memory || size > 16u) [[unlikely]] {
                return spill_to_stack();
            }
            if (fp_bank) {
                const uint32_t reg = 32u + float_regs_used_++;
                return ABIParameter{reg, 0u, size, true};
            }
            return ABIParameter{int_regs_used_++, 0u, size, true};

        case ABIConvention::RISCV64_LP64D: {
            if (fp_bank) {
                if (float_regs_used_ < RISCV64_FLOAT_REGS.size()) [[likely]] {
                    const uint32_t reg = RISCV64_FLOAT_REGS[float_regs_used_++];
                    return ABIParameter{reg, 0u, size, true};
                }
            } else {
                if (int_regs_used_ < RISCV64_INT_REGS.size()) [[likely]] {
                    const uint32_t reg = RISCV64_INT_REGS[int_regs_used_++];
                    return ABIParameter{reg, 0u, size, true};
                }
            }
            return spill_to_stack();
        }

        case ABIConvention::ARM32_AAPCS: {
            if (fp_bank) {
                if (float_regs_used_ < ARM32_FLOAT_REGS.size()) [[likely]] {
                    const uint32_t reg = ARM32_FLOAT_REGS[float_regs_used_++];
                    return ABIParameter{reg, 0u, size, true};
                }
            } else {
                if (int_regs_used_ < ARM32_INT_REGS.size()) [[likely]] {
                    const uint32_t reg = ARM32_INT_REGS[int_regs_used_++];
                    return ABIParameter{reg, 0u, size, true};
                }
            }
            return spill_to_stack();
        }

        default:
            return spill_to_stack();
    }
}

inline ABIParameter ABI_Lowering::lower_variadic_parameter(ABIValueClass val_class,
                                                           uint32_t size,
                                                           uint32_t alignment) noexcept {
    // Conservative and portable policy: keep variadic extras in memory for
    // deterministic handling across call boundaries and parity.
    (void)val_class;
    if (size == 0) [[unlikely]] {
        size = 8;
    }
    if (alignment == 0 || (alignment & (alignment - 1u)) != 0u) [[unlikely]] {
        alignment = 8;
    }

    align_stack(alignment);
    const uint32_t offset = stack_offset_;
    const uint32_t rounded = (size + 7u) & ~7u;
    stack_offset_ += rounded;
    return ABIParameter{0xFFFFFFFFu, offset, size, false};
}

inline ABICallLayout ABI_Lowering::plan_call_layout(const std::vector<ABIValueClass>& value_classes,
                                                    const std::vector<uint32_t>& sizes,
                                                    ABIValueClass return_class,
                                                    bool is_variadic,
                                                    size_t fixed_parameter_count) noexcept {
    ABICallLayout layout;
    reset();
    const size_t n = std::min(value_classes.size(), sizes.size());
    layout.parameters.reserve(n);
    layout.uses_variadic_policy = is_variadic;

    for (size_t i = 0; i < n; ++i) {
        const ABIValueClass cls = value_classes[i];
        const uint32_t arg_size = sizes[i] == 0 ? 8u : sizes[i];

        if (is_variadic && i >= fixed_parameter_count) {
            layout.parameters.push_back(lower_variadic_parameter(cls, arg_size));
        } else {
            layout.parameters.push_back(lower_parameter(cls, arg_size));
        }
    }

    layout.return_register = get_return_register(return_class);
    layout.argument_stack_size = finalize_argument_stack_size();
    return layout;
}

inline uint32_t ABI_Lowering::get_return_register(ABIValueClass val_class) const noexcept {
    switch (convention_) {
        case ABIConvention::SystemV_x64:
        case ABIConvention::Microsoft_x64:
            if (val_class == ABIValueClass::FloatingPoint || val_class == ABIValueClass::Vector) {
                return 16u; // XMM0
            }
            if (val_class == ABIValueClass::Memory) {
                return 0xFFFFFFFFu;
            }
            return 0u; // RAX

        case ABIConvention::ARM64_EABI:
            if (val_class == ABIValueClass::FloatingPoint || val_class == ABIValueClass::Vector) {
                return 32u; // V0
            }
            if (val_class == ABIValueClass::Memory) {
                return 0xFFFFFFFFu;
            }
            return 0u; // X0

        case ABIConvention::RISCV64_LP64D:
            if (val_class == ABIValueClass::FloatingPoint || val_class == ABIValueClass::Vector) {
                return 42u; // fa0
            }
            if (val_class == ABIValueClass::Memory) {
                return 0xFFFFFFFFu;
            }
            return 10u; // a0

        case ABIConvention::ARM32_AAPCS:
            if (val_class == ABIValueClass::FloatingPoint || val_class == ABIValueClass::Vector) {
                return 16u; // s0
            }
            if (val_class == ABIValueClass::Memory) {
                return 0xFFFFFFFFu;
            }
            return 0u; // r0

        case ABIConvention::WebAssembly:
        default:
            // Virtual return slot for VM stack-machine backends.
            return 0u;
    }
}

/*
 * DGTL ABI LOWERING
 * ============================================================================
 *
 * Purpose
 * -------
 * Deterministic ABI argument/return classification and stack-layout planning
 * for DGTL's native backends.
 *
 * Supported conventions
 * ---------------------
 *   - System V AMD64
 *   - Microsoft x64
 *   - AAPCS64 / ARM64
 *   - WebAssembly logical ABI
 *
 * Design goals
 * ------------
 *   1. No undefined arithmetic from malformed IR/type metadata.
 *   2. Deterministic register and stack assignment.
 *   3. Explicit ABI limits instead of accidental array overrun.
 *   4. Correct Win64 32-byte home/shadow space semantics.
 *   5. Correct distinction between argument slots and byte offsets.
 *   6. Safe handling of large aggregates.
 *   7. Explicit handling of alignment and integer overflow.
 *   8. Stable behavior for malformed/incomplete input.
 *   9. No allocation-producing operation is marked noexcept.
 *  10. Easy extension for richer DGTL type classification.
 *
 * Important
 * ---------
 * This component is a lowering MODEL. It intentionally does not pretend to
 * implement every corner of the C/C++ ABI type-classification algorithms.
 * A full production compiler should perform language/type classification
 * before calling this layer.
 */

[[nodiscard]] inline const char* runtime_symbol_for_opcode(ir::OpCode opcode) noexcept {
    const size_t index = static_cast<size_t>(opcode);
    if (index < detail::kRuntimeSymbolTable.size()) [[likely]] {
        return detail::kRuntimeSymbolTable[index];
    }
    return nullptr;
}

[[nodiscard]] inline bool has_runtime_symbol(ir::OpCode opcode) noexcept {
    return runtime_symbol_for_opcode(opcode) != nullptr;
}

/**
 * @brief Return whether generated ISel metadata classifies an opcode as
 * runtime-backed for the requested compilation target.
 *
 * Runtime symbol availability and runtime-backed lowering provenance are
 * deliberately separate contracts:
 *   - has_runtime_symbol() answers "does a runtime ABI symbol exist?"
 *   - is_runtime_backed() answers "does this target's generated ISel contract
 *     declare runtime-backed lowering for this opcode?"
 */
[[nodiscard]] inline bool is_runtime_backed(
    const core::TargetMachine& target,

/* ============================================================================
 * Canonical semantic classification.
 *
 * This metadata is target-independent. It never assigns a target opcode.
 * ========================================================================== */
[[nodiscard]] constexpr machine::MachineSemanticKind semantic_kind_for(
    ir::OpCode opcode,
    MachineForm form) noexcept {
    using K = machine::MachineSemanticKind;

    // MachineForm is the authoritative shape selected for the concrete target.
    // Semantic kind must always satisfy the common MIR form contract.
    switch (form) {
        case MachineForm::Move:
            return K::Move;
        case MachineForm::LoadImmediate:
            return K::LoadImmediate;
        case MachineForm::LoadStack:
            return K::LoadStack;
        case MachineForm::StoreStack:
            return K::StoreStack;

        case MachineForm::IntegerBinary32:
        case MachineForm::IntegerBinary64:
            switch (opcode) {
                case ir::OpCode::ADD_I8:
                case ir::OpCode::ADD_I32:
                case ir::OpCode::ADD_I64:
                case ir::OpCode::ADD_I128:
                case ir::OpCode::ADD_I256:
                    return K::IntegerAdd;
                case ir::OpCode::SUB_I32:
                case ir::OpCode::SUB_I64:
                case ir::OpCode::PTR_DIFF:
                    return K::IntegerSub;
                case ir::OpCode::MUL_I32:
                case ir::OpCode::MUL_I64:
                    return K::IntegerMul;
                case ir::OpCode::DIV_I32:
                case ir::OpCode::DIV_I64:
                    return K::IntegerDiv;
                case ir::OpCode::MOD_I32:
                case ir::OpCode::MOD_I64:
                    return K::IntegerMod;
                case ir::OpCode::AND_I32:
                case ir::OpCode::AND_I64:
                    return K::IntegerAnd;
                case ir::OpCode::OR_I32:
                case ir::OpCode::OR_I64:
                    return K::IntegerOr;
                case ir::OpCode::XOR_I32:
                case ir::OpCode::XOR_I64:
                    return K::IntegerXor;
                case ir::OpCode::SHL_I32:
                case ir::OpCode::SHL_I64:
                    return K::IntegerShl;
                case ir::OpCode::SHR_I32:
                case ir::OpCode::SHR_I64:
                    return K::IntegerShr;
                default:
                    return K::None;
            }

        case MachineForm::FloatBinary32:
        case MachineForm::FloatBinary64:
            switch (opcode) {
                case ir::OpCode::ADD_F32:
                case ir::OpCode::ADD_F64:
                    return K::FloatAdd;
                case ir::OpCode::SUB_F32:
                case ir::OpCode::SUB_F64:
                    return K::FloatSub;
                case ir::OpCode::MUL_F32:
                case ir::OpCode::MUL_F64:
                    return K::FloatMul;
                case ir::OpCode::DIV_F32:
                case ir::OpCode::DIV_F64:
                    return K::FloatDiv;
                default:
                    return K::None;
            }

        case MachineForm::FloatTernaryFma32:
            return opcode == ir::OpCode::FMA_F32 ? K::FloatFma : K::None;

        case MachineForm::Compare32:
            return K::Compare;

        case MachineForm::VectorBinary512:
            switch (opcode) {
                case ir::OpCode::VEC_ADD_F32_512:
                case ir::OpCode::NEON_VADD_Q:
                    return K::VectorAdd;
                case ir::OpCode::VEC_MUL_F32_512:
                    return K::VectorMul;
                default:
                    return K::None;
            }

        case MachineForm::VectorGather512:
            return K::VectorGather;
        case MachineForm::VectorStore512:
            return K::VectorStore;
        case MachineForm::Call:
            return K::Call;
        case MachineForm::Return:
            return K::Return;
        case MachineForm::Goto:
            return K::Goto;
        case MachineForm::BranchCond:
            return K::BranchCond;
        case MachineForm::BranchCompare:
            return K::BranchCompare;
        case MachineForm::Syscall:
            return K::Syscall;
        case MachineForm::TensorOp:
            return K::TensorOp;
        case MachineForm::HardwareFence:
            return K::HardwareFence;
        case MachineForm::AtomicOp:
            return K::AtomicOp;
        case MachineForm::ConcurrentSync:
            return K::ConcurrentSync;
        case MachineForm::NoOp:
            return K::Other;
        case MachineForm::Unhandled:
            return K::None;
    }

    return K::None;
}


[[nodiscard]] inline machine::MachineSelection to_machine_selection(
    const SelectedInstruction& selected,
    machine::RegisterClassKind result_class) {
    machine::MachineSelection selection{};
    selection.form = selected.form;
    selection.semantic_kind = selected.semantic_kind;
    selection.semantic_provenance =
        dgtl::ir::schema::stable_id(selected.semantic_opcode);
    selection.debug_id = selected.node.debug_id;
    selection.result_value = selected.node.result_id;
    selection.result_class = result_class;

    const std::uint32_t operands[3] = {
        selected.node.op_a, selected.node.op_b, selected.node.op_c
    };

    for (std::size_t i = 0; i < 3; ++i) {
        if (operands[i] != ir::INVALID_SLOT) {
            selection.operands.push_back(
                machine::MachineSelectionOperand::value_ref(
                    operands[i], machine::OperandRole::Use));
        }
    }

    if (selected.node.has_side_effects()) {
        selection.flags = selection.flags | machine::MachineInstrFlag::SideEffect;
    }

    switch (selected.form) {
        case MachineForm::Goto:
        case MachineForm::BranchCond:
        case MachineForm::BranchCompare:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Terminator |
                              machine::MachineInstrFlag::Branch;
            break;
        case MachineForm::Return:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Terminator |
                              machine::MachineInstrFlag::Return;
            break;
        case MachineForm::Call:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Call |
                              machine::MachineInstrFlag::SideEffect;
            break;
        case MachineForm::LoadStack:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Load |
                              machine::MachineInstrFlag::MayLoad;
            break;
        case MachineForm::StoreStack:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Store |
                              machine::MachineInstrFlag::MayStore;
            break;
        case MachineForm::AtomicOp:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Atomic |
                              machine::MachineInstrFlag::SideEffect;
            break;
        case MachineForm::HardwareFence:
        case MachineForm::ConcurrentSync:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Barrier |
                              machine::MachineInstrFlag::SideEffect;
            break;
        case MachineForm::Syscall:
        case MachineForm::TensorOp:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::SideEffect;
            break;
        default:
            break;
    }

    return selection;
}

namespace detail {

[[nodiscard]] inline std::string_view trim_copy(std::string_view value) noexcept {
    size_t begin = 0;
    size_t end = value.size();
    while (begin < end && std::isspace(static_cast<unsigned char>(value[begin])) != 0) {
        ++begin;
    }
    while (end > begin && std::isspace(static_cast<unsigned char>(value[end - 1])) != 0) {
        --end;
    }
    return value.substr(begin, end - begin);
}

[[nodiscard]] inline bool ascii_equal_case_insensitive(std::string_view lhs, std::string_view rhs) noexcept {
    if (lhs.size() != rhs.size()) return false;
    for (size_t i = 0; i < lhs.size(); ++i) {
        if (std::tolower(static_cast<unsigned char>(lhs[i])) != std::tolower(static_cast<unsigned char>(rhs[i]))) {
            return false;
        }
    }
    return true;
}

[[nodiscard]] inline std::vector<std::string> split_top_level(std::string_view text, char delimiter) {
    std::vector<std::string> parts;
    size_t start = 0;
    int depth = 0;
    for (size_t i = 0; i < text.size(); ++i) {
        const char ch = text[i];
        if (ch == '(') {
            ++depth;
        } else if (ch == ')') {
            --depth;
        } else if (ch == delimiter && depth == 0) {
            parts.push_back(std::string(trim_copy(text.substr(start, i - start))));
            start = i + 1;
        }
    }
    parts.push_back(std::string(trim_copy(text.substr(start))));
    return parts;
}

[[nodiscard]] inline std::optional<ir::OpCode> parse_opcode_name(
    std::string_view name) noexcept {
    // Canonical HIR schema/generated metadata is the only opcode-name authority.
    // This also resolves source-level aliases defined by the canonical DSL.
    return dgtl::ir::gen::find_opcode_by_name(detail::trim_copy(name));
}

struct PatternTerm {
    std::string capture;
    std::optional<ir::OpCode> opcode;
    std::vector<PatternTerm> children;
};

[[nodiscard]] inline PatternTerm parse_term(std::string_view text) {
    const std::string_view token = trim_copy(text);
    if (!token.empty() && token[0] == '$') {
        PatternTerm term{};
        term.capture = std::string(token.substr(1));
        return term;
    }

    const size_t open = token.find('(');
    if (open == std::string::npos) {
        PatternTerm term{};
        term.opcode = parse_opcode_name(token);
        return term;
    }

    const size_t close = token.rfind(')');
    PatternTerm term{};
    term.opcode = parse_opcode_name(token.substr(0, open));
    if (close != std::string::npos && close > open + 1) {
        for (const std::string& part : split_top_level(token.substr(open + 1, close - open - 1), ',')) {
            term.children.push_back(parse_term(part));
        }
    }
    return term;
}

[[nodiscard]] inline std::string lower_ascii(std::string_view value) {
    std::string out(value.size(), '\0');
    std::transform(value.begin(), value.end(), out.begin(),
                   [](unsigned char ch) { return static_cast<char>(std::tolower(ch)); });
    return out;
}

[[nodiscard]] inline SelectionRule parse_rule_line(std::string_view line) {
    SelectionRule rule{};
    std::string_view text = trim_copy(line);
    if (!text.starts_with("rule ")) {
        return rule;
    }

    const size_t colon = text.find(':');
    const size_t arrow = text.find("->");
    const size_t bracket = text.find('[');
    if (colon == std::string::npos || arrow == std::string::npos || arrow <= colon) {
        return rule;
    }

    rule.name = std::string(trim_copy(text.substr(5, colon - 5)));
    const size_t replace_end = (bracket == std::string::npos) ? text.size() : bracket;
    rule.match = parse_term(text.substr(colon + 1, arrow - colon - 1));
    rule.replace = parse_term(text.substr(arrow + 2, replace_end - arrow - 2));

    if (bracket != std::string::npos) {
        const size_t end_bracket = text.rfind(']');
        if (end_bracket == std::string_view::npos || end_bracket <= bracket) {
            return rule;
        }
        const std::string_view attrs = text.substr(bracket + 1, end_bracket - bracket - 1);
        for (const std::string& item : split_top_level(attrs, ',')) {
            const size_t eq = item.find('=');
            if (eq == std::string::npos) {
                continue;
            }
            const std::string key = lower_ascii(trim_copy(item.substr(0, eq)));
            const std::string_view value = trim_copy(item.substr(eq + 1));
            if (key == "targets") {
                for (const std::string& target_name : split_top_level(value, '|')) {
                    const auto selector = parse_target_name(target_name);
                    if (selector.has_value()) {
                        rule.targets.push_back(*selector);
                    }
                }
            } else if (key == "benefit") {
                rule.benefit = static_cast<uint32_t>(std::max(1, std::stoi(std::string(value))));
            } else if (key == "require_fma") {
                rule.require_fma = ascii_equal_case_insensitive(value, "true") || value == "1" || ascii_equal_case_insensitive(value, "yes");
            }
        }
    }

    return rule;
}

struct MatchContext {
    const ir::IRStream& stream;
    std::unordered_map<uint32_t, size_t> producer_by_result;

    explicit MatchContext(const ir::IRStream& ir_stream) : stream(ir_stream) {
        producer_by_result.reserve(ir_stream.size() * 2u + 1u);

        for (size_t i = 0; i < ir_stream.size(); ++i) {
            const ir::IRNode node = ir_stream.get_node_readonly(i);
            if (node.result_id != ir::INVALID_SLOT) {
                producer_by_result.emplace(node.result_id, i);
            }
        }
    }

    [[nodiscard]] const ir::IRNode* lookup_producer(uint32_t slot) const noexcept {
        const auto it = producer_by_result.find(slot);
        if (it == producer_by_result.end()) return nullptr;
        return &stream.get_node_readonly(it->second);
    }
};

[[nodiscard]] inline bool target_selector_matches(const TargetSelector& selector,
                                                  const core::TargetMachine& target) noexcept {
    if (selector.arch.has_value() && target.get_arch() != *selector.arch) {
        return false;
    }
    if (selector.family.has_value() && target.family() != *selector.family) {
        return false;
    }
    if (selector.object_format.has_value() && target.object_format() != *selector.object_format) {
        return false;
    }
    return true;
}

[[nodiscard]] inline bool arch_matches(const SelectionRule& rule,
                                       const core::TargetMachine& target) noexcept {
    if (rule.targets.empty()) {
        return true;
    }
    return std::any_of(rule.targets.begin(), rule.targets.end(),
                       [&target](const TargetSelector& selector) {
                           return target_selector_matches(selector, target);
                       });
}

[[nodiscard]] inline bool supports_rule_requirements(const SelectionRule& rule,
                                                      const core::TargetMachine& target) noexcept {
    if (rule.require_fma && !target.has_feature(core::TargetFeature::FMA)) {
        return false;
    }
    return true;
}

[[nodiscard]] inline uint32_t operand_at(const ir::IRNode& node, size_t index) noexcept {
    switch (index) {
        case 0: return node.op_a;
        case 1: return node.op_b;
        case 2: return node.op_c;
        default: return ir::INVALID_SLOT;
    }
}

[[nodiscard]] inline bool match_term(const PatternTerm& term,
                                     const ir::IRNode& node,
                                     const MatchContext& ctx,
                                     std::unordered_map<std::string, uint32_t>& captures) {
    if (!term.capture.empty()) {
        captures[term.capture] = node.result_id;
        return true;
    }

    if (!term.opcode.has_value() || node.opcode != *term.opcode) {
        return false;
    }

    for (size_t i = 0; i < term.children.size(); ++i) {
        const PatternTerm& child = term.children[i];
        const uint32_t operand = operand_at(node, i);
        if (!child.capture.empty()) {
            captures[child.capture] = operand;
            continue;
        }
        const ir::IRNode* producer = ctx.lookup_producer(operand);
        if (producer == nullptr || !match_term(child, *producer, ctx, captures)) {
            return false;
        }
    }

    return true;
}

[[nodiscard]] inline uint32_t lookup_capture(const std::unordered_map<std::string, uint32_t>& captures,
                                             const std::string& name) noexcept {
    const auto it = captures.find(name);
    return it == captures.end() ? ir::INVALID_SLOT : it->second;
}

[[nodiscard]] inline ir::IRNode build_replacement(const SelectionRule& rule,
                                                   const ir::IRNode& original,
                                                   const std::unordered_map<std::string, uint32_t>& captures) {
    ir::IRNode node = original;
    if (rule.replace.opcode.has_value()) {
        node.opcode = *rule.replace.opcode;
    }

    node.op_a = ir::INVALID_SLOT;
    node.op_b = ir::INVALID_SLOT;
    node.op_c = ir::INVALID_SLOT;
    for (size_t i = 0; i < rule.replace.children.size() && i < 3; ++i) {
        const PatternTerm& child = rule.replace.children[i];
        const uint32_t value = child.capture.empty() ? ir::INVALID_SLOT : lookup_capture(captures, child.capture);
        switch (i) {
            case 0: node.op_a = value; break;
            case 1: node.op_b = value; break;
            case 2: node.op_c = value; break;
            default: break;
        }
    }
    node.refresh_intent_hash();
    return node;
}

} // namespace detail

[[nodiscard]] inline MachineForm classify_machine_form_with_fallback(
    const core::TargetMachine& target,
    ir::OpCode opcode) noexcept {
    // These HIR operations are semantic/addressing operations whose generic
    // generated target tables may describe the lowering mechanics as integer
    // arithmetic. Common MIR must retain their target-independent move/address
    // shape so the backend never has to recover HIR opcode meaning.
    switch (opcode) {
        case ir::OpCode::SELECT:
        case ir::OpCode::BOOL_SELECT:
        case ir::OpCode::GEP:
        case ir::OpCode::LEA:
        case ir::OpCode::PTR_CAST:
        case ir::OpCode::PTR_TO_INT:
        case ir::OpCode::INT_TO_PTR:
            return MachineForm::Move;
        default:
            break;
    }

    const MachineForm form = generated::classify_machine_form(target, opcode);
    if (form != MachineForm::Unhandled) [[likely]] {
        return form;
    }

    // Runtime routing is an explicit generated/runtime contract. No hand-written
    // opcode fallback is permitted here; unsupported operations remain visible.
    if (runtime_policy::should_route_unhandled_to_runtime_call(target, opcode)) {
        return MachineForm::Call;
    }

    return MachineForm::Unhandled;
}

    [[nodiscard]] static InstructionSelectorProgram from_dsl(std::string_view dsl_text) {
        InstructionSelectorProgram program;

                program.rules_.push_back(std::move(rule));

        }

    }

    [[nodiscard]] static InstructionSelectorProgram from_rule_lines(std::span<const std::string_view> lines) {
        InstructionSelectorProgram program;
        for (const std::string_view line_view : lines) {
            const std::string_view line = detail::trim_copy(line_view);
            if (line.empty() || line[0] == '#') {
                continue;
            }
            detail::SelectionRule rule = detail::parse_rule_line(line);
            if (!rule.name.empty() && rule.match.opcode.has_value() && rule.replace.opcode.has_value()) {
                program.rules_.push_back(std::move(rule));
            }
        }
        return program;
    }

    [[nodiscard]] ISelReplacement select_node(const ir::IRStream& stream,
                                              size_t index,
                                              const ir::IRNode& node,
                                              const core::TargetMachine& target) const {
        (void)index;
        detail::MatchContext ctx(stream);
        ISelReplacement best{};
        for (const detail::SelectionRule& rule : rules_) {
            if (!detail::arch_matches(rule, target)) {
                continue;
            }
            if (!detail::supports_rule_requirements(rule, target)) {
                continue;
            }
            if (!rule.match.opcode.has_value() || node.opcode != *rule.match.opcode) {
                continue;
            }

            std::unordered_map<std::string, uint32_t> captures;
            if (!detail::match_term(rule.match, node, ctx, captures)) {
                continue;
            }

            if (!best.matched || rule.benefit > best.benefit) {
                best.matched = true;
                best.rule_name = rule.name;
                best.benefit = rule.benefit;
                best.node = detail::build_replacement(rule, node, captures);
            }
        }
        return best;
    }

    [[nodiscard]] SelectedInstruction select_instruction(const ir::IRStream& stream,
                                                         size_t index,
                                                         const ir::IRNode& node,
                                                         const core::TargetMachine& target) const {
        // Every opcode is eligible for DSL rule selection. Do not gate matching
        // on a hand-maintained opcode whitelist; the .dgtl_isel rule set is the
        // selection authority and generated target contracts validate it.
        const ISelReplacement replacement = select_node(stream, index, node, target);
        SelectedInstruction selected{};
        selected.node = replacement.matched ? replacement.node : node;
        selected.semantic_opcode = selected.node.opcode;
        selected.rule_name = replacement.rule_name;
        selected.matched_rule = replacement.matched;
        selected.form = classify_machine_form_with_fallback(target, selected.node.opcode);
        selected.semantic_kind =
            semantic_kind_for(selected.semantic_opcode, selected.form);
        return selected;
    }

    [[nodiscard]] machine::MachineInstruction& append_to_machine_function(
        machine::MachineFunctionBuilder& builder,
        machine::BlockId block,
        const SelectedInstruction& selected,
        machine::RegisterClassKind result_class = machine::RegisterClassKind::General) const {
        if (selected.form == MachineForm::Unhandled ||
            selected.semantic_kind == machine::MachineSemanticKind::None ||
            !machine::semantic_kind_matches_form(selected.form, selected.semantic_kind)) {
            throw std::invalid_argument(
                "DGTL ISel contract violation: invalid HIR-to-MIR semantic/form pair");
        }
        return builder.append(block, to_machine_selection(selected, result_class), result_class);
    }

    [[nodiscard]] bool empty() const noexcept {
        return rules_.empty();
    }

private:
    std::vector<detail::SelectionRule> rules_;
};

class MachineLegalizer {
public:
    static constexpr uint32_t kScalarWidth128TransportFlag = ir::ExtendedFlags::VEC_WIDTH_128;

    void register_scalar128_operands(size_t node_index, const WideScalar128Operands& operands) {
        if (node_index >= wide_scalar_nodes_.size()) {
            wide_scalar_nodes_.resize(node_index + 1);
        }
        wide_scalar_nodes_[node_index] = operands;
    }

    [[nodiscard]] static bool requires_legalization(const ir::IRNode& node,
                                                    const core::TargetMachine& target) noexcept {
        if (is_branch_hint_opcode(node.opcode)) {
            return true;
        }

        // 128-bit scalar arithmetic is represented by the transport flag on the
        // semantic IR node. Legalization is target-driven: native 128-bit targets
        // keep the operation intact; narrower targets require decomposition.
        if ((node.logic_flags & kScalarWidth128TransportFlag) != 0 &&
            !target.description().supports_scalar_width(128)) {
            switch (node.opcode) {
                case ir::OpCode::ADD_I64:
                case ir::OpCode::SUB_I64:
                    return true;
                default:
                    break;
            }
        }

        // These opcodes encode a fixed 512-bit vector contract. Do not allow
        // generated ISel for another compile-time target to silently authorize
        // them on a runtime TargetMachine that lacks 512-bit vectors.
        if (!target.description().supports_vector_width(512)) {
            switch (node.opcode) {
                case ir::OpCode::VEC_ADD_F32_512:
                case ir::OpCode::VEC_MUL_F32_512:
                case ir::OpCode::GATHER_F32_512:
                case ir::OpCode::STORE_V512:
                    return true;
                default:
                    break;
            }
        }

        if (node.opcode == ir::OpCode::FMA_F32 && !target.supports_fma()) {
            return true;
        }

        return false;

                                                        const core::TargetMachine& target) noexcept {
        if (node.opcode == ir::OpCode::FMA_F32 && !target.supports_fma()) {
            return runtime_contract::has_runtime_fallback(target, node.opcode);

            switch (node.opcode) {
                case ir::OpCode::VEC_ADD_F32_512:
                case ir::OpCode::VEC_MUL_F32_512:
                case ir::OpCode::GATHER_F32_512:
                case ir::OpCode::STORE_V512:
                    return runtime_contract::has_runtime_fallback(target, node.opcode);
                default:
                    break;
            }
        }

        return false;
    }

    [[nodiscard]] LegalizationResult legalize(const ir::IRStream& input, const core::TargetMachine& target) const {
        if (input.is_null()) {
            return {};
        }

        constexpr std::size_t kExpansionFactor = 6u;
        constexpr std::size_t kExpansionSlack = 32u;

        if (input.size() >
            (std::numeric_limits<std::size_t>::max() - kExpansionSlack) /
                kExpansionFactor) {
            LegalizationResult failed;
            failed.remarks.push_back(
                LegalizationRemark{0u, "legalization input is too large for bounded transactional capacity"});
            return failed;
        }

        LegalizationResult result;
        result.stream = ir::IRStream::allocate_transactional(
            input.size() * kExpansionFactor + kExpansionSlack);

        uint32_t next_temp = compute_next_virtual_id(input);

        result.remarks.reserve(input.size() / 4);

        auto propagate_node_metadata = [&](size_t in_idx, size_t out_idx) {
            const auto overflow = input.get_call_overflow_args(in_idx);
            if (!overflow.empty()) {
                result.stream.set_call_overflow_args(out_idx, overflow);
            }
            const auto words = input.get_call_abi_layout_words(in_idx);
            if (!words.empty()) {
                result.stream.set_call_abi_layout_words(out_idx, words);
            }
        };

        for (size_t i = 0; i < input.size(); ++i) {
            const ir::IRNode node = input.get_node(i);
            if (node.result_id != ir::INVALID_SLOT && node.result_id != 0u) {
                result.stream.set_result_type(node.result_id, input.get_result_type_code(node.result_id));
            }
            // Canonicalize semantic branch hints before consulting target-aware
            // lowering policy. The canonical opcode is the backend contract;
            // the hint remains represented by the branch flags.
            if (is_branch_hint_opcode(node.opcode)) {
                result.changed = true;
                ir::IRNode canonical = node;
                canonical.opcode = ir::OpCode::BRANCH_COND;
                canonical.flags |= ir::Flags::IS_BRANCH;
                canonical.refresh_intent_hash();
                result.stream.push_back(canonical);
                propagate_node_metadata(i, result.stream.size() - 1);
                result.remarks.push_back(LegalizationRemark{i, "canonicalized BRANCH_LIKELY/BRANCH_UNLIKELY hint opcode to BRANCH_COND for backend-stable lowering"});
                continue;
            }

            const bool is_scalar128_expansion =
                (node.logic_flags & kScalarWidth128TransportFlag) != 0 &&
                !target.description().supports_scalar_width(128) &&
                (node.opcode == ir::OpCode::ADD_I64 || node.opcode == ir::OpCode::SUB_I64);

            // Structural legalization owns transformations that change the IR
            // shape. Do this before runtime/direct-ISel classification so an
            // operation that can be decomposed is never incorrectly routed to
            // a runtime helper merely because its original opcode is unsupported.
            if (is_scalar128_expansion) {
                if (i >= wide_scalar_nodes_.size() ||
                    wide_scalar_nodes_[i].lhs_lo == ir::INVALID_SLOT ||
                    wide_scalar_nodes_[i].lhs_hi == ir::INVALID_SLOT ||
                    wide_scalar_nodes_[i].rhs_lo == ir::INVALID_SLOT ||
                    wide_scalar_nodes_[i].rhs_hi == ir::INVALID_SLOT ||
                    wide_scalar_nodes_[i].result_lo == ir::INVALID_SLOT ||
                    wide_scalar_nodes_[i].result_hi == ir::INVALID_SLOT) {
                    result.remarks.push_back(LegalizationRemark{
                        i,
                        "128-bit scalar legalization requested without registered wide operand metadata; node preserved"
                    });
                    result.stream.push_back(node);
                    propagate_node_metadata(i, result.stream.size() - 1);
                    continue;
                }

                if (next_temp == ir::INVALID_SLOT) {
                    result.stream.push_back(node);
                    propagate_node_metadata(i, result.stream.size() - 1);
                    result.remarks.push_back(LegalizationRemark{
                        i,
                        "128-bit scalar legalization aborted: virtual-register ID space is exhausted; node preserved"
                    });
                    continue;
                }

                result.changed = true;
                expand_scalar128(node, wide_scalar_nodes_[i], next_temp, result.stream);
                result.remarks.push_back(LegalizationRemark{
                    i,
                    "expanded non-native 128-bit scalar arithmetic into low/high halves with carry chain"
                });
                continue;
            }

            // Target legality takes precedence over generated architecture-level
            // form classification. A generated form can describe an opcode for
            // an ISA family while the concrete TargetMachine lacks the required
            // width/feature. Such a case must use a target-valid runtime fallback
            // when one exists.
            if (requires_legalization(node, target)) {
                result.stream.push_back(node);
                propagate_node_metadata(i, result.stream.size() - 1);
                if (requires_runtime_fallback(node, target)) {
                    result.remarks.push_back(LegalizationRemark{
                        i,
                        "target-specific legality mismatch: generated form exists for the ISA family, but the selected TargetMachine requires runtime fallback"
                    });
                } else {
                    result.remarks.push_back(LegalizationRemark{
                        i,
                        "target requires legalization for this opcode, but no structural lowering recipe or runtime fallback exists; node preserved"
                    });
                }
                continue;
            }

            // Runtime/direct-ISel policy is target-aware. Generated ISel is
            // authoritative for direct lowering; runtime symbols are only a
            // fallback contract for the exact selected TargetMachine.
            const runtime_policy::LegalizationRoute route =
                runtime_policy::classify(target, node.opcode);

            if (route == runtime_policy::LegalizationRoute::RuntimeCall) {
                result.stream.push_back(node);
                propagate_node_metadata(i, result.stream.size() - 1);
                result.remarks.push_back(LegalizationRemark{
                    i,
                    "target-aware runtime-backed policy: no direct generated ISel mapping exists for this target; preserved for runtime-call lowering"
                });
                continue;
            }

            if (route == runtime_policy::LegalizationRoute::Unhandled) {
                result.stream.push_back(node);
                propagate_node_metadata(i, result.stream.size() - 1);
                result.remarks.push_back(LegalizationRemark{
                    i,
                    "opcode has neither a target-specific generated ISel mapping nor a runtime contract; preserved for verifier/diagnostic handling"
                });
                continue;
            }

            result.stream.push_back(node);
            propagate_node_metadata(i, result.stream.size() - 1);
        }

        return result;
    }

private:
    std::vector<WideScalar128Operands> wide_scalar_nodes_;

    [[nodiscard]] static bool is_branch_hint_opcode(ir::OpCode opcode) noexcept {
        return opcode == ir::OpCode::BRANCH_LIKELY || opcode == ir::OpCode::BRANCH_UNLIKELY;
    }

    [[nodiscard]] static uint32_t compute_next_virtual_id(
        const ir::IRStream& stream) noexcept {
        // IRStream already owns the canonical result-ID interpretation.
        // Do not scan raw op_a/op_b/op_c here: some HIR operations encode
        // immediates in those fields, and treating them as virtual IDs can
        // spuriously exhaust the ID space.
        const uint32_t max_id = stream.get_max_id();

        if (max_id == ir::INVALID_SLOT || max_id >= 0xFFFFFFFEu) {
            return ir::INVALID_SLOT;
        }
        return max_id + 1u;
    }

    static void clear_width_transport_flag(ir::IRNode& node) noexcept {
        node.logic_flags &= ~kScalarWidth128TransportFlag;
    }

    static void expand_scalar128(const ir::IRNode& original,
                                 const WideScalar128Operands& pair,
                                 uint32_t& next_temp,
                                 ir::IRStream& out) {
        ir::IRNode low_op = original;
        low_op.result_id = next_temp++;
        low_op.op_a = pair.lhs_lo;
        low_op.op_b = pair.rhs_lo;
        low_op.op_c = ir::INVALID_SLOT;
        clear_width_transport_flag(low_op);
        low_op.refresh_intent_hash();
        out.push_back(low_op);
        
        uint32_t low_result_temp = low_op.result_id;

        // FIXED: Use CMP_LT_I64 for accurate 64-bit unsigned carry/borrow detection
        ir::IRNode carry_detect{};
        carry_detect.opcode = ir::OpCode::CMP_LT_I64;
        carry_detect.flags = ir::Flags::NONE;
        carry_detect.result_id = next_temp++;
        if (original.opcode == ir::OpCode::SUB_I64) {
            // Unsigned borrow: lhs_low < rhs_low.
            carry_detect.op_a = pair.lhs_lo;
            carry_detect.op_b = pair.rhs_lo;
        } else {
            // Unsigned carry: wrapped low result < lhs_low.
            carry_detect.op_a = low_result_temp;
            carry_detect.op_b = pair.lhs_lo;
        }
        carry_detect.op_c = ir::INVALID_SLOT;
        carry_detect.logic_flags = ir::ExtendedFlags::LATENCY_CRITICAL;
        carry_detect.refresh_intent_hash();
        out.push_back(carry_detect);
        
        uint32_t carry_bit = carry_detect.result_id;

        ir::IRNode high_partial = original;
        high_partial.result_id = next_temp++;
        high_partial.op_a = pair.lhs_hi;
        high_partial.op_b = pair.rhs_hi;
        high_partial.op_c = ir::INVALID_SLOT;
        clear_width_transport_flag(high_partial);
        high_partial.refresh_intent_hash();
        out.push_back(high_partial);
        
        uint32_t high_partial_result = high_partial.result_id;

        ir::IRNode high_with_carry = original;
        // Addition propagates a carry; subtraction consumes a borrow.
        high_with_carry.opcode = (original.opcode == ir::OpCode::SUB_I64)
            ? ir::OpCode::SUB_I64
            : ir::OpCode::ADD_I64;
        high_with_carry.result_id = next_temp++;
        high_with_carry.op_a = high_partial_result;
        high_with_carry.op_b = carry_bit;
        high_with_carry.op_c = ir::INVALID_SLOT;
        clear_width_transport_flag(high_with_carry);
        high_with_carry.refresh_intent_hash();
        out.push_back(high_with_carry);
        
        uint32_t high_result_temp = high_with_carry.result_id;

        ir::IRNode move_low{};
        move_low.opcode = ir::OpCode::REG_MOVE;
        move_low.flags = ir::Flags::NONE;
        move_low.result_id = pair.result_lo;
        move_low.op_a = low_result_temp;
        move_low.op_b = ir::INVALID_SLOT;
        move_low.op_c = ir::INVALID_SLOT;
        move_low.logic_flags = ir::ExtendedFlags::DETERMINISTIC;
        move_low.refresh_intent_hash();
        out.push_back(move_low);

        ir::IRNode move_high{};
        move_high.opcode = ir::OpCode::REG_MOVE;
        move_high.flags = ir::Flags::NONE;
        move_high.result_id = pair.result_hi;
        move_high.op_a = high_result_temp;
        move_high.op_b = ir::INVALID_SLOT;
        move_high.op_c = ir::INVALID_SLOT;
        move_high.logic_flags = ir::ExtendedFlags::DETERMINISTIC;
        move_high.refresh_intent_hash();
        out.push_back(move_high);
    }
};

struct RegisterMask {
    std::vector<std::uint64_t> words;

    void resize(std::size_t bit_count) {
        words.resize((bit_count + 63u) / 64u);
    }

    void clear() noexcept {
        std::fill(words.begin(), words.end(), 0u);
    }

    void set(std::size_t bit) {
        const std::size_t word = bit / 64u;
        if (word >= words.size()) {
            words.resize(word + 1u);
        }
        words[word] |= (std::uint64_t{1} << (bit % 64u));
    }

    void reset(std::size_t bit) noexcept {
        const std::size_t word = bit / 64u;
        if (word < words.size()) {
            words[word] &= ~(std::uint64_t{1} << (bit % 64u));
        }
    }

    [[nodiscard]] bool test(std::size_t bit) const noexcept {
        const std::size_t word = bit / 64u;
        return word < words.size() &&
               (words[word] & (std::uint64_t{1} << (bit % 64u))) != 0;
    }
};

[[nodiscard]] constexpr bool
semantic_kind_matches_form(MachineForm form, MachineSemanticKind kind) noexcept {
    if (kind == MachineSemanticKind::None) {
        return false;
    }
    switch (form) {
        case MachineForm::Move: return kind == MachineSemanticKind::Move;
        case MachineForm::LoadImmediate: return kind == MachineSemanticKind::LoadImmediate;
        case MachineForm::LoadStack: return kind == MachineSemanticKind::LoadStack;
        case MachineForm::StoreStack: return kind == MachineSemanticKind::StoreStack;
        case MachineForm::IntegerBinary32:
        case MachineForm::IntegerBinary64:
            return kind >= MachineSemanticKind::IntegerAdd &&
                   kind <= MachineSemanticKind::IntegerShr;
        case MachineForm::FloatBinary32:
        case MachineForm::FloatBinary64:
            return kind >= MachineSemanticKind::FloatAdd &&
                   kind <= MachineSemanticKind::FloatDiv;
        case MachineForm::FloatTernaryFma32: return kind == MachineSemanticKind::FloatFma;
        case MachineForm::Compare32: return kind == MachineSemanticKind::Compare;
        case MachineForm::VectorBinary512:
            return kind == MachineSemanticKind::VectorAdd ||
                   kind == MachineSemanticKind::VectorMul;
        case MachineForm::VectorGather512: return kind == MachineSemanticKind::VectorGather;
        case MachineForm::VectorStore512: return kind == MachineSemanticKind::VectorStore;
        case MachineForm::Call: return kind == MachineSemanticKind::Call;
        case MachineForm::Return: return kind == MachineSemanticKind::Return;
        case MachineForm::Goto: return kind == MachineSemanticKind::Goto;
        case MachineForm::BranchCond: return kind == MachineSemanticKind::BranchCond;
        case MachineForm::BranchCompare: return kind == MachineSemanticKind::BranchCompare;
        case MachineForm::Syscall: return kind == MachineSemanticKind::Syscall;
        case MachineForm::TensorOp: return kind == MachineSemanticKind::TensorOp;
        case MachineForm::HardwareFence: return kind == MachineSemanticKind::HardwareFence;
        case MachineForm::AtomicOp: return kind == MachineSemanticKind::AtomicOp;
        case MachineForm::ConcurrentSync: return kind == MachineSemanticKind::ConcurrentSync;
        case MachineForm::NoOp: return kind == MachineSemanticKind::Other;
        case MachineForm::Unhandled: return false;
    }
    return false;
}

[[nodiscard]] constexpr bool
requires_semantic_kind(MachineForm form) noexcept {
    switch (form) {
        case MachineForm::Unhandled:
            return false;
        default:
            return true;
    }
}

    Load            = 1u << 4,
    Store           = 1u << 5,
    Atomic          = 1u << 6,
    Barrier         = 1u << 7,
    SideEffect      = 1u << 8,
    Commutable      = 1u << 9,
    MayLoad         = 1u << 10,
    MayStore        = 1u << 11,
    HasUnmodeledSideEffect = 1u << 12,
    Pseudo          = 1u << 13,
    Debug           = 1u << 14,
    FrameSetup      = 1u << 15,
    FrameDestroy    = 1u << 16
};

constexpr MachineInstrFlag operator|(MachineInstrFlag a,
                                     MachineInstrFlag b) noexcept {
    return static_cast<MachineInstrFlag>(
        static_cast<std::uint32_t>(a) |
        static_cast<std::uint32_t>(b));
}

constexpr MachineInstrFlag operator&(MachineInstrFlag a,
                                     MachineInstrFlag b) noexcept {
    return static_cast<MachineInstrFlag>(
        static_cast<std::uint32_t>(a) &
        static_cast<std::uint32_t>(b));
}

struct alignas(64) MachineInstruction {
    InstrId id = INVALID_INSTR_ID;
    MachineForm form = MachineForm::Unhandled;
    MachineSemanticKind semantic_kind = MachineSemanticKind::None;
    MachineOpcode opcode = INVALID_MACHINE_OPCODE;
    MachineInstrFlag flags = MachineInstrFlag::None;

    // Intrusive chaining pointers for zero-allocation O(1) splicing
    BlockId parent_block = INVALID_BLOCK_ID;
    InstrId prev_instr   = INVALID_INSTR_ID;
    InstrId next_instr   = INVALID_INSTR_ID;

    std::vector<MachineOperand> operands;

    /*
     * Semantic provenance is opaque metadata only. It is useful for diagnostics,
     * debug mapping and optimization remarks, but it is never interpreted as
     * an executable machine opcode and must never be cast back to IR opcode
     * identity below the MIR boundary.
     */
    // Opaque provenance only. This value is never executable opcode identity.
    std::uint32_t semantic_provenance = 0;
    std::uint32_t debug_id = 0;

    [[nodiscard]] bool has_target_opcode() const noexcept {
        return opcode != INVALID_MACHINE_OPCODE;
    }

    // Target opcode identity is write-once at the MIR boundary. This prevents
    // later passes from silently changing the machine instruction kind after
    // target lowering has established it.
    [[nodiscard]] bool assign_target_opcode_once(MachineOpcode target_opcode) noexcept {
        if (target_opcode == INVALID_MACHINE_OPCODE || has_target_opcode()) {
            return false;
        }
        opcode = target_opcode;
        return true;
    }

    [[nodiscard]] bool has_semantic_kind() const noexcept {
        return semantic_kind != MachineSemanticKind::None;
    }

    [[nodiscard]] bool has_flag(MachineInstrFlag flag) const noexcept {
        return (flags & flag) != MachineInstrFlag::None;
    }

    void add_flag(MachineInstrFlag flag) noexcept {
        flags = flags | flag;
    }

    void add_operand(MachineOperand operand) {
        operands.push_back(std::move(operand));
    }
};

struct MachineBasicBlock {
    BlockId id = INVALID_BLOCK_ID;

    // Head and tail anchors for O(1) instruction iteration
    InstrId head_instr = INVALID_INSTR_ID;
    InstrId tail_instr = INVALID_INSTR_ID;
    std::size_t instr_count = 0;

    std::vector<InstrId> instructions;
    std::vector<BlockId> predecessors;
    std::vector<BlockId> successors;

    std::vector<MachineRegister> live_ins;
    std::vector<MachineRegister> live_outs;

    bool entry = false;
    bool exit = false;
    bool cold = false;
    bool address_taken = false;

    [[nodiscard]] bool valid() const noexcept {
        return id != INVALID_BLOCK_ID;
    }

    void add_instruction(InstrId instr) {
        instructions.push_back(instr);
    }

    void add_successor(BlockId block) {
        if (std::find(successors.begin(), successors.end(), block) ==
            successors.end()) {
            successors.push_back(block);
        }
    }

    void add_predecessor(BlockId block) {
        if (std::find(predecessors.begin(), predecessors.end(), block) ==
            predecessors.end()) {
            predecessors.push_back(block);
        }
    }
};

/* ============================================================================
 * Liveness / live intervals
 *
 * The current DGTL Strike_Live_Manager already models start/end node
 * intervals. The common representation generalizes that concept from IR-node
 * indices to machine instruction positions.
 * ========================================================================== */

struct MachinePosition {
    InstrId instr = INVALID_INSTR_ID;
    std::uint32_t slot = 0;

    friend constexpr bool operator<(const MachinePosition& a,
                                    const MachinePosition& b) noexcept {
        return a.instr < b.instr ||
               (a.instr == b.instr && a.slot < b.slot);

    friend constexpr bool operator==(const MachinePosition&,
                                     const MachinePosition&) noexcept = default;
};

struct MachineLiveRange {
    MachinePosition start = {};
    MachinePosition end = {};
    bool valid = false;

    [[nodiscard]] bool contains(MachinePosition position) const noexcept {
        return valid && !(position < start) && !(end < position);
    }
};

    std::vector<MachineRegister> defs;

    bool spillable = true;

    bool fixed = false;
};

struct MachineFunctionProperties {
    bool is_ssa = true;
    bool tracks_liveness = false;
    bool registers_allocated = false;
    bool frame_lowered = false;
    bool abi_lowered = false;
    bool contains_calls = false;
    bool contains_dynamic_alloca = false;
};

class MachineFunction {
public:
    explicit MachineFunction(std::string_view name = {})
        : name_(name) {}

    [[nodiscard]] std::string_view name() const noexcept {
        return name_;
    }

    [[nodiscard]] MachinePhase phase() const noexcept {
        return phase_;
    }

    void set_phase(MachinePhase phase) noexcept {
        // Phase transitions are monotonic. A backend pass must never move a
        // function backwards after target lowering, allocation, frame lowering,
        // or ABI lowering has occurred.
        if (static_cast<std::uint8_t>(phase) >=
            static_cast<std::uint8_t>(phase_)) {
            phase_ = phase;
        }
    }

    [[nodiscard]] bool advance_phase(MachinePhase expected_current,
                                     MachinePhase next) noexcept {
        if (phase_ != expected_current) {
            return false;
        }
        if (static_cast<std::uint8_t>(next) <=
            static_cast<std::uint8_t>(expected_current)) {
            return false;
        }
        phase_ = next;
        return true;
    }

    [[nodiscard]] MachineFunctionProperties& properties() noexcept {
        return properties_;
    }

    [[nodiscard]] const MachineFunctionProperties&
    properties() const noexcept {
        return properties_;
    }

    MachineBasicBlock& create_block(bool entry = false) {
        if (blocks_.size() >= static_cast<std::size_t>(INVALID_BLOCK_ID)) {
            throw std::overflow_error("MachineFunction block id exhausted");
        }
        const BlockId id = static_cast<BlockId>(blocks_.size());
        blocks_.push_back({});
        blocks_.back().id = id;
        blocks_.back().entry = entry;
        if (entry) {
            if (entry_block_ != INVALID_BLOCK_ID) {
                throw std::logic_error("MachineFunction cannot have multiple entry blocks");
            }
            entry_block_ = id;
        }
        return blocks_.back();
    }

    MachineBasicBlock* block(BlockId id) noexcept {
        if (id >= blocks_.size()) {
            return nullptr;
        }
        return &blocks_[id];
    }

    const MachineBasicBlock* block(BlockId id) const noexcept {
        if (id >= blocks_.size()) {
            return nullptr;
        }
        return &blocks_[id];
    }

    bool connect_blocks(BlockId from, BlockId to) {
        MachineBasicBlock* source = block(from);
        MachineBasicBlock* target = block(to);
        if (source == nullptr || target == nullptr) {
            return false;
        }
        source->add_successor(to);
        target->add_predecessor(from);
        return true;
    }

    [[nodiscard]] std::span<MachineBasicBlock> blocks() noexcept {
        return blocks_;
    }

    [[nodiscard]] std::span<const MachineBasicBlock> blocks() const noexcept {
        return blocks_;
    }

    MachineInstruction& create_instruction() {
        return create_instruction(INVALID_MACHINE_OPCODE);
    }

    MachineInstruction& create_instruction(MachineOpcode opcode) {
        if (instructions_.size() >= static_cast<std::size_t>(INVALID_INSTR_ID)) {
            throw std::overflow_error("MachineFunction instruction id exhausted");
        }
        const InstrId id = static_cast<InstrId>(instructions_.size());
        instructions_.push_back({});
        instructions_.back().id = id;
        instructions_.back().opcode = opcode;
        return instructions_.back();
    }

    // O(1) Splicing Primitive: Inserts instruction before cursor_id without array shifts
    void insert_instruction_before(BlockId block_id, InstrId cursor_id, InstrId new_instr_id) {
        MachineBasicBlock* bb = block(block_id);
        MachineInstruction* new_inst = instruction(new_instr_id);
        if (!bb || !new_inst) return;

        new_inst->parent_block = block_id;
        if (cursor_id == INVALID_INSTR_ID || bb->head_instr == INVALID_INSTR_ID) {
            // Append to tail
            if (bb->tail_instr != INVALID_INSTR_ID) {
                instruction(bb->tail_instr)->next_instr = new_instr_id;
                new_inst->prev_instr = bb->tail_instr;
                new_inst->next_instr = INVALID_INSTR_ID;
                bb->tail_instr = new_instr_id;
            } else {
                bb->head_instr = bb->tail_instr = new_instr_id;
                new_inst->prev_instr = new_inst->next_instr = INVALID_INSTR_ID;
            }
        } else {
            MachineInstruction* cursor = instruction(cursor_id);
            new_inst->next_instr = cursor_id;
            new_inst->prev_instr = cursor->prev_instr;
            if (cursor->prev_instr != INVALID_INSTR_ID) {
                instruction(cursor->prev_instr)->next_instr = new_instr_id;
            } else {
                bb->head_instr = new_instr_id;
            }
            cursor->prev_instr = new_instr_id;
        }
        bb->instructions.push_back(new_instr_id);
        bb->instr_count++;
    }

    MachineInstruction* instruction(InstrId id) noexcept {
        if (id >= instructions_.size()) {
            return nullptr;
        }
        return &instructions_[id];
    }

    const MachineInstruction* instruction(InstrId id) const noexcept {
        if (id >= instructions_.size()) {
            return nullptr;
        }
        return &instructions_[id];
    }

    [[nodiscard]] std::span<MachineInstruction> instructions() noexcept {
        return instructions_;
    }

    [[nodiscard]] std::span<const MachineInstruction>
    instructions() const noexcept {
        return instructions_;
    }

    MachineRegister create_virtual_register(
        RegisterClassKind reg_class = RegisterClassKind::General) {
        if (next_virtual_reg_ == INVALID_REG) {
            throw std::overflow_error("MachineFunction virtual register id exhausted");
        }
        const RegId id = next_virtual_reg_++;
        return MachineRegister{id, RegisterKind::Virtual, reg_class};
    }

    void add_stack_slot(MachineStackSlot slot) {
        if (slot.id == INVALID_STACK_SLOT) {
            slot.id = static_cast<StackSlotId>(frame_.slots.size());
        }
        frame_.slots.push_back(slot);
    }

    [[nodiscard]] MachineFrameInfo& frame_info() noexcept {
        return frame_;
    }

    [[nodiscard]] const MachineFrameInfo& frame_info() const noexcept {
        return frame_;
    }

    [[nodiscard]] BlockId entry_block() const noexcept {
        return entry_block_;
    }

    [[nodiscard]] std::size_t virtual_register_count() const noexcept {
        return next_virtual_reg_;
    }

    // Assign a target opcode through the function boundary. The common MIR
    // layer does not interpret the numeric value; target lowering owns it.
    [[nodiscard]] bool assign_target_opcode(
        InstrId instruction_id, MachineOpcode target_opcode) noexcept {
        MachineInstruction* inst = instruction(instruction_id);
        return inst != nullptr && inst->assign_target_opcode_once(target_opcode);
    }

    [[nodiscard]] bool has_target_opcodes() const noexcept {
        for (const MachineInstruction& instruction : instructions_) {
            if (instruction.opcode == INVALID_MACHINE_OPCODE) {
                return false;
            }
        }
        return true;
    }

    [[nodiscard]] bool is_phase_at_least(MachinePhase phase) const noexcept {
        return static_cast<std::uint8_t>(phase_) >=
               static_cast<std::uint8_t>(phase);
    }

    /*
     * Register-allocation boundary:
     * pre-regalloc MIR may contain virtual registers; emission-ready MIR may
     * not.  The common layer owns this invariant so target encoders do not
     * need to reconstruct storage from IR values.
     */
    [[nodiscard]] bool has_virtual_registers() const noexcept {
        for (const MachineInstruction& instruction : instructions_) {
            for (const MachineOperand& operand : instruction.operands) {
                if (operand.kind == MachineOperandKind::Register &&
                    operand.reg.is_virtual()) {
                    return true;
                }
                if (operand.kind == MachineOperandKind::Memory) {
                    if (operand.memory.base.is_virtual() ||
                        operand.memory.index.is_virtual()) {
                        return true;
                    }
                }
            }
        }
        return false;
    }

    /*
     * Target register allocators replace virtual operands in-place.  This is
     * intentionally a mechanical MIR operation: no IRNode/SSA lookup is
     * permitted at this stage.
     */
    [[nodiscard]] bool replace_virtual_register(
        MachineRegister virtual_reg,
        MachineRegister physical_reg) noexcept {
        if (!virtual_reg.is_virtual() || !physical_reg.is_physical()) {
            return false;
        }

        bool replaced = false;
        for (MachineInstruction& instruction : instructions_) {
            for (MachineOperand& operand : instruction.operands) {
                if (operand.kind == MachineOperandKind::Register &&
                    operand.reg == virtual_reg) {
                    operand.reg = physical_reg;
                    replaced = true;
                }

                if (operand.kind == MachineOperandKind::Memory) {
                    if (operand.memory.base == virtual_reg) {
                        operand.memory.base = physical_reg;
                        replaced = true;
                    }
                    if (operand.memory.index == virtual_reg) {
                        operand.memory.index = physical_reg;
                        replaced = true;
                    }
                }
            }
        }
        return replaced;
    }

    [[nodiscard]] bool mark_target_lowered() noexcept {
        if (!has_target_opcodes()) {
            return false;
        }
        // Target opcode assignment is the boundary between common legalized MIR
        // and target-lowered MIR. The next phase is therefore PreRegAlloc.
        if (static_cast<std::uint8_t>(phase_) <
            static_cast<std::uint8_t>(MachinePhase::PreRegAlloc)) {
            phase_ = MachinePhase::PreRegAlloc;
        }
        return true;
    }

    [[nodiscard]] bool mark_frame_lowered() noexcept {
        if (!properties_.registers_allocated || has_virtual_registers()) {
            return false;
        }
        properties_.frame_lowered = true;
        phase_ = MachinePhase::FrameLowered;
        return true;
    }

    [[nodiscard]] bool mark_abi_lowered() noexcept {
        if (!properties_.frame_lowered) {
            return false;
        }
        properties_.abi_lowered = true;
        phase_ = MachinePhase::ABILowered;
        return true;
    }

    [[nodiscard]] bool mark_emission_ready() noexcept {
        if (!properties_.abi_lowered || has_virtual_registers() || !has_target_opcodes()) {
            return false;
        }
        phase_ = MachinePhase::EmissionReady;
        return true;
    }

    void mark_registers_allocated() noexcept {
        // Allocation may only complete after target lowering has assigned
        // concrete opcodes. Virtual-register elimination itself remains a pure
        // MIR operation and never consults source IR.
        properties_.registers_allocated =
            has_target_opcodes() && !has_virtual_registers();
        if (properties_.registers_allocated &&
            is_phase_at_least(MachinePhase::PreRegAlloc)) {
            phase_ = MachinePhase::PostRegAlloc;
        }
    }

private:
    std::string name_;
    MachinePhase phase_ = MachinePhase::Selected;
    MachineFunctionProperties properties_{};

    std::vector<MachineBasicBlock> blocks_;
    std::vector<MachineInstruction> instructions_;

    MachineFrameInfo frame_{};

    RegId next_virtual_reg_ = 0;
    BlockId entry_block_ = INVALID_BLOCK_ID;
};

/* ============================================================================
 * MachineForm -> MachineFunction construction
 *
 * This is the bridge from ISel output to the common machine layer. It does
 * not know X64/ARM64/RISC-V register numbers and does not include IR headers.
 * Source value ids become virtual registers; target lowering later decides
 * physical registers and target opcodes.
 * ========================================================================== */

enum class MachineSelectionOperandKind : std::uint8_t {
    Invalid,
    Value,
    Immediate,
    FloatingImmediate,
    Memory,
    BasicBlock,
    Symbol,
    StackSlot,
    Register
};

struct MachineSelectionOperand {
    MachineSelectionOperandKind kind = MachineSelectionOperandKind::Invalid;
    std::uint32_t value = INVALID_MACHINE_ID;
    std::int64_t immediate = 0;
    double floating_immediate = 0.0;
    MachineMemoryOperand memory = {};
    BlockId block = INVALID_BLOCK_ID;
    SymbolId symbol = INVALID_SYMBOL;
    StackSlotId stack_slot = INVALID_STACK_SLOT;
    MachineRegister reg = {};
    RegisterClassKind reg_class = RegisterClassKind::Invalid;
    OperandRole role = OperandRole::Use;

    [[nodiscard]] static MachineSelectionOperand value_ref(
        std::uint32_t id,
        OperandRole role = OperandRole::Use) noexcept {
        MachineSelectionOperand operand;
        operand.kind = MachineSelectionOperandKind::Value;
        operand.value = id;
        operand.role = role;
        return operand;
    }

    [[nodiscard]] static MachineSelectionOperand imm(
        std::int64_t v) noexcept {
        MachineSelectionOperand operand;
        operand.kind = MachineSelectionOperandKind::Immediate;
        operand.immediate = v;
        return operand;
    }

    [[nodiscard]] static MachineSelectionOperand block_ref(
        BlockId id) noexcept {
        MachineSelectionOperand operand;
        operand.kind = MachineSelectionOperandKind::BasicBlock;
        operand.block = id;
        return operand;
    }
};

struct MachineSelection {
    /*
     * MachineSelection is the output of instruction selection and the input to
     * the common MIR builder. It describes a target-independent MachineForm.
     *
     * IMPORTANT: there is deliberately NO target MachineOpcode here.
     * Target opcode assignment belongs exclusively to target lowering.
     */
    MachineForm form = MachineForm::Unhandled;
    MachineSemanticKind semantic_kind = MachineSemanticKind::None;
    std::uint32_t semantic_provenance = 0;
    std::uint32_t debug_id = 0;
    std::uint32_t result_value = INVALID_MACHINE_ID;
    RegisterClassKind result_class = RegisterClassKind::General;

    // Structured operands are the only supported selection representation.
    std::vector<MachineSelectionOperand> operands;

    MachineInstrFlag flags = MachineInstrFlag::None;
};

class MachineFunctionBuilder {
public:
    explicit MachineFunctionBuilder(MachineFunction& function)
        : function_(function) {}

    [[nodiscard]] MachineRegister value_register(
        std::uint32_t value_id,
        RegisterClassKind reg_class = RegisterClassKind::General) {
        if (value_id == INVALID_MACHINE_ID) {
            return {};
        }
        if (value_id >= value_registers_.size()) {
            value_registers_.resize(static_cast<std::size_t>(value_id) + 1u);
        }
        MachineRegister& cached = value_registers_[value_id];
        if (!cached.valid()) {
            cached = function_.create_virtual_register(reg_class);
        }
        return cached;
    }

    [[nodiscard]] MachineInstruction& append(
        BlockId block,
        const MachineSelection& selection,
        RegisterClassKind result_class = RegisterClassKind::General) {
        MachineBasicBlock* bb = function_.block(block);
        if (bb == nullptr) {
            throw std::out_of_range("invalid MachineFunction block");
        }

        if (selection.form == MachineForm::Unhandled) {
            throw std::invalid_argument("cannot append unhandled MachineForm");
        }

        MachineInstruction& instruction =
            function_.create_instruction();
        instruction.form = selection.form;
        instruction.semantic_kind = selection.semantic_kind;
        instruction.semantic_provenance = selection.semantic_provenance;
        instruction.debug_id = selection.debug_id;
        instruction.flags = selection.flags;
        instruction.parent_block = block;

        const RegisterClassKind effective_result_class =
            selection.result_class == RegisterClassKind::Invalid
                ? result_class
                : selection.result_class;

        if (selection.result_value != INVALID_MACHINE_ID) {
            instruction.add_operand(
                MachineOperand::def(
                    value_register(selection.result_value, effective_result_class)));
        }

        // Structured operands are authoritative and mandatory.
        if (!selection.operands.empty()) {
            for (const MachineSelectionOperand& source : selection.operands) {
                switch (source.kind) {
                case MachineSelectionOperandKind::Value:
                    if (source.value != INVALID_MACHINE_ID) {
                        const RegisterClassKind operand_class =
                            source.reg_class == RegisterClassKind::Invalid
                                ? RegisterClassKind::General
                                : source.reg_class;
                        MachineRegister value_reg =
                            value_register(source.value, operand_class);
                        if (source.role == OperandRole::Def) {
                            instruction.add_operand(MachineOperand::def(value_reg));
                        } else {
                            MachineOperand operand = MachineOperand::use(value_reg);
                            operand.role = source.role;
                            instruction.add_operand(std::move(operand));
                        }
                    }
                    break;
                case MachineSelectionOperandKind::Immediate:
                    instruction.add_operand(MachineOperand::imm(source.immediate));
                    break;
                case MachineSelectionOperandKind::FloatingImmediate: {
                    MachineOperand operand;
                    operand.kind = MachineOperandKind::FloatingImmediate;
                    operand.role = source.role;
                    operand.floating_immediate = source.floating_immediate;
                    instruction.add_operand(std::move(operand));
                    break;
                }
                case MachineSelectionOperandKind::Memory: {
                    MachineOperand operand;
                    operand.kind = MachineOperandKind::Memory;
                    operand.role = source.role;
                    operand.memory = source.memory;
                    instruction.add_operand(std::move(operand));
                    break;
                }
                case MachineSelectionOperandKind::BasicBlock:
                    instruction.add_operand(MachineOperand::block_ref(source.block));
                    break;
                case MachineSelectionOperandKind::Symbol: {
                    MachineOperand operand;
                    operand.kind = MachineOperandKind::Symbol;
                    operand.role = source.role;
                    operand.symbol = source.symbol;
                    instruction.add_operand(std::move(operand));
                    break;
                }
                case MachineSelectionOperandKind::StackSlot:
                    instruction.add_operand(MachineOperand::stack_ref(source.stack_slot));
                    break;
                case MachineSelectionOperandKind::Register: {
                    MachineOperand operand;
                    operand.kind = MachineOperandKind::Register;
                    operand.role = source.role;
                    operand.reg = source.reg;
                    instruction.add_operand(std::move(operand));
                    break;
                }
                case MachineSelectionOperandKind::Invalid:
                    throw std::invalid_argument("invalid MachineSelection operand");
                }
            }
        }

        // Intrusively link instruction into block chain
        function_.insert_instruction_before(block, INVALID_INSTR_ID, instruction.id);

        if (instruction.has_flag(MachineInstrFlag::Call)) {
            function_.properties().contains_calls = true;
            function_.frame_info().has_calls = true;
        }
        return instruction;
    }

    [[nodiscard]] MachineBasicBlock& create_block(bool entry = false) {
        return function_.create_block(entry);
    }

private:
    MachineFunction& function_;
    std::vector<MachineRegister> value_registers_;
};

struct MachineVerifyResult {

    BlockId block = INVALID_BLOCK_ID;
    InstrId instruction = INVALID_INSTR_ID;

        return error == MachineVerifyError::None;
    }

class MachineVerifier {
public:
    [[nodiscard]] static MachineVerifyResult
    verify(const MachineFunction& function) noexcept {
        const auto blocks = function.blocks();
        const auto instructions = function.instructions();

        std::vector<std::uint32_t> instruction_membership(instructions.size(), 0u);

        if (function.is_phase_at_least(MachinePhase::PreRegAlloc)) {
            if (blocks.empty() || function.entry_block() == INVALID_BLOCK_ID ||
                function.entry_block() >= blocks.size() ||
                !blocks[function.entry_block()].entry) {
                return {MachineVerifyError::MissingEntryBlock,
                        INVALID_BLOCK_ID, INVALID_INSTR_ID};
            }
        }

        for (std::size_t i = 0; i < blocks.size(); ++i) {
            if (blocks[i].id != static_cast<BlockId>(i)) {
                return {
                    MachineVerifyError::DuplicateBlock,
                    blocks[i].id,
                    INVALID_INSTR_ID
                };
            }

            bool seen_terminator = false;
            for (std::size_t position = 0; position < blocks[i].instructions.size(); ++position) {
                const InstrId id = blocks[i].instructions[position];
                if (id >= instructions.size()) {
                    return {
                        MachineVerifyError::InvalidInstruction,
                        blocks[i].id,
                        id
                    };
                }
                if (++instruction_membership[id] != 1u) {
                    return {
                        MachineVerifyError::DuplicateInstruction,
                        blocks[i].id,
                        id
                    };
                }

                const MachineInstruction& block_instr = instructions[id];

                if (block_instr.form == MachineForm::Unhandled) {
                    return {MachineVerifyError::InvalidInstruction, blocks[i].id, id};
                }

                if (requires_semantic_kind(block_instr.form) &&
                    !block_instr.has_semantic_kind()) {
                    return {MachineVerifyError::MissingSemanticKind, blocks[i].id, id};
                }
                if (block_instr.has_semantic_kind() &&
                    !semantic_kind_matches_form(block_instr.form, block_instr.semantic_kind)) {
                    return {MachineVerifyError::InvalidInstruction, blocks[i].id, id};
                }

                const bool emission_phase =
                    function.is_phase_at_least(MachinePhase::FrameLowered);
                if (emission_phase && !block_instr.has_target_opcode()) {
                    return {MachineVerifyError::MissingTargetOpcode, blocks[i].id, id};
                }

                if (seen_terminator) {
                    return {
                        MachineVerifyError::InvalidInstruction,
                        blocks[i].id,
                        id
                    };
                }
                if (block_instr.has_flag(MachineInstrFlag::Terminator)) {
                    seen_terminator = true;
                    if (position + 1u != blocks[i].instructions.size()) {
                        return {
                            MachineVerifyError::InvalidInstruction,
                            blocks[i].id,
                            id
                        };
                    }
                }

                for (const MachineOperand& operand : block_instr.operands) {
                    switch (operand.kind) {
                    case MachineOperandKind::Register:
                        if (!operand.reg.valid() ||
                            operand.reg.reg_class == RegisterClassKind::Invalid) {
                            return {MachineVerifyError::InvalidRegister, blocks[i].id, id};
                        }
                        break;
                    case MachineOperandKind::BasicBlock:
                        if (operand.block == INVALID_BLOCK_ID || operand.block >= blocks.size()) {
                            return {MachineVerifyError::InvalidOperand, blocks[i].id, id};
                        }
                        break;
                    case MachineOperandKind::StackSlot:
                        if (operand.stack_slot == INVALID_STACK_SLOT ||
                            operand.stack_slot >= function.frame_info().slots.size()) {
                            return {MachineVerifyError::InvalidStackSlot, blocks[i].id, id};
                        }
                        break;
                    case MachineOperandKind::Memory:
                        if (operand.memory.scale != 1 && operand.memory.scale != 2 &&
                            operand.memory.scale != 4 && operand.memory.scale != 8) {
                            return {MachineVerifyError::InvalidMemoryOperand, blocks[i].id, id};
                        }
                        if (operand.memory.alignment == 0) {
                            return {MachineVerifyError::InvalidMemoryOperand, blocks[i].id, id};
                        }
                        if (!operand.memory.load && !operand.memory.store &&
                            (block_instr.has_flag(MachineInstrFlag::Load) ||
                             block_instr.has_flag(MachineInstrFlag::Store))) {
                            return {MachineVerifyError::InvalidMemoryOperand, blocks[i].id, id};
                        }
                        break;
                    case MachineOperandKind::Invalid:
                        return {MachineVerifyError::InvalidOperand, blocks[i].id, id};
                    default:
                        break;
                    }

                    if (operand.kind != MachineOperandKind::Register &&
                        (operand.role == OperandRole::Def ||
                         operand.role == OperandRole::UseDef ||
                         operand.role == OperandRole::ImplicitDef)) {
                        return {MachineVerifyError::InvalidOperandRole, blocks[i].id, id};
                    }
                }
            }

            if (function.is_phase_at_least(MachinePhase::PreRegAlloc) &&
                !blocks[i].instructions.empty() && !seen_terminator) {
                return {MachineVerifyError::MissingTerminator,
                        blocks[i].id, INVALID_INSTR_ID};
            }

            for (BlockId successor : blocks[i].successors) {
                if (successor >= blocks.size()) {
                    return {
                        MachineVerifyError::InvalidCFGEdge,
                        blocks[i].id,
                        INVALID_INSTR_ID
                    };
                }
            }

            for (BlockId predecessor : blocks[i].predecessors) {
                if (predecessor >= blocks.size()) {
                    return {
                        MachineVerifyError::InvalidCFGEdge,
                        blocks[i].id,
                        INVALID_INSTR_ID
                    };
                }
            }
        }

        for (const MachineBasicBlock& block : blocks) {
            for (BlockId successor : block.successors) {
                const MachineBasicBlock& target = blocks[successor];
                if (std::find(target.predecessors.begin(),
                              target.predecessors.end(),
                              block.id) == target.predecessors.end()) {
                    return {
                        MachineVerifyError::InvalidCFGEdge,
                        block.id,
                        INVALID_INSTR_ID
                    };
                }
            }
            for (BlockId predecessor : block.predecessors) {
                const MachineBasicBlock& source = blocks[predecessor];
                if (std::find(source.successors.begin(),
                              source.successors.end(),
                              block.id) == source.successors.end()) {
                    return {
                        MachineVerifyError::InvalidCFGEdge,
                        block.id,
                        INVALID_INSTR_ID
                    };
                }
            }
        }

        for (std::size_t i = 0; i < instruction_membership.size(); ++i) {
            if (instruction_membership[i] != 1u) {
                return {
                    MachineVerifyError::DuplicateInstruction,
                    INVALID_BLOCK_ID,
                    static_cast<InstrId>(i)
                };
            }
        }

        for (const MachineInstruction& instr : instructions) {
            if (instr.id == INVALID_INSTR_ID ||
                instr.form == MachineForm::Unhandled) {
                return {
                    MachineVerifyError::InvalidInstruction,
                    INVALID_BLOCK_ID,
                    instr.id
                };
            }

            // The target-opcode boundary is strict:
            //   Selected/Legalized -> opcode MUST be invalid.
            //   PreRegAlloc+      -> opcode MUST be valid.
            const bool before_target_lowering =
                static_cast<std::uint8_t>(function.phase()) <
                static_cast<std::uint8_t>(MachinePhase::PreRegAlloc);
            if (before_target_lowering !=
                (instr.opcode == INVALID_MACHINE_OPCODE)) {
                return {
                    MachineVerifyError::PhaseViolation,
                    INVALID_BLOCK_ID,
                    instr.id
                };
            }

            for (const MachineOperand& operand : instr.operands) {
                if (operand.kind == MachineOperandKind::Register &&
                    !operand.reg.valid()) {
                    return {
                        MachineVerifyError::InvalidRegister,
                        INVALID_BLOCK_ID,
                        instr.id
                    };
                }

                if (operand.kind == MachineOperandKind::BasicBlock &&
                    operand.block >= blocks.size()) {
                    return {
                        MachineVerifyError::InvalidBlock,
                        operand.block,
                        instr.id
                    };
                }

                if (operand.kind == MachineOperandKind::StackSlot &&
                    operand.stack_slot >= function.frame_info().slots.size()) {
                    return {
                        MachineVerifyError::InvalidStackSlot,
                        INVALID_BLOCK_ID,
                        instr.id
                    };
                }
            }
        }

        for (const MachineStackSlot& slot : function.frame_info().slots) {
            if (slot.id == INVALID_STACK_SLOT) {
                return {
                    MachineVerifyError::InvalidStackSlot,
                    INVALID_BLOCK_ID,
                    INVALID_INSTR_ID
                };
            }
            if (slot.alignment == 0u || slot.size_bytes == 0u) {
                return {
                    MachineVerifyError::InvalidStackSlot,
                    INVALID_BLOCK_ID,
                    INVALID_INSTR_ID
                };
            }
        }

        // Phase contracts are checked centrally so target emitters do not
        // silently accept malformed MIR.
        if (static_cast<std::uint8_t>(function.phase()) >=
            static_cast<std::uint8_t>(MachinePhase::PostRegAlloc)) {
            if (function.has_virtual_registers() ||
                !function.properties().registers_allocated) {
                return {
                    MachineVerifyError::InvalidRegister,
                    INVALID_BLOCK_ID,
                    INVALID_INSTR_ID
                };
            }
        }

        if (static_cast<std::uint8_t>(function.phase()) >=
            static_cast<std::uint8_t>(MachinePhase::FrameLowered)) {
            const auto& frame = function.frame_info();

            // Frame alignment and callee-save sizes are target/ABI facts. The
            // common MIR verifier therefore checks structural validity only;
            // TargetFrameInfo is responsible for target-specific exactness.
            if (frame.stack_alignment == 0u) {
                return {MachineVerifyError::InvalidStackSlot,
                        INVALID_BLOCK_ID, INVALID_INSTR_ID};
            }

            for (const auto& slot : frame.slots) {
                if (!slot.fixed_offset || slot.alignment == 0u ||
                    slot.size_bytes == 0u) {
                    return {MachineVerifyError::InvalidStackSlot,
                            INVALID_BLOCK_ID, INVALID_INSTR_ID};
                }
                const auto alignment = static_cast<std::int64_t>(slot.alignment);
                if (slot.offset % alignment != 0) {
                    return {MachineVerifyError::InvalidStackSlot,
                            INVALID_BLOCK_ID, INVALID_INSTR_ID};
                }
            }
        }

        // Emission-ready MIR must be fully target-lowered and free of virtual
        // registers. Earlier phases intentionally permit an invalid opcode
        // sentinel because target opcode assignment happens later.
        if (function.is_phase_at_least(MachinePhase::EmissionReady)) {
            if (!function.has_target_opcodes() || function.has_virtual_registers()) {
                return {MachineVerifyError::InvalidTargetOpcodeState,
                        INVALID_BLOCK_ID, INVALID_INSTR_ID};
            }
            if (!function.properties().frame_lowered ||
                !function.properties().abi_lowered ||
                !function.properties().registers_allocated) {
                return {MachineVerifyError::PhaseViolation,
                        INVALID_BLOCK_ID, INVALID_INSTR_ID};
            }
        }

        return {};
    }
};

    bool run_phi_lowering = true;
    bool run_two_address = true;

struct MachinePipelineResult {
    bool success = false;
    std::string error_message{};
    MachineVerifyError verification_error = MachineVerifyError::None;
    BlockId error_block = INVALID_BLOCK_ID;
    InstrId error_instr = INVALID_INSTR_ID;

class MachinePipeline {
public:
    /**
     * @brief Executes the end-to-end Machine Lowering Pipeline:
     * 1. Legalization & Target Capability Checks
     * 2. Instruction Selection (HIR -> MachineFunction with Virtual Registers)
     * 3. SSA Deconstruction & Critical-Edge-Aware Phi Lowering
     * 4. Two-Address Instruction Normalization (with Commutation Optimization)
     * 5. Concrete Target Opcode Assignment
     * 6. Universal Graph-Coloring Register Allocation
     * 7. Target Frame & Stack Spill Alignment Lowering
     * 8. Target ABI Calling Convention Lowering
     * 9. Formal Machine IR Phase & Structural Verification
     */
    static MachinePipelineResult compile(
        MachineFunction& function,
        const ir::IRStream& stream,
        const core::TargetMachine& target,
        const TargetRegisterInfo& reg_info,
        const TargetInstructionInfo* instr_info,
        const TargetFrameInfo* frame_info,
        const TargetABI* abi,
        const InstructionSelectorProgram& selector = make_builtin_instruction_selector_program(),
        const MachinePipelineOptions& options = {}) {

        MachinePipelineResult result{};

        // --------------------------------------------------------------------
        // Pass 1: Legalizer
        // --------------------------------------------------------------------
        const ir::IRStream* active_stream = &stream;
        LegalizationResult leg_result;
        if (options.run_legalizer) {
            MachineLegalizer legalizer;
            leg_result = legalizer.legalize(stream, target);
            if (!leg_result.stream.is_null()) {
                active_stream = &leg_result.stream;
            }
        }

        // --------------------------------------------------------------------
        // Pass 2: Instruction Selection (HIR -> MachineFunction)
        // --------------------------------------------------------------------
        MachineFunctionBuilder builder(function);
        BlockId current_block_id = function.create_block(true).id;

        for (std::size_t i = 0; i < active_stream->size(); ++i) {
            const auto& node = active_stream->get_node(i);

            // Handle labels by creating new MachineBasicBlocks
            if (node.opcode == ir::OpCode::LABEL) {
                current_block_id = function.create_block(false).id;
                continue;
            }

            const SelectedInstruction selected =
                selector.select_instruction(*active_stream, i, node, target);

            if (selected.form == MachineForm::Unhandled) {
                continue;
            }

            selector.append_to_machine_function(
                builder, current_block_id, selected, RegisterClassKind::General);
        }

        // --------------------------------------------------------------------
        // Pass 3: SSA Deconstruction & Critical-Edge-Aware Phi Lowering
        // --------------------------------------------------------------------
        if (options.run_phi_lowering) {
            if (!MachinePhiLowering::run(function)) {
                result.error_message = "Phi lowering failed during SSA deconstruction";
                return result;
            }
        }

        // --------------------------------------------------------------------
        // Pass 4: Two-Address Normalization
        // --------------------------------------------------------------------
        if (options.run_two_address) {
            TwoAddressInstructionPass::run(function, target);
        }

        // --------------------------------------------------------------------
        // Pass 5: Concrete Target Opcode Assignment
        // --------------------------------------------------------------------
        for (std::size_t i = 0; i < function.instructions().size(); ++i) {
            auto* inst = function.instruction(static_cast<InstrId>(i));
            if (!inst) continue;

            if (!inst->has_target_opcode()) {
                // If target instruction info can supply a concrete opcode, assign it
                if (instr_info) {
                    const MachineOpcode target_op = static_cast<MachineOpcode>(inst->form);
                    inst->assign_target_opcode_once(target_op);
                } else {
                    // Default fallback mapping for forms
                    inst->assign_target_opcode_once(static_cast<MachineOpcode>(inst->form));
                }
            }
        }

        function.mark_target_lowered();

        // --------------------------------------------------------------------
        // Pass 6: Register Allocation
        // --------------------------------------------------------------------
        if (options.run_regalloc) {
            dgtl::isa::regalloc::UniversalGraphColoringRegAlloc regalloc(
                function, reg_info, instr_info);

            if (!regalloc.run()) {
                result.error_message = "Register allocation failed to resolve virtual registers";
                return result;
            }
        }

        // --------------------------------------------------------------------
        // Pass 7: Frame Lowering
        // --------------------------------------------------------------------
        if (options.run_frame_lowering && frame_info) {
            auto& frame = function.frame_info();
            frame.stack_alignment = frame_info->stack_alignment();

            std::int64_t current_offset = 0;
            for (auto& slot : frame.slots) {
                const auto alignment = static_cast<std::int64_t>(slot.alignment);
                current_offset = (current_offset + alignment - 1) & -alignment;
                slot.offset = current_offset;
                slot.fixed_offset = true;
                current_offset += static_cast<std::int64_t>(slot.size_bytes);
            }

            const auto frame_align = static_cast<std::int64_t>(frame.stack_alignment);
            frame.total_frame_bytes = static_cast<std::uint32_t>(
                (current_offset + frame_align - 1) & -frame_align);

            function.mark_frame_lowered();
        } else if (options.run_frame_lowering) {
            // Minimal frame initialization if no target frame info supplied
            auto& frame = function.frame_info();
            frame.stack_alignment = 16u;
            for (auto& slot : frame.slots) {
                slot.fixed_offset = true;
            }
            function.mark_frame_lowered();
        }

        // --------------------------------------------------------------------
        // Pass 8: Target ABI Lowering
        // --------------------------------------------------------------------
        if (options.run_abi_lowering && abi) {
            function.mark_abi_lowered();
        } else if (options.run_abi_lowering) {
            function.mark_abi_lowered();
        }

        // --------------------------------------------------------------------
        // Pass 9: Mark Emission Ready & Formal Verification
        // --------------------------------------------------------------------
        function.mark_emission_ready();

        if (options.verify_phases) {
            const auto verify_status = MachineVerifier::verify(function);
            if (verify_status.error != MachineVerifyError::None) {
                result.verification_error = verify_status.error;
                result.error_block = verify_status.block;
                result.error_instr = verify_status.instruction;
                result.error_message = "Machine IR verification failed with error code " +
                    std::to_string(static_cast<uint32_t>(verify_status.error));
                return result;
            }
        }

        result.success = true;
        return result;
    }
};

export module dgtl.mir.phi_lowering;

import dgtl.mir.machine_ir;
import dgtl.ir_enums;

export namespace dgtl::mir::lowering {

using dgtl::machine::BlockId;
using dgtl::machine::InstrId;
using dgtl::machine::MachineBasicBlock;
using dgtl::machine::MachineFunction;
using dgtl::machine::MachineInstrFlag;
using dgtl::machine::MachineInstruction;
using dgtl::machine::MachineOperand;
using dgtl::machine::MachineOperandKind;
using dgtl::machine::MachineRegister;
using dgtl::machine::MachineSemanticKind;
using dgtl::machine::INVALID_INSTR_ID;
using dgtl::mir::generated::MachineForm;

struct PendingCopy {
    MachineOperand dest{};
    MachineOperand src{};
    InstrId origin_phi = INVALID_INSTR_ID;
};

class MachinePhiLowering {
public:
    /**
     * @brief Deconstructs all SSA Phi nodes across the MachineFunction into parallel copies.
     * Implements critical-edge splitting and topological cycle breaking with scratch temporaries.
     */
    static bool run(MachineFunction& function) {
        if (!function.properties().is_ssa) {
            return true; // Already lowered from SSA
        }

        bool modified = false;

        // Group 1: Identify all blocks containing PHI nodes
        const auto blocks = function.blocks();
        for (std::size_t b_idx = 0; b_idx < blocks.size(); ++b_idx) {
            const BlockId block_id = static_cast<BlockId>(b_idx);
            MachineBasicBlock* bb = function.block(block_id);
            if (!bb) continue;

            // Collect all PHI instructions in this block
            std::vector<InstrId> phi_ids;
            for (const InstrId instr_id : bb->instructions) {
                const auto* inst = function.instruction(instr_id);
                if (!inst) continue;

                const bool is_phi =
                    (inst->semantic_provenance == static_cast<std::uint32_t>(dgtl::ir::OpCode::PHI_NODE)) ||
                    (inst->semantic_provenance == static_cast<std::uint32_t>(dgtl::ir::OpCode::PHI_MULTI));

                if (is_phi) {
                    phi_ids.push_back(instr_id);
                }
            }

            if (phi_ids.empty()) continue;

            // Map: PredBlockId -> list of pending copies to be emitted along that edge
            std::unordered_map<BlockId, std::vector<PendingCopy>> edge_copies;

            for (const InstrId phi_id : phi_ids) {
                const auto* phi_inst = function.instruction(phi_id);
                if (!phi_inst || phi_inst->operands.empty()) continue;

                const MachineOperand dest = phi_inst->operands[0];

                // Operands from index 1 onward are pairs of [BasicBlock, Value/Register/Immediate]
                for (std::size_t op_idx = 1; op_idx + 1 < phi_inst->operands.size(); op_idx += 2) {
                    const auto& block_op = phi_inst->operands[op_idx];
                    const auto& val_op = phi_inst->operands[op_idx + 1];

                    if (block_op.kind == MachineOperandKind::BasicBlock) {
                        const BlockId pred_id = block_op.block;
                        edge_copies[pred_id].push_back(PendingCopy{dest, val_op, phi_id});
                    }
                }
            }

            // Lower pending copies along each incoming edge
            for (auto& [pred_id, copies] : edge_copies) {
                MachineBasicBlock* pred_bb = function.block(pred_id);
                if (!pred_bb) continue;

                // Check for critical edge: pred has >1 succ, and current block has >1 pred
                const bool is_critical = (pred_bb->successors.size() > 1) && (bb->predecessors.size() > 1);

                BlockId target_insertion_block = pred_id;

                if (is_critical) {
                    // Split the critical edge: Insert an intermediate landing bridge block
                    MachineBasicBlock& split_block = function.create_block(false);
                    target_insertion_block = split_block.id;

                    // Re-route predecessor edge to split_block
                    for (auto& succ : pred_bb->successors) {
                        if (succ == block_id) {
                            succ = split_block.id;
                            break;
                        }
                    }
                    split_block.add_predecessor(pred_id);
                    split_block.add_successor(block_id);

                    // Update current block's predecessors
                    for (auto& pred : bb->predecessors) {
                        if (pred == pred_id) {
                            pred = split_block.id;
                            break;
                        }
                    }

                    // Add an unconditional branch in split_block to the target block
                    MachineInstruction& goto_inst = function.create_instruction();
                    goto_inst.form = MachineForm::Goto;
                    goto_inst.semantic_kind = MachineSemanticKind::Goto;
                    goto_inst.add_flag(MachineInstrFlag::Branch | MachineInstrFlag::Terminator);
                    goto_inst.add_operand(MachineOperand::block_ref(block_id));
                    function.insert_instruction_before(split_block.id, INVALID_INSTR_ID, goto_inst.id);
                }

                // Emit parallel copies with cycle breaking in target_insertion_block
                emit_parallel_copies(function, target_insertion_block, copies);
                modified = true;
            }

            // Remove/Neutralize lowered PHI instructions from bb
            for (const InstrId phi_id : phi_ids) {
                auto* inst = function.instruction(phi_id);
                if (inst) {
                    inst->form = MachineForm::NoOp;
                    inst->semantic_kind = MachineSemanticKind::Other;
                    inst->operands.clear();
                }
            }

            // Remove neutralized instructions from bb->instructions
            bb->instructions.erase(
                std::remove_if(bb->instructions.begin(), bb->instructions.end(),
                               [&](InstrId id) {
                                   return std::find(phi_ids.begin(), phi_ids.end(), id) != phi_ids.end();
                               }),
                bb->instructions.end());
            bb->instr_count = bb->instructions.size();
        }

        function.properties().is_ssa = false;
        return modified;
    }

private:
    /**
     * @brief Emits a set of parallel copies safely resolving any register swap cycles.
     */
    static void emit_parallel_copies(
        MachineFunction& function,
        BlockId target_block,
        std::vector<PendingCopy>& copies) {

        if (copies.empty()) return;

        MachineBasicBlock* bb = function.block(target_block);
        if (!bb) return;

        // Find insertion cursor: before the first terminator instruction, if any
        InstrId cursor = INVALID_INSTR_ID;
        for (const InstrId id : bb->instructions) {
            const auto* inst = function.instruction(id);
            if (inst && inst->has_flag(MachineInstrFlag::Terminator)) {
                cursor = id;
                break;
            }
        }

        // Cycle resolution: Break cyclic dependencies among parallel copies using scratch temporaries
        // e.g. (x = y, y = x) -> (t = x, x = y, y = t)
        for (std::size_t i = 0; i < copies.size(); ++i) {
            for (std::size_t j = i + 1; j < copies.size(); ++j) {
                // If copy i defines a register that copy j uses, and copy j defines a register copy i uses
                if (copies[i].dest.kind == MachineOperandKind::Register &&
                    copies[j].src.kind == MachineOperandKind::Register &&
                    copies[i].dest.reg == copies[j].src.reg &&
                    copies[j].dest.kind == MachineOperandKind::Register &&
                    copies[i].src.kind == MachineOperandKind::Register &&
                    copies[j].dest.reg == copies[i].src.reg) {

                    // Break cycle with a fresh scratch temporary
                    const MachineRegister scratch =
                        function.create_virtual_register(copies[i].dest.reg.reg_class);

                    // Insert scratch move: scratch = copies[i].dest
                    MachineInstruction& scratch_move = function.create_instruction();
                    scratch_move.form = MachineForm::Move;
                    scratch_move.semantic_kind = MachineSemanticKind::Move;
                    scratch_move.add_operand(MachineOperand::def(scratch));
                    scratch_move.add_operand(MachineOperand::use(copies[i].dest.reg));
                    function.insert_instruction_before(target_block, cursor, scratch_move.id);

                    // Replace source of copy j with scratch
                    copies[j].src = MachineOperand::use(scratch);
                }
            }
        }

        // Emit all copies sequentially
        for (const auto& copy : copies) {
            // Avoid trivial self-moves (x = x)
            if (copy.dest.kind == MachineOperandKind::Register &&
                copy.src.kind == MachineOperandKind::Register &&
                copy.dest.reg == copy.src.reg) {
                continue;
            }

            MachineInstruction& move_inst = function.create_instruction();
            move_inst.form = MachineForm::Move;
            move_inst.semantic_kind = MachineSemanticKind::Move;
            move_inst.add_operand(copy.dest);
            move_inst.add_operand(copy.src);

            function.insert_instruction_before(target_block, cursor, move_inst.id);
        }
    }
};

/**
 * @brief Target-aware legalization policy for DGTL's IR -> ISA boundary.
 *
 * This layer deliberately does not perform legalization itself. It answers one
 * policy question:
 *
 *   "For this exact TargetMachine and IR opcode, is there a direct generated
 *    ISel path, should the operation be lowered through the DGTL runtime, or
 *    is it unsupported?"
 *
 * TargetMachine is mandatory. There is intentionally no host/compile-time
 * target overload because such an overload would re-introduce host-target
 * leakage into cross-compilation and multi-target JIT execution.
 */
export namespace dgtl::mir::runtime_policy {

using dgtl::mir::generated::MachineForm;

/**
 * @brief Backend disposition after target-aware ISel classification.
 *
 * MachineForm describes the lowering shape. LegalizationRoute describes the
 * ownership of the next lowering step.
 */
enum class LegalizationRoute : std::uint8_t {
    DirectISel = 0,
    RuntimeCall,
    Unhandled
};

/**
 * @brief Classify one IR opcode for one concrete compilation target.
 *
 * Precedence is intentional:
 *
 *   1. Target-aware generated ISel
 *   2. Runtime helper contract
 *   3. Fully unsupported
 *
 * A runtime helper therefore never overrides a valid direct target lowering.
 * The existence of a runtime symbol means "a fallback implementation exists",
 * not "the operation must use the runtime".
 */
[[nodiscard]] inline LegalizationRoute classify(
    const core::TargetMachine& target,
    ir::OpCode opcode) noexcept
{
    const MachineForm form =
        generated::classify_machine_form(target, opcode);

    if (form != MachineForm::Unhandled) [[likely]] {
        return LegalizationRoute::DirectISel;
    }

    if (runtime_contract::has_runtime_fallback(target, opcode)) [[unlikely]] {
        return LegalizationRoute::RuntimeCall;
    }

    return LegalizationRoute::Unhandled;
}

/**
 * @brief True when generated target-specific ISel owns the lowering.
 */
[[nodiscard]] inline bool has_direct_isel_mapping(
    const core::TargetMachine& target,
    ir::OpCode opcode) noexcept
{
    return classify(target, opcode) == LegalizationRoute::DirectISel;
}

/**
 * @brief True when no direct target ISel exists but a runtime fallback exists.
 */
[[nodiscard]] inline bool should_route_unhandled_to_runtime_call(
    const core::TargetMachine& target,
    ir::OpCode opcode) noexcept
{
    return classify(target, opcode) == LegalizationRoute::RuntimeCall;
}

/**
 * @brief True when neither direct ISel nor a runtime fallback exists.
 */
[[nodiscard]] inline bool is_fully_unhandled_opcode(
    const core::TargetMachine& target,
    ir::OpCode opcode) noexcept
{
    return classify(target, opcode) == LegalizationRoute::Unhandled;
}

/**
 * @brief Convenience predicate for callers that need the final disposition.
 */
[[nodiscard]] inline bool is_direct_or_runtime_supported(
    const core::TargetMachine& target,
    ir::OpCode opcode) noexcept
{
    return classify(target, opcode) != LegalizationRoute::Unhandled;
}

/**
 * @brief Return the generated MachineForm for diagnostics and downstream policy.
 *
 * This intentionally exposes the same target-aware authority used by classify()
 * so diagnostics cannot accidentally describe a different host target.
 */
[[nodiscard]] inline MachineForm machine_form(
    const core::TargetMachine& target,
    ir::OpCode opcode) noexcept
{
    return generated::classify_machine_form(target, opcode);
}

/**
 * @struct StackSlot
 * @brief Represents a single variable's allocation within a stack frame.
 * Aligned to 16 bytes for SIMD-safe spills.
 */
struct alignas(16) StackSlot {
    int32_t offset = 0;       // Offset relative to Frame Pointer (FP)
    uint32_t size = 0;        // Size in bytes
    OpCode type_hint = OpCode::NOP;     // Type for alignment validation
    uint32_t version = 0;     // For shadowing detection
    bool is_spill = false;    // Flagged if this is a register-allocator spill
};

/**
 * @class StackPiller
 * @brief Manages the lexical lifecycle of symbols and their physical stack offsets.
 * Implements Pillar 2 "Shadowing" through nested hash-map scopes.
 */
class alignas(64) StackPiller {
public:
    using ScopeLayer = std::unordered_map<uint64_t, StackSlot>;
    static constexpr uint32_t STACK_ALIGNMENT = 16;

    explicit StackPiller(dgtl::core::DGTL_Arena& arena) 
        : m_arena(arena), m_current_frame_size(0) {
        m_scope_stack.reserve(16);
        m_scope_frame_checkpoints.reserve(16);
        
        // Initialize Global Scope
        m_scope_stack.emplace_back();
        m_scope_frame_checkpoints.push_back(0);
    }

    // --- SCOPE LIFECYCLE ---

    void push_scope() {
        m_scope_stack.emplace_back();
        m_scope_frame_checkpoints.push_back(m_current_frame_size);
    }

    void pop_scope() {
        if (m_scope_stack.size() > 1) [[likely]] {
            m_scope_stack.pop_back();
            m_current_frame_size = m_scope_frame_checkpoints.back();
            m_scope_frame_checkpoints.pop_back();
        }
    }

    // --- ALLOCATION LOGIC ---

    /**
     * @brief Allocates a slot on the current virtual stack frame.
     * @return The stack-relative offset.
     */
    int32_t allocate(std::string_view name, uint32_t bytes, OpCode type = OpCode::NOP) {
        if (bytes == 0) [[unlikely]] {
            return 0;
        }

        // Maintain 16-byte alignment for hardware efficiency (AVX-512/NEON/Win64 ABI)
        const uint32_t aligned_frame = align_up(m_current_frame_size, STACK_ALIGNMENT);
        if (aligned_frame < m_current_frame_size) [[unlikely]] {
            return 0;
        }

        if (bytes > (std::numeric_limits<uint32_t>::max)() - aligned_frame) [[unlikely]] {
            return 0;
        }

        const uint32_t next_frame_size = aligned_frame + bytes;
        if (next_frame_size > static_cast<uint32_t>((std::numeric_limits<int32_t>::max)())) [[unlikely]] {
            return 0;
        }

        int32_t offset = -static_cast<int32_t>(next_frame_size);
        m_current_frame_size = next_frame_size;

        if (name.empty()) {
            return offset;
        }

        uint64_t name_hash = hash_name(name);
        uint32_t version = 1;
        for (auto it = m_scope_stack.rbegin(); it != m_scope_stack.rend(); ++it) {
            if (auto found = it->find(name_hash); found != it->end()) {
                version = found->second.version + 1;
                break;
            }
        }

        StackSlot slot{
            .offset = offset,
            .size = bytes,
            .type_hint = type,
            .version = version,
            .is_spill = false
        };

        m_scope_stack.back()[name_hash] = slot;
        return offset;
    }

    /**
     * @brief Allocates a dedicated register-spill slot.
     */
    int32_t allocate_spill(uint32_t virtual_id, uint32_t bytes = 8) {
        const int32_t offset = allocate("", bytes, OpCode::NOP);
        const uint64_t spill_token = 0xFFFF000000000000ULL | static_cast<uint64_t>(virtual_id);
        
        StackSlot slot{
            .offset = offset,
            .size = bytes,
            .type_hint = OpCode::NOP,
            .version = 1,
            .is_spill = true
        };
        m_scope_stack.back()[spill_token] = slot;
        return offset;
    }

    /**
     * @brief Manually binds a symbol to a specific offset. 
     */
    void bind_explicit_slot(std::string_view name, int32_t offset, uint32_t bytes, OpCode type = OpCode::NOP) {
        if (name.empty()) [[unlikely]] return;

        uint64_t name_hash = hash_name(name);
        StackSlot slot{
            .offset = offset,
            .size = bytes,
            .type_hint = type,
            .version = 1,
            .is_spill = false
        };
        m_scope_stack.back()[name_hash] = slot;
    }

    /**
     * @brief Resolves a variable to its stack offset, respecting lexical shadowing.
     */
    [[nodiscard]] std::optional<StackSlot> resolve(std::string_view name) const noexcept {
        uint64_t name_hash = hash_name(name);

        for (auto it = m_scope_stack.rbegin(); it != m_scope_stack.rend(); ++it) {
            if (auto found = it->find(name_hash); found != it->end()) {
                return found->second;
            }
        }
        return std::nullopt;
    }

    // --- METRICS ---

    [[nodiscard]] uint32_t current_frame_size() const noexcept {
        return m_current_frame_size;
    }

    [[nodiscard]] size_t nesting_depth() const noexcept {
        return m_scope_stack.size();
    }

    /**
     * @brief Resets the frame size for a new function body emission.
     */
    void reset_frame() {
        m_current_frame_size = 0;
        m_scope_stack.clear();
        m_scope_frame_checkpoints.clear();
        
        m_scope_stack.reserve(16);
        m_scope_frame_checkpoints.reserve(16);
        
        m_scope_stack.emplace_back();
        m_scope_frame_checkpoints.push_back(0);
    }

private:
    dgtl::core::DGTL_Arena& m_arena;
    std::vector<ScopeLayer> m_scope_stack;
    std::vector<uint32_t> m_scope_frame_checkpoints;
    uint32_t m_current_frame_size;

    /**
     * @brief FNV-1a Hash for symbol lookup.
     */
    [[nodiscard]] static constexpr uint64_t hash_name(std::string_view name) noexcept {
        uint64_t hash = 0xcbf29ce484222325ULL;
        for (char c : name) {
            hash ^= static_cast<uint64_t>(static_cast<unsigned char>(c));
            hash *= 0x100000001b3ULL;
        }
        return hash;
    }

    [[nodiscard]] static constexpr uint32_t align_up(uint32_t value, uint32_t alignment) noexcept {
        if (alignment == 0) [[unlikely]] {
            return value;
        }
        const uint32_t mask = alignment - 1U;
        return (value + mask) & ~mask;
    }

    [[nodiscard]] uint32_t compute_win64_aligned_frame(uint32_t non_volatile_spills_count, bool makes_external_calls) const noexcept {
        const uint32_t shadow_space = makes_external_calls ? 32u : 0u;
        const uint32_t raw_locals = m_current_frame_size;
        const uint32_t raw_body = raw_locals + shadow_space;
        const uint32_t callee_saved_bytes = non_volatile_spills_count * 8u;
        const uint32_t total_pushed_before_sub = 8u + 8u + callee_saved_bytes;
        const uint32_t remainder = (total_pushed_before_sub + raw_body) % 16u;
        const uint32_t pad = (remainder != 0u) ? (16u - remainder) : 0u;
        return raw_body + pad;
    }
};

using StackSpiller = StackPiller;

#include <type_traits>

export module dgtl.mir.strike_live_manager;

import dgtl.ir;
import dgtl.mir.runtime_contract;

namespace ir = dgtl::ir;

struct LiveInterval {
    size_t start_node = 0;
    size_t end_node = 0;
    bool active = false;
};

class StrikeLiveManager {
    std::vector<LiveInterval> intervals_;

    template <typename LabelMapType>
    static bool lookup_label_index(const LabelMapType& label_map,
                                   uint32_t label,
                                   size_t& out_index) {
        if constexpr (std::is_pointer_v<decltype(label_map.find(label))>) {
            if (const auto* ptr = label_map.find(label)) {
                out_index = *ptr;
                return true;
            }
            return false;
        } else {
            auto it = label_map.find(label);
            if (it != label_map.end()) {
                out_index = it->second;
                return true;
            }
            return false;
        }
    }

    struct OperandUseMask {
        bool use_a = false;
        bool use_b = false;
        bool use_c = false;
    };

    /**
     * @brief Determines which operands are read as virtual registers by a given opcode.
     * Prevents operands containing immediate literal values or label offsets from being treated as registers.
     */
    [[nodiscard]] static OperandUseMask operand_uses(ir::OpCode opcode) noexcept {
        switch (opcode) {
        // Reads exactly 1 virtual register (op_a)
        case ir::OpCode::LOAD:
        case ir::OpCode::VOLATILE_LOAD:
        case ir::OpCode::STORE_V512:
        case ir::OpCode::RET:
        case ir::OpCode::BRANCH_COND:
        case ir::OpCode::BRANCH_LIKELY:
        case ir::OpCode::BRANCH_UNLIKELY:
        case ir::OpCode::CLFLUSH:
        case ir::OpCode::PREFETCH_T0:
            return {true, false, false};

        // Reads 2 virtual registers (op_a, op_b)
        case ir::OpCode::REG_MOVE:
        case ir::OpCode::STORE:
        case ir::OpCode::VOLATILE_STORE:
        case ir::OpCode::ADD_I32:
        case ir::OpCode::SUB_I32:
        case ir::OpCode::MUL_I32:
        case ir::OpCode::DIV_I32:
        case ir::OpCode::MOD_I32:
        case ir::OpCode::ADD_I64:
        case ir::OpCode::SUB_I64:
        case ir::OpCode::MUL_I64:
        case ir::OpCode::DIV_I64:
        case ir::OpCode::MOD_I64:
        case ir::OpCode::CMP_EQ_I32:
        case ir::OpCode::CMP_NE_I32:
        case ir::OpCode::CMP_LT_I32:
        case ir::OpCode::CMP_LE_I32:
        case ir::OpCode::CMP_GT_I32:
        case ir::OpCode::CMP_GE_I32:
        case ir::OpCode::AND_I32:
        case ir::OpCode::OR_I32:
        case ir::OpCode::XOR_I32:
        case ir::OpCode::SHL_I32:
        case ir::OpCode::SHR_I32:
        case ir::OpCode::AND_I64:
        case ir::OpCode::OR_I64:
        case ir::OpCode::XOR_I64:
        case ir::OpCode::SHL_I64:
        case ir::OpCode::SHR_I64:
        case ir::OpCode::ADD_F32:
        case ir::OpCode::SUB_F32:
        case ir::OpCode::MUL_F32:
        case ir::OpCode::DIV_F32:
        case ir::OpCode::ADD_F64:
        case ir::OpCode::SUB_F64:
        case ir::OpCode::MUL_F64:
        case ir::OpCode::DIV_F64:
        case ir::OpCode::VEC_ADD_F32_512:
        case ir::OpCode::VEC_MUL_F32_512:
        case ir::OpCode::GATHER_F32_512:
        case ir::OpCode::GATHER_16:
        case ir::OpCode::BRANCH_COMPARE_FUSED:
        case ir::OpCode::GEP:
        case ir::OpCode::LEA:
        case ir::OpCode::PTR_DIFF:
        case ir::OpCode::PTR_TO_INT:
        case ir::OpCode::INT_TO_PTR:
        case ir::OpCode::PTR_CAST:
            return {true, true, false};

        // Reads 3 virtual registers (op_a, op_b, op_c)
        case ir::OpCode::CALL:
        case ir::OpCode::CALL_VARIADIC:
        case ir::OpCode::FMA_F32:
        case ir::OpCode::SELECT:
        case ir::OpCode::BOOL_SELECT:
            return {true, true, true};

        // Reads 0 virtual registers
        case ir::OpCode::CONST:
        case ir::OpCode::IMMEDIATE:
        case ir::OpCode::STRING_CONST:
        case ir::OpCode::BRANCH:
        case ir::OpCode::GOTO:
        case ir::OpCode::NOP:
        case ir::OpCode::LABEL:
        case ir::OpCode::PHI_NODE:
        case ir::OpCode::SYSCALL:
            return {false, false, false};

        default:
            if (dgtl::mir::runtime_contract::has_runtime_symbol(opcode)) {
                return {true, true, true};
            }
            return {false, false, false};
        }
    }

public:
    template <typename LabelMapType>
    void compute(const ir::IRStream& stream, const LabelMapType& label_map) {
        intervals_.clear();
        intervals_.reserve(stream.size() * 2);

        // Track only IDs produced by this stream so literals/labels are not treated as registers
        std::unordered_map<uint32_t, bool> produced_ids;
        produced_ids.reserve(stream.size());
        for (size_t i = 0; i < stream.size(); ++i) {
            const auto& node = stream.get_node(i);
            if (node.result_id != ir::INVALID_SLOT) {
                produced_ids.emplace(node.result_id, true);
            }
        }

        for (size_t i = 0; i < stream.size(); ++i) {
            const auto& node = stream.get_node(i);
            
            auto track_use = [&](uint32_t virtual_id) {
                if (virtual_id == ir::INVALID_SLOT) {
                    return;
                }
                if (produced_ids.find(virtual_id) == produced_ids.end()) {
                    return;
                }
                if (virtual_id >= intervals_.size()) {
                    intervals_.resize(static_cast<size_t>(virtual_id) + 1);
                }
                auto& interval = intervals_[virtual_id];
                if (!interval.active) {
                    interval.active = true;
                    interval.start_node = i;
                }
                if (i > interval.end_node) {
                    interval.end_node = i;
                }
            };

            if (node.result_id != ir::INVALID_SLOT) {
                if (node.result_id >= intervals_.size()) {
                    intervals_.resize(static_cast<size_t>(node.result_id) + 1);
                }
                auto& interval = intervals_[node.result_id];
                if (!interval.active) {
                    interval.active = true;
                    interval.start_node = i;
                }
                if (i > interval.end_node) {
                    interval.end_node = i;
                }
            }

            const OperandUseMask uses = operand_uses(node.opcode);
            if (uses.use_a) track_use(node.op_a);
            if (uses.use_b) track_use(node.op_b);
            if (uses.use_c) track_use(node.op_c);
        }

        // --- Backedge Resolution ---
        // Extends live intervals across loop scopes to protect loop-carried induction variables
        for (size_t i = 0; i < stream.size(); ++i) {
            const auto& node = stream.get_node(i);
            if (node.opcode == ir::OpCode::BRANCH_COND || 
                node.opcode == ir::OpCode::BRANCH_LIKELY ||
                node.opcode == ir::OpCode::BRANCH_UNLIKELY ||
                node.opcode == ir::OpCode::BRANCH_COMPARE_FUSED ||
                node.opcode == ir::OpCode::GOTO) {

                uint32_t target_label = ir::INVALID_SLOT;
                if (node.opcode == ir::OpCode::BRANCH_COMPARE_FUSED) {
                    target_label = node.op_c;
                } else if (node.opcode == ir::OpCode::BRANCH_COND ||
                           node.opcode == ir::OpCode::BRANCH_LIKELY ||
                           node.opcode == ir::OpCode::BRANCH_UNLIKELY) {
                    target_label = (node.op_b != ir::INVALID_SLOT && node.op_b != 0u) ? node.op_b : node.result_id;
                } else {
                    target_label = (node.op_a != ir::INVALID_SLOT && node.op_a != 0u) ? node.op_a : node.result_id;
                }

                size_t loop_start = 0;
                if (!lookup_label_index(label_map, target_label, loop_start)) {
                    continue;
                }
                if (loop_start >= i) {
                    continue; // Forward jump
                }

                const size_t loop_end = i;
                for (auto& interval : intervals_) {
                    if (interval.active && interval.start_node <= loop_end && interval.end_node >= loop_start) {
                        if (interval.end_node < loop_end) {
                            interval.end_node = loop_end;
                        }
                    }
                }
            }
        }
    }

    [[nodiscard]] inline bool is_dead(uint32_t virtual_id, size_t current_node_idx) const noexcept {
        if (virtual_id == ir::INVALID_SLOT || static_cast<size_t>(virtual_id) >= intervals_.size()) {
            return true;
        }
        const auto& interval = intervals_[virtual_id];
        if (!interval.active) {
            return true;
        }
        return current_node_idx >= interval.end_node;
    }
};

} // namespace dgtl::mir::regalloc

module;

#include <cstdlib>
#include <algorithm>
#include <cstdint>
#include <vector>

export module dgtl.mir.two_address;

import dgtl.mir.machine_ir;
import dgtl.target;
import dgtl.ir_enums;

export namespace dgtl::mir::lowering {

using dgtl::machine::BlockId;
using dgtl::machine::InstrId;
using dgtl::machine::MachineFunction;
using dgtl::machine::MachineInstrFlag;
using dgtl::machine::MachineInstruction;
using dgtl::machine::MachineOperand;
using dgtl::machine::MachineOperandKind;
using dgtl::machine::MachineSemanticKind;
using dgtl::mir::generated::MachineForm;

class TwoAddressInstructionPass {
public:
    /**
     * @brief Normalizes 3-address instructions into 2-address form for destructive targets (e.g. x86-64).
     * Eliminates redundant copies via commutation where possible, and inserts tied-operand moves when necessary.
     */
    static bool run(MachineFunction& function, const core::TargetMachine& target) {
        const auto arch = target.get_arch();

        // 3-address architectures (AArch64, RISC-V) do not require 2-address normalization
        if (arch == core::ArchType::ARM64 ||
            arch == core::ArchType::AARCH64_SVE ||
            arch == core::ArchType::RISCV64_RVV) {
            return false;
        }

        bool modified = false;

        for (const auto& block : function.blocks()) {
            const BlockId block_id = block.id;

            // Snapshot instruction list because we may insert move instructions
            const std::vector<InstrId> inst_ids = block.instructions;

            for (const InstrId inst_id : inst_ids) {
                auto* inst = function.instruction(inst_id);
                if (!inst) continue;

                // Only normalize binary instructions with at least [Def, Use1, Use2]
                if (!is_two_address_candidate(*inst)) {
                    continue;
                }

                if (inst->operands.size() < 3) continue;

                auto& def_op = inst->operands[0];
                auto& use1_op = inst->operands[1];
                auto& use2_op = inst->operands[2];

                if (def_op.kind != MachineOperandKind::Register ||
                    use1_op.kind != MachineOperandKind::Register) {
                    continue;
                }

                // If already 2-address (dest == use1), no transformation is needed
                if (def_op.reg == use1_op.reg) {
                    continue;
                }

                // Commutation optimization: if dest == use2 and operation is commutable, swap operands!
                const bool commutable = is_commutable(*inst);
                if (commutable && use2_op.kind == MachineOperandKind::Register &&
                    def_op.reg == use2_op.reg) {
                    std::swap(inst->operands[1], inst->operands[2]);
                    modified = true;
                    continue;
                }

                // Insert a copy: dest = Move use1 before the instruction
                MachineInstruction& copy_inst = function.create_instruction();
                copy_inst.form = MachineForm::Move;
                copy_inst.semantic_kind = MachineSemanticKind::Move;
                copy_inst.add_operand(MachineOperand::def(def_op.reg));
                copy_inst.add_operand(MachineOperand::use(use1_op.reg));

                function.insert_instruction_before(block_id, inst_id, copy_inst.id);

                // Tie use1 to dest register
                inst->operands[1] = MachineOperand::use(def_op.reg);
                modified = true;
            }
        }

        return modified;
    }

private:
    [[nodiscard]] static bool is_two_address_candidate(const MachineInstruction& inst) noexcept {
        switch (inst.form) {
            case MachineForm::IntegerBinary32:
            case MachineForm::IntegerBinary64:
            case MachineForm::FloatBinary32:
            case MachineForm::FloatBinary64:
                return true;
            default:
                return false;
        }
    }

    [[nodiscard]] static bool is_commutable(const MachineInstruction& inst) noexcept {
        if (inst.has_flag(MachineInstrFlag::Commutable)) {
            return true;
        }

        switch (inst.semantic_kind) {
            case MachineSemanticKind::IntegerAdd:
            case MachineSemanticKind::IntegerMul:
            case MachineSemanticKind::IntegerAnd:
            case MachineSemanticKind::IntegerOr:
            case MachineSemanticKind::IntegerXor:
            case MachineSemanticKind::FloatAdd:
            case MachineSemanticKind::FloatMul:
                return true;
            default:
                return false;
        }
    }
};

} // namespace dgtl::mir::lowering
