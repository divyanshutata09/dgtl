cmake_minimum_required(VERSION 3.20)
project(DGTL_ENGINE LANGUAGES CXX)

# --- GLOBAL CONFIGURATION ---
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# --- PRODUCTION COMPILER FLAGS (UCRT64 / G++) ---
if(CMAKE_COMPILER_IS_GNUCXX)
    add_compile_options(
        -Ofast               # Peak performance for HFT/AI kernels
        -march=native        # Auto-detects AVX2/FMA instructions
        -flto=auto           # Uses all 12 threads for Link-Time Optimization
        -fexceptions         # Required for your BIR 2.1 transactional catch-blocks
        -fpermissive         # Handles low-level memory casts
        -Wall                
        -Wno-unused-parameter
    )
    
    add_link_options(-flto=auto -fuse-linker-plugin)
endif()

# --- INCLUDE DIRECTORIES ---
# Maps "dgtl/..." path to your /include/ folder
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include)

# --- AUTOMATIC SOURCE COLLECTION ---
# This grabs ALL .cpp files in src/ and its subfolders (ir, pass, analysis, core, etc.)
file(GLOB_RECURSE LIB_SOURCES "src/*.cpp")

# Remove main.cpp from LIB_SOURCES if it exists there, to avoid double-definition of main()
list(FILTER LIB_SOURCES EXCLUDE REGEX "src/main.cpp")

# --- TARGETS ---

# 1. The Core Engine Library (Everything in src/)
add_library(DGTL_CORE STATIC ${LIB_SOURCES})

# 2. Main Compiler Executable (Zenith/Aura)
# Assumes your main entry point is src/main.cpp
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
    add_executable(AuraCore src/main.cpp)
    target_link_libraries(AuraCore PRIVATE DGTL_CORE)
endif()

# 3. Robustness Suite (Auto-collect all tests)
file(GLOB_RECURSE TEST_SOURCES "tests/*.cpp")

foreach(test_file ${TEST_SOURCES})
    get_filename_component(test_name ${test_file} NAME_WE)
    add_executable(${test_name} ${test_file})
    target_link_libraries(${test_name} PRIVATE DGTL_CORE)
endforeach()
