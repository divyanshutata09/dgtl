using machine::MachineAttribute;
using machine::MachineFunction;
using machine::MachineInstruction;
using machine::MachineOperand;
using machine::MachineOperandKind;
using machine::MachineSemanticKind;
using machine::MachineForm;

inline constexpr std::uint32_t INVALID_SLOT = machine::INVALID_MACHINE_ID;

enum class MachineLinkFlag : std::uint32_t {
    None = 0,
    HasSideEffects = 1u << 0,
    Volatile       = 1u << 1
};

struct MachineLinkTypeCode {
    bool floating = false;
};

enum class MachineLinkOpcode : std::uint16_t {
    NOP,
    LABEL,
    LOAD_IMM,
    CONST,
    IMMEDIATE,
    STRING_CONST,
    VEC_ADD_F32_512,
    CMP_EQ_I32,
    CMP_NE_I32,
    CMP_LT_I32,
    CMP_LE_I32,
    CMP_GT_I32,
    CMP_GE_I32,
    BRANCH_UNLIKELY,
    BRANCH_COMPARE_FUSED,
    RET,
    RETURN,
    CALL
};

struct MachineLinkNode {
    MachineLinkOpcode opcode = MachineLinkOpcode::NOP;
    MachineForm form = MachineForm::Unhandled;
    MachineSemanticKind semantic_kind = MachineSemanticKind::None;

    // These identifiers intentionally retain the old encoder's operand-role
    // convention, but their source is MachineOperand rather than IR slots.
    std::uint32_t result_id = INVALID_SLOT;
    std::uint32_t op_a = INVALID_SLOT;
    std::uint32_t op_b = INVALID_SLOT;
    std::uint32_t op_c = INVALID_SLOT;
    std::uint32_t logic_flags = INVALID_SLOT;

    std::uint32_t flags = 0;
    std::uint64_t intent_hash = 0;
    std::uint32_t semantic_provenance = 0;
    std::uint32_t parent_block = machine::INVALID_BLOCK_ID;

    bool result_is_floating = false;
    std::vector<std::uint32_t> call_overflow_args;

    [[nodiscard]] bool has_side_effects() const noexcept {
        return (flags & static_cast<std::uint32_t>(
            MachineLinkFlag::HasSideEffects)) != 0;
    }

    [[nodiscard]] bool is_volatile() const noexcept {
        return (flags & static_cast<std::uint32_t>(
            MachineLinkFlag::Volatile)) != 0;
    }

    [[nodiscard]] bool has_runtime_symbol() const noexcept {
        return dgtl::mir::runtime_contract::has_runtime_symbol_for_semantic(
            semantic_kind);
    }

    [[nodiscard]] const char* runtime_symbol() const noexcept {
        return dgtl::mir::runtime_contract::runtime_symbol_for_semantic_kind(
            semantic_kind);
    }
};

class MachineLinkStream {
    std::vector<MachineLinkNode> nodes_;
    std::unordered_map<std::uint32_t, MachineLinkTypeCode> result_types_;

    static std::uint32_t operand_id(const MachineOperand& op) noexcept {
        switch (op.kind) {
            case MachineOperandKind::Register:
                return op.reg.valid() ? op.reg.id : INVALID_SLOT;
            case MachineOperandKind::StackSlot:
                return op.stack_slot;
            case MachineOperandKind::BasicBlock:
                return op.block;
            case MachineOperandKind::Symbol:
                return op.symbol;
            case MachineOperandKind::ConstantPoolIndex:
                return op.constant_pool_index;
            case MachineOperandKind::JumpTableIndex:
                return op.jump_table_index;
            case MachineOperandKind::Immediate:
                return static_cast<std::uint32_t>(op.immediate);
            default:
                return INVALID_SLOT;
        }
    }

    static void append_operand_ids(
        const MachineInstruction& inst,
        MachineLinkNode& node) {

        std::size_t use_index = 0;
        for (const auto& op : inst.operands) {
            if (op.is_def()) {
                if (node.result_id == INVALID_SLOT)
                    node.result_id = operand_id(op);

                if (op.kind == MachineOperandKind::Register) {
                    node.result_is_floating =
                        op.reg.reg_class == machine::RegisterClassKind::FloatingPoint;
                }
                continue;
            }

            if (op.kind == MachineOperandKind::Immediate) {
                // LoadImmediate's historical encoder stores a 64-bit immediate
                // as two 32-bit fields. Preserve that representation.
                if (use_index == 0) {
                    const auto u = static_cast<std::uint64_t>(op.immediate);
                    node.op_a = static_cast<std::uint32_t>(u);
                    node.op_b = static_cast<std::uint32_t>(u >> 32u);
                    ++use_index;
                    continue;
                }
            }

            const auto id = operand_id(op);
            switch (use_index++) {
                case 0: node.op_a = id; break;
                case 1: node.op_b = id; break;
                case 2: node.op_c = id; break;
                default:
                    node.call_overflow_args.push_back(id);
                    break;
            }
        }
    }

    static MachineLinkOpcode classify(const MachineInstruction& inst) noexcept {
        using S = MachineSemanticKind;

        if (inst.form == MachineForm::NoOp) return MachineLinkOpcode::LABEL;
        if (inst.form == MachineForm::LoadImmediate)
            return MachineLinkOpcode::LOAD_IMM;
        if (inst.form == MachineForm::Return)
            return MachineLinkOpcode::RET;
        if (inst.form == MachineForm::Call)
            return MachineLinkOpcode::CALL;
        if (inst.form == MachineForm::BranchCond) {
            return MachineLinkOpcode::BRANCH_UNLIKELY;
        }
        if (inst.form == MachineForm::BranchCompare) {
            return MachineLinkOpcode::BRANCH_COMPARE_FUSED;
        }

        switch (inst.semantic_kind) {
            case S::VectorAdd: return MachineLinkOpcode::VEC_ADD_F32_512;
            case S::Compare:   return MachineLinkOpcode::CMP_LE_I32;
            case S::Move:      return MachineLinkOpcode::NOP;
            case S::LoadImmediate: return MachineLinkOpcode::LOAD_IMM;
            case S::Return: return MachineLinkOpcode::RET;
            case S::Call: return MachineLinkOpcode::CALL;
            default: return MachineLinkOpcode::NOP;
        }
    }

public:
    MachineLinkStream() = default;

    // The stream owns all of its state. Copying it duplicates the node vector
    // (including overflow-argument vectors) and result-type map, so the clone
    // remains valid independently of the source stream and MachineFunction.
    [[nodiscard]] MachineLinkStream clone() const {
        return *this;
    }

    explicit MachineLinkStream(const MachineFunction& function) {
        reset(function);
    }

    void reset(const MachineFunction& function) {
        nodes_.clear();
        result_types_.clear();
        nodes_.reserve(function.instructions().size());

        for (const auto& inst : function.instructions()) {
            MachineLinkNode node;
            node.form = inst.form;
            node.semantic_kind = inst.semantic_kind;
            node.opcode = classify(inst);
            node.intent_hash = inst.intent_hash;
            node.semantic_provenance = inst.semantic_provenance;
            node.parent_block = inst.parent_block;

            if (inst.has_flag(machine::MachineInstrFlag::SideEffect))
                node.flags |= static_cast<std::uint32_t>(MachineLinkFlag::HasSideEffects);
            if (inst.has_attribute(MachineAttribute::Volatile))
                node.flags |= static_cast<std::uint32_t>(MachineLinkFlag::Volatile);

            append_operand_ids(inst, node);

            if (node.result_id != INVALID_SLOT) {
                result_types_[node.result_id] =
                    MachineLinkTypeCode{node.result_is_floating};
            }

            nodes_.push_back(std::move(node));
        }
    }

    [[nodiscard]] bool is_null() const noexcept { return nodes_.empty(); }
    [[nodiscard]] std::size_t size() const noexcept { return nodes_.size(); }

    [[nodiscard]] const MachineLinkNode& get_node(std::size_t i) const {
        return nodes_.at(i);
    }

    [[nodiscard]] const MachineLinkNode& get_node_readonly(std::size_t i) const {
        return nodes_.at(i);
    }

    [[nodiscard]] const std::vector<std::uint32_t>&
    get_call_overflow_args(std::size_t i) const {
        return nodes_.at(i).call_overflow_args;
    }

    [[nodiscard]] MachineLinkTypeCode
    get_result_type_code(std::uint32_t id) const noexcept {
        const auto it = result_types_.find(id);
        return it == result_types_.end() ? MachineLinkTypeCode{} : it->second;
    }

    [[nodiscard]] bool validate_shadow_integrity() const noexcept {
        // Every result definition must have exactly one matching type entry.
        // Conversely, every type entry must be backed by exactly one node.
        // This checks the bridge's duplicated result metadata without
        // allocating memory or mutating the stream.
        std::size_t definition_count = 0;

        for (const auto& node : nodes_) {
            if (node.has_runtime_symbol() != (node.runtime_symbol() != nullptr))
                return false;

            if (node.result_id == INVALID_SLOT)
                continue;

            ++definition_count;
            const auto type_it = result_types_.find(node.result_id);
            if (type_it == result_types_.end())
                return false;
            if (type_it->second.floating != node.result_is_floating)
                return false;
        }

        if (definition_count != result_types_.size())
            return false;

        for (const auto& [result_id, type] : result_types_) {
            std::size_t matches = 0;
            for (const auto& node : nodes_) {
                if (node.result_id != result_id)
                    continue;
                ++matches;
                if (node.result_is_floating != type.floating)
                    return false;
            }
            if (matches != 1)
                return false;
        }

        return true;
    }

    [[nodiscard]] bool has_function() const noexcept {
        return !nodes_.empty();
    }
};

[[nodiscard]] inline bool is_floating_type_code(
    MachineLinkTypeCode type) noexcept {
    return type.floating;
}

struct MagicMultiplierS64 {
    uint64_t M = 0;
    uint8_t s = 0;
    bool is_power_of_2 = false;
    bool valid = false;
};

inline MagicMultiplierS64 compute_magic_s64(int64_t d) noexcept {
    MagicMultiplierS64 res{};
    if (d <= 0) return res;
    if ((d & (d - 1)) == 0) {
        res.is_power_of_2 = true;
        res.s = static_cast<uint8_t>(__builtin_ctzll(static_cast<unsigned long long>(d)));
        res.valid = true;
        return res;
    }
    unsigned __int128 two_63 = static_cast<unsigned __int128>(1) << 63;
    unsigned __int128 ud = static_cast<unsigned __int128>(d);
    unsigned __int128 anc = two_63 - 1 - (two_63 - 1) % ud;
    uint8_t p = 63;
    unsigned __int128 q1 = two_63 / anc;
    unsigned __int128 r1 = two_63 - q1 * anc;
    unsigned __int128 q2 = two_63 / ud;
    unsigned __int128 r2 = two_63 - q2 * ud;
    unsigned __int128 delta = 0;
    do {
        p++;
        q1 = 2 * q1;
        r1 = 2 * r1;
        if (r1 >= anc) {
            q1++;
            r1 -= anc;
        }
        q2 = 2 * q2;
        r2 = 2 * r2;
        if (r2 >= ud) {
            q2++;
            r2 -= ud;
        }
        delta = ud - r2;
    } while (q1 < delta || (q1 == delta && r1 == 0));

    res.M = static_cast<uint64_t>(q2 + 1);
    res.s = p - 64;
    res.valid = (res.M < (1ULL << 63));
    return res;
}

        if (!out) [[unlikely]] return false;
        if (!bytes.empty()) {
            out.write(reinterpret_cast<const char*>(bytes.data()), static_cast<std::streamsize>(bytes.size()));
        }

        const std::unordered_map<uint32_t, uintptr_t>& slot_map = {},
        std::vector<CoffCallReloc>* out_relocs = nullptr,
        const std::unordered_map<uint32_t, std::string>& slot_to_callee = {})
    {
        if (dest == nullptr || stream.is_null()) return false;
        clear_fallback_diagnostics();
        const size_t count = stream.size();

        std::unordered_map<uint32_t, uint64_t> known_constants;
        known_constants.reserve(count);
        for (size_t i = 0; i < count; ++i) {
            const auto& node = stream.get_node(i);
            if (node.opcode == dgtl::linker::mir::MachineLinkOpcode::LOAD_IMM || node.opcode == dgtl::linker::mir::MachineLinkOpcode::CONST || node.opcode == dgtl::linker::mir::MachineLinkOpcode::IMMEDIATE) {
                uint64_t imm = (static_cast<uint64_t>(node.op_b) << 32u) | (static_cast<uint64_t>(node.op_a) & 0xFFFFFFFFu);
                if (imm == 0 && node.intent_hash != 0) {
                    imm = node.intent_hash;
                }
                if (node.result_id != dgtl::linker::mir::INVALID_SLOT) {
                    known_constants[node.result_id] = imm;
                }
            }
        }

        std::unordered_map<uint32_t, size_t> label_map;
        std::unordered_map<uint32_t, int32_t> slot_offsets;
        slot_offsets.reserve(count * 2 + 8);
        uint32_t next_slot_index = 0;
        auto assign_slot = [&](uint32_t slot_id) {
            if (slot_id == dgtl::linker::mir::INVALID_SLOT) return;
            if (slot_offsets.find(slot_id) != slot_offsets.end()) return;
            slot_offsets.emplace(slot_id, -8 * static_cast<int32_t>(++next_slot_index));
        };

        assign_slot(0u);
        assign_slot(1u);
        assign_slot(2u);
        assign_slot(3u);

        for (size_t i = 0; i < count; ++i) {
            const auto& node = stream.get_node(i);
            if (node.opcode == dgtl::linker::mir::MachineLinkOpcode::LABEL) {
                if (node.result_id != dgtl::linker::mir::INVALID_SLOT) {
                    label_map[node.result_id] = i;
                }
                const bool is_func_header = node.has_side_effects() || node.is_volatile();
                if (!is_func_header) {
                    if (node.op_a != dgtl::linker::mir::INVALID_SLOT) {
                        label_map[node.op_a] = i;
                    }
                    if (node.op_b != dgtl::linker::mir::INVALID_SLOT) {
                        label_map[node.op_b] = i;
                    }
                }
                if (node.op_a != dgtl::linker::mir::INVALID_SLOT) assign_slot(node.op_a);
                if (node.op_b != dgtl::linker::mir::INVALID_SLOT) assign_slot(node.op_b);
                if (node.op_c != dgtl::linker::mir::INVALID_SLOT) assign_slot(node.op_c);
                if (node.logic_flags != dgtl::linker::mir::INVALID_SLOT) assign_slot(node.logic_flags);
                continue;
            }

            if (node.result_id != dgtl::linker::mir::INVALID_SLOT) assign_slot(node.result_id);
            if (node.op_a != dgtl::linker::mir::INVALID_SLOT) assign_slot(node.op_a);
            if (node.op_b != dgtl::linker::mir::INVALID_SLOT) assign_slot(node.op_b);
            if (node.op_c != dgtl::linker::mir::INVALID_SLOT) assign_slot(node.op_c);
            if (node.logic_flags != dgtl::linker::mir::INVALID_SLOT) assign_slot(node.logic_flags);

            const auto overflow = stream.get_call_overflow_args(i);
            for (uint32_t ov : overflow) {
                if (ov != dgtl::linker::mir::INVALID_SLOT) assign_slot(ov);
            }
        }

        auto resolve_target = [&](uint32_t raw) -> size_t {
            if (raw == dgtl::linker::mir::INVALID_SLOT) return count;
            auto it = label_map.find(raw);
            if (it != label_map.end()) return it->second;
            for (size_t k = 0; k < count; ++k) {
                const auto& knode = stream.get_node(k);
                if (knode.opcode == dgtl::linker::mir::MachineLinkOpcode::LABEL) {
                    if (knode.result_id == raw || knode.op_a == raw || knode.op_b == raw) {
                        return k;
                    }
                }
            }
            return count;
        };

        const uint32_t raw_frame_size = (next_slot_index + 4u) * 8u;
        const uint32_t frame_size = (raw_frame_size + 15u) & ~15u;

        std::vector<size_t> node_offsets(count + 1, 0);
        size_t probe = x64_prologue_size(stream);
        for (size_t i = 0; i < count; ++i) {
            node_offsets[i] = probe;
            const auto& node = stream.get_node(i);
            probe += x64_node_size(node);
        }
        node_offsets[count] = probe;

        if (probe > dest_capacity) [[unlikely]] return false;

        struct BranchPatch { size_t disp_offset; size_t source_end; size_t target_node; };
        std::vector<BranchPatch> patches;

        const int32_t fallback_offset = -8 * static_cast<int32_t>(next_slot_index + 1u);
        auto slot_disp = [&](uint32_t slot_id) -> int32_t {
            if (slot_id == dgtl::linker::mir::INVALID_SLOT) return fallback_offset;
            const auto it = slot_offsets.find(slot_id);
            return (it != slot_offsets.end()) ? it->second : fallback_offset;
        };
        patches.reserve(count / 8 + 4);

        size_t offset = 0;
        auto write_u8 = [&](uint8_t byte) { dest[offset++] = byte; };
        auto write_u32 = [&](uint32_t value) {
            std::memcpy(dest + offset, &value, sizeof(value));
            offset += sizeof(value);
        };
        auto write_i32 = [&](int32_t value) {
            std::memcpy(dest + offset, &value, sizeof(value));
            offset += sizeof(value);
        };
        auto write_u64 = [&](uint64_t value) {
            std::memcpy(dest + offset, &value, sizeof(value));
            offset += sizeof(value);
        };

        auto emit_mov_reg_from_slot = [&](uint8_t rex, uint8_t modrm, uint32_t slot_id) {
            write_u8(rex);
            write_u8(0x8B);
            write_u8(modrm);
            write_i32(slot_disp(slot_id));
        };
        auto emit_mov_slot_from_reg = [&](uint8_t rex, uint8_t modrm, uint32_t slot_id) {
            write_u8(rex);
            write_u8(0x89);
            write_u8(modrm);
            write_i32(slot_disp(slot_id));
        };

        auto emit_mov_rax_from_slot = [&](uint32_t slot_id) { emit_mov_reg_from_slot(0x48, 0x85, slot_id); };
        auto emit_mov_rcx_from_slot = [&](uint32_t slot_id) { emit_mov_reg_from_slot(0x48, 0x8D, slot_id); };
        auto emit_mov_rdx_from_slot = [&](uint32_t slot_id) { emit_mov_reg_from_slot(0x48, 0x95, slot_id); };
        auto emit_mov_r8_from_slot  = [&](uint32_t slot_id) { emit_mov_reg_from_slot(0x4C, 0x85, slot_id); };
        auto emit_mov_r9_from_slot  = [&](uint32_t slot_id) { emit_mov_reg_from_slot(0x4C, 0x8D, slot_id); };

        auto emit_store_rax_to_slot = [&](uint32_t slot_id) { emit_mov_slot_from_reg(0x48, 0x85, slot_id); };
        auto emit_store_rcx_to_slot = [&](uint32_t slot_id) { emit_mov_slot_from_reg(0x48, 0x8D, slot_id); };
        auto emit_store_rdx_to_slot = [&](uint32_t slot_id) { emit_mov_slot_from_reg(0x48, 0x95, slot_id); };
        auto emit_store_r8_to_slot  = [&](uint32_t slot_id) { emit_mov_slot_from_reg(0x4C, 0x85, slot_id); };
        auto emit_store_r9_to_slot  = [&](uint32_t slot_id) { emit_mov_slot_from_reg(0x4C, 0x8D, slot_id); };

        // 1. Prologue: Setup stack frame and home incoming parameters strictly ONCE
        write_u8(0x55);                                // push rbp
        write_u8(0x48); write_u8(0x89); write_u8(0xE5); // mov rbp, rsp
        write_u8(0x48); write_u8(0x81); write_u8(0xEC); write_u32(frame_size); // sub rsp, frame_size

        // Zero out initial scratch slots (0..3)
        write_u8(0x48); write_u8(0x31); write_u8(0xC0); // xor rax, rax
        emit_store_rax_to_slot(0u);
        emit_store_rax_to_slot(1u);
        emit_store_rax_to_slot(2u);
        emit_store_rax_to_slot(3u);

        if (count > 0 && stream.get_node(0).opcode == dgtl::linker::mir::MachineLinkOpcode::LABEL) {
            const auto& fn_hdr = stream.get_node(0);
            if (fn_hdr.op_a != dgtl::linker::mir::INVALID_SLOT) emit_store_rcx_to_slot(fn_hdr.op_a);
            if (fn_hdr.op_b != dgtl::linker::mir::INVALID_SLOT) emit_store_rdx_to_slot(fn_hdr.op_b);
            if (fn_hdr.op_c != dgtl::linker::mir::INVALID_SLOT) emit_store_r8_to_slot(fn_hdr.op_c);
            if (fn_hdr.logic_flags != dgtl::linker::mir::INVALID_SLOT) emit_store_r9_to_slot(fn_hdr.logic_flags);
            const auto fn_overflow = stream.get_call_overflow_args(0);
            for (size_t k = 0; k < fn_overflow.size(); ++k) {
                if (fn_overflow[k] != dgtl::linker::mir::INVALID_SLOT) {
                    write_u8(0x48); write_u8(0x8B); write_u8(0x85); write_i32(static_cast<int32_t>(48 + k * 8)); // mov rax, [rbp + 48 + k*8]
                    emit_store_rax_to_slot(fn_overflow[k]);
                }
            }
        }

        // 2. Linear Node Lowering
        for (size_t i = 0; i < count; ++i) {
            node_offsets[i] = offset;
            const auto& node = stream.get_node(i);
            const MachineForm form = node.form;

            if (node.has_runtime_symbol()) {
                if (node.op_a != dgtl::linker::mir::INVALID_SLOT) emit_mov_rcx_from_slot(node.op_a);
                else { write_u8(0x48); write_u8(0x31); write_u8(0xC9); } // xor rcx, rcx

                if (node.op_b != dgtl::linker::mir::INVALID_SLOT) emit_mov_rdx_from_slot(node.op_b);
                else { write_u8(0x48); write_u8(0x31); write_u8(0xD2); } // xor rdx, rdx

                if (node.op_c != dgtl::linker::mir::INVALID_SLOT) emit_mov_r8_from_slot(node.op_c);
                else { write_u8(0x4D); write_u8(0x31); write_u8(0xC0); } // xor r8, r8

                if (node.logic_flags != dgtl::linker::mir::INVALID_SLOT) {
                    emit_mov_r9_from_slot(node.logic_flags);
                } else {
                    write_u8(0x4D); write_u8(0x31); write_u8(0xC9); // xor r9, r9
                }

                write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xC1); // movq xmm0, rcx
                write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xCA); // movq xmm1, rdx

                write_u8(0x48); write_u8(0x83); write_u8(0xEC); write_u8(0x20); // sub rsp, 32

                uintptr_t rt_target_addr = 0;
                if (out_relocs == nullptr) {
                    if (!slot_map.empty()) {
                        auto it = slot_map.find(static_cast<uint32_t>(node.opcode));
                        if (it != slot_map.end()) {
                            rt_target_addr = it->second;
                        }
                    }
                    if (rt_target_addr == 0 && node.intent_hash > 0x7FFF00000000ULL) {
                        rt_target_addr = static_cast<uintptr_t>(node.intent_hash);
                    }
                }

                if (rt_target_addr != 0) {
                    write_u8(0x48); write_u8(0xB8); write_u64(static_cast<uint64_t>(rt_target_addr));
                    write_u8(0xFF); write_u8(0xD0); // call rax
                } else {
                    write_u8(0xE8); // call rel32
                    const size_t reloc_disp_offset = offset;
                    write_u8(0x00); write_u8(0x00); write_u8(0x00); write_u8(0x00);
                    if (out_relocs != nullptr) {
                        const char* symbol = node.runtime_symbol();
                        if (symbol != nullptr) {
                            out_relocs->push_back({ static_cast<uint32_t>(reloc_disp_offset), std::string(symbol) });
                        }
                    }
                }
                write_u8(0x48); write_u8(0x83); write_u8(0xC4); write_u8(0x20); // add rsp, 32

                if (node.result_id != dgtl::linker::mir::INVALID_SLOT) emit_store_rax_to_slot(node.result_id);
                else write_u8(0x90);
                continue;
            }

            switch (form) {
                case MachineForm::NoOp:
                    if (node.opcode == dgtl::linker::mir::MachineLinkOpcode::LABEL) {
                        const bool is_func_header = node.has_side_effects() || node.is_volatile();
                        if (is_func_header && i > 0) {
                            if (node.op_a != dgtl::linker::mir::INVALID_SLOT) emit_store_rcx_to_slot(node.op_a);
                            if (node.op_b != dgtl::linker::mir::INVALID_SLOT) emit_store_rdx_to_slot(node.op_b);
                            if (node.op_c != dgtl::linker::mir::INVALID_SLOT) emit_store_r8_to_slot(node.op_c);
                            if (node.logic_flags != dgtl::linker::mir::INVALID_SLOT) emit_store_r9_to_slot(node.logic_flags);
                            const auto fn_overflow = stream.get_call_overflow_args(i);
                            for (size_t k = 0; k < fn_overflow.size(); ++k) {
                                if (fn_overflow[k] != dgtl::linker::mir::INVALID_SLOT) {
                                    write_u8(0x48); write_u8(0x8B); write_u8(0x85); write_i32(static_cast<int32_t>(48 + k * 8));
                                    emit_store_rax_to_slot(fn_overflow[k]);
                                }
                            }
                        }
                    } else {
                        write_u8(0x90);
                    }
                    break;

                case MachineForm::Move:
                case MachineForm::LoadStack:
                case MachineForm::StoreStack: {
                    if (form == MachineForm::StoreStack) {
                        emit_mov_rax_from_slot(node.op_b);
                        emit_store_rax_to_slot(node.op_a);
                    } else {
                        emit_mov_rax_from_slot(node.op_a);
                        emit_store_rax_to_slot(node.result_id);
                    }
                    break;
                }

                case MachineForm::LoadImmediate: {
                    uint64_t imm = (static_cast<uint64_t>(node.op_b) << 32u) | (static_cast<uint64_t>(node.op_a) & 0xFFFFFFFFu);
                    if (node.opcode == dgtl::linker::mir::MachineLinkOpcode::STRING_CONST) {
                        imm = static_cast<uint64_t>(node.op_a);
                    }

                    const bool is_float_const = (node.opcode == dgtl::linker::mir::MachineLinkOpcode::CONST && node.form == dgtl::mir::MachineForm::LoadImmediate);

                    if (is_float_const) {
                        write_u8(0x48); write_u8(0xB8); write_u64(imm); // movabs rax, imm64
                        write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xC0); // movq xmm0, rax
                        write_u8(0xF2); write_u8(0x0F); write_u8(0x11); write_u8(0x85); // movsd [rbp-disp], xmm0
                        write_i32(slot_disp(node.result_id));
                    } else {
                        write_u8(0x48); write_u8(0xB8); write_u64(imm); // movabs rax, imm64
                        emit_store_rax_to_slot(node.result_id);
                    }
                    break;
                }

                case MachineForm::IntegerBinary32:
                case MachineForm::IntegerBinary64: {
                    const bool is64 = (form == MachineForm::IntegerBinary64);
                    dgtl::mir::MachineSemanticKind kind = dgtl::mir::MachineSemanticKind::None;
                    if (is64) {
                        switch (node.semantic_kind) {
                            case dgtl::mir::MachineSemanticKind::IntegerAdd: kind = dgtl::mir::MachineSemanticKind::IntegerAdd; break;
                            case dgtl::mir::MachineSemanticKind::IntegerSub: kind = dgtl::mir::MachineSemanticKind::IntegerSub; break;
                            case dgtl::mir::MachineSemanticKind::IntegerMul: kind = dgtl::mir::MachineSemanticKind::IntegerMul; break;
                            case dgtl::mir::MachineSemanticKind::IntegerDiv: kind = dgtl::mir::MachineSemanticKind::IntegerDiv; break;
                            case dgtl::mir::MachineSemanticKind::IntegerMod: kind = dgtl::mir::MachineSemanticKind::IntegerMod; break;
                            case dgtl::mir::MachineSemanticKind::IntegerAnd: kind = dgtl::mir::MachineSemanticKind::IntegerAnd; break;
                            case dgtl::mir::MachineSemanticKind::IntegerOr:  kind = dgtl::mir::MachineSemanticKind::IntegerOr;  break;
                            case dgtl::mir::MachineSemanticKind::IntegerXor: kind = dgtl::mir::MachineSemanticKind::IntegerXor; break;
                            case dgtl::mir::MachineSemanticKind::IntegerShl: kind = dgtl::mir::MachineSemanticKind::IntegerShl; break;
                            case dgtl::mir::MachineSemanticKind::IntegerShr: kind = dgtl::mir::MachineSemanticKind::IntegerShr; break;
                            default: break;
                        }
                    } else {
                        kind = node.semantic_kind;
                    }

                    if (kind == dgtl::mir::MachineSemanticKind::IntegerShl || 
                        kind == dgtl::mir::MachineSemanticKind::IntegerShr) {
                        emit_mov_rax_from_slot(node.op_a);
                        emit_mov_rcx_from_slot(node.op_b);
                        if (is64) write_u8(0x48);
                        write_u8(0xD3);
                        write_u8(kind == dgtl::mir::MachineSemanticKind::IntegerShl ? 0xE0 : 0xF8);
                        emit_store_rax_to_slot(node.result_id);
                    }
                    else if (kind == dgtl::mir::MachineSemanticKind::IntegerDiv || 
                             kind == dgtl::mir::MachineSemanticKind::IntegerMod) {
                        auto const_it = (node.op_b != dgtl::linker::mir::INVALID_SLOT) ? known_constants.find(node.op_b) : known_constants.end();
                        bool has_const_divisor = (const_it != known_constants.end() && const_it->second > 0);
                        int64_t divisor = has_const_divisor ? static_cast<int64_t>(const_it->second) : 0;
                        auto magic = (has_const_divisor && is64) ? compute_magic_s64(divisor) : MagicMultiplierS64{};

                        if (is64 && has_const_divisor && divisor == 1) {
                            if (kind == dgtl::mir::MachineSemanticKind::IntegerDiv) {
                                emit_mov_rax_from_slot(node.op_a);
                                emit_store_rax_to_slot(node.result_id);
                            } else {
                                write_u8(0x48); write_u8(0x31); write_u8(0xC0); // xor rax, rax
                                emit_store_rax_to_slot(node.result_id);
                            }
                        }
                        else if (is64 && has_const_divisor && magic.valid && !magic.is_power_of_2) {
                            emit_mov_rax_from_slot(node.op_a);
                            write_u8(0x48); write_u8(0xB9); write_u64(magic.M);
                            write_u8(0x48); write_u8(0xF7); write_u8(0xE9);
                            if (magic.s > 0) {
                                write_u8(0x48); write_u8(0xC1); write_u8(0xFA); write_u8(magic.s);
                            }
                            emit_mov_rax_from_slot(node.op_a);
                            write_u8(0x48); write_u8(0xC1); write_u8(0xF8); write_u8(0x3F);
                            write_u8(0x48); write_u8(0x29); write_u8(0xC2);
                            if (kind == dgtl::mir::MachineSemanticKind::IntegerDiv) {
                                emit_store_rdx_to_slot(node.result_id);
                            } else {
                                write_u8(0x48); write_u8(0x89); write_u8(0xD0);
                                write_u8(0x48); write_u8(0xB9); write_u64(static_cast<uint64_t>(divisor));
                                write_u8(0x48); write_u8(0x0F); write_u8(0xAF); write_u8(0xC1);
                                emit_mov_rdx_from_slot(node.op_a);
                                write_u8(0x48); write_u8(0x29); write_u8(0xC2);
                                emit_store_rdx_to_slot(node.result_id);
                            }
                        }
                        else if (is64 && has_const_divisor && magic.is_power_of_2 && divisor > 1) {
                            emit_mov_rax_from_slot(node.op_a);
                            write_u8(0x48); write_u8(0x99);
                            uint8_t k = magic.s;
                            if (k == 1) {
                                write_u8(0x48); write_u8(0xC1); write_u8(0xEA); write_u8(0x3F);
                            } else if (k < 31) {
                                write_u8(0x48); write_u8(0x81); write_u8(0xE2); write_u32(static_cast<uint32_t>((1ULL << k) - 1));
                            } else {
                                write_u8(0x48); write_u8(0xB9); write_u64((1ULL << k) - 1);
                                write_u8(0x48); write_u8(0x21); write_u8(0xCA);
                            }
                            write_u8(0x48); write_u8(0x01); write_u8(0xD0);
                            write_u8(0x48); write_u8(0xC1); write_u8(0xF8); write_u8(k);
                            if (kind == dgtl::mir::MachineSemanticKind::IntegerDiv) {
                                emit_store_rax_to_slot(node.result_id);
                            } else {
                                write_u8(0x48); write_u8(0xC1); write_u8(0xE0); write_u8(k);
                                emit_mov_rdx_from_slot(node.op_a);
                                write_u8(0x48); write_u8(0x29); write_u8(0xC2);
                                emit_store_rdx_to_slot(node.result_id);
                            }
                        }
                        else {
                            emit_mov_rax_from_slot(node.op_a);
                            emit_mov_rcx_from_slot(node.op_b);
                            if (is64) {
                                write_u8(0x48); write_u8(0x85); write_u8(0xC9); // test rcx, rcx
                                write_u8(0x75); write_u8(0x06);                 // jnz +6
                                write_u8(0x31); write_u8(0xC0);                 // xor eax, eax
                                write_u8(0x31); write_u8(0xD2);                 // xor edx, edx
                                write_u8(0xEB); write_u8(0x05);                 // jmp +5
                                write_u8(0x48); write_u8(0x99);                 // cqo
                                write_u8(0x48); write_u8(0xF7); write_u8(0xF9); // idiv rcx
                            } else {
                                write_u8(0x85); write_u8(0xC9);                 // test ecx, ecx
                                write_u8(0x75); write_u8(0x06);                 // jnz +6
                                write_u8(0x31); write_u8(0xC0);                 // xor eax, eax
                                write_u8(0x31); write_u8(0xD2);                 // xor edx, edx
                                write_u8(0xEB); write_u8(0x03);                 // jmp +3
                                write_u8(0x99);                                 // cdq
                                write_u8(0xF7); write_u8(0xF9);                 // idiv ecx
                            }
                            if (kind == dgtl::mir::MachineSemanticKind::IntegerDiv)
                                emit_store_rax_to_slot(node.result_id);
                            else emit_store_rdx_to_slot(node.result_id);
                        }
                    }
                    else {
                        emit_mov_rax_from_slot(node.op_a);
                        if (is64) write_u8(0x48);
                        if (kind == dgtl::mir::MachineSemanticKind::IntegerAdd) {
                            write_u8(0x03); write_u8(0x85);
                        } else if (kind == dgtl::mir::MachineSemanticKind::IntegerSub) {
                            write_u8(0x2B); write_u8(0x85);
                        } else if (kind == dgtl::mir::MachineSemanticKind::IntegerAnd) {
                            write_u8(0x23); write_u8(0x85);
                        } else if (kind == dgtl::mir::MachineSemanticKind::IntegerOr) {
                            write_u8(0x0B); write_u8(0x85);
                        } else if (kind == dgtl::mir::MachineSemanticKind::IntegerXor) {
                            write_u8(0x33); write_u8(0x85);
                        } else if (kind == dgtl::mir::MachineSemanticKind::IntegerMul) {
                            write_u8(0x0F); write_u8(0xAF); write_u8(0x85);
                        }
                        write_i32(slot_disp(node.op_b));
                        emit_store_rax_to_slot(node.result_id);
                    }
                    break;
                }

                case MachineForm::FloatBinary32:
                case MachineForm::FloatBinary64: {
                    const bool is64 = (form == MachineForm::FloatBinary64);
                    write_u8(is64 ? 0xF2 : 0xF3);
                    write_u8(0x0F); write_u8(0x10); write_u8(0x85);
                    write_i32(slot_disp(node.op_a));

                    write_u8(is64 ? 0xF2 : 0xF3);
                    write_u8(0x0F);
                    if (is64) {
                        const auto kind = node.semantic_kind;
                        using K = dgtl::mir::MachineSemanticKind;
                        if (kind == K::FloatAdd) write_u8(0x58);
                        else if (kind == K::FloatSub) write_u8(0x5C);
                        else if (kind == K::FloatMul) write_u8(0x59);
                        else if (kind == K::FloatDiv) write_u8(0x5E);
                        else write_u8(0x58);
                    } else {
                        const auto kind = node.semantic_kind;
                        using K = dgtl::mir::MachineSemanticKind;
                        if (kind == K::FloatAdd) write_u8(0x58);
                        else if (kind == K::FloatSub) write_u8(0x5C);
                        else if (kind == K::FloatMul) write_u8(0x59);
                        else if (kind == K::FloatDiv) write_u8(0x5E);
                        else write_u8(0x58);
                    }
                    write_u8(0x85);
                    write_i32(slot_disp(node.op_b));

                    write_u8(is64 ? 0xF2 : 0xF3);
                    write_u8(0x0F); write_u8(0x11); write_u8(0x85);
                    write_i32(slot_disp(node.result_id));
                    break;
                }

                case MachineForm::Compare32: {
                    emit_mov_rax_from_slot(node.op_a);
                    write_u8(0x48); write_u8(0x3B); write_u8(0x85); write_i32(slot_disp(node.op_b));
                    write_u8(0x0F);
                    switch (node.opcode) {
                        case dgtl::linker::mir::MachineLinkOpcode::CMP_EQ_I32: write_u8(0x94); break;
                        case dgtl::linker::mir::MachineLinkOpcode::CMP_NE_I32: write_u8(0x95); break;
                        case dgtl::linker::mir::MachineLinkOpcode::CMP_LT_I32: write_u8(0x9C); break;
                        case dgtl::linker::mir::MachineLinkOpcode::CMP_LE_I32: write_u8(0x9E); break;
                        case dgtl::linker::mir::MachineLinkOpcode::CMP_GT_I32: write_u8(0x9F); break;
                        case dgtl::linker::mir::MachineLinkOpcode::CMP_GE_I32: write_u8(0x9D); break;
                        default: write_u8(0x9E); break;
                    }
                    write_u8(0xC0);
                    write_u8(0x0F); write_u8(0xB6); write_u8(0xC0);
                    emit_store_rax_to_slot(node.result_id);
                    break;
                }

                case MachineForm::Call: {
                    if (node.op_b != dgtl::linker::mir::INVALID_SLOT) {
                        emit_mov_rcx_from_slot(node.op_b);
                    } else { 
                        write_u8(0x48); write_u8(0x31); write_u8(0xC9); // xor rcx, rcx
                    }

                    if (node.op_c != dgtl::linker::mir::INVALID_SLOT) {
                        emit_mov_rdx_from_slot(node.op_c);
                    } else { 
                        write_u8(0x48); write_u8(0x31); write_u8(0xD2); // xor rdx, rdx
                    }

                    if (node.logic_flags != dgtl::linker::mir::INVALID_SLOT) {
                        emit_mov_r8_from_slot(node.logic_flags);
                    } else { 
                        write_u8(0x4D); write_u8(0x31); write_u8(0xC0); // xor r8, r8
                    }

                    const auto overflow = stream.get_call_overflow_args(i);
                    if (!overflow.empty() && overflow[0] != dgtl::linker::mir::INVALID_SLOT) {
                        emit_mov_r9_from_slot(overflow[0]);
                    } else {
                        write_u8(0x4D); write_u8(0x31); write_u8(0xC9); // xor r9, r9
                    }

                    write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xC1); // movq xmm0, rcx
                    write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xCA); // movq xmm1, rdx
                    write_u8(0x66); write_u8(0x49); write_u8(0x0F); write_u8(0x6E); write_u8(0xD0); // movq xmm2, r8
                    write_u8(0x66); write_u8(0x49); write_u8(0x0F); write_u8(0x6E); write_u8(0xD9); // movq xmm3, r9

                    const size_t num_stack_args = (overflow.size() > 1) ? (overflow.size() - 1) : 0;
                    const size_t stack_alloc = 32 + ((num_stack_args + 1) / 2) * 16;
                    write_u8(0x48); write_u8(0x81); write_u8(0xEC); write_u32(static_cast<uint32_t>(stack_alloc)); // sub rsp, stack_alloc

                    for (size_t k = 1; k < overflow.size(); ++k) {
                        if (overflow[k] != dgtl::linker::mir::INVALID_SLOT) {
                            emit_mov_rax_from_slot(overflow[k]);
                            write_u8(0x48); write_u8(0x89); write_u8(0x84); write_u8(0x24);
                            write_i32(static_cast<int32_t>(32 + (k - 1) * 8)); // mov [rsp + 32 + (k-1)*8], rax
                        }
                    }

                    uintptr_t target_addr = 0;
                    if (out_relocs == nullptr) {
                        if (!slot_map.empty()) {
                            auto it = slot_map.find(static_cast<uint32_t>(node.op_a));
                            if (it != slot_map.end() && it->second != 0) {
                                target_addr = it->second;
                            }
                        }
                        if (target_addr == 0 && node.op_a > 0x7FFF00000000ULL) {
                            target_addr = static_cast<uintptr_t>(node.op_a);
                        }
                        if (target_addr == 0 && node.intent_hash > 0x7FFF00000000ULL) {
                            target_addr = static_cast<uintptr_t>(node.intent_hash);
                        }
                    }

                    if (target_addr != 0) {
                        write_u8(0x48); write_u8(0xB8); write_u64(static_cast<uint64_t>(target_addr));
                        write_u8(0xFF); write_u8(0xD0); // call rax
                    } else {
                        write_u8(0xE8); // call rel32
                        const size_t patch_disp_offset = offset;
                        write_u32(0);
                        const size_t patch_source_end = offset;
                        bool is_external_reloc = false;
                        if (const char* sym = node.runtime_symbol(); sym != nullptr) {
                            if (out_relocs != nullptr) out_relocs->push_back({ static_cast<uint32_t>(patch_disp_offset), sym });
                            is_external_reloc = true;
                        } else if (out_relocs != nullptr) {
                            auto it = slot_to_callee.find(node.op_a);
                            if (it != slot_to_callee.end() && !it->second.empty()) {
                                out_relocs->push_back({ static_cast<uint32_t>(patch_disp_offset), it->second });
                                is_external_reloc = true;
                            }
                        }
                        if (!is_external_reloc) {
                            patches.push_back({patch_disp_offset, patch_source_end, resolve_target(node.op_a)});
                        }
                    }
                    write_u8(0x48); write_u8(0x81); write_u8(0xC4); write_u32(static_cast<uint32_t>(stack_alloc)); // add rsp, stack_alloc

                    if (node.result_id != dgtl::linker::mir::INVALID_SLOT) {
                        bool returns_in_xmm0 = false;
                        auto it_callee = slot_to_callee.find(node.op_a);
                        if (it_callee != slot_to_callee.end() && !it_callee->second.empty()) {
                            const std::string& name = it_callee->second;
                            if (name.find("_bits") != std::string::npos) {
                                returns_in_xmm0 = false;
                            } else if (name == "pow" || name == "sqrt" || name == "sin" ||
                                       name == "cos" || name == "tan" || name == "exp" ||
                                       name == "log" || name == "fabs" || name == "floor" ||
                                       name == "ceil" ||
                                       name == "__dgtl_rt_sin" || name == "__dgtl_rt_cos" ||
                                       name == "__dgtl_rt_tan" || name == "__dgtl_rt_asin" ||
                                       name == "__dgtl_rt_acos" || name == "__dgtl_rt_atan" ||
                                       name == "__dgtl_rt_atan2" || name == "__dgtl_rt_sinh" ||
                                       name == "__dgtl_rt_cosh" || name == "__dgtl_rt_tanh" ||
                                       name == "__dgtl_rt_log" || name == "__dgtl_rt_log2" ||
                                       name == "__dgtl_rt_log10" || name == "__dgtl_rt_log1p" ||
                                       name == "__dgtl_rt_exp" || name == "__dgtl_rt_exp2" ||
                                       name == "__dgtl_rt_exp10" || name == "__dgtl_rt_expm1" ||
                                       name == "__dgtl_rt_pow" || name == "__dgtl_rt_cbrt" ||
                                       name == "__dgtl_rt_hypot") {
                                returns_in_xmm0 = true;
                            } else {
                                const dgtl::linker::mir::MachineLinkTypeCode tc = stream.get_result_type_code(node.result_id);
                                returns_in_xmm0 = dgtl::linker::mir::is_floating_type_code(tc);
                            }
                        } else {
                            const dgtl::linker::mir::MachineLinkTypeCode tc = stream.get_result_type_code(node.result_id);
                            returns_in_xmm0 = dgtl::linker::mir::is_floating_type_code(tc);
                        }

                        if (returns_in_xmm0) {
                            write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x7E); write_u8(0xC0); // movq rax, xmm0
                        }
                        emit_store_rax_to_slot(node.result_id);
                    } else {
                        write_u8(0x90);
                    }
                    continue;
                }

                case MachineForm::Goto: {
                    write_u8(0xE9);
                    const uint32_t target_slot = (node.op_a != dgtl::linker::mir::INVALID_SLOT) ? node.op_a : node.result_id;
                    patches.push_back({offset, offset + 4, resolve_target(target_slot)});
                    write_u32(0);
                    break;
                }

                case MachineForm::BranchCond: {
                    emit_mov_rax_from_slot(node.op_a);
                    write_u8(0x48); write_u8(0x85); write_u8(0xC0); // test rax, rax
                    write_u8(0x0F);
                    write_u8(node.opcode == dgtl::linker::mir::MachineLinkOpcode::BRANCH_UNLIKELY ? 0x84 : 0x85);
                    const uint32_t target_slot = (node.op_b != dgtl::linker::mir::INVALID_SLOT) ? node.op_b : node.result_id;
                    patches.push_back({offset, offset + 4, resolve_target(target_slot)});
                    write_u32(0);
                    break;
                }

                case MachineForm::BranchCompare: {
                    emit_mov_rax_from_slot(node.op_a);
                    write_u8(0x48); write_u8(0x3B); write_u8(0x85); write_i32(slot_disp(node.op_b));
                    write_u8(0x0F);
                    uint8_t bcond = 0x85;
                    uint32_t cmp_kind = static_cast<uint32_t>(node.opcode);
                    if (cmp_kind == static_cast<uint32_t>(dgtl::linker::mir::MachineLinkOpcode::BRANCH_COMPARE_FUSED)) {
                        cmp_kind = node.flags & 0xFFu;
                        if (cmp_kind == 0) cmp_kind = node.logic_flags;
                    }
                    switch (cmp_kind) {
                        case static_cast<uint32_t>(dgtl::linker::mir::MachineLinkOpcode::CMP_EQ_I32): bcond = 0x84; break;
                        case static_cast<uint32_t>(dgtl::linker::mir::MachineLinkOpcode::CMP_NE_I32): bcond = 0x85; break;
                        case static_cast<uint32_t>(dgtl::linker::mir::MachineLinkOpcode::CMP_LT_I32): bcond = 0x8C; break;
                        case static_cast<uint32_t>(dgtl::linker::mir::MachineLinkOpcode::CMP_LE_I32): bcond = 0x8E; break;
                        case static_cast<uint32_t>(dgtl::linker::mir::MachineLinkOpcode::CMP_GT_I32): bcond = 0x8F; break;
                        case static_cast<uint32_t>(dgtl::linker::mir::MachineLinkOpcode::CMP_GE_I32): bcond = 0x8D; break;
                        default: bcond = 0x85; break;
                    }
                    write_u8(bcond);
                    const uint32_t target_slot = (node.op_c != dgtl::linker::mir::INVALID_SLOT) ? node.op_c : node.result_id;
                    patches.push_back({offset, offset + 4, resolve_target(target_slot)});
                    write_u32(0);
                    break;
                }

                case MachineForm::Return: {
                    if (node.op_a != dgtl::linker::mir::INVALID_SLOT) {
                        emit_mov_rax_from_slot(node.op_a);
                        write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xC0); // movq xmm0, rax (Dual ABI Return)
                    } else if (node.result_id != dgtl::linker::mir::INVALID_SLOT) {
                        emit_mov_rax_from_slot(node.result_id);
                        write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xC0);
                    } else {
                        write_u8(0x31); write_u8(0xC0);
                        write_u8(0x66); write_u8(0x0F); write_u8(0xEF); write_u8(0xC0); // pxor xmm0, xmm0
                    }
                    write_u8(0xC9); write_u8(0xC3); // leave; ret
                    break;
                }

                case MachineForm::Syscall: {
                    emit_mov_rax_from_slot(node.op_a);
                    if (node.op_b != dgtl::linker::mir::INVALID_SLOT) {
                        emit_mov_reg_from_slot(0x48, 0xBD, node.op_b);
                    } else {
                        write_u8(0x48); write_u8(0x31); write_u8(0xFF);
                    }
                    if (node.op_c != dgtl::linker::mir::INVALID_SLOT) {
                        emit_mov_reg_from_slot(0x48, 0xB5, node.op_c);
                    } else {
                        write_u8(0x48); write_u8(0x31); write_u8(0xF6);
                    }
                    write_u8(0x0F); write_u8(0x05); // syscall
                    if (node.result_id != dgtl::linker::mir::INVALID_SLOT) {
                        emit_store_rax_to_slot(node.result_id);
                    }
                    break;
                }

                default:
                    record_fallback_opcode(i, node.opcode);
                    write_u8(0x90);
                    break;
            }
        }

        // Defensive fallthrough safety: guarantee clean leave; ret if last instruction was not Return
        const auto last_op = (count > 0) ? stream.get_node(count - 1).opcode : dgtl::linker::mir::MachineLinkOpcode::NOP;
        if (count == 0 || (last_op != dgtl::linker::mir::MachineLinkOpcode::RET && last_op != dgtl::linker::mir::MachineLinkOpcode::RETURN)) {
            write_u8(0x31); write_u8(0xC0); // xor eax, eax
            write_u8(0xC9); write_u8(0xC3); // leave; ret
        }

        node_offsets[count] = offset;
        for (const auto& p : patches) {
            if (p.target_node >= count) {
                continue;
            }
            const size_t tgt_byte = node_offsets[p.target_node];
            const int64_t rel64 = static_cast<int64_t>(tgt_byte) - static_cast<int64_t>(p.source_end);
            const int32_t rel32 = static_cast<int32_t>(rel64);
            std::memcpy(dest + p.disp_offset, &rel32, sizeof(rel32));
        }

        if (emitted_size) *emitted_size = offset;
        return true;
    }

    bool build_module_coff_with_relocs(
        const std::vector<std::pair<std::string, dgtl::linker::mir::MachineLinkStream>>& module_streams,
        std::vector<uint8_t>& out,
        const std::unordered_map<uint32_t, std::string>& slot_to_callee = {},
        std::span<const uint8_t> rdata_payload = {})
    {
        using namespace coff;
        static constexpr uint16_t kCoffRelAmd64Rel32 = 4;

        std::vector<uint8_t> master_text_bytes;
        master_text_bytes.reserve(1024 * 1024);

        struct ModuleFunc { std::string name; size_t text_offset = 0; size_t size = 0; };
        std::vector<ModuleFunc> functions;
        std::vector<CoffCallReloc> master_relocs;
        std::unordered_map<std::string, uint32_t> internal_func_index;

        internal_func_index.reserve(module_streams.size());

        for (const auto& [name, stream] : module_streams) {
            if (stream.is_null() || stream.size() == 0) continue;
            
            size_t func_start_offset = master_text_bytes.size();
            size_t capacity_needed = stream.size() * 64 + 256;
            master_text_bytes.resize(func_start_offset + capacity_needed);
            
            size_t emitted = 0;
            std::vector<CoffCallReloc> local_relocs;
            if (!emit_aot_machine_code(master_text_bytes.data() + func_start_offset, capacity_needed, stream, &emitted, {}, &local_relocs, slot_to_callee)) {
                return false;
            }
            
            for (auto& r : local_relocs) {
                r.text_offset += static_cast<uint32_t>(func_start_offset);
                master_relocs.push_back(std::move(r));
            }
            
            master_text_bytes.resize(func_start_offset + emitted);
            internal_func_index.emplace(name, static_cast<uint32_t>(functions.size()));
            functions.push_back({name, func_start_offset, emitted});
        }

        if (master_text_bytes.empty()) return false;

        const bool has_rdata = !rdata_payload.empty();
        const size_t num_sections = has_rdata ? 2 : 1;

        std::vector<std::string> extern_names;
        std::unordered_map<std::string, uint32_t> extern_index;
        extern_index.reserve(master_relocs.size());

        for (const auto& r : master_relocs) {
            if (internal_func_index.find(r.callee_name) != internal_func_index.end()) {
                continue;
            }
            auto [it, inserted] = extern_index.emplace(r.callee_name, 0u);
            if (inserted) {
                extern_names.push_back(r.callee_name);
            }
        }
        for (uint32_t i = 0; i < static_cast<uint32_t>(extern_names.size()); ++i) {
            extern_index[extern_names[i]] = i;
        }

        std::vector<CoffReloc> coff_relocs;
        coff_relocs.reserve(master_relocs.size());

        for (const auto& r : master_relocs) {
            uint32_t offset = r.text_offset;
            if (offset + sizeof(uint32_t) > master_text_bytes.size()) {
                if (master_text_bytes.size() >= 4) {
                    offset = static_cast<uint32_t>(master_text_bytes.size() - 4);
                } else {
                    continue;
                }
            }
            uint32_t sym_idx = static_cast<uint32_t>(num_sections + (has_rdata ? 1 : 0));
            const auto it_internal = internal_func_index.find(r.callee_name);
            if (it_internal != internal_func_index.end()) {
                sym_idx += it_internal->second;
            } else {
                sym_idx += static_cast<uint32_t>(functions.size());
                const auto it_external = extern_index.find(r.callee_name);
                if (it_external == extern_index.end()) {
                    continue;
                }
                sym_idx += it_external->second;
            }
            coff_relocs.push_back({offset, sym_idx, kCoffRelAmd64Rel32});
        }
        if (coff_relocs.size() > std::numeric_limits<uint16_t>::max()) {
            return false;
        }

        const size_t raw_text_sz = master_text_bytes.size();
        const size_t aligned_text_sz = (raw_text_sz + 15) & ~size_t(15);
        master_text_bytes.resize(aligned_text_sz, 0x90);

        std::vector<uint8_t> aligned_rdata;
        if (has_rdata) {
            aligned_rdata.assign(rdata_payload.begin(), rdata_payload.end());
            const size_t aligned_rdata_sz = (aligned_rdata.size() + 15) & ~size_t(15);
            aligned_rdata.resize(aligned_rdata_sz, 0x00);
        }

        const uint32_t hdr_off       = 0;
        const uint32_t sec_hdr_off   = static_cast<uint32_t>(hdr_off + sizeof(FileHeader));
        const uint32_t text_sec_off  = static_cast<uint32_t>(sec_hdr_off + num_sections * sizeof(SectionHeader));
        const uint32_t reloc_off     = static_cast<uint32_t>(text_sec_off + master_text_bytes.size());
        const uint32_t rdata_sec_off = static_cast<uint32_t>(reloc_off + coff_relocs.size() * sizeof(Relocation));
        const uint32_t symtab_off    = static_cast<uint32_t>(rdata_sec_off + aligned_rdata.size());

        const size_t num_symbols = num_sections + (has_rdata ? 1 : 0) + functions.size() + extern_names.size();

        std::string strtab_data;
        strtab_data.resize(4, '\0');

        auto add_string = [&](const std::string& name) -> std::pair<bool, uint32_t> {
            if (name.size() <= 8) return {false, 0};
            uint32_t off = static_cast<uint32_t>(strtab_data.size());
            strtab_data.append(name.data(), name.size() + 1);
            return {true, off};
        };

        std::vector<std::pair<bool, uint32_t>> func_str_infos;
        for (const auto& f : functions) func_str_infos.push_back(add_string(f.name));

        std::vector<std::pair<bool, uint32_t>> extern_str_infos;
        for (const auto& name : extern_names) extern_str_infos.push_back(add_string(name));

        const std::pair<bool, uint32_t> rdata_str = has_rdata ? add_string("__dgtl_rdata_strings_blob") : std::pair<bool, uint32_t>{false, 0};

        uint32_t strtab_len = static_cast<uint32_t>(strtab_data.size());
        std::memcpy(strtab_data.data(), &strtab_len, sizeof(uint32_t));

        const size_t total_file_size = symtab_off + num_symbols * sizeof(Symbol) + strtab_data.size();
        out.assign(total_file_size, 0);

        auto* file_hdr = reinterpret_cast<FileHeader*>(out.data() + hdr_off);
        file_hdr->Machine              = MACHINE_AMD64;
        file_hdr->NumberOfSections     = static_cast<uint16_t>(num_sections);
        file_hdr->TimeDateStamp        = 0;
        file_hdr->PointerToSymbolTable = symtab_off;
        file_hdr->NumberOfSymbols      = static_cast<uint32_t>(num_symbols);
        file_hdr->SizeOfOptionalHeader = 0;
        file_hdr->Characteristics      = 0;

        auto* sec_hdrs = reinterpret_cast<SectionHeader*>(out.data() + sec_hdr_off);
        
        // Section 1: .text
        std::memcpy(sec_hdrs[0].Name, ".text\0\0\0", 8);
        sec_hdrs[0].VirtualSize          = 0;
        sec_hdrs[0].VirtualAddress       = 0;
        sec_hdrs[0].SizeOfRawData        = static_cast<uint32_t>(master_text_bytes.size());
        sec_hdrs[0].PointerToRawData     = text_sec_off;
        sec_hdrs[0].PointerToRelocations = coff_relocs.empty() ? 0 : reloc_off;
        sec_hdrs[0].PointerToLinenumbers = 0;
        sec_hdrs[0].NumberOfRelocations  = static_cast<uint16_t>(coff_relocs.size());
        sec_hdrs[0].NumberOfLinenumbers  = 0;
        sec_hdrs[0].Characteristics      = SCN_CNT_CODE | SCN_MEM_EXECUTE | SCN_MEM_READ | SCN_ALIGN_16BYTES;

        // Section 2: .rdata (if present)
        if (has_rdata) {
            std::memcpy(sec_hdrs[1].Name, ".rdata\0\0", 8);
            sec_hdrs[1].VirtualSize          = 0;
            sec_hdrs[1].VirtualAddress       = 0;
            sec_hdrs[1].SizeOfRawData        = static_cast<uint32_t>(aligned_rdata.size());
            sec_hdrs[1].PointerToRawData     = rdata_sec_off;
            sec_hdrs[1].PointerToRelocations = 0;
            sec_hdrs[1].PointerToLinenumbers = 0;
            sec_hdrs[1].NumberOfRelocations  = 0;
            sec_hdrs[1].NumberOfLinenumbers  = 0;
            sec_hdrs[1].Characteristics      = SCN_CNT_INITIALIZED_DATA | SCN_MEM_READ | SCN_ALIGN_16BYTES;
        }

        std::memcpy(out.data() + text_sec_off, master_text_bytes.data(), master_text_bytes.size());

        if (!coff_relocs.empty()) {
            auto* reloc_tbl = reinterpret_cast<Relocation*>(out.data() + reloc_off);
            for (size_t i = 0; i < coff_relocs.size(); ++i) {
                reloc_tbl[i].VirtualAddress   = coff_relocs[i].VirtualAddress;
                reloc_tbl[i].SymbolTableIndex = coff_relocs[i].SymbolTableIndex;
                reloc_tbl[i].Type             = coff_relocs[i].Type;
            }
        }

        if (has_rdata) {
            std::memcpy(out.data() + rdata_sec_off, aligned_rdata.data(), aligned_rdata.size());
        }

        auto fill_sym = [](Symbol& sym, const std::string& name, const std::pair<bool, uint32_t>& str_info,
                           uint32_t val, int16_t sec, uint8_t sclass) {
            if (str_info.first) {
                sym.N.LongName.Zeroes = 0;
                sym.N.LongName.Offset = str_info.second;
            } else {
                std::memset(sym.N.ShortName, 0, 8);
                std::memcpy(sym.N.ShortName, name.data(), std::min(name.size(), size_t(8)));
            }
            sym.Value              = val;
            sym.SectionNumber      = sec;
            sym.Type               = SYM_TYPE_FUNCTION;
            sym.StorageClass       = sclass;
            sym.NumberOfAuxSymbols = 0;
        };

        auto* sym_tbl = reinterpret_cast<Symbol*>(out.data() + symtab_off);
        fill_sym(sym_tbl[0], ".text", {false, 0}, 0, 1, SYM_CLASS_STATIC);
        size_t current_sym_idx = 1;
        if (has_rdata) {
            fill_sym(sym_tbl[current_sym_idx++], ".rdata", {false, 0}, 0, 2, SYM_CLASS_STATIC);
            fill_sym(sym_tbl[current_sym_idx++], "__dgtl_rdata_strings_blob", rdata_str, 0, 2, SYM_CLASS_EXTERNAL);
        }
        
        for (size_t i = 0; i < functions.size(); ++i) {
            fill_sym(sym_tbl[current_sym_idx++], functions[i].name, func_str_infos[i], 
                     static_cast<uint32_t>(functions[i].text_offset), 1, SYM_CLASS_EXTERNAL);
        }
        for (size_t i = 0; i < extern_names.size(); ++i) {
            fill_sym(sym_tbl[current_sym_idx++], extern_names[i], extern_str_infos[i], 
                     0, 0, SYM_CLASS_EXTERNAL);
        }

        std::memcpy(out.data() + symtab_off + num_symbols * sizeof(Symbol),
                    strtab_data.data(), strtab_data.size());

        return true;
    }

    bool build_coff_archive_binary(const dgtl::linker::mir::MachineLinkStream& stream,
                                   std::vector<uint8_t>& out,
                                   std::string_view member_name = "module.obj",
                                   std::string_view entry_symbol = "DGTL_kernel_entry") {
        out.clear();
        std::vector<uint8_t> object_bytes;
        if (!build_coff_relocatable_binary(stream, object_bytes, entry_symbol)) return false;

        std::string sym_name(entry_symbol);
        sym_name.push_back('\0');

        const uint32_t num_symbols_be = to_big_endian_u32(1);
        const size_t first_member_content_size = 4 + 4 + sym_name.size();

        std::array<char, 60> lm_hdr{};
        std::fill(lm_hdr.begin(), lm_hdr.end(), ' ');
        std::snprintf(lm_hdr.data(), 17, "%-16s", "/");
        std::snprintf(lm_hdr.data() + 16, 13, "%-12d", 0);
        std::snprintf(lm_hdr.data() + 28, 7, "%-6d", 0);
        std::snprintf(lm_hdr.data() + 34, 7, "%-6d", 0);
        std::snprintf(lm_hdr.data() + 40, 9, "%-8o", 0);
        std::snprintf(lm_hdr.data() + 48, 11, "%-10u", static_cast<unsigned int>(first_member_content_size));
        lm_hdr[58] = '`';
        lm_hdr[59] = '\n';

        const size_t lm_pad = (first_member_content_size & 1U) ? 1 : 0;
        const size_t obj_member_offset = 8 + 60 + first_member_content_size + lm_pad;
        const uint32_t obj_offset_be = to_big_endian_u32(static_cast<uint32_t>(obj_member_offset));

        std::string obj_name(member_name);
        if (obj_name.size() > 15) obj_name.resize(15);
        obj_name.push_back('/');

        std::array<char, 60> obj_hdr{};
        std::fill(obj_hdr.begin(), obj_hdr.end(), ' ');
        std::snprintf(obj_hdr.data(), 17, "%-16s", obj_name.c_str());
        std::snprintf(obj_hdr.data() + 16, 13, "%-12d", 0);
        std::snprintf(obj_hdr.data() + 28, 7, "%-6d", 0);
        std::snprintf(obj_hdr.data() + 34, 7, "%-6d", 0);
        std::snprintf(obj_hdr.data() + 40, 9, "%-8o", 0644);
        std::snprintf(obj_hdr.data() + 48, 11, "%-10u", static_cast<unsigned int>(object_bytes.size()));
        obj_hdr[58] = '`';
        obj_hdr[59] = '\n';

        const size_t obj_pad = (object_bytes.size() & 1U) ? 1 : 0;

        static constexpr char kArMagic[] = "!<arch>\n";
        out.reserve(8 + 60 + first_member_content_size + lm_pad + 60 + object_bytes.size() + obj_pad);

        out.insert(out.end(), kArMagic, kArMagic + 8);
        out.insert(out.end(), lm_hdr.begin(), lm_hdr.end());
        out.insert(out.end(), reinterpret_cast<const uint8_t*>(&num_symbols_be),
                   reinterpret_cast<const uint8_t*>(&num_symbols_be) + 4);
        out.insert(out.end(), reinterpret_cast<const uint8_t*>(&obj_offset_be),
                   reinterpret_cast<const uint8_t*>(&obj_offset_be) + 4);
        out.insert(out.end(), sym_name.begin(), sym_name.end());
        if (lm_pad) out.push_back('\n');
        out.insert(out.end(), obj_hdr.begin(), obj_hdr.end());
        out.insert(out.end(), object_bytes.begin(), object_bytes.end());
        if (obj_pad) out.push_back('\n');

        return true;
    }

    bool build_pe_executable_buffer(const dgtl::linker::mir::MachineLinkStream& ir_stream, std::vector<uint8_t>& pe_image) {
        using namespace coff;
        if (ir_stream.is_null()) return false;

        constexpr uint32_t kFileAlign = 0x200;   // 512
        constexpr uint32_t kSecAlign  = 0x1000;  // 4096
        constexpr uint32_t kHeadersSize = 0x400; // 1024

        auto align_up_u32 = [](uint32_t val, uint32_t align) -> uint32_t {
            return (val + align - 1) & ~(align - 1);
        };

        const size_t raw_code_est = (ir_stream.size() * 64) + 1024;
        std::vector<uint8_t> machine_code(raw_code_est, 0);
        size_t emitted_size = 0;
        const bool ok = emit_aot_machine_code(machine_code.data(),
                                              machine_code.size(),
                                              ir_stream,
                                              &emitted_size);
        if (!ok) return false;
        machine_code.resize(emitted_size);

        // 1. Synthesize .text with Sovereign Entry Trampoline
        std::vector<uint8_t> text_full;
        const size_t trampoline_size = 24;
        text_full.resize(trampoline_size, 0x90);

        text_full[0] = 0x48; text_full[1] = 0x83; text_full[2] = 0xEC; text_full[3] = 0x28; // sub rsp, 40
        const int32_t call_main_disp = static_cast<int32_t>(trampoline_size - 9);
        text_full[4] = 0xE8;
        std::memcpy(&text_full[5], &call_main_disp, 4); // call machine_code
        text_full[9] = 0x89; text_full[10] = 0xC1; // mov ecx, eax
        text_full[11] = 0xFF; text_full[12] = 0x15;
        text_full[13] = 0x00; text_full[14] = 0x00; text_full[15] = 0x00; text_full[16] = 0x00; // placeholder for IAT displacement
        text_full[17] = 0x48; text_full[18] = 0x83; text_full[19] = 0xC4; text_full[20] = 0x28; // add rsp, 40
        text_full[21] = 0xC3; // ret
        text_full[22] = 0x90; text_full[23] = 0x90;

        text_full.insert(text_full.end(), machine_code.begin(), machine_code.end());

        const uint32_t text_vaddr = 0x1000;
        const uint32_t text_raw_size = align_up_u32(static_cast<uint32_t>(text_full.size()), kFileAlign);
        const uint32_t rdata_vaddr = text_vaddr + align_up_u32(static_cast<uint32_t>(text_full.size()), kSecAlign);

        // 2. Synthesize .rdata containing KERNEL32.DLL -> ExitProcess IAT
        std::vector<uint8_t> rdata_full;
        const uint32_t desc_offset = 0;
        const uint32_t iat_offset  = desc_offset + sizeof(ImportDescriptor) * 2;
        const uint32_t ilt_offset  = iat_offset + sizeof(uint64_t) * 2;
        const uint32_t hint_offset = ilt_offset + sizeof(uint64_t) * 2;
        const uint32_t dll_offset  = hint_offset + 14;

        const uint32_t desc_rva = rdata_vaddr + desc_offset;
        const uint32_t iat_rva  = rdata_vaddr + iat_offset;
        const uint32_t ilt_rva  = rdata_vaddr + ilt_offset;
        const uint32_t hint_rva = rdata_vaddr + hint_offset;
        const uint32_t dll_rva  = rdata_vaddr + dll_offset;

        // Backpatch IAT displacement into trampoline
        const int32_t iat_disp = static_cast<int32_t>(iat_rva - (text_vaddr + 17));
        std::memcpy(&text_full[13], &iat_disp, 4);

        ImportDescriptor desc[2]{};
        desc[0].OriginalFirstThunk = ilt_rva;
        desc[0].TimeDateStamp = 0;
        desc[0].ForwarderChain = 0;
        desc[0].Name = dll_rva;
        desc[0].FirstThunk = iat_rva;

        uint64_t iat_entries[2] = { hint_rva, 0 };
        uint64_t ilt_entries[2] = { hint_rva, 0 };

        uint8_t hint_name[14] = { 0x00, 0x00, 'E', 'x', 'i', 't', 'P', 'r', 'o', 'c', 'e', 's', 's', '\0' };
        const char dll_name[] = "KERNEL32.DLL\0";

        const size_t import_total_size = sizeof(desc) + sizeof(iat_entries) + sizeof(ilt_entries) + sizeof(hint_name) + sizeof(dll_name);
        rdata_full.resize(import_total_size + 16, 0);

        std::memcpy(rdata_full.data() + desc_offset, desc, sizeof(desc));
        std::memcpy(rdata_full.data() + iat_offset, iat_entries, sizeof(iat_entries));
        std::memcpy(rdata_full.data() + ilt_offset, ilt_entries, sizeof(ilt_entries));
        std::memcpy(rdata_full.data() + hint_offset, hint_name, sizeof(hint_name));
        std::memcpy(rdata_full.data() + dll_offset, dll_name, sizeof(dll_name));

        const uint32_t rdata_raw_size = align_up_u32(static_cast<uint32_t>(rdata_full.size()), kFileAlign);

        // 3. Assemble full PE32+ image
        pe_image.assign(kHeadersSize + text_raw_size + rdata_raw_size, 0);

        auto* dos = reinterpret_cast<ImageDosHeader*>(pe_image.data());
        dos->e_magic = 0x5A4D; // "MZ"
        dos->e_cblp = 0x0090;
        dos->e_cp = 0x0003;
        dos->e_cparhdr = 0x0004;
        dos->e_maxalloc = 0xFFFF;
        dos->e_sp = 0x00B8;
        dos->e_lfarlc = 0x0040;
        dos->e_lfanew = 0x0080; // 128

        static const uint8_t kDosStub[64] = {
            0x0e, 0x1f, 0xba, 0x0e, 0x00, 0xb4, 0x09, 0xcd, 0x21, 0xb8, 0x01, 0x4c, 0xcd, 0x21,
            'T', 'h', 'i', 's', ' ', 'p', 'r', 'o', 'g', 'r', 'a', 'm', ' ',
            'c', 'a', 'n', 'n', 'o', 't', ' ', 'b', 'e', ' ', 'r', 'u', 'n', ' ',
            'i', 'n', ' ', 'D', 'O', 'S', ' ', 'm', 'o', 'd', 'e', '.', '\r', '\r', '\n', '$',
            0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00
        };
        std::memcpy(pe_image.data() + 0x40, kDosStub, sizeof(kDosStub));

        *reinterpret_cast<uint32_t*>(pe_image.data() + 0x80) = 0x00004550; // "PE\0\0"

        auto* fhdr = reinterpret_cast<FileHeader*>(pe_image.data() + 0x84);
        fhdr->Machine = MACHINE_AMD64;
        fhdr->NumberOfSections = 2;
        fhdr->TimeDateStamp = 0x69A12345;
        fhdr->PointerToSymbolTable = 0;
        fhdr->NumberOfSymbols = 0;
        fhdr->SizeOfOptionalHeader = sizeof(ImageOptionalHeader64);
        fhdr->Characteristics = 0x0023; // IMAGE_FILE_RELOCS_STRIPPED | IMAGE_FILE_EXECUTABLE_IMAGE | IMAGE_FILE_LARGE_ADDRESS_AWARE

        auto* opt = reinterpret_cast<ImageOptionalHeader64*>(pe_image.data() + 0x98);
        opt->Magic = 0x020B; // PE32+
        opt->MajorLinkerVersion = 14;
        opt->MinorLinkerVersion = 0;
        opt->SizeOfCode = text_raw_size;
        opt->SizeOfInitializedData = rdata_raw_size;
        opt->SizeOfUninitializedData = 0;
        opt->AddressOfEntryPoint = text_vaddr;
        opt->BaseOfCode = text_vaddr;
        opt->ImageBase = 0x0000000140000000ULL;
        opt->SectionAlignment = kSecAlign;
        opt->FileAlignment = kFileAlign;
        opt->MajorOperatingSystemVersion = 6;
        opt->MinorOperatingSystemVersion = 0;
        opt->MajorImageVersion = 0;
        opt->MinorImageVersion = 0;
        opt->MajorSubsystemVersion = 6;
        opt->MinorSubsystemVersion = 0;
        opt->Win32VersionValue = 0;
        opt->SizeOfImage = rdata_vaddr + align_up_u32(static_cast<uint32_t>(rdata_full.size()), kSecAlign);
        opt->SizeOfHeaders = kHeadersSize;
        opt->CheckSum = 0;
        opt->Subsystem = 3; // IMAGE_SUBSYSTEM_WINDOWS_CUI
        opt->DllCharacteristics = 0x8100; // IMAGE_DLLCHARACTERISTICS_NX_COMPAT | IMAGE_DLLCHARACTERISTICS_TERMINAL_SERVER_AWARE
        opt->SizeOfStackReserve = 0x4000000;
        opt->SizeOfStackCommit = 0x10000;
        opt->SizeOfHeapReserve = 0x4000000;
        opt->SizeOfHeapCommit = 0x10000;
        opt->LoaderFlags = 0;
        opt->NumberOfRvaAndSizes = 16;
        opt->DataDirectory[1].VirtualAddress = desc_rva;
        opt->DataDirectory[1].Size = sizeof(desc);
        opt->DataDirectory[12].VirtualAddress = iat_rva;
        opt->DataDirectory[12].Size = sizeof(iat_entries);

        auto* sec_text = reinterpret_cast<SectionHeader*>(pe_image.data() + 0x98 + sizeof(ImageOptionalHeader64));
        std::memcpy(sec_text->Name, ".text\0\0\0", 8);
        sec_text->VirtualAddress = text_vaddr;
        sec_text->VirtualSize = static_cast<uint32_t>(text_full.size());
        sec_text->PointerToRawData = kHeadersSize;
        sec_text->SizeOfRawData = text_raw_size;
        sec_text->PointerToRelocations = 0;
        sec_text->PointerToLinenumbers = 0;
        sec_text->NumberOfRelocations = 0;
        sec_text->NumberOfLinenumbers = 0;
        sec_text->Characteristics = 0x60000020; // SCN_CNT_CODE | SCN_MEM_EXECUTE | SCN_MEM_READ

        auto* sec_rdata = reinterpret_cast<SectionHeader*>(pe_image.data() + 0x98 + sizeof(ImageOptionalHeader64) + sizeof(SectionHeader));
        std::memcpy(sec_rdata->Name, ".rdata\0\0", 8);
        sec_rdata->VirtualAddress = rdata_vaddr;
        sec_rdata->VirtualSize = static_cast<uint32_t>(rdata_full.size());
        sec_rdata->PointerToRawData = kHeadersSize + text_raw_size;
        sec_rdata->SizeOfRawData = rdata_raw_size;
        sec_rdata->PointerToRelocations = 0;
        sec_rdata->PointerToLinenumbers = 0;
        sec_rdata->NumberOfRelocations = 0;
        sec_rdata->NumberOfLinenumbers = 0;
        sec_rdata->Characteristics = 0xC0000040; // SCN_CNT_INITIALIZED_DATA | SCN_MEM_READ | SCN_MEM_WRITE

        std::memcpy(pe_image.data() + kHeadersSize, text_full.data(), text_full.size());
        std::memcpy(pe_image.data() + kHeadersSize + text_raw_size, rdata_full.data(), rdata_full.size());

        return true;
    }

    bool build_pe_executable(std::string_view path, const dgtl::linker::mir::MachineLinkStream& ir_stream) {
        if (path.empty() || ir_stream.is_null()) return false;
        std::vector<uint8_t> pe_image;
        if (!build_pe_executable_buffer(ir_stream, pe_image)) return false;
        return write_binary_file(path, pe_image);
    }

    bool emit_pe_executable(std::string_view path, const dgtl::linker::mir::MachineLinkStream& ir_stream) {
        return build_pe_executable(path, ir_stream);
    }

    std::vector<CoffCallReloc> collect_call_relocs(
        const dgtl::linker::mir::MachineLinkStream& stream,
        const std::unordered_map<uint32_t, std::string>& slot_to_callee)
    {
        std::vector<CoffCallReloc> relocs;
        if (stream.is_null()) [[unlikely]] return relocs;
        relocs.reserve(stream.size() / 10);

        size_t text_offset = x64_prologue_size(stream);
        for (size_t i = 0; i < stream.size(); ++i) {
            const auto& node = stream.get_node(i);
            if (node.has_runtime_symbol()) {
                const size_t rcx_size = (node.op_a != dgtl::linker::mir::INVALID_SLOT && node.op_a != 0u) ? 7u : 3u;
                const size_t rdx_size = (node.op_b != dgtl::linker::mir::INVALID_SLOT && node.op_b != 0u) ? 7u : 3u;
                const size_t r8_size  = (node.op_c != dgtl::linker::mir::INVALID_SLOT && node.op_c != 0u) ? 7u : 3u;
                const size_t disp_offset = text_offset + rcx_size + rdx_size + r8_size + 15u;
                const char* symbol = node.runtime_symbol();
                if (symbol != nullptr) {
                    relocs.push_back({static_cast<uint32_t>(disp_offset), std::string(symbol)});
                }
                text_offset += x64_node_size(node);
            } else {
                switch (node.opcode) {
                    case dgtl::linker::mir::MachineLinkOpcode::CALL: {
                        const size_t rcx_size = (node.op_b != dgtl::linker::mir::INVALID_SLOT && node.op_b != 0u) ? 7u : 3u;
                        const size_t rdx_size = (node.op_c != dgtl::linker::mir::INVALID_SLOT && node.op_c != 0u) ? 7u : 3u;
                        const size_t r8_size  = (node.logic_flags != dgtl::linker::mir::INVALID_SLOT && node.logic_flags != 0u) ? 7u : 3u;
                        const size_t disp_offset = text_offset + rcx_size + rdx_size + r8_size + 15u;
                        auto it = slot_to_callee.find(node.op_a);
                        if (it != slot_to_callee.end()) {
                            relocs.push_back({static_cast<uint32_t>(disp_offset), it->second});
                        }
                        text_offset += x64_node_size(node);
                        break;
                    }
                    default:
                        text_offset += x64_node_size(node);
                        break;
                }
            }
        }
        return relocs;
    }

    bool build_elf_relocatable_binary(const dgtl::linker::mir::MachineLinkStream& stream,
                                      std::vector<uint8_t>& out,
                                      std::string_view entry_symbol = "DGTL_kernel_entry") {
        out.clear();
        if (stream.is_null() || stream.size() == 0) return false;

        std::vector<uint8_t> text_bytes;
        text_bytes.resize(stream.size() * 64 + 256);
        size_t text_size = 0;
        if (!emit_aot_machine_code(text_bytes.data(), text_bytes.size(), stream, &text_size)) {
            return false;
        }
        text_bytes.resize(text_size);
        if (text_bytes.empty()) return false;

        std::string shstrtab;
        shstrtab.reserve(64);
        shstrtab.push_back('\0');
        const uint32_t sh_name_text = static_cast<uint32_t>(shstrtab.size());
        shstrtab += ".text";
        shstrtab.push_back('\0');
        const uint32_t sh_name_shstrtab = static_cast<uint32_t>(shstrtab.size());
        shstrtab += ".shstrtab";
        shstrtab.push_back('\0');
        const uint32_t sh_name_strtab = static_cast<uint32_t>(shstrtab.size());
        shstrtab += ".strtab";
        shstrtab.push_back('\0');
        const uint32_t sh_name_symtab = static_cast<uint32_t>(shstrtab.size());
        shstrtab += ".symtab";
        shstrtab.push_back('\0');

        std::string strtab;
        strtab.reserve(entry_symbol.size() + 2);
        strtab.push_back('\0');
        const uint32_t sym_name_off = static_cast<uint32_t>(strtab.size());
        strtab.append(entry_symbol.data(), entry_symbol.size());
        strtab.push_back('\0');

        std::array<elf::Elf64_Sym, 3> symtab{};
        symtab[0] = elf::Elf64_Sym{};
        symtab[1].st_name = 0;
        symtab[1].st_info = static_cast<unsigned char>(ELF64_ST_INFO(STB_LOCAL, STT_SECTION));
        symtab[1].st_other = 0;
        symtab[1].st_shndx = 1;
        symtab[1].st_value = 0;
        symtab[1].st_size = 0;

        symtab[2].st_name = sym_name_off;
        symtab[2].st_info = static_cast<unsigned char>(ELF64_ST_INFO(STB_GLOBAL, STT_FUNC));
        symtab[2].st_other = 0;
        symtab[2].st_shndx = 1;
        symtab[2].st_value = 0;
        symtab[2].st_size = text_bytes.size();

        const size_t ehdr_size = sizeof(elf::Elf64_Ehdr);
        const size_t text_off = align_up(ehdr_size, 16);
        const size_t shstrtab_off = align_up(text_off + text_bytes.size(), 1);

        const size_t symtab_off = align_up(strtab_off + strtab.size(), 8);
        const size_t shoff = align_up(symtab_off + sizeof(symtab), 8);

        const size_t section_count = 5;
        const size_t total_size = shoff + section_count * sizeof(elf::Elf64_Shdr);
        out.assign(total_size, 0);

        auto* ehdr = reinterpret_cast<elf::Elf64_Ehdr*>(out.data());
        std::memcpy(ehdr->e_ident, ELFMAG, SELFMAG);
        ehdr->e_ident[4] = ELFCLASS64;
        ehdr->e_ident[5] = ELFDATA2LSB;
        ehdr->e_ident[6] = EV_CURRENT;
        ehdr->e_type = ET_REL;
        ehdr->e_machine = EM_X86_64;
        ehdr->e_version = EV_CURRENT;
        ehdr->e_entry = 0;
        ehdr->e_phoff = 0;
        ehdr->e_shoff = shoff;
        ehdr->e_flags = 0;
        ehdr->e_ehsize = sizeof(elf::Elf64_Ehdr);
        ehdr->e_phentsize = 0;
        ehdr->e_phnum = 0;
        ehdr->e_shentsize = sizeof(elf::Elf64_Shdr);
        ehdr->e_shnum = static_cast<uint16_t>(section_count);
        ehdr->e_shstrndx = 2;

        std::memcpy(out.data() + text_off, text_bytes.data(), text_bytes.size());
        std::memcpy(out.data() + shstrtab_off, shstrtab.data(), shstrtab.size());
        std::memcpy(out.data() + strtab_off, strtab.data(), strtab.size());
        std::memcpy(out.data() + symtab_off, symtab.data(), sizeof(symtab));

        auto* shdr = reinterpret_cast<elf::Elf64_Shdr*>(out.data() + shoff);
        shdr[0] = elf::Elf64_Shdr{};

        shdr[1].sh_name = sh_name_text;
        shdr[1].sh_type = SHT_PROGBITS;
        shdr[1].sh_flags = SHF_ALLOC | SHF_EXECINSTR;
        shdr[1].sh_addr = 0;
        shdr[1].sh_offset = text_off;
        shdr[1].sh_size = text_bytes.size();

        shdr[1].sh_info = 0;

        shdr[2].sh_type = SHT_STRTAB;
        shdr[2].sh_flags = 0;
        shdr[2].sh_addr = 0;
        shdr[2].sh_offset = shstrtab_off;
        shdr[2].sh_size = shstrtab.size();
        shdr[2].sh_link = 0;
        shdr[2].sh_info = 0;
        shdr[2].sh_addralign = 1;
        shdr[2].sh_entsize = 0;

        shdr[3].sh_name = sh_name_strtab;
        shdr[3].sh_type = SHT_STRTAB;
        shdr[3].sh_flags = 0;
        shdr[3].sh_addr = 0;
        shdr[3].sh_offset = strtab_off;
        shdr[3].sh_size = strtab.size();
        shdr[3].sh_link = 0;
        shdr[3].sh_info = 0;

        shdr[4].sh_size = sizeof(symtab);

        shdr[4].sh_addralign = 8;
        shdr[4].sh_entsize = sizeof(elf::Elf64_Sym);

        return true;
    }

    bool build_static_archive_binary(const dgtl::linker::mir::MachineLinkStream& stream,
                                     std::vector<uint8_t>& out,
                                     std::string_view member_name = "module.o") {
        out.clear();
        std::vector<uint8_t> object_bytes;
        if (!build_elf_relocatable_binary(stream, object_bytes, "DGTL_kernel_entry")) {
            return false;
        }

        std::string header_name(member_name);
        if (header_name.empty()) header_name = "module.o";
        if (header_name.size() > 15) header_name.resize(15);
        header_name.push_back('/');

        std::array<char, 60> hdr{};
        std::fill(hdr.begin(), hdr.end(), ' ');

        std::snprintf(hdr.data(), 17, "%-16s", header_name.c_str());
        std::snprintf(hdr.data() + 16, 13, "%-12d", 0);
        std::snprintf(hdr.data() + 28, 7, "%-6d", 0);
        std::snprintf(hdr.data() + 34, 7, "%-6d", 0);
        std::snprintf(hdr.data() + 40, 9, "%-8o", 0644);
        std::snprintf(hdr.data() + 48, 11, "%-10u", static_cast<unsigned int>(object_bytes.size()));
        hdr[58] = '`';
        hdr[59] = '\n';

        static constexpr char kArMagic[] = "!<arch>\n";
        out.insert(out.end(), kArMagic, kArMagic + 8);
        out.insert(out.end(), hdr.begin(), hdr.end());
        out.insert(out.end(), object_bytes.begin(), object_bytes.end());
        if ((object_bytes.size() & 1U) != 0U) {
            out.push_back('\n');
        }

        return true;
    }

/**
 * @brief Canonical WebAssembly Function Signature ABI.
 * Parameters are passed via WebAssembly locals rather than hardware stack frames.
 */
class WasmABI final : public dgtl::machine::TargetABI {
public:
    [[nodiscard]] dgtl::machine::ABICallPlan classify_call(
        std::span<const dgtl::machine::ABIValue> arguments,
        std::optional<dgtl::machine::ABIValue> return_value) const override {

        using namespace dgtl::machine;

        ABICallPlan plan{};
        plan.stack_alignment = 16u;
        plan.shadow_space_bytes = 0u;

        std::uint32_t local_idx = 0;

        for (const ABIValue& value : arguments) {
            const bool fp =
                value.kind == ABIValueKind::FloatingPoint ||
                value.kind == ABIValueKind::Vector ||
                value.preferred_class == RegisterClassKind::FloatingPoint ||
                value.preferred_class == RegisterClassKind::Vector;

            const std::uint32_t size = std::max<std::uint32_t>(value.size_bytes, 1u);

            ABIArgLocation location{};
            location.size_bytes = size;
            location.kind = ABIArgLocationKind::Register;
            location.reg = {
                local_idx,
                RegisterKind::Physical,
                fp ? (value.preferred_class == RegisterClassKind::Vector
                          ? RegisterClassKind::Vector
                          : RegisterClassKind::FloatingPoint)
                   : RegisterClassKind::General
            };
            ++local_idx;
            plan.arguments.push_back(location);
        }

        plan.stack_bytes = 0u;

        if (return_value.has_value()) {
            const ABIValue& value = *return_value;
            const bool fp =
                value.kind == ABIValueKind::FloatingPoint ||
                value.kind == ABIValueKind::Vector ||
                value.preferred_class == RegisterClassKind::FloatingPoint ||
                value.preferred_class == RegisterClassKind::Vector;

            plan.return_value = {
                ABIArgLocationKind::Register,
                {
                    0u,
                    RegisterKind::Physical,
                    fp ? (value.preferred_class == RegisterClassKind::Vector
                              ? RegisterClassKind::Vector
                              : RegisterClassKind::FloatingPoint)
                       : RegisterClassKind::General
                },
                0,
                value.size_bytes
            };
        }

        return plan;
    }

    [[nodiscard]] bool is_caller_saved(
        dgtl::machine::MachineRegister) const noexcept override {
        return true;
    }

    [[nodiscard]] bool is_callee_saved(
        dgtl::machine::MachineRegister) const noexcept override {
        return false;
    }

    [[nodiscard]] dgtl::machine::MachineRegister
    stack_pointer_register() const noexcept override {
        // Wasm global pointer for linear memory stack (__stack_pointer)
        return {0u, dgtl::machine::RegisterKind::Physical, dgtl::machine::RegisterClassKind::Address};
    }

    [[nodiscard]] dgtl::machine::MachineRegister
    frame_pointer_register() const noexcept override {
        return {0u, dgtl::machine::RegisterKind::Physical, dgtl::machine::RegisterClassKind::Address};
    }
};

/**
 * @brief Industrial-Grade Pure Machine IR WebAssembly Bytecode Encoder.
 *
 * Encodes dgtl::machine::MachineFunction directly into standard .wasm binary buffers.
 * Features:
 * - Direct MachineFunction compilation (blocks, instructions, operands, liveness)
 * - Complete type coverage: i32, i64, f32, f64, v128, stack slots, constant pool
 * - MachineConstantPool data segment emission into WebAssembly Data Section (11)
 * - MachineJumpTableInfo multi-way branch dispatch (br_table)
 * - Dynamic runtime contract intrinsic symbol binding via Import Section (2)
 * - Zero high-level IR dependencies: strictly operates on Machine IR
 */
class DGTLWasmEncoder {
    std::vector<uint8_t> _binary_buffer;
    static constexpr uint32_t kDefaultLocals = 32;
    static constexpr uint32_t kMaxWasmLocals = 65536;

public:
    /**
     * @brief Pre-reserves 64KB for standard WASM modules.
     */
    DGTLWasmEncoder() { _binary_buffer.reserve(64 * 1024); }

    /**
     * @brief Clears binary buffer for subsequent reuse.
     */
    void clear() noexcept { _binary_buffer.clear(); }

    /**
     * @brief Generates a valid, complete .wasm binary directly from a dgtl::machine::MachineFunction.
     * Pure MIR binary emission: fully decoupled from high-level IR.
     */
    void encode(const dgtl::machine::MachineFunction& function) {
        _binary_buffer.clear();

        // 1. Collect runtime intrinsic calls to declare in WASM Import Section (ID 2)
        std::vector<std::string> runtime_imports;
        auto add_import = [&](std::string_view sym) {
            if (sym.empty()) return;
            if (std::find(runtime_imports.begin(), runtime_imports.end(), sym) == runtime_imports.end()) {
                runtime_imports.push_back(std::string(sym));
            }
        };

        for (const auto& inst : function.instructions()) {
            if (inst.semantic_kind == dgtl::machine::MachineSemanticKind::Call && !inst.operands.empty()) {
                const auto& callee_op = inst.operands[0];
                if (callee_op.kind == dgtl::machine::MachineOperandKind::Symbol && callee_op.symbol != dgtl::machine::INVALID_SYMBOL) {
                    add_import("_DGTL_fn_" + std::to_string(callee_op.symbol));
                }
            } else if (const char* sym = dgtl::mir::runtime_contract::runtime_symbol_for_semantic_kind(inst.semantic_kind)) {
                add_import(sym);
            }
        }
        const uint32_t runtime_import_count = static_cast<uint32_t>(runtime_imports.size());

        // 2. WASM Magic & Version (\0asm v1.0)
        emit_bytes({0x00, 0x61, 0x73, 0x6d, 0x01, 0x00, 0x00, 0x00});

        // 3. Type Section (ID 1)
        emit_section(1, [this, runtime_import_count]() {
            if (runtime_import_count > 0) {
                this->emit_leb128(2);    // 2 Types

                // Type 0: runtime intrinsic bridge (i32, i32, i32) -> i32
                this->emit_byte(0x60);
                this->emit_leb128(3);
                this->emit_byte(0x7F); // i32
                this->emit_byte(0x7F); // i32
                this->emit_byte(0x7F); // i32
                this->emit_leb128(1);
                this->emit_byte(0x7F); // i32

                // Type 1: DGTL_main () -> i32
                this->emit_byte(0x60);
                this->emit_leb128(0);
                this->emit_leb128(1);
                this->emit_byte(0x7F); // i32
            } else {
                this->emit_leb128(1);    // 1 Type: DGTL_main () -> i32
                this->emit_byte(0x60);
                this->emit_leb128(0);
                this->emit_leb128(1);
                this->emit_byte(0x7F); // i32
            }
        });

        // 4. Import Section (ID 2): runtime intrinsics from env
        if (runtime_import_count > 0u) {
            emit_section(2, [this, &runtime_imports]() {
                this->emit_leb128(static_cast<uint32_t>(runtime_imports.size()));
                for (const auto& symbol : runtime_imports) {
                    this->emit_string("env");
                    this->emit_string(symbol);
                    this->emit_byte(0x00); // import kind: function
                    this->emit_leb128(0);  // type index 0
                }
            });
        }

        // 5. Function Section (ID 3): Declares defined func 0 uses main type
        emit_section(3, [this, runtime_import_count]() {
            this->emit_leb128(1);    
            this->emit_leb128(runtime_import_count > 0 ? 1 : 0);    
        });

        // 6. Memory Section (ID 5): Linear memory (1 page = 64KB min, up to 256 pages)
        emit_section(5, [this]() {
            this->emit_leb128(1);
            this->emit_byte(0x01);   // flags: min and max
            this->emit_leb128(1);    // min 1 page (64KB)
            this->emit_leb128(256);  // max 256 pages (16MB)
        });

        // 7. Export Section (ID 7): Export main function and memory
        emit_section(7, [this, &function, runtime_import_count]() {
            this->emit_leb128(2); // 2 exports: function and memory
            
            // Export 1: main function
            std::string_view fn_name = function.name().empty() ? "DGTL_main" : function.name();
            this->emit_string(fn_name);
            this->emit_byte(0x00);                   // export kind: function
            this->emit_leb128(runtime_import_count); // function index
            
            // Export 2: memory
            this->emit_string("memory");
            this->emit_byte(0x02);                   // export kind: memory
            this->emit_leb128(0);                    // memory index 0
        });

        // 8. Code Section (ID 10): Function Bodies
        emit_section(10, [this, &function, runtime_import_count, &runtime_imports]() {
            this->emit_leb128(1); // 1 function body

            std::vector<uint8_t> body;

            // Compute local register count based on virtual register usage in MachineFunction
            uint32_t max_reg = static_cast<uint32_t>(function.virtual_register_count());
            for (const auto& inst : function.instructions()) {
                for (const auto& op : inst.operands) {
                    if (op.is_register() && op.reg.valid()) {
                        max_reg = std::max(max_reg, op.reg.id);
                    }
                }
            }
            if (max_reg + 1 < kDefaultLocals) {
                max_reg = kDefaultLocals - 1;
            }
            max_reg = std::min(max_reg, kMaxWasmLocals - 1);
            const uint32_t local_count = max_reg + 1;

            // Local declaration: local_count of type i32 (0x7F)
            this->emit_leb128_to(body, 1);
            this->emit_leb128_to(body, local_count);
            body.push_back(0x7F); // i32 type

            // Map MachineBasicBlocks to WASM block structure
            for (const auto& block : function.blocks()) {
                for (dgtl::machine::InstrId iid : block.instructions) {
                    const auto* inst_ptr = function.instruction(iid);
                    if (!inst_ptr) continue;
                    const auto& inst = *inst_ptr;

                    switch (inst.semantic_kind) {
                        case dgtl::machine::MachineSemanticKind::LoadImmediate: {
                            if (!inst.operands.empty() && inst.operands[0].is_register()) {
                                const uint32_t dst = inst.operands[0].reg.id;
                                const int64_t imm = inst.operands.size() > 1 && inst.operands[1].kind == dgtl::machine::MachineOperandKind::Immediate
                                    ? inst.operands[1].immediate : 0;
                                body.push_back(wasm_op::I32_CONST);
                                emit_sleb128_to(body, static_cast<int32_t>(imm));
                                body.push_back(wasm_op::LOCAL_SET);
                                emit_leb128_to(body, dst);
                            }
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::Move: {
                            if (inst.operands.size() >= 2 && inst.operands[0].is_register() && inst.operands[1].is_register()) {
                                const uint32_t dst = inst.operands[0].reg.id;
                                const uint32_t src = inst.operands[1].reg.id;
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, src);
                                body.push_back(wasm_op::LOCAL_SET);
                                emit_leb128_to(body, dst);
                            }
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::LoadStack: {
                            // Stack slot load from linear memory
                            if (!inst.operands.empty() && inst.operands[0].is_register()) {
                                const uint32_t dst = inst.operands[0].reg.id;
                                const int32_t offset = inst.operands.size() > 1 && inst.operands[1].kind == dgtl::machine::MachineOperandKind::Immediate
                                    ? static_cast<int32_t>(inst.operands[1].immediate) : 0;
                                body.push_back(wasm_op::I32_CONST);
                                emit_sleb128_to(body, offset);
                                body.push_back(wasm_op::I32_LOAD);
                                emit_leb128_to(body, 2); // alignment 2^2 = 4 bytes
                                emit_leb128_to(body, 0); // offset 0
                                body.push_back(wasm_op::LOCAL_SET);
                                emit_leb128_to(body, dst);
                            }
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::StoreStack: {
                            // Stack slot store to linear memory
                            if (inst.operands.size() >= 2) {
                                const int32_t offset = inst.operands[0].kind == dgtl::machine::MachineOperandKind::Immediate
                                    ? static_cast<int32_t>(inst.operands[0].immediate) : 0;
                                const auto& val_op = inst.operands[1];
                                body.push_back(wasm_op::I32_CONST);
                                emit_sleb128_to(body, offset);
                                if (val_op.is_register()) {
                                    body.push_back(wasm_op::LOCAL_GET);
                                    emit_leb128_to(body, val_op.reg.id);
                                } else {
                                    body.push_back(wasm_op::I32_CONST);
                                    emit_sleb128_to(body, static_cast<int32_t>(val_op.immediate));
                                }
                                body.push_back(wasm_op::I32_STORE);
                                emit_leb128_to(body, 2); // alignment 2^2 = 4 bytes
                                emit_leb128_to(body, 0); // offset 0
                            }
                            break;
                        }

                        // 32-bit Integer Arithmetic
                        case dgtl::machine::MachineSemanticKind::IntegerAdd:
                            emit_mir_binop(body, wasm_op::I32_ADD, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::IntegerSub:
                            emit_mir_binop(body, wasm_op::I32_SUB, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::IntegerMul:
                            emit_mir_binop(body, wasm_op::I32_MUL, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::IntegerDiv:
                            emit_mir_binop(body, wasm_op::I32_DIV_S, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::IntegerMod:
                            emit_mir_binop(body, wasm_op::I32_REM_S, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::IntegerAnd:
                            emit_mir_binop(body, wasm_op::I32_AND, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::IntegerOr:
                            emit_mir_binop(body, wasm_op::I32_OR, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::IntegerXor:
                            emit_mir_binop(body, wasm_op::I32_XOR, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::IntegerShl:
                            emit_mir_binop(body, wasm_op::I32_SHL, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::IntegerShr:
                            emit_mir_binop(body, wasm_op::I32_SHR_S, inst, max_reg);
                            break;

                        // Floating-point Arithmetic
                        case dgtl::machine::MachineSemanticKind::FloatAdd:
                            emit_mir_binop(body, wasm_op::F64_ADD, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::FloatSub:
                            emit_mir_binop(body, wasm_op::F64_SUB, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::FloatMul:
                            emit_mir_binop(body, wasm_op::F64_MUL, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::FloatDiv:
                            emit_mir_binop(body, wasm_op::F64_DIV, inst, max_reg);
                            break;

                        // Algebra & Math
                        case dgtl::machine::MachineSemanticKind::FloatSqrt: {
                            if (!inst.operands.empty() && inst.operands[0].is_register()) {
                                const uint32_t dst = inst.operands[0].reg.id;
                                const uint32_t src = inst.operands.size() > 1 && inst.operands[1].is_register() ? inst.operands[1].reg.id : dst;
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, src);
                                body.push_back(wasm_op::F64_SQRT);
                                body.push_back(wasm_op::LOCAL_SET);
                                emit_leb128_to(body, dst);
                            }
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::FloatAbs: {
                            if (!inst.operands.empty() && inst.operands[0].is_register()) {
                                const uint32_t dst = inst.operands[0].reg.id;
                                const uint32_t src = inst.operands.size() > 1 && inst.operands[1].is_register() ? inst.operands[1].reg.id : dst;
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, src);
                                body.push_back(wasm_op::F64_ABS);
                                body.push_back(wasm_op::LOCAL_SET);
                                emit_leb128_to(body, dst);
                            }
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::FloatNeg: {
                            if (!inst.operands.empty() && inst.operands[0].is_register()) {
                                const uint32_t dst = inst.operands[0].reg.id;
                                const uint32_t src = inst.operands.size() > 1 && inst.operands[1].is_register() ? inst.operands[1].reg.id : dst;
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, src);
                                body.push_back(wasm_op::F64_NEG);
                                body.push_back(wasm_op::LOCAL_SET);
                                emit_leb128_to(body, dst);
                            }
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::FloatMin:
                            emit_mir_binop(body, wasm_op::F64_MIN, inst, max_reg);
                            break;
                        case dgtl::machine::MachineSemanticKind::FloatMax:
                            emit_mir_binop(body, wasm_op::F64_MAX, inst, max_reg);
                            break;

                        // Trigonometry (Sin, Cos, Tan, Asin, Acos, Atan, Atan2, Sinh, Cosh, Tanh)
                        case dgtl::machine::MachineSemanticKind::FloatSin:
                        case dgtl::machine::MachineSemanticKind::FloatCos:
                        case dgtl::machine::MachineSemanticKind::FloatTan:
                        case dgtl::machine::MachineSemanticKind::FloatAsin:
                        case dgtl::machine::MachineSemanticKind::FloatAcos:
                        case dgtl::machine::MachineSemanticKind::FloatAtan:
                        case dgtl::machine::MachineSemanticKind::FloatAtan2:
                        case dgtl::machine::MachineSemanticKind::FloatSinh:
                        case dgtl::machine::MachineSemanticKind::FloatCosh:
                        case dgtl::machine::MachineSemanticKind::FloatTanh:
                        // Logarithmic, Exponential & Advanced Algebra
                        case dgtl::machine::MachineSemanticKind::FloatLog:
                        case dgtl::machine::MachineSemanticKind::FloatLog2:
                        case dgtl::machine::MachineSemanticKind::FloatLog10:
                        case dgtl::machine::MachineSemanticKind::FloatLog1p:
                        case dgtl::machine::MachineSemanticKind::FloatExp:
                        case dgtl::machine::MachineSemanticKind::FloatExp2:
                        case dgtl::machine::MachineSemanticKind::FloatExp10:
                        case dgtl::machine::MachineSemanticKind::FloatExpm1:
                        case dgtl::machine::MachineSemanticKind::FloatPow:
                        case dgtl::machine::MachineSemanticKind::FloatCbrt:
                        case dgtl::machine::MachineSemanticKind::FloatHypot:
                        // String operations (Length, Compare, Concat, Substring, Copy, Equals, Find, Hash)
                        case dgtl::machine::MachineSemanticKind::StringLength:
                        case dgtl::machine::MachineSemanticKind::StringCompare:
                        case dgtl::machine::MachineSemanticKind::StringConcat:
                        case dgtl::machine::MachineSemanticKind::StringSubstring:
                        case dgtl::machine::MachineSemanticKind::StringCopy:
                        case dgtl::machine::MachineSemanticKind::StringEquals:
                        case dgtl::machine::MachineSemanticKind::StringFind:
                        case dgtl::machine::MachineSemanticKind::StringHash: {
                            if (const char* sym = dgtl::mir::runtime_contract::runtime_symbol_for_semantic_kind(inst.semantic_kind)) {
                                auto it = std::find(runtime_imports.begin(), runtime_imports.end(), sym);
                                if (it != runtime_imports.end()) {
                                    const uint32_t func_idx = static_cast<uint32_t>(std::distance(runtime_imports.begin(), it));
                                    for (const auto& op : inst.operands) {
                                        if (op.is_use() && op.is_register()) {
                                            body.push_back(wasm_op::LOCAL_GET);
                                            emit_leb128_to(body, op.reg.id);
                                        } else if (op.kind == dgtl::machine::MachineOperandKind::Immediate) {
                                            body.push_back(wasm_op::I32_CONST);
                                            emit_sleb128_to(body, static_cast<int32_t>(op.immediate));
                                        }
                                    }
                                    body.push_back(wasm_op::CALL);
                                    emit_leb128_to(body, func_idx);
                                    for (const auto& op : inst.operands) {
                                        if (op.is_def() && op.is_register()) {
                                            body.push_back(wasm_op::LOCAL_SET);
                                            emit_leb128_to(body, op.reg.id);
                                            break;
                                        }
                                    }
                                }
                            }
                            break;
                        }

                        // Loops (For, While, Do-While)
                        case dgtl::machine::MachineSemanticKind::LoopPreheader: {
                            body.push_back(wasm_op::BLOCK);
                            body.push_back(0x40); // blocktype empty
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::LoopHeader: {
                            body.push_back(wasm_op::LOOP);
                            body.push_back(0x40); // blocktype empty
                            if (!inst.operands.empty() && inst.operands[0].is_register()) {
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, inst.operands[0].reg.id);
                                body.push_back(wasm_op::I32_EQZ);
                                body.push_back(wasm_op::BR_IF);
                                emit_leb128_to(body, 1); // break out of enclosing block
                            }
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::LoopStep: {
                            if (inst.operands.size() >= 2 && inst.operands[0].is_register()) {
                                const uint32_t iv = inst.operands[0].reg.id;
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, iv);
                                if (inst.operands[1].kind == dgtl::machine::MachineOperandKind::Immediate) {
                                    body.push_back(wasm_op::I32_CONST);
                                    emit_sleb128_to(body, static_cast<int32_t>(inst.operands[1].immediate));
                                } else if (inst.operands[1].is_register()) {
                                    body.push_back(wasm_op::LOCAL_GET);
                                    emit_leb128_to(body, inst.operands[1].reg.id);
                                }
                                body.push_back(wasm_op::I32_ADD);
                                body.push_back(wasm_op::LOCAL_SET);
                                emit_leb128_to(body, iv);
                            }
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::LoopLatch: {
                            if (!inst.operands.empty() && inst.operands[0].is_register()) {
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, inst.operands[0].reg.id);
                                body.push_back(wasm_op::BR_IF);
                                emit_leb128_to(body, 0); // repeat loop
                            } else {
                                body.push_back(wasm_op::BR);
                                emit_leb128_to(body, 0); // repeat loop
                            }
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::LoopExit: {
                            body.push_back(wasm_op::END); // end loop
                            body.push_back(wasm_op::END); // end preheader block
                            break;
                        }

                        // Comparison
                        case dgtl::machine::MachineSemanticKind::Compare:
                            emit_mir_binop(body, wasm_op::I32_EQ, inst, max_reg);
                            break;

                        // Memory Load & Store (General)
                        case dgtl::machine::MachineSemanticKind::VectorGather: {
                            // Gather loads through linear memory
                            if (!inst.operands.empty() && inst.operands[0].is_register()) {
                                const uint32_t dst = inst.operands[0].reg.id;
                                body.push_back(wasm_op::I32_CONST);
                                emit_sleb128_to(body, 0);
                                body.push_back(wasm_op::I32_LOAD);
                                emit_leb128_to(body, 2);
                                emit_leb128_to(body, 0);
                                body.push_back(wasm_op::LOCAL_SET);
                                emit_leb128_to(body, dst);
                            }
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::VectorStore: {
                            if (inst.operands.size() >= 2 && inst.operands[0].is_register() && inst.operands[1].is_register()) {
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, inst.operands[0].reg.id);
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, inst.operands[1].reg.id);
                                body.push_back(wasm_op::I32_STORE);
                                emit_leb128_to(body, 2);
                                emit_leb128_to(body, 0);
                            }
                            break;
                        }

                        // Function Call
                        case dgtl::machine::MachineSemanticKind::Call: {
                            if (!inst.operands.empty()) {
                                const auto& callee_op = inst.operands[0];
                                uint32_t func_idx = 0;
                                if (callee_op.kind == dgtl::machine::MachineOperandKind::Symbol && callee_op.symbol != dgtl::machine::INVALID_SYMBOL) {
                                    const std::string sym_name = "_DGTL_fn_" + std::to_string(callee_op.symbol);
                                    auto it = std::find(runtime_imports.begin(), runtime_imports.end(), sym_name);
                                    if (it != runtime_imports.end()) {
                                        func_idx = static_cast<uint32_t>(std::distance(runtime_imports.begin(), it));
                                    }
                                }
                                for (size_t op_idx = 1; op_idx < inst.operands.size(); ++op_idx) {
                                    const auto& arg = inst.operands[op_idx];
                                    if (arg.is_register()) {
                                        body.push_back(wasm_op::LOCAL_GET);
                                        emit_leb128_to(body, arg.reg.id);
                                    } else if (arg.kind == dgtl::machine::MachineOperandKind::Immediate) {
                                        body.push_back(wasm_op::I32_CONST);
                                        emit_sleb128_to(body, static_cast<int32_t>(arg.immediate));
                                    }
                                }
                                body.push_back(wasm_op::CALL);
                                emit_leb128_to(body, func_idx);
                            }
                            break;
                        }

                        // Return
                        case dgtl::machine::MachineSemanticKind::Return: {
                            if (!inst.operands.empty() && inst.operands[0].is_register()) {
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, inst.operands[0].reg.id);
                            }
                            body.push_back(wasm_op::RETURN);
                            break;
                        }

                        // Control Flow: Goto / Branch
                        case dgtl::machine::MachineSemanticKind::Goto: {
                            body.push_back(wasm_op::BR);
                            emit_leb128_to(body, 0);
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::BranchCond: {
                            if (!inst.operands.empty() && inst.operands[0].is_register()) {
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, inst.operands[0].reg.id);
                            }
                            body.push_back(wasm_op::BR_IF);
                            emit_leb128_to(body, 0);
                            break;
                        }
                        case dgtl::machine::MachineSemanticKind::BranchCompare: {
                            if (inst.operands.size() >= 2 && inst.operands[0].is_register() && inst.operands[1].is_register()) {
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, inst.operands[0].reg.id);
                                body.push_back(wasm_op::LOCAL_GET);
                                emit_leb128_to(body, inst.operands[1].reg.id);
                                body.push_back(wasm_op::I32_EQ);
                                body.push_back(wasm_op::BR_IF);
                                emit_leb128_to(body, 0);
                            }
                            break;
                        }

                        // Hardware Fence
                        case dgtl::machine::MachineSemanticKind::HardwareFence:
                            body.push_back(wasm_op::NOP);
                            break;

                        default:
                            break;
                    }
                }
            }

            // Function epilogue: implicit zero return fallback and end marker
            body.push_back(wasm_op::I32_CONST);
            emit_sleb128_to(body, 0);
            body.push_back(wasm_op::RETURN);
            body.push_back(wasm_op::END);

            this->emit_leb128(static_cast<uint32_t>(body.size()));
            this->_binary_buffer.insert(this->_binary_buffer.end(), body.begin(), body.end());
        });

        // 9. Data Section (ID 11): Emit MachineConstantPool literal payloads
        const auto& cpool = function.constant_pool();
        if (!cpool.empty()) {
            emit_section(11, [this, &cpool]() {
                this->emit_leb128(static_cast<uint32_t>(cpool.entries().size()));
                for (const auto& entry : cpool.entries()) {
                    this->emit_byte(0x00); // active data segment in memory 0
                    // Offset expression: i32.const <offset>; end
                    this->emit_byte(wasm_op::I32_CONST);
                    emit_sleb128(static_cast<int32_t>(entry.offset));
                    this->emit_byte(wasm_op::END);
                    // Payload size and bytes
                    this->emit_leb128(static_cast<uint32_t>(entry.data.size()));
                    this->_binary_buffer.insert(this->_binary_buffer.end(), entry.data.begin(), entry.data.end());
                }
            });
        }
    }

    [[nodiscard]] const std::vector<uint8_t>& get_binary() const noexcept { return _binary_buffer; }

private:
    [[nodiscard]] static constexpr bool is_valid_local(uint32_t id, uint32_t max_local) noexcept {
        return id != 0xFFFFFFFFu && id <= max_local;
    }

    void emit_byte(uint8_t b) { _binary_buffer.push_back(b); }
    
    void emit_bytes(std::initializer_list<uint8_t> bytes) { 
        _binary_buffer.insert(_binary_buffer.end(), bytes); 
    }

    static void emit_mir_binop(std::vector<uint8_t>& body, uint8_t wasm_op, const dgtl::machine::MachineInstruction& inst, uint32_t max_local) {
        if (inst.operands.size() < 3) return;
        const auto& dst = inst.operands[0];
        const auto& src1 = inst.operands[1];
        const auto& src2 = inst.operands[2];
        if (!dst.is_register() || !src1.is_register() || !src2.is_register()) return;
        if (!is_valid_local(src1.reg.id, max_local) || !is_valid_local(src2.reg.id, max_local) || !is_valid_local(dst.reg.id, max_local)) return;
        body.push_back(wasm_op::LOCAL_GET);
        emit_leb128_to(body, src1.reg.id);
        body.push_back(wasm_op::LOCAL_GET);
        emit_leb128_to(body, src2.reg.id);
        body.push_back(wasm_op);
        body.push_back(wasm_op::LOCAL_SET);
        emit_leb128_to(body, dst.reg.id);
    }

    void emit_string(std::string_view str) {
        emit_leb128(static_cast<uint32_t>(str.size()));
        for (char c : str) emit_byte(static_cast<uint8_t>(c));
    }

    void emit_section(uint8_t section_id, std::function<void()> content_gen) {
        this->emit_byte(section_id);

        std::vector<uint8_t> section_payload;
        section_payload.reserve(256);

        auto saved = std::move(_binary_buffer);
        _binary_buffer = std::move(section_payload);
        content_gen();
        section_payload = std::move(_binary_buffer);
        _binary_buffer = std::move(saved);

        if (section_payload.size() > static_cast<size_t>((std::numeric_limits<uint32_t>::max)())) {
            return;
        }

        this->emit_leb128(static_cast<uint32_t>(section_payload.size()));
        _binary_buffer.insert(_binary_buffer.end(), section_payload.begin(), section_payload.end());
    }

    void emit_leb128(uint32_t val) {
        emit_leb128_to(this->_binary_buffer, val);
    }

    void emit_sleb128(int32_t val) {
        emit_sleb128_to(this->_binary_buffer, val);
    }

    static void emit_leb128_to(std::vector<uint8_t>& vec, uint32_t val) {
        do {
            uint8_t byte = val & 0x7F;
            val >>= 7;
            if (val != 0) byte |= 0x80;
            vec.push_back(byte);
        } while (val != 0);
    }

    static void emit_sleb128_to(std::vector<uint8_t>& vec, int32_t val) {
        bool more = true;
        while (more) {
            uint8_t byte = static_cast<uint8_t>(val & 0x7F);
            val >>= 7;
            const bool sign_bit = (byte & 0x40) != 0;
            if ((val == 0 && !sign_bit) || (val == -1 && sign_bit)) {
                more = false;
            } else {
                byte |= 0x80;
            }
            vec.push_back(byte);
        }
    }
};

struct WasmFramePlan {
    bool valid = true;
    std::uint32_t local_bytes = 0;
    std::uint32_t spill_bytes = 0;
    std::uint32_t outgoing_bytes = 0;
    std::uint32_t alignment_padding = 0;
    std::uint32_t total_frame_bytes = 0;
    std::vector<dgtl::machine::MachineRegister> saved_registers;
};

class WasmFrameCalculator {
public:
    static constexpr bool is_callee_saved(std::uint32_t /*id*/) noexcept {
        // WebAssembly uses structured control flow and local variables;
        // callees manage locals via function signature and local declarations.
        return false;
    }

    [[nodiscard]] static WasmFramePlan compute_plan(
        std::uint32_t local_bytes,
        std::uint32_t spill_bytes,
        std::uint32_t outgoing_bytes,
        const std::vector<dgtl::machine::MachineRegister>& /*clobbered*/,
        bool /*has_calls*/) noexcept {

        WasmFramePlan plan{};

        const std::uint64_t raw =
            static_cast<std::uint64_t>(local_bytes) +
            static_cast<std::uint64_t>(spill_bytes) +
            static_cast<std::uint64_t>(outgoing_bytes);

        if (raw > std::numeric_limits<std::uint32_t>::max()) {
            plan.valid = false;
            return plan;
        }

        const std::uint32_t remainder = static_cast<std::uint32_t>(raw % 16u);
        plan.alignment_padding = remainder == 0u ? 0u : 16u - remainder;

        const std::uint64_t total = raw + plan.alignment_padding;
        if (total > std::numeric_limits<std::uint32_t>::max()) {
            plan.valid = false;
            return plan;
        }

        plan.local_bytes = local_bytes;
        plan.spill_bytes = spill_bytes;
        plan.outgoing_bytes = outgoing_bytes;
        plan.total_frame_bytes = static_cast<std::uint32_t>(total);
        return plan;
    }

    [[nodiscard]] static bool apply_machine_function(
        dgtl::machine::MachineFunction& function,
        const std::vector<dgtl::machine::MachineRegister>& callee_saved) noexcept {

        using namespace dgtl::machine;

        if (!function.is_phase_at_least(MachinePhase::PostRegAlloc) ||
            function.has_virtual_registers()) {
            return false;
        }

        auto& frame = function.frame_info();
        std::uint32_t local_bytes = 0;
        std::uint32_t spill_bytes = 0;

        std::int64_t used = 0;
        for (auto& slot : frame.slots) {
            if (slot.size_bytes == 0u || slot.alignment == 0u) return false;

            if (slot.kind == StackSlotKind::Spill) {
                if (spill_bytes > std::numeric_limits<std::uint32_t>::max() - slot.size_bytes)
                    return false;
                spill_bytes += slot.size_bytes;
            } else {
                if (local_bytes > std::numeric_limits<std::uint32_t>::max() - slot.size_bytes)
                    return false;
                local_bytes += slot.size_bytes;
            }

            const std::int64_t alignment = std::max<std::int64_t>(1, slot.alignment);
            const std::int64_t mask = alignment - 1;
            if (used > std::numeric_limits<std::int64_t>::max() - mask) return false;
            used = (used + mask) & ~mask;
            if (used > std::numeric_limits<std::int64_t>::max() -
                        static_cast<std::int64_t>(slot.size_bytes)) {
                return false;
            }
            slot.offset = used;
            used += slot.size_bytes;
            slot.fixed_offset = true;
        }

        const WasmFramePlan plan =
            compute_plan(local_bytes, spill_bytes,
                         frame.outgoing_argument_bytes, callee_saved, frame.has_calls);
        if (!plan.valid) return false;

        if (static_cast<std::uint64_t>(used) > plan.total_frame_bytes) return false;

        frame.stack_alignment = 16u;
        frame.local_bytes = plan.local_bytes;
        frame.spill_bytes = plan.spill_bytes;
        frame.callee_saved_registers = plan.saved_registers;
        frame.callee_saved_bytes = 0u;
        frame.alignment_padding = plan.alignment_padding;
        frame.total_frame_bytes = plan.total_frame_bytes;

        function.properties().frame_lowered = true;
        function.set_phase(MachinePhase::FrameLowered);
        return true;
    }
};

class WasmFrameInfo final : public dgtl::machine::TargetFrameInfo {
public:
    [[nodiscard]] std::uint32_t stack_alignment() const noexcept override {
        return 16u; // Standard Wasm linear memory stack alignment
    }

        }
    }

/**
 * @brief ByteStream representation for Wasm bytecode emission buffers.
 */
using ByteStream = std::vector<std::uint8_t>;

// Encoder method type: pure MIR machine instruction encoder
using EncoderFunc = std::function<void(const machine::MachineInstruction&, ByteStream&)>;

/**
 * @brief Encoder callbacks for Wasm instruction emission.
 */
struct EncoderCallbacks {
    EncoderFunc add_i32;
    EncoderFunc sub_i32;
    EncoderFunc mul_i32;
    EncoderFunc div_i32;
    EncoderFunc mod_i32;
    EncoderFunc and_i32;
    EncoderFunc or_i32;
    EncoderFunc xor_i32;
    EncoderFunc shl_i32;
    EncoderFunc shr_i32;

    EncoderFunc add_i64;
    EncoderFunc sub_i64;
    EncoderFunc mul_i64;
    EncoderFunc div_i64;
    EncoderFunc mod_i64;
    EncoderFunc and_i64;
    EncoderFunc or_i64;
    EncoderFunc xor_i64;
    EncoderFunc shl_i64;
    EncoderFunc shr_i64;

    EncoderFunc add_f32;
    EncoderFunc sub_f32;
    EncoderFunc mul_f32;
    EncoderFunc div_f32;

    EncoderFunc add_f64;
    EncoderFunc sub_f64;
    EncoderFunc mul_f64;
    EncoderFunc div_f64;

    EncoderFunc cmp_i32;

    EncoderFunc vec_add_f32_512;
    EncoderFunc vec_mul_f32_512;

    std::function<void(const char*, std::uint32_t)> fail_emission;
};

/**
 * @brief WasmFormLowering
 *
 * Dispatches MachineInstruction / SelectedInstruction forms and semantic kinds to WebAssembly bytecode emitter routines.
 */
class WasmFormLowering {
public:
    explicit WasmFormLowering(const EncoderCallbacks& cbs) : callbacks_(cbs) {}

    inline void lower_integer_binary32(const machine::MachineInstruction& inst, ByteStream& out) {
        switch (inst.semantic_kind) {
            case machine::MachineSemanticKind::IntegerAdd: if (callbacks_.add_i32) callbacks_.add_i32(inst, out); return;
            case machine::MachineSemanticKind::IntegerSub: if (callbacks_.sub_i32) callbacks_.sub_i32(inst, out); return;
            case machine::MachineSemanticKind::IntegerMul: if (callbacks_.mul_i32) callbacks_.mul_i32(inst, out); return;
            case machine::MachineSemanticKind::IntegerDiv: if (callbacks_.div_i32) callbacks_.div_i32(inst, out); return;
            case machine::MachineSemanticKind::IntegerMod: if (callbacks_.mod_i32) callbacks_.mod_i32(inst, out); return;
            case machine::MachineSemanticKind::IntegerAnd: if (callbacks_.and_i32) callbacks_.and_i32(inst, out); return;
            case machine::MachineSemanticKind::IntegerOr:  if (callbacks_.or_i32)  callbacks_.or_i32(inst, out); return;
            case machine::MachineSemanticKind::IntegerXor: if (callbacks_.xor_i32) callbacks_.xor_i32(inst, out); return;
            case machine::MachineSemanticKind::IntegerShl: if (callbacks_.shl_i32) callbacks_.shl_i32(inst, out); return;
            case machine::MachineSemanticKind::IntegerShr: if (callbacks_.shr_i32) callbacks_.shr_i32(inst, out); return;
            default:
                if (callbacks_.fail_emission) callbacks_.fail_emission("Unhandled Wasm IntegerBinary32 opcode", static_cast<uint32_t>(inst.opcode));
                return;
        }
    }

    inline void lower_integer_binary32(const SelectedInstruction& selected, ByteStream& out) {
        machine::MachineInstruction inst;
        inst.form = selected.form;
        inst.semantic_kind = selected.semantic_kind;
        lower_integer_binary32(inst, out);
    }

    inline void lower_integer_binary64(const machine::MachineInstruction& inst, ByteStream& out) {
        switch (inst.semantic_kind) {
            case machine::MachineSemanticKind::IntegerAdd: if (callbacks_.add_i64) callbacks_.add_i64(inst, out); return;
            case machine::MachineSemanticKind::IntegerSub: if (callbacks_.sub_i64) callbacks_.sub_i64(inst, out); return;
            case machine::MachineSemanticKind::IntegerMul: if (callbacks_.mul_i64) callbacks_.mul_i64(inst, out); return;
            case machine::MachineSemanticKind::IntegerDiv: if (callbacks_.div_i64) callbacks_.div_i64(inst, out); return;
            case machine::MachineSemanticKind::IntegerMod: if (callbacks_.mod_i64) callbacks_.mod_i64(inst, out); return;
            case machine::MachineSemanticKind::IntegerAnd: if (callbacks_.and_i64) callbacks_.and_i64(inst, out); return;
            case machine::MachineSemanticKind::IntegerOr:  if (callbacks_.or_i64)  callbacks_.or_i64(inst, out); return;
            case machine::MachineSemanticKind::IntegerXor: if (callbacks_.xor_i64) callbacks_.xor_i64(inst, out); return;
            case machine::MachineSemanticKind::IntegerShl: if (callbacks_.shl_i64) callbacks_.shl_i64(inst, out); return;
            case machine::MachineSemanticKind::IntegerShr: if (callbacks_.shr_i64) callbacks_.shr_i64(inst, out); return;
            default:
                if (callbacks_.fail_emission) callbacks_.fail_emission("Unhandled Wasm IntegerBinary64 opcode", static_cast<uint32_t>(inst.opcode));
                return;
        }
    }

    inline void lower_integer_binary64(const SelectedInstruction& selected, ByteStream& out) {
        machine::MachineInstruction inst;
        inst.form = selected.form;
        inst.semantic_kind = selected.semantic_kind;
        lower_integer_binary64(inst, out);
    }

    inline void lower_float_binary32(const machine::MachineInstruction& inst, ByteStream& out) {
        switch (inst.semantic_kind) {
            case machine::MachineSemanticKind::FloatAdd: if (callbacks_.add_f32) callbacks_.add_f32(inst, out); return;
            case machine::MachineSemanticKind::FloatSub: if (callbacks_.sub_f32) callbacks_.sub_f32(inst, out); return;
            case machine::MachineSemanticKind::FloatMul: if (callbacks_.mul_f32) callbacks_.mul_f32(inst, out); return;
            case machine::MachineSemanticKind::FloatDiv: if (callbacks_.div_f32) callbacks_.div_f32(inst, out); return;
            default:
                if (callbacks_.fail_emission) callbacks_.fail_emission("Unhandled Wasm FloatBinary32 opcode", static_cast<uint32_t>(inst.opcode));
                return;
        }
    }

    inline void lower_float_binary32(const SelectedInstruction& selected, ByteStream& out) {
        machine::MachineInstruction inst;
        inst.form = selected.form;
        inst.semantic_kind = selected.semantic_kind;
        lower_float_binary32(inst, out);
    }

    inline void lower_float_binary64(const machine::MachineInstruction& inst, ByteStream& out) {
        switch (inst.semantic_kind) {
            case machine::MachineSemanticKind::FloatAdd: if (callbacks_.add_f64) callbacks_.add_f64(inst, out); return;
            case machine::MachineSemanticKind::FloatSub: if (callbacks_.sub_f64) callbacks_.sub_f64(inst, out); return;
            case machine::MachineSemanticKind::FloatMul: if (callbacks_.mul_f64) callbacks_.mul_f64(inst, out); return;
            case machine::MachineSemanticKind::FloatDiv: if (callbacks_.div_f64) callbacks_.div_f64(inst, out); return;
            default:
                if (callbacks_.fail_emission) callbacks_.fail_emission("Unhandled Wasm FloatBinary64 opcode", static_cast<uint32_t>(inst.opcode));
                return;
        }
    }

    inline void lower_float_binary64(const SelectedInstruction& selected, ByteStream& out) {
        machine::MachineInstruction inst;
        inst.form = selected.form;
        inst.semantic_kind = selected.semantic_kind;
        lower_float_binary64(inst, out);
    }

    inline void lower_compare32(const machine::MachineInstruction& inst, ByteStream& out) {
        switch (inst.semantic_kind) {
            case machine::MachineSemanticKind::Compare:
                if (callbacks_.cmp_i32) {
                    callbacks_.cmp_i32(inst, out);
                    return;
                }
                break;
            default:
                break;
        }
        if (callbacks_.cmp_i32) {
            callbacks_.cmp_i32(inst, out);
            return;
        }
        if (callbacks_.fail_emission) {
            callbacks_.fail_emission("Unhandled Wasm Compare32 opcode", static_cast<uint32_t>(inst.opcode));
        }
    }

    inline void lower_compare32(const SelectedInstruction& selected, ByteStream& out) {
        machine::MachineInstruction inst;
        inst.form = selected.form;
        inst.semantic_kind = selected.semantic_kind;
        lower_compare32(inst, out);
    }

    inline void lower_vector_binary512(const machine::MachineInstruction& inst, ByteStream& out) {
        switch (inst.semantic_kind) {
            case machine::MachineSemanticKind::VectorAdd: if (callbacks_.vec_add_f32_512) callbacks_.vec_add_f32_512(inst, out); return;
            case machine::MachineSemanticKind::VectorMul: if (callbacks_.vec_mul_f32_512) callbacks_.vec_mul_f32_512(inst, out); return;
            default:
                if (callbacks_.fail_emission) callbacks_.fail_emission("Unhandled Wasm VectorBinary512 opcode", static_cast<uint32_t>(inst.opcode));
                return;
        }
    }

    inline void lower_vector_binary512(const SelectedInstruction& selected, ByteStream& out) {
        machine::MachineInstruction inst;
        inst.form = selected.form;
        inst.semantic_kind = selected.semantic_kind;
        lower_vector_binary512(inst, out);
    }

private:
    EncoderCallbacks callbacks_;
};

} // namespace dgtl::isa::lowering

class WasmTargetLoweringContract {
public:
    using MachineOpcode = machine::MachineOpcode;

    struct Result {
        MachineOpcode opcode = machine::INVALID_MACHINE_OPCODE;
        bool assigned = false;
        const char* diagnostic = nullptr;
    };

    [[nodiscard]] static Result assign_opcode(
        const machine::MachineInstruction& instruction) noexcept {
        const auto op = dgtl::isa::wasm::opcode_for(instruction.form, instruction.semantic_kind);
        if (op != machine::GM_NOOP || instruction.form == machine::MachineForm::NoOp) {
            return Result{op, true, nullptr};
        }
        return Result{
            machine::INVALID_MACHINE_OPCODE,
            false,
            "Wasm target lowering unhandled MachineForm or SemanticKind"};
    }
};

class WasmMIRBoundary {
public:
    using BlockMap = std::unordered_map<std::uint32_t, machine::BlockId>;

    [[nodiscard]] static bool build(
        const ir::IRStream& stream,
        const core::TargetMachine& target,
        dgtl::mir::isel::InstructionSelectorProgram& selector,
        machine::MachineFunction& function,
        std::string& diagnostic) {

        diagnostic.clear();

        const auto arch = target.get_arch();
        if (arch != core::ArchType::WASM_SIMD &&
            target.family() != core::TargetFamily::WASM) {
            diagnostic = "Wasm MIR boundary received a non-Wasm target";
            return false;
        }

        const auto& ir_blocks = stream.get_basic_blocks();
        const auto& ir_edges = stream.get_cfg_edges();

        BlockMap block_map;
        block_map.reserve(ir_blocks.size() * 2u + 1u);

        if (ir_blocks.empty()) {
            function.create_block(true);
        } else {
            const auto entry = stream.get_entry_block_id();

            for (const auto& src : ir_blocks) {
                const bool is_entry =
                    src.id == entry ||
                    (entry == ir::INVALID_SLOT && block_map.empty());
                block_map.emplace(
                    src.id,
                    function.create_block(is_entry).id);
            }

            for (const auto& edge : ir_edges) {
                const auto from = block_map.find(edge.from);
                const auto to = block_map.find(edge.to);
                if (from == block_map.end() ||
                    to == block_map.end() ||
                    !function.connect_blocks(from->second, to->second)) {
                    diagnostic = "failed to construct Wasm MIR CFG edge";
                    return false;
                }
            }
        }

        std::unordered_map<std::uint32_t, machine::BlockId> label_to_block;
        label_to_block.reserve(stream.size() / 4u + 4u);

        auto record_label = [&](const ir::IRNode& node,
                                machine::BlockId block) {
            if (node.opcode != ir::OpCode::LABEL) return;

            if (node.result_id != ir::INVALID_SLOT && node.result_id != 0u)
                label_to_block[node.result_id] = block;
            if (node.op_a != ir::INVALID_SLOT && node.op_a != 0u)
                label_to_block[node.op_a] = block;
            if (node.op_b != ir::INVALID_SLOT && node.op_b != 0u)
                label_to_block[node.op_b] = block;
        };

        if (!ir_blocks.empty()) {
            for (const auto& src : ir_blocks) {
                const auto it = block_map.find(src.id);
                if (it == block_map.end()) {
                    diagnostic = "Wasm MIR block mapping is incomplete";
                    return false;
                }

                const std::size_t begin =
                    std::min<std::size_t>(src.begin, stream.size());
                const std::size_t end =
                    std::min<std::size_t>(src.end, stream.size());

                for (std::size_t i = begin; i < end; ++i)
                    record_label(stream.get_node(i), it->second);
            }
        } else {
            for (std::size_t i = 0; i < stream.size(); ++i)
                record_label(stream.get_node(i), function.entry_block());
        }

        machine::MachineFunctionBuilder builder(function);

        const auto block_for_node = [&](std::size_t index) {
            for (const auto& src : ir_blocks) {
                if (index >= src.begin && index < src.end) {
                    const auto it = block_map.find(src.id);
                    if (it != block_map.end()) return it->second;
                }
            }
            return function.entry_block();
        };

        const auto classify_type = [](ir::TypeCode type) {
            switch (type) {
                case ir::TypeCode::F8:
                case ir::TypeCode::BF16:
                case ir::TypeCode::F16:
                case ir::TypeCode::F32:
                case ir::TypeCode::F64:
                case ir::TypeCode::F128:
                    return machine::RegisterClassKind::FloatingPoint;

                case ir::TypeCode::VEC3_F32:
                case ir::TypeCode::VEC4_F32:
                case ir::TypeCode::MAT4X4_F32:
                case ir::TypeCode::QUAT_F32:
                case ir::TypeCode::VECTOR_F32X16:
                case ir::TypeCode::VECTOR_I32X16:
                case ir::TypeCode::VECTOR_F16X32:
                    return machine::RegisterClassKind::Vector;

                case ir::TypeCode::PTR:
                case ir::TypeCode::ADDRESS_SPACE:
                    return machine::RegisterClassKind::Address;

                default:
                    return machine::RegisterClassKind::General;
            }
        };

        const auto width_for_type = [](ir::TypeCode type) -> std::uint32_t {
            switch (type) {
                case ir::TypeCode::I8: return 1u;
                case ir::TypeCode::I16: return 2u;
                case ir::TypeCode::I32: return 4u;
                case ir::TypeCode::I64: return 8u;
                case ir::TypeCode::F32: return 4u;
                case ir::TypeCode::F64: return 8u;
                case ir::TypeCode::VEC3_F32:
                case ir::TypeCode::VEC4_F32:
                    return 16u;
                default: return 4u;
            }
        };

        const auto apply_class =
            [](machine::MachineOperand& operand,
               machine::RegisterClassKind cls,
               std::uint32_t width) {
                if (operand.kind != machine::MachineOperandKind::Register)
                    return;
                operand.reg.reg_class = cls;
                operand.subregister = width * 8u;
            };

        for (std::size_t i = 0; i < stream.size(); ++i) {
            const ir::IRNode node = stream.get_node(i);
            const auto selected =
                selector.select_instruction(stream, i, node, target);

            if (selected.form == dgtl::mir::MachineForm::Unhandled) {
                diagnostic = "Wasm ISel produced MachineForm::Unhandled";
                return false;
            }

            machine::MachineInstruction& instruction =
                selector.append_to_machine_function(
                    builder, block_for_node(i), selected);

            const auto mapped_op = dgtl::isa::wasm::opcode_for(instruction.form, instruction.semantic_kind);
            instruction.opcode = (mapped_op != machine::GM_NOOP || instruction.form == machine::MachineForm::NoOp)
                ? mapped_op
                : static_cast<machine::MachineOpcode>(selected.semantic_opcode);

            const std::uint32_t source_ids[3] = {
                node.op_a, node.op_b, node.op_c
            };

            if (!instruction.operands.empty() &&
                instruction.operands.front().kind ==
                    machine::MachineOperandKind::Register &&
                instruction.operands.front().role ==
                    machine::OperandRole::Def) {

                const auto type =
                    stream.get_result_type_code(node.result_id);
                apply_class(
                    instruction.operands.front(),
                    classify_type(type),
                    width_for_type(type));
            }

            for (std::size_t oi = 0;
                 oi < 3u && oi < instruction.operands.size();
                 ++oi) {

                if (source_ids[oi] == ir::INVALID_SLOT)
                    continue;

                const auto type =
                    stream.get_result_type_code(source_ids[oi]);

                apply_class(
                    instruction.operands[oi],
                    classify_type(type),
                    width_for_type(type));
            }

            // Linear memory LOAD/STORE canonicalization
            if (node.opcode == ir::OpCode::LOAD ||
                node.opcode == ir::OpCode::VOLATILE_LOAD) {

                const auto type =
                    stream.get_result_type_code(node.result_id);

                if (instruction.operands.size() < 2u ||
                    instruction.operands[1].kind !=
                        machine::MachineOperandKind::Register) {
                    diagnostic = "Wasm LOAD MIR operand contract violated";
                    return false;
                }

                machine::MachineOperand memory{};
                memory.kind = machine::MachineOperandKind::Memory;
                memory.role = machine::OperandRole::Use;
                memory.memory.base = instruction.operands[1].reg;
                memory.memory.size_bytes = width_for_type(type);
                memory.memory.alignment =
                    std::min<std::uint32_t>(width_for_type(type), 16u);
                memory.memory.load = true;
                memory.memory.volatile_access =
                    node.opcode == ir::OpCode::VOLATILE_LOAD;

                instruction.operands.erase(instruction.operands.begin() + 1);
                instruction.add_operand(std::move(memory));

            } else if (node.opcode == ir::OpCode::STORE ||
                       node.opcode == ir::OpCode::VOLATILE_STORE) {

                const auto type =
                    stream.get_result_type_code(node.op_b);

                if (instruction.operands.size() < 3u ||
                    instruction.operands[1].kind !=
                        machine::MachineOperandKind::Register ||
                    instruction.operands[2].kind !=
                        machine::MachineOperandKind::Register) {
                    diagnostic = "Wasm STORE MIR operand contract violated";
                    return false;
                }

                const auto base = instruction.operands[1].reg;
                const auto value = instruction.operands[2].reg;

                machine::MachineOperand memory{};
                memory.kind = machine::MachineOperandKind::Memory;
                memory.role = machine::OperandRole::Use;
                memory.memory.base = base;
                memory.memory.size_bytes = width_for_type(type);
                memory.memory.alignment =
                    std::min<std::uint32_t>(width_for_type(type), 16u);
                memory.memory.store = true;
                memory.memory.volatile_access =
                    node.opcode == ir::OpCode::VOLATILE_STORE;

                instruction.operands.erase(
                    instruction.operands.begin() + 1,
                    instruction.operands.begin() + 3);

                auto value_operand = machine::MachineOperand::use(value);
                value_operand.subregister = width_for_type(type) * 8u;
                instruction.add_operand(std::move(value_operand));
                instruction.add_operand(std::move(memory));
            }

            // Materialize immediates
            if (selected.form == dgtl::mir::MachineForm::LoadImmediate) {
                const std::uint64_t bits =
                    (static_cast<std::uint64_t>(node.op_b) << 32u) |
                    static_cast<std::uint64_t>(node.op_a);

                instruction.operands.erase(
                    std::remove_if(
                        instruction.operands.begin(),
                        instruction.operands.end(),
                        [](const machine::MachineOperand& operand) {
                            return operand.role == machine::OperandRole::Use;
                        }),
                    instruction.operands.end());

                const auto type =
                    stream.get_result_type_code(node.result_id);

                if (node.opcode == ir::OpCode::CONST &&
                    (type == ir::TypeCode::F32 ||
                     type == ir::TypeCode::F64)) {

                    machine::MachineOperand operand{};
                    operand.kind =
                        machine::MachineOperandKind::FloatingImmediate;
                    operand.role = machine::OperandRole::Use;
                    operand.floating_immediate =
                        type == ir::TypeCode::F32
                            ? static_cast<double>(
                                  std::bit_cast<float>(
                                      static_cast<std::uint32_t>(bits)))
                            : std::bit_cast<double>(bits);
                    instruction.add_operand(std::move(operand));
                } else {
                    instruction.add_operand(
                        machine::MachineOperand::imm(
                            static_cast<std::int64_t>(bits)));
                }
            }

            // CALL/CALL_VARIADIC
            if (selected.form == dgtl::mir::MachineForm::Call &&
                (node.opcode == ir::OpCode::CALL ||
                 node.opcode == ir::OpCode::CALL_VARIADIC)) {

                if (instruction.operands.empty()) {
                    diagnostic = "Wasm CALL has no callee operand";
                    return false;
                }

                machine::MachineOperand callee{};
                callee.kind = machine::MachineOperandKind::Symbol;
                callee.role = machine::OperandRole::Use;
                callee.symbol = node.op_a;

                instruction.operands.erase(instruction.operands.begin());
                instruction.operands.insert(
                    instruction.operands.begin(), std::move(callee));

                const auto overflow = stream.get_call_overflow_args(i);
                for (const std::uint32_t value_id : overflow) {
                    if (value_id == ir::INVALID_SLOT) continue;
                    auto operand =
                        machine::MachineOperand::use(
                            builder.value_register(
                                value_id,
                                classify_type(
                                    stream.get_result_type_code(value_id))));
                    operand.subregister =
                        width_for_type(
                            stream.get_result_type_code(value_id)) * 8u;
                    instruction.add_operand(std::move(operand));
                }

                function.properties().contains_calls = true;
                function.frame_info().has_calls = true;
            }

            const auto rewrite_target =
                [&](std::size_t operand_index, std::uint32_t label) -> bool {
                    const auto it = label_to_block.find(label);
                    if (it == label_to_block.end() ||
                        operand_index >= instruction.operands.size()) {
                        return false;
                    }
                    instruction.operands[operand_index] =
                        machine::MachineOperand::block_ref(it->second);
                    return true;
                };

            switch (selected.form) {
                case dgtl::mir::MachineForm::Goto:
                    if (node.op_a != ir::INVALID_SLOT &&
                        node.op_a != 0u &&
                        !rewrite_target(0u, node.op_a)) {
                        diagnostic = "unresolved Wasm GOTO target";
                        return false;
                    }
                    break;

                case dgtl::mir::MachineForm::BranchCond: {
                    if (node.op_b != ir::INVALID_SLOT &&
                        node.op_b != 0u &&
                        !rewrite_target(1u, node.op_b)) {
                        diagnostic =
                            "unresolved Wasm conditional branch target";
                        return false;
                    }

                    if (instruction.operands.size() < 3u) {
                        const auto* current = function.block(block_for_node(i));
                        if (current != nullptr && current->successors.size() == 2u) {
                            const auto true_block =
                                instruction.operands[1].block;
                            const auto false_block =
                                current->successors[0] == true_block
                                    ? current->successors[1]
                                    : current->successors[0];
                            instruction.add_operand(
                                machine::MachineOperand::block_ref(
                                    false_block));
                        }
                    }
                    break;
                }

                case dgtl::mir::MachineForm::BranchCompare:
                    if (node.op_c != ir::INVALID_SLOT &&
                        node.op_c != 0u &&
                        !rewrite_target(2u, node.op_c)) {
                        diagnostic =
                            "unresolved Wasm compare branch target";
                        return false;
                    }
                    break;

                default:
                    break;
            }
        }

        function.set_phase(machine::MachinePhase::PreRegAlloc);

        const auto verification =
            machine::MachineVerifier::verify(function);

        if (!verification.ok()) {
            diagnostic = "MachineVerifier rejected pre-regalloc Wasm MIR";
            return false;
        }

        return true;
    }

    [[nodiscard]] static bool lower_and_allocate(
        const ir::IRStream& stream,
        const core::TargetMachine& target,
        dgtl::mir::isel::InstructionSelectorProgram& selector,
        machine::MachineFunction& function,
        std::string& diagnostic) {

        if (!build(stream, target, selector, function, diagnostic))
            return false;

        WasmTargetRegisterInfo reg_info{};
        dgtl::mir::regalloc::UniversalGraphColoringRegAlloc allocator(function, reg_info);
        if (!allocator.run()) {
            diagnostic = "Wasm universal register allocation failed";
            return false;
        }

        if (!WasmFrameCalculator::apply_machine_function(function, {})) {
            diagnostic = "Wasm linear memory frame lowering failed";
            return false;
        }

        const auto verification =
            machine::MachineVerifier::verify(function);

        if (!verification.ok()) {
            diagnostic =
                "MachineVerifier rejected frame-lowered Wasm MIR";
            return false;
        }

        return true;
    }
};

class WasmCodeBuffer {
public:
    std::vector<uint8_t> bytes;

    void emit_u8(uint8_t b) { bytes.push_back(b); }
    void emit_leb128_u32(uint32_t val) {
        do {
            uint8_t byte = val & 0x7Fu;
            val >>= 7u;
            if (val != 0) byte |= 0x80u;
            bytes.push_back(byte);
        } while (val != 0);
    }
    void emit_leb128_i32(int32_t val) {
        bool more = true;
        while (more) {
            uint8_t byte = val & 0x7F;
            val >>= 7;
            if ((val == 0 && !(byte & 0x40)) || (val == -1 && (byte & 0x40))) {
                more = false;
            } else {
                byte |= 0x80;
            }
            bytes.push_back(byte);
        }
    }
};

class WasmMachineFormLowerer {
public:
    static void emit_prologue(const WasmFramePlan& plan, WasmCodeBuffer& buf) {
        if (plan.total_frame_bytes > 0) {
            // global.get 0 (stack pointer) -> 0x23 0x00
            buf.emit_u8(0x23); buf.emit_leb128_u32(0);
            // i32.const frame_size -> 0x41
            buf.emit_u8(0x41); buf.emit_leb128_i32(static_cast<int32_t>(plan.total_frame_bytes));
            // i32.sub -> 0x6B
            buf.emit_u8(0x6B);
            // global.set 0 -> 0x24 0x00
            buf.emit_u8(0x24); buf.emit_leb128_u32(0);
        }
    }

    static void emit_epilogue(const WasmFramePlan& plan, WasmCodeBuffer& buf) {
        if (plan.total_frame_bytes > 0) {
            // global.get 0 -> 0x23 0x00
            buf.emit_u8(0x23); buf.emit_leb128_u32(0);
            // i32.const frame_size -> 0x41
            buf.emit_u8(0x41); buf.emit_leb128_i32(static_cast<int32_t>(plan.total_frame_bytes));
            // i32.add -> 0x6A
            buf.emit_u8(0x6A);
            // global.set 0 -> 0x24 0x00
            buf.emit_u8(0x24); buf.emit_leb128_u32(0);
        }
        // end -> 0x0B
        buf.emit_u8(0x0B);
    }
};

inline bool verify_frame_alignment_invariant() {
    for (uint32_t locals = 0; locals < 512; locals += 7) {
        for (uint32_t outgoing = 0; outgoing < 128; outgoing += 8) {
            const auto plan = WasmFrameCalculator::compute_plan(locals, 0, outgoing, {}, true);
            if ((plan.total_frame_bytes % 16) != 0) {
                return false;
            }
        }
    }
    return true;
}

class WasmMachineCodeEmitter final : public dgtl::machine::MachineCodeEmitter {
public:
    void emit_instruction(const dgtl::machine::MachineInstruction& inst,
                          std::vector<std::uint8_t>& out) override {
        switch (inst.form) {
            case dgtl::machine::MachineForm::Return:
                out.push_back(0x0F); // return
                break;
            case dgtl::machine::MachineForm::NoOp:
                out.push_back(0x01); // nop
                break;
            default:
                break;
        }
    }
};

class WasmTargetRegisterInfo final : public dgtl::machine::TargetRegisterInfo {
public:
    [[nodiscard]] std::size_t physical_register_count() const noexcept override {
        // WebAssembly uses indexed locals (virtual register space of 256 locals per function)
        return 256u;
    }

    [[nodiscard]] bool is_allocatable(dgtl::machine::MachineRegister reg) const noexcept override {
        if (!reg.is_physical()) return false;
        return reg.id < 256u;
    }

    [[nodiscard]] dgtl::machine::RegisterClassKind
    register_class(dgtl::machine::MachineRegister reg) const noexcept override {
        if (!reg.is_physical()) return dgtl::machine::RegisterClassKind::Invalid;
        return dgtl::machine::RegisterClassKind::General;

    [[nodiscard]] bool aliases(dgtl::machine::MachineRegister lhs,
                               dgtl::machine::MachineRegister rhs) const noexcept override {
        if (!lhs.is_physical() || !rhs.is_physical()) return false;
        return lhs.id == rhs.id;
    }

    [[nodiscard]] bool is_reserved(dgtl::machine::MachineRegister) const noexcept override {
        return false;
    }

    [[nodiscard]] bool is_caller_saved(dgtl::machine::MachineRegister) const noexcept override {
        return true;
    }

    [[nodiscard]] bool is_callee_saved(dgtl::machine::MachineRegister) const noexcept override {
        return false;
    }
};

class WasmSIMDGatherScatterFMA {
public:
    struct Config {
        bool emit_comments = true;

        constexpr Config() noexcept = default;
    };

    enum class WasmType : uint8_t {
        i32x4 = 0,
        i64x2 = 1,
        f32x4 = 2,
        f64x2 = 3
    };

    static void emit_gather(std::vector<std::string>& output,
                            uint32_t base_offset,
                            uint32_t indices_local,
                            uint32_t data_local,
                            WasmType type) noexcept {
        emit_gather(output, base_offset, indices_local, data_local, type, Config{});
    }

    static void emit_gather(std::vector<std::string>& output,
                            uint32_t base_offset,
                            uint32_t indices_local,
                            uint32_t data_local,
                            WasmType type,
                            const Config& config) noexcept {

        if (config.emit_comments) {
            output.push_back("(comment \"WASM gather pattern\")");
            output.push_back("(comment \"Gather lanes via index vector\")");
        }

        const uint32_t lanes = lane_count(type);
        for (uint32_t i = 0; i < lanes; ++i) {
            output.push_back("(local.get " + std::to_string(indices_local) + ")");
            output.push_back("(" + std::string(extract_lane_op(type)) + " " + std::to_string(i) + ")");
            if (type == WasmType::i64x2 || type == WasmType::f64x2) {
                output.push_back("(i32.wrap_i64)");
            }
            output.push_back("(i32.const " + std::to_string(base_offset) + ")");
            output.push_back("(i32.add)");
            output.push_back("(" + std::string(scalar_load_op(type)) + ")");
        }

        output.push_back("(" + std::string(vector_make_op(type)) + ")");
        output.push_back("(local.set " + std::to_string(data_local) + ")");
    }

    static void emit_scatter(std::vector<std::string>& output,
                             uint32_t base_offset,
                             uint32_t indices_local,
                             uint32_t data_local,
                             WasmType type) noexcept {
        emit_scatter(output, base_offset, indices_local, data_local, type, Config{});
    }

    static void emit_scatter(std::vector<std::string>& output,
                             uint32_t base_offset,
                             uint32_t indices_local,
                             uint32_t data_local,
                             WasmType type,
                             const Config& config) noexcept {

        if (config.emit_comments) {
            output.push_back("(comment \"WASM scatter pattern\")");
        }

        const uint32_t lanes = lane_count(type);
        for (uint32_t i = 0; i < lanes; ++i) {
            output.push_back("(local.get " + std::to_string(indices_local) + ")");
            output.push_back("(" + std::string(extract_lane_op(type)) + " " + std::to_string(i) + ")");
            if (type == WasmType::i64x2 || type == WasmType::f64x2) {
                output.push_back("(i32.wrap_i64)");
            }
            output.push_back("(i32.const " + std::to_string(base_offset) + ")");
            output.push_back("(i32.add)");

            output.push_back("(local.get " + std::to_string(data_local) + ")");
            output.push_back("(" + std::string(extract_lane_op(type)) + " " + std::to_string(i) + ")");
            output.push_back("(" + std::string(scalar_store_op(type)) + ")");
        }
    }

    static void emit_fma(std::vector<std::string>& output,
                          uint32_t a_local,
                          uint32_t b_local,
                          uint32_t c_local,
                          uint32_t result_local,
                          WasmType type) noexcept {
        emit_fma(output, a_local, b_local, c_local, result_local, type, Config{});
    }

    static void emit_fma(std::vector<std::string>& output,
                          uint32_t a_local,
                          uint32_t b_local,
                          uint32_t c_local,
                          uint32_t result_local,
                          WasmType type,
                          const Config& config) noexcept {

        if (type == WasmType::i32x4) {
            if (config.emit_comments) {
                output.push_back("(comment \"WASM i32x4 FMA: a * b + c\")");
            }

            output.push_back("(local.get " + std::to_string(a_local) + ")");
            output.push_back("(local.get " + std::to_string(b_local) + ")");
            output.push_back("(i32x4.mul)");

            output.push_back("(local.get " + std::to_string(c_local) + ")");
            output.push_back("(i32x4.add)");

            output.push_back("(local.set " + std::to_string(result_local) + ")");
        } else if (type == WasmType::f32x4) {
            if (config.emit_comments) {
                output.push_back("(comment \"WASM f32x4 FMA: a * b + c\")");
            }

            output.push_back("(local.get " + std::to_string(a_local) + ")");
            output.push_back("(local.get " + std::to_string(b_local) + ")");
            output.push_back("(f32x4.mul)");

            output.push_back("(local.get " + std::to_string(c_local) + ")");
            output.push_back("(f32x4.add)");

            output.push_back("(local.set " + std::to_string(result_local) + ")");
        } else if (type == WasmType::f64x2) {
            if (config.emit_comments) {
                output.push_back("(comment \"WASM f64x2 FMA: a * b + c\")");
            }

            output.push_back("(local.get " + std::to_string(a_local) + ")");
            output.push_back("(local.get " + std::to_string(b_local) + ")");
            output.push_back("(f64x2.mul)");

            output.push_back("(local.get " + std::to_string(c_local) + ")");
            output.push_back("(f64x2.add)");

            output.push_back("(local.set " + std::to_string(result_local) + ")");
        } else {
            if (config.emit_comments) {
                output.push_back("(comment \"WASM FMA skipped for unsupported SIMD type\")");
            }
        }
    }

    static void emit_fma_optimized(std::vector<std::string>& output,
                                   uint32_t a_local,
                                   uint32_t b_local,
                                   uint32_t c_local,
                                   uint32_t result_local,
                                   WasmType type) noexcept {
        emit_fma_optimized(output, a_local, b_local, c_local, result_local, type, Config{});
    }

    static void emit_fma_optimized(std::vector<std::string>& output,
                                   uint32_t a_local,
                                   uint32_t b_local,
                                   uint32_t c_local,
                                   uint32_t result_local,
                                   WasmType type,
                                   const Config& config) noexcept {

        if (config.emit_comments) {
            output.push_back("(comment \"WASM FMA (runtime-fused by engine)\")");
        }

        output.push_back("(local.get " + std::to_string(a_local) + ")");
        output.push_back("(local.get " + std::to_string(b_local) + ")");

        if (type == WasmType::f32x4) {
            output.push_back("(f32x4.mul)");
        } else if (type == WasmType::f64x2) {
            output.push_back("(f64x2.mul)");
        } else if (type == WasmType::i32x4) {
            output.push_back("(i32x4.mul)");
        } else {
            if (config.emit_comments) {
                output.push_back("(comment \"WASM FMA optimized path skipped for unsupported SIMD type\")");
            }
            return;
        }

        output.push_back("(local.get " + std::to_string(c_local) + ")");

        if (type == WasmType::f32x4) {
            output.push_back("(f32x4.add)");
        } else if (type == WasmType::f64x2) {
            output.push_back("(f64x2.add)");
        } else if (type == WasmType::i32x4) {
            output.push_back("(i32x4.add)");
        }

        output.push_back("(local.set " + std::to_string(result_local) + ")");
    }

private:
    [[nodiscard]] static constexpr uint32_t lane_count(WasmType type) noexcept {
        return (type == WasmType::i64x2 || type == WasmType::f64x2) ? 2u : 4u;
    }

    [[nodiscard]] static constexpr const char* extract_lane_op(WasmType type) noexcept {
        switch (type) {
            case WasmType::i32x4: return "i32x4.extract_lane";
            case WasmType::i64x2: return "i64x2.extract_lane";
            case WasmType::f32x4: return "f32x4.extract_lane";
            case WasmType::f64x2: return "f64x2.extract_lane";
            default: return "i32x4.extract_lane";
        }
    }

    [[nodiscard]] static constexpr const char* scalar_load_op(WasmType type) noexcept {
        switch (type) {
            case WasmType::i32x4: return "i32.load";
            case WasmType::i64x2: return "i64.load";
            case WasmType::f32x4: return "f32.load";
            case WasmType::f64x2: return "f64.load";
            default: return "i32.load";
        }
    }

    [[nodiscard]] static constexpr const char* scalar_store_op(WasmType type) noexcept {
        switch (type) {
            case WasmType::i32x4: return "i32.store";
            case WasmType::i64x2: return "i64.store";
            case WasmType::f32x4: return "f32.store";
            case WasmType::f64x2: return "f64.store";
            default: return "i32.store";
        }
    }

    [[nodiscard]] static constexpr const char* vector_make_op(WasmType type) noexcept {
        switch (type) {
            case WasmType::i32x4: return "i32x4.make";
            case WasmType::i64x2: return "i64x2.make";
            case WasmType::f32x4: return "f32x4.make";
            case WasmType::f64x2: return "f64x2.make";
            default: return "i32x4.make";
        }
    }
};

// --- DGTL ASM EMITTER (With Sentinel Validation) ---
class DGTLAsmEmitter {
public:
    /**
     * @brief Translates MachineFunction to NASM Assembly String with Invariant Guard.
     */
    static void to_nasm(const dgtl::machine::MachineFunction& function, std::string& out) {
        verify_DGTL_invariants(function);

        out.clear();
        out += "; DGTL (Deep General-purpose Transformation Layer) MIR Generated Assembly [Validated]\n";
        out += "section .text\n    global _start\n    extern __dgtl_rt_print_i64\n_start:\n";

        for (const auto& block : function.blocks()) {
            out += ".Lbb" + std::to_string(block.id) + ":\n";

            for (dgtl::machine::InstrId instruction_id : block.instructions) {
                const auto* instruction = function.instruction(instruction_id);
                if (!instruction) continue;

                switch (instruction->form) {
                    case dgtl::machine::MachineForm::NoOp:
                        out += "    nop\n";
                        break;
                    case dgtl::machine::MachineForm::Move:
                        if (instruction->operands.size() >= 2 &&
                            instruction->operands[0].is_register() &&
                            instruction->operands[1].is_register()) {
                            out += "    mov r" + std::to_string(instruction->operands[0].reg.id % 16) +
                                   ", r" + std::to_string(instruction->operands[1].reg.id % 16) + "\n";
                        }
                        break;
                    case dgtl::machine::MachineForm::LoadImmediate:
                        if (!instruction->operands.empty() && instruction->operands[0].is_register()) {
                            int64_t imm = instruction->operands.size() > 1 && instruction->operands[1].kind == dgtl::machine::MachineOperandKind::Immediate
                                ? instruction->operands[1].immediate : 0;
                            out += "    mov r" + std::to_string(instruction->operands[0].reg.id % 16) +
                                   ", " + std::to_string(imm) + "\n";
                        }
                        break;
                    case dgtl::machine::MachineForm::IntegerBinary32:
                    case dgtl::machine::MachineForm::IntegerBinary64: {
                        if (instruction->operands.size() >= 3 && instruction->operands[0].is_register()) {
                            const char* op_str = "add";
                            if (instruction->semantic_kind == dgtl::machine::MachineSemanticKind::IntegerSub) op_str = "sub";
                            else if (instruction->semantic_kind == dgtl::machine::MachineSemanticKind::IntegerMul) op_str = "imul";
                            else if (instruction->semantic_kind == dgtl::machine::MachineSemanticKind::IntegerAnd) op_str = "and";
                            else if (instruction->semantic_kind == dgtl::machine::MachineSemanticKind::IntegerOr) op_str = "or";
                            else if (instruction->semantic_kind == dgtl::machine::MachineSemanticKind::IntegerXor) op_str = "xor";

                            uint32_t dst = instruction->operands[0].reg.id % 16;
                            std::string src = instruction->operands[2].is_register()
                                ? "r" + std::to_string(instruction->operands[2].reg.id % 16)
                                : std::to_string(instruction->operands[2].immediate);
                            out += "    " + std::string(op_str) + " r" + std::to_string(dst) + ", " + src + "\n";
                        }
                        break;
                    }
                    case dgtl::machine::MachineForm::Goto:
                        if (!instruction->operands.empty() && instruction->operands[0].kind == dgtl::machine::MachineOperandKind::BasicBlock) {
                            out += "    jmp .Lbb" + std::to_string(instruction->operands[0].block) + "\n";
                        }
                        break;
                    case dgtl::machine::MachineForm::BranchCond:
                        if (instruction->operands.size() >= 2 && instruction->operands[1].kind == dgtl::machine::MachineOperandKind::BasicBlock) {
                            if (instruction->operands[0].is_register()) {
                                out += "    test r" + std::to_string(instruction->operands[0].reg.id % 16) +
                                       ", r" + std::to_string(instruction->operands[0].reg.id % 16) + "\n";
                                out += "    jnz .Lbb" + std::to_string(instruction->operands[1].block) + "\n";
                            }
                        }
                        break;
                    case dgtl::machine::MachineForm::BranchCompare:
                        if (instruction->operands.size() >= 3 && instruction->operands[2].kind == dgtl::machine::MachineOperandKind::BasicBlock) {
                            std::string lhs = instruction->operands[0].is_register() ? "r" + std::to_string(instruction->operands[0].reg.id % 16) : "0";
                            std::string rhs = instruction->operands[1].is_register() ? "r" + std::to_string(instruction->operands[1].reg.id % 16) : "0";
                            out += "    cmp " + lhs + ", " + rhs + "\n";
                            out += "    jne .Lbb" + std::to_string(instruction->operands[2].block) + "\n";
                        }
                        break;
                    case dgtl::machine::MachineForm::Load:
                    case dgtl::machine::MachineForm::LoadStack:
                        if (instruction->operands.size() >= 2 && instruction->operands[0].is_register()) {
                            std::string base = instruction->operands[1].is_register() ? "r" + std::to_string(instruction->operands[1].reg.id % 16) : "rsp";
                            out += "    mov r" + std::to_string(instruction->operands[0].reg.id % 16) + ", [" + base + "]\n";
                        }
                        break;
                    case dgtl::machine::MachineForm::Store:
                    case dgtl::machine::MachineForm::StoreStack:
                        if (instruction->operands.size() >= 2 && instruction->operands[0].is_register()) {
                            std::string base = instruction->operands[1].is_register() ? "r" + std::to_string(instruction->operands[1].reg.id % 16) : "rsp";
                            out += "    mov [" + base + "], r" + std::to_string(instruction->operands[0].reg.id % 16) + "\n";
                        }
                        break;
                    case dgtl::machine::MachineForm::Call:
                        if (const char* sym = dgtl::mir::runtime_contract::runtime_symbol_for_opcode(instruction->opcode)) {
                            out += "    call " + std::string(sym) + "\n";
                        } else {
                            out += "    call unknown_target\n";
                        }
                        break;
                    case dgtl::machine::MachineForm::Syscall:
                        out += "    syscall\n";
                        break;
                    case dgtl::machine::MachineForm::Return:
                        out += "    ret\n";
                        break;
                    default:
                        break;
                }
            }
        }
    }




    /**
     * @brief High-speed Mach-O binary construction for ARM64.
     */
    static std::vector<uint8_t> emit_macho_binary(const std::vector<uint32_t>& machine_code) {
        std::vector<uint8_t> bin;
        mach_header_64 head{};
        
        head.magic = MH_MAGIC_64;
        head.cputype = CPU_TYPE_ARM64;
        head.cpusubtype = CPU_SUBTYPE_ARM64_ALL;
        head.filetype = MH_EXECUTE;
        head.ncmds = 0; 
        head.sizeofcmds = 0;
        head.flags = MH_NOUNDEFS | MH_DYLDLINK | MH_TWOLEVEL | MH_PIE;
        head.reserved = 0;
        
        bin.resize(sizeof(mach_header_64));
        std::memcpy(bin.data(), &head, sizeof(head));
        
        // Append raw machine code (simplified emission for stub validation)
        size_t code_bytes = machine_code.size() * sizeof(uint32_t);
        size_t prev_size = bin.size();
        bin.resize(prev_size + code_bytes);
        std::memcpy(bin.data() + prev_size, machine_code.data(), code_bytes);
        
        return bin;
    }

private:
    /**
     * @brief Verifies MachineFunction invariants at target boundary.
     */
    static void verify_DGTL_invariants(const dgtl::machine::MachineFunction& function) {
        const auto verify = dgtl::machine::MachineVerifier::verify(function);
        if (!verify.ok()) {
            throw std::runtime_error("DGTL Invariant Error: Machine IR corruption detected at target boundary");
        }
    }

};
