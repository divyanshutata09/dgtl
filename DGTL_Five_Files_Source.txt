    inline void record_fallback_opcode(size_t node_index, dgtl::linker::mir::MachineLinkOpcode opcode) {
        std::string msg;
        msg.reserve(48);
        msg.append("node#").append(std::to_string(node_index))

    void reset_mapping_state() noexcept {
        output_ptr = nullptr;
        fd_handle = -1;
        file_size = 0;
    }

    void release_mapping() noexcept {
        if (!output_ptr || output_ptr == DGTL_MAP_FAILED) {
            output_ptr = nullptr;
            if (fd_handle != -1) {
#if defined(_WIN32)
                CloseHandle(reinterpret_cast<HANDLE>(fd_handle));
#else
                close(static_cast<int>(fd_handle));
#endif
                fd_handle = -1;
            }
            return;
        }

#if defined(_WIN32)
        UnmapViewOfFile(output_ptr);
        if (fd_handle != -1) CloseHandle(reinterpret_cast<HANDLE>(fd_handle));
#else
        munmap(output_ptr, file_size);
        if (fd_handle != -1) close(static_cast<int>(fd_handle));
#endif

        output_ptr = nullptr;
        fd_handle = -1;
        file_size = 0;
    }

    static inline void write_leb128(std::vector<uint8_t>& buf, uint64_t value) {
        do {
            uint8_t byte = value & 0x7F;
            value >>= 7;
            if (value != 0) byte |= 0x80;
            buf.push_back(byte);
        } while (value != 0);
    }

    [[nodiscard]] static constexpr size_t align_up(size_t value, size_t alignment) noexcept {
        if (alignment <= 1) [[unlikely]] {
            return value;
        }
        const size_t rem = value % alignment;
        return rem == 0 ? value : (value + (alignment - rem));
    }

    [[nodiscard]] static bool write_binary_file(std::string_view path, const std::vector<uint8_t>& bytes) {
        if (path.empty()) [[unlikely]] return false;
        std::ofstream out(path.data(), std::ios::out | std::ios::binary | std::ios::trunc);
        if (!out) [[unlikely]] return false;
        
        if (!bytes.empty()) {
            out.write(reinterpret_cast<const char*>(bytes.data()), static_cast<std::streamsize>(bytes.size()));
        }
        
        out.close();
        return out.good();
    }

    [[nodiscard]] static constexpr uint32_t to_big_endian_u32(uint32_t val) noexcept {
#if defined(_MSC_VER)
        return _byteswap_ulong(val);
#elif defined(__GNUC__) || defined(__clang__)
        return __builtin_bswap32(val);
#else
        return ((val & 0xFF000000u) >> 24) |
               ((val & 0x00FF0000u) >> 8)  |
               ((val & 0x0000FF00u) << 8)  |
               ((val & 0x000000FFu) << 24);
#endif
    }

    [[nodiscard]] static constexpr size_t x64_node_size(const dgtl::linker::mir::MachineLinkNode& node) noexcept {
        using namespace dgtl::isa::lowering;
        const MachineForm form = node.form;

        if (node.has_runtime_symbol()) {
            const size_t arg1_size = (node.op_b != dgtl::linker::mir::INVALID_SLOT) ? 7 : 2;
            const size_t arg2_size = (node.op_c != dgtl::linker::mir::INVALID_SLOT) ? 7 : 2;
            const size_t arg3_size = (node.logic_flags != dgtl::linker::mir::INVALID_SLOT && node.logic_flags != 0u) ? 7 : 3;
            const size_t result_size = (node.result_id != dgtl::linker::mir::INVALID_SLOT) ? 7 : 1;
            return arg1_size + arg2_size + arg3_size + 10 + 4 + 5 + 4 + result_size;
        }

        switch (form) {
            case MachineForm::NoOp:
                if (node.opcode == dgtl::linker::mir::MachineLinkOpcode::LABEL) {
                    size_t lbl_bytes = 0;
                    if (node.has_side_effects() || node.is_volatile()) {
                        if (node.op_a != dgtl::linker::mir::INVALID_SLOT) lbl_bytes += 7;
                        if (node.op_b != dgtl::linker::mir::INVALID_SLOT) lbl_bytes += 7;
                        if (node.op_c != dgtl::linker::mir::INVALID_SLOT) lbl_bytes += 7;
                        if (node.logic_flags != dgtl::linker::mir::INVALID_SLOT && node.logic_flags != 0u) lbl_bytes += 7;
                    }
                    return lbl_bytes;
                }
                return 1;
            case MachineForm::Move:
            case MachineForm::LoadStack:
            case MachineForm::StoreStack:
                return 14;
            case MachineForm::LoadImmediate: {
                const bool is_float_const = (node.opcode == dgtl::linker::mir::MachineLinkOpcode::CONST && node.form == dgtl::mir::MachineForm::LoadImmediate);
                return is_float_const ? 23 : 17;
            }
            case MachineForm::IntegerBinary32:
            case MachineForm::IntegerBinary64: {
                const bool is64 = (form == MachineForm::IntegerBinary64);
                const auto kind = is64 ? static_cast<dgtl::mir::MachineSemanticKind>(node.semantic_kind)
                                       : node.semantic_kind;
                if (kind == dgtl::mir::MachineSemanticKind::IntegerShl || kind == dgtl::mir::MachineSemanticKind::IntegerShr) {
                    return is64 ? 24 : 23;
                }
                if (kind == dgtl::mir::MachineSemanticKind::IntegerMul) {
                    return is64 ? 22 : 21;
                }
                if (kind == dgtl::mir::MachineSemanticKind::IntegerDiv || kind == dgtl::mir::MachineSemanticKind::IntegerMod) {
                    return is64 ? 80 : 40;
                }
                return is64 ? 21 : 20;
            }
            case MachineForm::FloatBinary32:
            case MachineForm::FloatBinary64:
                return 24;
            case MachineForm::FloatTernaryFma32:
                return 32;
            case MachineForm::VectorBinary512:
                return 28;
            case MachineForm::VectorStore512:
                return 22;
            case MachineForm::VectorGather512:
                return 32;
            case MachineForm::Compare32:
                return 27;
            case MachineForm::Call: {
                const size_t arg1_size = (node.op_b != dgtl::linker::mir::INVALID_SLOT) ? 7 : 3;
                const size_t arg2_size = (node.op_c != dgtl::linker::mir::INVALID_SLOT) ? 7 : 3;
                const size_t arg3_size = (node.logic_flags != dgtl::linker::mir::INVALID_SLOT && node.logic_flags != 0u) ? 7 : 3;
                const size_t result_size = (node.result_id != dgtl::linker::mir::INVALID_SLOT) ? 24 : 1;
                return arg1_size + arg2_size + arg3_size + 24 + 4 + 10 + 4 + result_size + 32;
            }
            case MachineForm::Return:
                return (node.op_a != dgtl::linker::mir::INVALID_SLOT) ? 14 : 8;
            case MachineForm::Goto:
                return 5;
            case MachineForm::BranchCond:
                return 16;
            case MachineForm::BranchCompare:
                return 20;

            }
            default:
                return 1;
        }

    [[nodiscard]] static bool build_wasm_binary(const dgtl::linker::mir::MachineLinkStream& stream,
                                                 std::vector<uint8_t>& out) {
        out.clear();
        if (stream.is_null() || stream.size() == 0 || !stream.validate_shadow_integrity()) [[unlikely]] {
            return false;
        }
        // The WASM encoder is intentionally MIR-only. Lower the IR stream through
        // the dedicated WASM MIR boundary first, then encode the resulting
        // MachineFunction. This keeps the linker from bypassing the canonical
        // target-aware ISel contract.
        core::TargetMachine target(core::ArchType::WASM_SIMD);
        auto selector_lines = dgtl::isa::rule_lines(target);
        dgtl::mir::isel::InstructionSelectorProgram selector =
            dgtl::mir::isel::InstructionSelectorProgram::from_rule_lines(selector_lines);

        dgtl::machine::MachineFunction function("DGTL_main");
        std::string diagnostic;
        if (!dgtl::isa::lowering::WasmMIRBoundary::build(
                stream, target, selector, function, diagnostic)) {
            return false;
        }

        dgtl::exec::DGTLWasmEncoder encoder;
        encoder.encode(function);
        out = encoder.get_binary();
        return !out.empty();
    }

    [[nodiscard]] static bool inspect_elf_object_from_memory(
        const void* data,
        size_t size,
        dgtl::execution::format::Object_Format& out_object) {
        if (data == nullptr || size == 0) {
            return false;
        }
        try {
            dgtl::execution::format::Object_Loader loader;
            auto obj = loader.load_from_memory(data, size);
            if (!obj) {
                return false;
            }

            for (const auto& s : obj->get_symbols()) {
                if (s.is_extern) {
                    out_object.add_extern_symbol(s.name, s.binding);
                } else {
                    out_object.add_symbol(s.name, s.address, s.size, s.binding, s.type);
                }
            }
            for (const auto& r : obj->get_relocations()) {
                out_object.add_relocation(r.offset, r.symbol_name, r.addend, r.type);
            }
            out_object.get_text_section() = obj->get_text_section();
            out_object.get_data_section() = obj->get_data_section();
            out_object.get_rodata_section() = obj->get_rodata_section();
            out_object.get_bss_section() = obj->get_bss_section();
            return true;
        } catch (...) {
            return false;
        }
    }

    [[nodiscard]] static bool inspect_binary_format_from_memory(
        const void* data,
        size_t size,
        dgtl::execution::format::Binary_Format& out_binary) {
        if (data == nullptr || size == 0) return false;
        try {
            dgtl::execution::format::Binary_Loader loader;
            auto bin = loader.load_from_memory(data, size);
            if (!bin) return false;
            out_binary = *bin;
            return true;
        } catch (...) {
            return false;
        }
    }

    [[nodiscard]] static bool build_cached_machine_code(
        dgtl::execution::cache::Code_Cache& cache,
        std::string_view key,
        const dgtl::linker::mir::MachineLinkStream& stream,
        std::vector<uint8_t>& out_bytes) {
        if (!key.empty() && cache.retrieve_copy(key, out_bytes)) {
            return true;
        }
        DGTLLinker linker;
        if (!linker.build_raw_x64_machine_code(stream, out_bytes)) {
            return false;
        }
        if (!key.empty()) {
            (void)cache.store(key, out_bytes.data(), out_bytes.size());
        }
        return true;
    }

    bool build_pe_executable(std::string_view path, const dgtl::linker::mir::MachineLinkStream& ir_stream) {

    bool build_raw_x64_machine_code(const dgtl::linker::mir::MachineLinkStream& stream,
                                    std::vector<uint8_t>& out_bytes,
                                    const std::unordered_map<uint32_t, uintptr_t>& slot_map = {}) {
        out_bytes.resize(stream.size() * 64 + 1024);
        size_t emitted_size = 0;
        if (!emit_machine_code(out_bytes.data(), out_bytes.size(), stream, &emitted_size, slot_map)) {
            return false;
        }
        out_bytes.resize(emitted_size);
        return true;
    }

    bool emit_machine_code(uint8_t* dest,
                           size_t dest_capacity,
                           const dgtl::linker::mir::MachineLinkStream& stream,
                           size_t* emitted_size = nullptr,
                           const std::unordered_map<uint32_t, uintptr_t>& slot_map = {},
                           std::vector<CoffCallReloc>* out_relocs = nullptr,
                           const std::unordered_map<uint32_t, std::string>& slot_to_callee = {}) noexcept {
        if (stream.is_null() || !dest || stream.size() == 0) [[unlikely]] return false;
        clear_fallback_diagnostics();
        const size_t count = stream.size();

        std::unordered_map<uint32_t, uint64_t> known_constants;
        known_constants.reserve(count);
        for (size_t i = 0; i < count; ++i) {
            const auto& node = stream.get_node(i);
            if (node.opcode == dgtl::linker::mir::MachineLinkOpcode::LOAD_IMM || node.opcode == dgtl::linker::mir::MachineLinkOpcode::CONST || node.opcode == dgtl::linker::mir::MachineLinkOpcode::IMMEDIATE) {
                uint64_t imm = (static_cast<uint64_t>(node.op_b) << 32u) | (static_cast<uint64_t>(node.op_a) & 0xFFFFFFFFu);
                if (imm == 0 && node.intent_hash != 0) {
                    imm = node.intent_hash;
                }
                if (node.result_id != dgtl::linker::mir::INVALID_SLOT) {
                    known_constants[node.result_id] = imm;
                }
            }
        }

        std::unordered_map<uint32_t, size_t> label_map;
        label_map.reserve(count / 4 + 4);

        std::unordered_map<uint32_t, int32_t> slot_offsets;
        slot_offsets.reserve(count * 2 + 8);
        uint32_t next_slot_index = 0;
        auto assign_slot = [&](uint32_t slot_id) {
            if (slot_id == dgtl::linker::mir::INVALID_SLOT) return;
            if (slot_offsets.find(slot_id) != slot_offsets.end()) return;
            slot_offsets.emplace(slot_id, -8 * static_cast<int32_t>(++next_slot_index));
        };

        assign_slot(2u);
        assign_slot(3u);

        for (size_t i = 0; i < count; ++i) {
            const auto& node = stream.get_node(i);
            if (node.opcode == dgtl::linker::mir::MachineLinkOpcode::LABEL) {
                if (node.result_id != dgtl::linker::mir::INVALID_SLOT && node.result_id != 0u) {
                    label_map[node.result_id] = i;
                }
                const bool is_func_header = node.has_side_effects() || node.is_volatile();
                if (!is_func_header) {
                    if (node.op_a != dgtl::linker::mir::INVALID_SLOT && node.op_a != 0u) {
                        label_map[node.op_a] = i;
                    }
                    if (node.op_b != dgtl::linker::mir::INVALID_SLOT && node.op_b != 0u) {
                        label_map[node.op_b] = i;
                    }
                }
                continue;
            }

            assign_slot(node.result_id);
            assign_slot(node.op_a);
            assign_slot(node.op_b);
            assign_slot(node.op_c);
            if (node.logic_flags != 0 && node.logic_flags != dgtl::linker::mir::INVALID_SLOT) {
                assign_slot(node.logic_flags);
            }
        }

        auto resolve_target = [&](uint32_t raw) -> size_t {
            auto it = label_map.find(raw);
            if (it != label_map.end()) return it->second;
            for (size_t k = 0; k < count; ++k) {
                const auto& knode = stream.get_node(k);
                if (knode.opcode == dgtl::linker::mir::MachineLinkOpcode::LABEL) {
                    if (knode.result_id == raw || knode.op_a == raw || knode.op_b == raw) {
                        return k;
                    }
                }
            }
            return count;
        };

        const uint32_t raw_frame_size = (next_slot_index + 2u) * 8u;
        const uint32_t frame_size = (raw_frame_size + 15u) & ~15u;

        std::vector<size_t> node_offsets(count + 1, 0);
        size_t probe = x64_prologue_size(stream);
        for (size_t i = 0; i < count; ++i) {
            node_offsets[i] = probe;
            const auto& node = stream.get_node(i);
            probe += x64_node_size(node);
        }
        node_offsets[count] = probe;

        if (probe > dest_capacity) [[unlikely]] return false;

        struct BranchPatch { size_t disp_offset; size_t source_end; size_t target_node; };
        std::vector<BranchPatch> patches;

        const int32_t fallback_offset = -8 * static_cast<int32_t>(next_slot_index + 1u);
        auto slot_disp = [&](uint32_t slot_id) -> int32_t {
            if (slot_id == dgtl::linker::mir::INVALID_SLOT) return fallback_offset;
            const auto it = slot_offsets.find(slot_id);
            return (it != slot_offsets.end()) ? it->second : fallback_offset;
        };
        patches.reserve(count / 8 + 4);

        size_t offset = 0;
        auto write_u8 = [&](uint8_t byte) { dest[offset++] = byte; };
        auto write_u32 = [&](uint32_t value) {
            std::memcpy(dest + offset, &value, sizeof(value));
            offset += sizeof(value);
        };
        auto write_i32 = [&](int32_t value) {
            std::memcpy(dest + offset, &value, sizeof(value));
            offset += sizeof(value);
        };
        auto write_u64 = [&](uint64_t value) {
            std::memcpy(dest + offset, &value, sizeof(value));
            offset += sizeof(value);
        };
        auto emit_mov_reg_from_slot = [&](uint8_t rex, uint8_t modrm, uint32_t slot_id) {
            write_u8(rex);
            write_u8(0x8B);
            write_u8(modrm);
            write_i32(slot_disp(slot_id));
        };
        auto emit_mov_slot_from_reg = [&](uint8_t rex, uint8_t modrm, uint32_t slot_id) {
            write_u8(rex);
            write_u8(0x89);
            write_u8(modrm);
            write_i32(slot_disp(slot_id));
        };
        auto emit_mov_rax_from_slot = [&](uint32_t slot_id) { emit_mov_reg_from_slot(0x48, 0x85, slot_id); };
        auto emit_mov_rcx_from_slot = [&](uint32_t slot_id) { emit_mov_reg_from_slot(0x48, 0x8D, slot_id); };
        auto emit_mov_rdx_from_slot = [&](uint32_t slot_id) { emit_mov_reg_from_slot(0x48, 0x95, slot_id); };
        auto emit_mov_r8_from_slot  = [&](uint32_t slot_id) { emit_mov_reg_from_slot(0x4C, 0x85, slot_id); };
        auto emit_mov_r9_from_slot  = [&](uint32_t slot_id) { emit_mov_reg_from_slot(0x4C, 0x8D, slot_id); };
        auto emit_store_rax_to_slot = [&](uint32_t slot_id) { emit_mov_slot_from_reg(0x48, 0x85, slot_id); };
        auto emit_store_rdx_to_slot = [&](uint32_t slot_id) { emit_mov_slot_from_reg(0x48, 0x95, slot_id); };
        auto emit_store_rcx_to_slot = [&](uint32_t slot_id) { emit_mov_slot_from_reg(0x48, 0x8D, slot_id); };
        auto emit_store_r8_to_slot  = [&](uint32_t slot_id) { emit_mov_slot_from_reg(0x4C, 0x85, slot_id); };
        auto emit_store_r9_to_slot  = [&](uint32_t slot_id) { emit_mov_slot_from_reg(0x4C, 0x8D, slot_id); };

        write_u8(0x55);
        write_u8(0x48); write_u8(0x89); write_u8(0xE5);
        write_u8(0x48); write_u8(0x81); write_u8(0xEC); write_u32(frame_size);
        if (count > 0 && stream.get_node(0).opcode == dgtl::linker::mir::MachineLinkOpcode::LABEL) {
            const auto& fn_hdr = stream.get_node(0);
            if (fn_hdr.op_a != dgtl::linker::mir::INVALID_SLOT) emit_store_rcx_to_slot(fn_hdr.op_a);
            if (fn_hdr.op_b != dgtl::linker::mir::INVALID_SLOT) emit_store_rdx_to_slot(fn_hdr.op_b);
            if (fn_hdr.op_c != dgtl::linker::mir::INVALID_SLOT) emit_store_r8_to_slot(fn_hdr.op_c);
            if (fn_hdr.logic_flags != dgtl::linker::mir::INVALID_SLOT && fn_hdr.logic_flags != 0u) emit_store_r9_to_slot(fn_hdr.logic_flags);
            const auto fn_overflow = stream.get_call_overflow_args(0);
            for (size_t k = 0; k < fn_overflow.size(); ++k) {
                if (fn_overflow[k] != dgtl::linker::mir::INVALID_SLOT && fn_overflow[k] != 0u) {
                    write_u8(0x48); write_u8(0x8B); write_u8(0x85); write_i32(static_cast<int32_t>(48 + k * 8)); // mov rax, [rbp + 48 + k*8]
                    emit_store_rax_to_slot(fn_overflow[k]);
                }
            }
        }

        for (size_t i = 0; i < count; ++i) {
            node_offsets[i] = offset;
            const auto& node = stream.get_node(i);
            using namespace dgtl::isa::lowering;
            const MachineForm form = node.form;

            if (node.has_runtime_symbol()) {
                if (node.op_a != dgtl::linker::mir::INVALID_SLOT && node.op_a != 0u) emit_mov_rcx_from_slot(node.op_a);
                else { write_u8(0x48); write_u8(0x31); write_u8(0xC9); } // xor rcx, rcx
                if (node.op_b != dgtl::linker::mir::INVALID_SLOT && node.op_b != 0u) emit_mov_rdx_from_slot(node.op_b);
                else { write_u8(0x48); write_u8(0x31); write_u8(0xD2); } // xor rdx, rdx
                if (node.op_c != dgtl::linker::mir::INVALID_SLOT && node.op_c != 0u) emit_mov_r8_from_slot(node.op_c);
                else { write_u8(0x4D); write_u8(0x31); write_u8(0xC0); } // xor r8, r8
                if (node.logic_flags != dgtl::linker::mir::INVALID_SLOT && node.logic_flags != 0u) {
                    emit_mov_reg_from_slot(0x4C, 0x8D, node.logic_flags); // mov r9, [rbp-disp]
                }
                else { write_u8(0x4D); write_u8(0x31); write_u8(0xC9); } // xor r9, r9

                write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xC1); // movq xmm0, rcx
                write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xCA); // movq xmm1, rdx

                write_u8(0x48); write_u8(0x83); write_u8(0xEC); write_u8(0x20); // sub rsp, 32

                uintptr_t rt_target_addr = 0;
                if (out_relocs == nullptr) {
                    if (!slot_map.empty()) {
                        auto it = slot_map.find(static_cast<uint32_t>(node.opcode));
                        if (it != slot_map.end()) {
                            rt_target_addr = it->second;
                        }
                    }
                    if (rt_target_addr == 0 && node.intent_hash > 0x7FFF00000000ULL) {
                        rt_target_addr = static_cast<uintptr_t>(node.intent_hash);
                    }
                }

                if (rt_target_addr != 0) {
                    write_u8(0x48); write_u8(0xB8); write_u64(static_cast<uint64_t>(rt_target_addr));
                    write_u8(0xFF); write_u8(0xD0);
                } else {
                    write_u8(0xE8);
                    const size_t reloc_disp_offset = offset;
                    write_u8(0x00); write_u8(0x00); write_u8(0x00); write_u8(0x00);
                    if (out_relocs != nullptr) {
                        const char* symbol = node.runtime_symbol();
                        if (symbol != nullptr) {
                            out_relocs->push_back({ static_cast<uint32_t>(reloc_disp_offset), std::string(symbol) });
                        }
                    }
                }
                write_u8(0x48); write_u8(0x83); write_u8(0xC4); write_u8(0x20); // add rsp, 32

                if (node.result_id != dgtl::linker::mir::INVALID_SLOT) emit_store_rax_to_slot(node.result_id);
                else write_u8(0x90);
                continue;
            }

            switch (form) {
                case MachineForm::NoOp:
                    if (node.opcode == dgtl::linker::mir::MachineLinkOpcode::LABEL) {
                        const bool is_func_header = node.has_side_effects() || node.is_volatile();
                        if (is_func_header && i > 0) {
                            if (node.op_a != dgtl::linker::mir::INVALID_SLOT) emit_store_rcx_to_slot(node.op_a);
                            if (node.op_b != dgtl::linker::mir::INVALID_SLOT) emit_store_rdx_to_slot(node.op_b);
                            if (node.op_c != dgtl::linker::mir::INVALID_SLOT) emit_store_r8_to_slot(node.op_c);
                            if (node.logic_flags != dgtl::linker::mir::INVALID_SLOT && node.logic_flags != 0u) emit_store_r9_to_slot(node.logic_flags);
                            const auto fn_overflow = stream.get_call_overflow_args(i);
                            for (size_t k = 0; k < fn_overflow.size(); ++k) {
                                if (fn_overflow[k] != dgtl::linker::mir::INVALID_SLOT && fn_overflow[k] != 0u) {
                                    write_u8(0x48); write_u8(0x8B); write_u8(0x85);
                                    emit_store_rax_to_slot(fn_overflow[k]);
                                }
                            }
                        }
                    } else {
                        write_u8(0x90);
                    }
                    break;

                case MachineForm::Move:
                case MachineForm::LoadStack:
                case MachineForm::StoreStack: {
                    if (form == MachineForm::StoreStack) {
                        emit_mov_rax_from_slot(node.op_b);
                        emit_store_rax_to_slot(node.op_a);
                    } else if (node.result_id == dgtl::dgtl::linker::mir::INVALID_SLOT || node.result_id == 0xFFFFFFFFu) {
                        // Side-effecting register copy (REG_MOVE): op_a = dst, op_b = src
                        emit_mov_rax_from_slot(node.op_b);
                        emit_store_rax_to_slot(node.op_a);
                    } else {
                        emit_mov_rax_from_slot(node.op_a);
                        emit_store_rax_to_slot(node.result_id);
                    }
                    break;
                }

                case MachineForm::LoadImmediate: {
                    uint64_t imm = (static_cast<uint64_t>(node.op_b) << 32u) | (static_cast<uint64_t>(node.op_a) & 0xFFFFFFFFu);
                    if (imm == 0 && node.intent_hash != 0) {
                        imm = node.intent_hash;
                    }

                    const bool is_float_const = (node.opcode == dgtl::linker::mir::MachineLinkOpcode::CONST && node.form == dgtl::mir::MachineForm::LoadImmediate);

                    if (is_float_const) {
                        write_u8(0x48); write_u8(0xB8); write_u64(imm); // movabs rax, imm64
                        write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xC0); // movq xmm0, rax
                        write_u8(0xF2); write_u8(0x0F); write_u8(0x11); write_u8(0x85); // movsd [rbp-disp], xmm0
                        write_i32(slot_disp(node.result_id));
                    } else {
                        write_u8(0x48); write_u8(0xB8); write_u64(imm); // movabs rax, imm64
                        emit_store_rax_to_slot(node.result_id);
                    }
                    break;
                }

                case MachineForm::IntegerBinary32:
                case MachineForm::IntegerBinary64: {
                    const bool is64 = (form == MachineForm::IntegerBinary64);
                    dgtl::mir::MachineSemanticKind kind = dgtl::mir::MachineSemanticKind::None;
                    if (is64) {
                        switch (node.semantic_kind) {
                            case dgtl::mir::MachineSemanticKind::IntegerAdd: kind = dgtl::mir::MachineSemanticKind::IntegerAdd; break;
                            case dgtl::mir::MachineSemanticKind::IntegerSub: kind = dgtl::mir::MachineSemanticKind::IntegerSub; break;
                            case dgtl::mir::MachineSemanticKind::IntegerMul: kind = dgtl::mir::MachineSemanticKind::IntegerMul; break;
                            case dgtl::mir::MachineSemanticKind::IntegerDiv: kind = dgtl::mir::MachineSemanticKind::IntegerDiv; break;
                            case dgtl::mir::MachineSemanticKind::IntegerMod: kind = dgtl::mir::MachineSemanticKind::IntegerMod; break;
                            case dgtl::mir::MachineSemanticKind::IntegerAnd: kind = dgtl::mir::MachineSemanticKind::IntegerAnd; break;
                            case dgtl::mir::MachineSemanticKind::IntegerOr:  kind = dgtl::mir::MachineSemanticKind::IntegerOr;  break;
                            case dgtl::mir::MachineSemanticKind::IntegerXor: kind = dgtl::mir::MachineSemanticKind::IntegerXor; break;
                            case dgtl::mir::MachineSemanticKind::IntegerShl: kind = dgtl::mir::MachineSemanticKind::IntegerShl; break;
                            case dgtl::mir::MachineSemanticKind::IntegerShr: kind = dgtl::mir::MachineSemanticKind::IntegerShr; break;
                            default: break;
                        }
                    } else {
                        kind = node.semantic_kind;
                    }
                    if (kind == dgtl::mir::MachineSemanticKind::IntegerShl || 
                        kind == dgtl::mir::MachineSemanticKind::IntegerShr) {
                        emit_mov_rax_from_slot(node.op_a);
                        emit_mov_rcx_from_slot(node.op_b);
                        if (is64) write_u8(0x48);
                        write_u8(0xD3);
                        write_u8(kind == dgtl::mir::MachineSemanticKind::IntegerShl ? 0xE0 : 0xF8);
                        emit_store_rax_to_slot(node.result_id);
                    }
                    else if (kind == dgtl::mir::MachineSemanticKind::IntegerDiv || 
                             kind == dgtl::mir::MachineSemanticKind::IntegerMod) {
                        auto const_it = (node.op_b != dgtl::linker::mir::INVALID_SLOT) ? known_constants.find(node.op_b) : known_constants.end();
                        bool has_const_divisor = (const_it != known_constants.end() && const_it->second > 0);
                        int64_t divisor = has_const_divisor ? static_cast<int64_t>(const_it->second) : 0;
                        auto magic = (has_const_divisor && is64) ? dgtl::linker::driver::compute_magic_s64(divisor) : dgtl::linker::driver::MagicMultiplierS64{};

                        if (is64 && has_const_divisor && divisor == 1) {
                            if (kind == dgtl::mir::MachineSemanticKind::IntegerDiv) {
                                emit_mov_rax_from_slot(node.op_a);
                                emit_store_rax_to_slot(node.result_id);
                            } else {
                                write_u8(0x48); write_u8(0x31); write_u8(0xC0); // xor rax, rax
                                emit_store_rax_to_slot(node.result_id);
                            }
                        }
                        else if (is64 && has_const_divisor && magic.valid && !magic.is_power_of_2) {
                            emit_mov_rax_from_slot(node.op_a);
                            write_u8(0x48); write_u8(0xB9); write_u64(magic.M);
                            write_u8(0x48); write_u8(0xF7); write_u8(0xE9);
                            if (magic.s > 0) {
                                write_u8(0x48); write_u8(0xC1); write_u8(0xFA); write_u8(magic.s);
                            }
                            emit_mov_rax_from_slot(node.op_a);
                            write_u8(0x48); write_u8(0xC1); write_u8(0xF8); write_u8(0x3F);
                            write_u8(0x48); write_u8(0x29); write_u8(0xC2);
                            if (kind == dgtl::mir::MachineSemanticKind::IntegerDiv) {
                                emit_store_rdx_to_slot(node.result_id);
                            } else {
                                write_u8(0x48); write_u8(0x89); write_u8(0xD0);
                                write_u8(0x48); write_u8(0xB9); write_u64(static_cast<uint64_t>(divisor));
                                write_u8(0x48); write_u8(0x0F); write_u8(0xAF); write_u8(0xC1);
                                emit_mov_rdx_from_slot(node.op_a);
                                write_u8(0x48); write_u8(0x29); write_u8(0xC2);
                                emit_store_rdx_to_slot(node.result_id);
                            }
                        }
                        else if (is64 && has_const_divisor && magic.is_power_of_2 && divisor > 1) {
                            emit_mov_rax_from_slot(node.op_a);
                            write_u8(0x48); write_u8(0x99);
                            uint8_t k = magic.s;
                            if (k == 1) {
                                write_u8(0x48); write_u8(0xC1); write_u8(0xEA); write_u8(0x3F);
                            } else if (k < 31) {
                                write_u8(0x48); write_u8(0x81); write_u8(0xE2); write_u32(static_cast<uint32_t>((1ULL << k) - 1));
                            } else {
                                write_u8(0x48); write_u8(0xB9); write_u64((1ULL << k) - 1);
                                write_u8(0x48); write_u8(0x21); write_u8(0xCA);
                            }
                            write_u8(0x48); write_u8(0x01); write_u8(0xD0);
                            write_u8(0x48); write_u8(0xC1); write_u8(0xF8); write_u8(k);
                            if (kind == dgtl::mir::MachineSemanticKind::IntegerDiv) {
                                emit_store_rax_to_slot(node.result_id);
                            } else {
                                write_u8(0x48); write_u8(0xC1); write_u8(0xE0); write_u8(k);
                                emit_mov_rdx_from_slot(node.op_a);
                                write_u8(0x48); write_u8(0x29); write_u8(0xC2);
                                emit_store_rdx_to_slot(node.result_id);
                            }
                        }
                        else {
                            emit_mov_rax_from_slot(node.op_a);
                            if (is64) {
                                write_u8(0x48); write_u8(0x99);
                                write_u8(0x48); write_u8(0xF7); write_u8(0xBD);
                            } else {
                                write_u8(0x99);
                                write_u8(0xF7); write_u8(0xBD);
                            }
                            write_i32(slot_disp(node.op_b));
                            if (kind == dgtl::mir::MachineSemanticKind::IntegerDiv)
                                emit_store_rax_to_slot(node.result_id);
                            else emit_store_rdx_to_slot(node.result_id);
                        }
                    }
                    else {
                        emit_mov_rax_from_slot(node.op_a);
                        if (is64) {
                            write_u8(0x48);
                        }
                        if (kind == dgtl::mir::MachineSemanticKind::IntegerAdd) {
                            write_u8(0x03); write_u8(0x85);
                        } else if (kind == dgtl::mir::MachineSemanticKind::IntegerSub) {
                            write_u8(0x2B); write_u8(0x85);
                        } else if (kind == dgtl::mir::MachineSemanticKind::IntegerAnd) {
                            write_u8(0x23); write_u8(0x85);
                        } else if (kind == dgtl::mir::MachineSemanticKind::IntegerOr) {
                            write_u8(0x0B); write_u8(0x85);
                        } else if (kind == dgtl::mir::MachineSemanticKind::IntegerXor) {
                            write_u8(0x33); write_u8(0x85);
                        } else if (kind == dgtl::mir::MachineSemanticKind::IntegerMul) {
                            write_u8(0x0F); write_u8(0xAF); write_u8(0x85);
                        }
                        write_i32(slot_disp(node.op_b));
                        emit_store_rax_to_slot(node.result_id);
                    }
                    break;
                }

                case MachineForm::FloatBinary32:
                case MachineForm::FloatBinary64: {
                    const bool is64 = (form == MachineForm::FloatBinary64);
                    write_u8(is64 ? 0xF2 : 0xF3);
                    write_u8(0x0F); write_u8(0x10); write_u8(0x85);
                    write_i32(slot_disp(node.op_a));

                    write_u8(is64 ? 0xF2 : 0xF3);
                    write_u8(0x0F);
                    if (is64) {
                        const auto kind = node.semantic_kind;
                        using K = dgtl::mir::MachineSemanticKind;
                        if (kind == K::FloatAdd) write_u8(0x58);
                        else if (kind == K::FloatSub) write_u8(0x5C);
                        else if (kind == K::FloatMul) write_u8(0x59);
                        else if (kind == K::FloatDiv) write_u8(0x5E);
                        else write_u8(0x58);
                    } else {
                        const auto kind = node.semantic_kind;
                        using K = dgtl::mir::MachineSemanticKind;
                        if (kind == K::FloatAdd) write_u8(0x58);
                        else if (kind == K::FloatSub) write_u8(0x5C);
                        else if (kind == K::FloatMul) write_u8(0x59);
                        else if (kind == K::FloatDiv) write_u8(0x5E);
                        else write_u8(0x58);
                    }
                    write_u8(0x85);
                    write_i32(slot_disp(node.op_b));

                    write_u8(is64 ? 0xF2 : 0xF3);
                    write_u8(0x0F); write_u8(0x11); write_u8(0x85);
                    write_i32(slot_disp(node.result_id));
                    break;
                }

                case MachineForm::FloatTernaryFma32: {
                    write_u8(0xF3); write_u8(0x0F); write_u8(0x10); write_u8(0x85); write_i32(slot_disp(node.op_a));
                    write_u8(0xF3); write_u8(0x0F); write_u8(0x10); write_u8(0x8D); write_i32(slot_disp(node.op_b));
                    write_u8(0xC4); write_u8(0xE2); write_u8(0x79); write_u8(0x98); write_u8(0x85); write_i32(slot_disp(node.op_c));
                    write_u8(0xF3); write_u8(0x0F); write_u8(0x11); write_u8(0x85); write_i32(slot_disp(node.result_id));
                    break;
                }

                case MachineForm::VectorBinary512: {
                    write_u8(0x62); write_u8(0xF1); write_u8(0x7C); write_u8(0x48);
                    write_u8(node.opcode == dgtl::linker::mir::MachineLinkOpcode::VEC_ADD_F32_512 ? 0x58 : 0x59);
                    write_u8(0x85); write_i32(slot_disp(node.op_b));
                    write_u8(0x62); write_u8(0xF1); write_u8(0x7C); write_u8(0x48); write_u8(0x11); write_u8(0x85);
                    write_i32(slot_disp(node.result_id));
                    break;
                }

                case MachineForm::VectorStore512: {
                    write_u8(0x62); write_u8(0xF1); write_u8(0x7C); write_u8(0x48); write_u8(0x11); write_u8(0x85);
                    write_i32(slot_disp(node.op_a));
                    break;
                }

                case MachineForm::VectorGather512: {
                    write_u8(0x62); write_u8(0xF2); write_u8(0x7D); write_u8(0x48); write_u8(0x38); write_u8(0x85);
                    write_i32(slot_disp(node.op_a));
                    break;
                }

                case MachineForm::Compare32: {
                    emit_mov_rax_from_slot(node.op_a);
                    write_u8(0x48); write_u8(0x3B); write_u8(0x85); write_i32(slot_disp(node.op_b));
                    write_u8(0x0F);
                    switch (node.opcode) {
                        case dgtl::linker::mir::MachineLinkOpcode::CMP_EQ_I32: write_u8(0x94); break;
                        case dgtl::linker::mir::MachineLinkOpcode::CMP_NE_I32: write_u8(0x95); break;
                        case dgtl::linker::mir::MachineLinkOpcode::CMP_LT_I32: write_u8(0x9C); break;
                        case dgtl::linker::mir::MachineLinkOpcode::CMP_LE_I32: write_u8(0x9E); break;
                        case dgtl::linker::mir::MachineLinkOpcode::CMP_GT_I32: write_u8(0x9F); break;
                        case dgtl::linker::mir::MachineLinkOpcode::CMP_GE_I32: write_u8(0x9D); break;
                        default: write_u8(0x9E); break;
                    }
                    write_u8(0xC0);
                    write_u8(0x0F); write_u8(0xB6); write_u8(0xC0);
                    emit_store_rax_to_slot(node.result_id);
                    break;
                }

                case MachineForm::Call: {
                    if (node.op_b != dgtl::linker::mir::INVALID_SLOT && node.op_b != 0u) {
                        emit_mov_rcx_from_slot(node.op_b);
                    } else { 
                        write_u8(0x48); write_u8(0x31); write_u8(0xC9); // xor rcx, rcx
                    }

                    if (node.op_c != dgtl::linker::mir::INVALID_SLOT && node.op_c != 0u) {
                        emit_mov_rdx_from_slot(node.op_c);
                    } else { 
                        write_u8(0x48); write_u8(0x31); write_u8(0xD2); // xor rdx, rdx
                    }

                    if (node.logic_flags != dgtl::linker::mir::INVALID_SLOT && node.logic_flags != 0u) {
                        emit_mov_r8_from_slot(node.logic_flags);
                    } else { 
                        write_u8(0x4D); write_u8(0x31); write_u8(0xC0); // xor r8, r8
                    }

                    const auto overflow = stream.get_call_overflow_args(i);
                    if (!overflow.empty() && overflow[0] != dgtl::linker::mir::INVALID_SLOT && overflow[0] != 0u) {
                        emit_mov_r9_from_slot(overflow[0]); // mov r9, [rbp-disp]
                    } else {
                        write_u8(0x4D); write_u8(0x31); write_u8(0xC9); // xor r9, r9
                    }

                    write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xC1); // movq xmm0, rcx
                    write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xCA); // movq xmm1, rdx
                    write_u8(0x66); write_u8(0x49); write_u8(0x0F); write_u8(0x6E); write_u8(0xD0); // movq xmm2, r8
                    write_u8(0x66); write_u8(0x49); write_u8(0x0F); write_u8(0x6E); write_u8(0xD9); // movq xmm3, r9

                    const size_t num_stack_args = (overflow.size() > 1) ? (overflow.size() - 1) : 0;
                    const size_t stack_alloc = 32 + ((num_stack_args + 1) / 2) * 16;
                    write_u8(0x48); write_u8(0x81); write_u8(0xEC); write_u32(static_cast<uint32_t>(stack_alloc)); // sub rsp, stack_alloc

                    for (size_t k = 1; k < overflow.size(); ++k) {
                        if (overflow[k] != dgtl::linker::mir::INVALID_SLOT && overflow[k] != 0u) {
                            emit_mov_rax_from_slot(overflow[k]);
                            write_u8(0x48); write_u8(0x89); write_u8(0x84); write_u8(0x24);
                            write_i32(static_cast<int32_t>(32 + (k - 1) * 8)); // mov [rsp + 32 + (k-1)*8], rax
                        }
                    }

                    uintptr_t target_addr = 0;
                    if (out_relocs == nullptr) {
                        if (!slot_map.empty()) {
                            auto it = slot_map.find(static_cast<uint32_t>(node.op_a));
                            if (it != slot_map.end() && it->second != 0) {
                                target_addr = it->second;
                            }
                        }
                        if (target_addr == 0 && node.op_a > 0x7FFF00000000ULL) {
                            target_addr = static_cast<uintptr_t>(node.op_a);
                        }
                        if (target_addr == 0 && node.intent_hash > 0x7FFF00000000ULL) {
                            target_addr = static_cast<uintptr_t>(node.intent_hash);
                        }
                    }

                    if (target_addr != 0) {
                        write_u8(0x48); write_u8(0xB8); write_u64(static_cast<uint64_t>(target_addr));
                        write_u8(0xFF); write_u8(0xD0);
                    } else {
                        write_u8(0xE8);
                        const size_t patch_disp_offset = offset;
                        write_u32(0);
                        const size_t patch_source_end = offset;
                        bool is_external_reloc = false;
                        if (out_relocs != nullptr) {
                            auto it = slot_to_callee.find(node.op_a);
                            if (it != slot_to_callee.end() && !it->second.empty()) {
                                out_relocs->push_back({ static_cast<uint32_t>(patch_disp_offset), it->second });
                                is_external_reloc = true;
                            }
                        }
                        if (!is_external_reloc) {
                            patches.push_back({patch_disp_offset, patch_source_end, resolve_target(node.op_a)});
                        }
                    }
                    write_u8(0x48); write_u8(0x81); write_u8(0xC4); write_u32(static_cast<uint32_t>(stack_alloc)); // add rsp, stack_alloc

                    if (node.result_id != dgtl::linker::mir::INVALID_SLOT) {
                        bool returns_in_xmm0 = false;
                        auto it_callee = slot_to_callee.find(node.op_a);
                        if (it_callee != slot_to_callee.end() && !it_callee->second.empty()) {
                            const std::string& name = it_callee->second;
                            if (name.find("_bits") != std::string::npos) {
                                returns_in_xmm0 = false;
                            } else if (name == "pow" || name == "sqrt" || name == "sin" ||
                                       name == "cos" || name == "tan" || name == "exp" ||
                                       name == "log" || name == "fabs" || name == "floor" ||
                                       name == "ceil") {
                                returns_in_xmm0 = true;
                            } else {
                                const dgtl::linker::mir::MachineLinkTypeCode tc = stream.get_result_type_code(node.result_id);
                                returns_in_xmm0 = dgtl::linker::mir::is_floating_type_code(tc);
                            }
                        } else {
                            const dgtl::linker::mir::MachineLinkTypeCode tc = stream.get_result_type_code(node.result_id);
                            returns_in_xmm0 = dgtl::linker::mir::is_floating_type_code(tc);
                        }

                        if (returns_in_xmm0) {
                            write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x7E); write_u8(0xC0); // movq rax, xmm0
                        }
                        emit_store_rax_to_slot(node.result_id);
                    } else {
                        write_u8(0x90);
                    }
                    continue;
                }

                case MachineForm::Goto: {
                    write_u8(0xE9);
                    const uint32_t target_slot = (node.op_a != dgtl::linker::mir::INVALID_SLOT && node.op_a != 0u) ? node.op_a : node.result_id;
                    patches.push_back({offset, offset + 4, resolve_target(target_slot)});
                    write_u32(0);
                    break;
                }

                case MachineForm::BranchCond: {
                    emit_mov_rax_from_slot(node.op_a);
                    write_u8(0x48); write_u8(0x85); write_u8(0xC0); // test rax, rax
                    write_u8(0x0F);
                    write_u8(node.opcode == dgtl::linker::mir::MachineLinkOpcode::BRANCH_UNLIKELY ? 0x84 : 0x85);
                    const uint32_t target_slot = (node.op_b != dgtl::linker::mir::INVALID_SLOT && node.op_b != 0u) ? node.op_b : node.result_id;
                    patches.push_back({offset, offset + 4, resolve_target(target_slot)});
                    write_u32(0);
                    break;
                }

                case MachineForm::BranchCompare: {
                    emit_mov_rax_from_slot(node.op_a);
                    write_u8(0x48); write_u8(0x3B); write_u8(0x85); write_i32(slot_disp(node.op_b));
                    write_u8(0x0F); write_u8(0x85);
                    const uint32_t target_slot = (node.op_c != dgtl::linker::mir::INVALID_SLOT && node.op_c != 0u) ? node.op_c : node.result_id;
                    patches.push_back({offset, offset + 4, resolve_target(target_slot)});
                    write_u32(0);
                    break;
                }

                case MachineForm::Return: {
                    if (node.op_a != dgtl::linker::mir::INVALID_SLOT) {
                        emit_mov_rax_from_slot(node.op_a);
                        write_u8(0x66); write_u8(0x48); write_u8(0x0F); write_u8(0x6E); write_u8(0xC0); // movq xmm0, rax (Dual ABI Return)
                    } else {
                        write_u8(0x31); write_u8(0xC0);
                        write_u8(0x66); write_u8(0x0F); write_u8(0xEF); write_u8(0xC0); // pxor xmm0, xmm0
                    }
                    write_u8(0xC9); write_u8(0xC3);
                    break;
                }

                case MachineForm::Syscall: {
                    emit_mov_rax_from_slot(node.op_a);
                    if (node.op_b != dgtl::linker::mir::INVALID_SLOT) {
                        emit_mov_reg_from_slot(0x48, 0xBD, node.op_b);
                    } else {
                        write_u8(0x48); write_u8(0x31); write_u8(0xFF);
                    }
                    if (node.op_c != dgtl::linker::mir::INVALID_SLOT) {
                        emit_mov_reg_from_slot(0x48, 0xB5, node.op_c);
                    } else {
                        write_u8(0x48); write_u8(0x31); write_u8(0xF6);
                    }
                    write_u8(0x0F); write_u8(0x05);
                    if (node.result_id != dgtl::linker::mir::INVALID_SLOT) {
                        emit_store_rax_to_slot(node.result_id);
                    }
                    break;
                }

                default:
                    record_fallback_opcode(i, node.opcode);
                    write_u8(0x90);
                    break;
            }
        }

        // Defensive fallthrough safety: guarantee clean leave; ret if last instruction was not Return
        if (count == 0 || stream.get_node(count - 1).opcode != dgtl::linker::mir::MachineLinkOpcode::RET) {
            write_u8(0x31); write_u8(0xC0); // xor eax, eax
            write_u8(0xC9); write_u8(0xC3); // leave; ret
        }

        node_offsets[count] = offset;
        for (const auto& p : patches) {
            if (p.target_node >= count) {
                continue; // Do not patch external calls; preserve zero addend for external COFF relocations
            }
            const size_t tgt_byte = node_offsets[p.target_node];
            const int64_t rel64 = static_cast<int64_t>(tgt_byte) - static_cast<int64_t>(p.source_end);
            const int32_t rel32 = static_cast<int32_t>(rel64);
            std::memcpy(dest + p.disp_offset, &rel32, sizeof(rel32));
        }

        if (emitted_size) *emitted_size = offset;
        return true;
    }

}; // class DGTLLinker

} // namespace dgtl::linker

enum class BinaryFormatType : uint8_t {
	ELF64 = 0,
	MachO64 = 1,
	WASM = 2,
	PE64 = 3,
	DSL_PAYLOAD = 4
};

struct Binary_Format {
	BinaryFormatType format_type = BinaryFormatType::ELF64;
	uint64_t entry_point = 0;
	std::vector<uint8_t> data;

	[[nodiscard]] bool is_valid() const noexcept {
		if (data.empty()) {
			return false;
		}

		switch (format_type) {
			case BinaryFormatType::ELF64:
				return data.size() >= 64 &&
				       data[0] == 0x7f && data[1] == 'E' && data[2] == 'L' && data[3] == 'F' &&
				       data[4] == 2u && data[5] == 1u;
			case BinaryFormatType::MachO64:
				return data.size() >= 32 && data[0] == 0xcf && data[1] == 0xfa && data[2] == 0xed && data[3] == 0xfe;
			case BinaryFormatType::PE64:
				return data.size() >= 64 && data[0] == 'M' && data[1] == 'Z';
			case BinaryFormatType::WASM:
				return data.size() >= 8 && data[0] == 0x00 && data[1] == 0x61 && data[2] == 0x73 && data[3] == 0x6d;
			case BinaryFormatType::DSL_PAYLOAD: {
				if (data.size() < 8) {
					return false;
				}
				uint32_t magic = 0;
				std::memcpy(&magic, data.data(), sizeof(uint32_t));
				return magic == 0x4D595A4E;
			}
			default:
				return false;
		}
	}

	[[nodiscard]] const void* get_code_ptr() const noexcept { return data.data(); }
	[[nodiscard]] void* get_code_ptr() noexcept { return data.data(); }
	[[nodiscard]] size_t get_code_size() const noexcept { return data.size(); }
	[[nodiscard]] uint64_t get_entry_point() const noexcept { return entry_point; }
};

	PCRelative = 1,

	GOT = 3,
	PLT = 4

class Object_Format {
public:
	Object_Format() = default;
	~Object_Format() = default;

	Object_Format(const Object_Format&) = delete;
	Object_Format& operator=(const Object_Format&) = delete;

	[[nodiscard]] std::optional<uint64_t> find_symbol_address(std::string_view name) const noexcept {
		auto it = symbol_map_.find(name);
		if (it != symbol_map_.end()) {
			return it->second;
		}
		return std::nullopt;
	}

	void add_symbol(std::string_view name, uint64_t addr, uint64_t size,
					uint8_t binding = 0, uint8_t type = 0) {
		std::string str_name{name};
		symbols.push_back({str_name, addr, size, binding, type, 0, false});
		symbol_map_.try_emplace(std::move(str_name), addr);
	}

	void add_extern_symbol(std::string_view name, uint8_t binding = 1) {
		symbols.push_back({std::string(name), 0, 0, binding, 0, 0, true});
	}

	void add_relocation(uint64_t offset, std::string_view symbol,
						int64_t addend = 0, RelocationType type = RelocationType::Absolute) {
		relocations.push_back({offset, std::string(symbol), addend, type});
	}

	[[nodiscard]] dgtl::object::ObjectFile to_object_file(
		dgtl::object::ObjectFormat fmt = dgtl::object::ObjectFormat::ELF) const {
		dgtl::object::ObjectFile obj(fmt);

		if (!text_section.data.empty()) {
			auto& s = obj.add_section(".text", text_section.alignment,
				text_section.flags, dgtl::object::SectionKind::Text);
			s.address = text_section.base_address;
			s.bytes = text_section.data;
		}
		if (!data_section.data.empty()) {
			auto& s = obj.add_section(".data", data_section.alignment,
				data_section.flags, dgtl::object::SectionKind::Data);
			s.address = data_section.base_address;
			s.bytes = data_section.data;
		}
		if (!rodata_section.data.empty()) {
			auto& s = obj.add_section(".rodata", rodata_section.alignment,
				rodata_section.flags, dgtl::object::SectionKind::RData);
			s.address = rodata_section.base_address;
			s.bytes = rodata_section.data;
		}
		if (bss_section.bss_size != 0u) {
			auto& s = obj.add_section(".bss", bss_section.alignment,
				bss_section.flags | dgtl::object::SectionFlags::Write |
				dgtl::object::SectionFlags::ZeroFill,
				dgtl::object::SectionKind::BSS);
			s.address = bss_section.base_address;
			s.bss_size = bss_section.bss_size;
		}

		for (const auto& sym : symbols) {
			if (sym.is_extern) {
				obj.add_external_symbol(sym.name, sym.binding, sym.type);
			} else {
				obj.add_symbol(sym.name, sym.address, sym.size, sym.binding, sym.type);
			}
		}

		for (const auto& rel : relocations) {
			dgtl::object::RelocationKind rk = dgtl::object::RelocationKind::Absolute64;
			switch (rel.type) {
				case RelocationType::PCRelative:
					rk = dgtl::object::RelocationKind::PCRel32; break;
				case RelocationType::Relative:
					rk = dgtl::object::RelocationKind::Relative64; break;
				case RelocationType::GOT:
					rk = dgtl::object::RelocationKind::GOT64; break;
				case RelocationType::PLT:
					rk = dgtl::object::RelocationKind::PLT64; break;
				case RelocationType::Absolute:
				default:
					rk = dgtl::object::RelocationKind::Absolute64; break;
			}
			obj.add_relocation(rel.offset, rel.symbol_name, rel.addend, 0u, rk);
		}

		return obj;
	}

	[[nodiscard]] Section& get_text_section() noexcept { return text_section; }
	[[nodiscard]] Section& get_data_section() noexcept { return data_section; }
	[[nodiscard]] Section& get_rodata_section() noexcept { return rodata_section; }
	[[nodiscard]] Section& get_bss_section() noexcept { return bss_section; }

	[[nodiscard]] const Section& get_text_section() const noexcept { return text_section; }
	[[nodiscard]] const Section& get_data_section() const noexcept { return data_section; }
	[[nodiscard]] const Section& get_rodata_section() const noexcept { return rodata_section; }
	[[nodiscard]] const Section& get_bss_section() const noexcept { return bss_section; }

	[[nodiscard]] const std::vector<Symbol>& get_symbols() const noexcept { return symbols; }
	[[nodiscard]] const std::vector<Relocation>& get_relocations() const noexcept { return relocations; }
	[[nodiscard]] size_t symbol_count() const noexcept { return symbols.size(); }
	[[nodiscard]] size_t relocation_count() const noexcept { return relocations.size(); }

private:
	Section text_section;
	Section data_section;
	Section rodata_section;
	Section bss_section;

	std::vector<Symbol> symbols;
	std::vector<Relocation> relocations;
	std::unordered_map<std::string, uint64_t, StringViewHash, StringViewEqual> symbol_map_;
};

class Binary_Loader {
public:
	Binary_Loader() = default;
	~Binary_Loader() = default;

	Binary_Loader(const Binary_Loader&) = delete;
	Binary_Loader& operator=(const Binary_Loader&) = delete;

	[[nodiscard]] std::unique_ptr<Binary_Format> load(const std::string& filepath) {
		auto binary = std::make_unique<Binary_Format>();

		std::ifstream file(filepath, std::ios::binary | std::ios::ate);
		if (!file.is_open()) {
			throw std::runtime_error("Failed to open binary file: " + filepath);
		}

		const std::streamsize size = file.tellg();
		if (size < 64) {
			throw std::runtime_error("Binary file too small");
		}

		file.seekg(0, std::ios::beg);
		binary->data.resize(static_cast<size_t>(size));
		if (!file.read(reinterpret_cast<char*>(binary->data.data()), size)) {
			throw std::runtime_error("Failed to read binary file");
		}

		detect_format(binary.get());
		if (!binary->is_valid()) {
			throw std::runtime_error("Invalid binary format");
		}

		parse_entry_point(binary.get());
		return binary;
	}

	[[nodiscard]] bool load_frontend_payload(const std::string& filepath) {
		std::ifstream file(filepath, std::ios::binary | std::ios::ate);
		if (!file.is_open()) {
			return false;
		}

		const std::streamsize size = file.tellg();
		if (size < 8) {
			return false;
		}

		file.seekg(0, std::ios::beg);
		payload_buffer_.resize(static_cast<size_t>(size));
		if (!file.read(payload_buffer_.data(), size)) {
			return false;
		}

		const char* ptr = payload_buffer_.data();
		const char* end = ptr + size;

		const uint32_t magic = read_u32(ptr);
		ptr += sizeof(uint32_t);
		if (magic != 0x4D595A4E) {
			return false;
		}

		const uint32_t string_count = read_u32(ptr);
		ptr += sizeof(uint32_t);

		payload_string_ids_.clear();
		payload_string_ids_.reserve(string_count);

		for (uint32_t i = 0; i < string_count; ++i) {
			if (ptr + (sizeof(uint32_t) * 2) > end) {
				break;
			}

			const uint32_t id = read_u32(ptr);
			ptr += sizeof(uint32_t);

			const uint32_t len = read_u32(ptr);
			ptr += sizeof(uint32_t);

			if (ptr + len > end) {
				break;
			}
			payload_string_ids_.push_back(id);
			ptr += len;
		}

		return true;
	}

	[[nodiscard]] const std::vector<uint32_t>& get_payload_string_ids() const noexcept {
		return payload_string_ids_;
	}

	[[nodiscard]] std::unique_ptr<Binary_Format> load_from_memory(const void* data, size_t size) {
		if (data == nullptr || size < 64) {
			throw std::runtime_error("Invalid binary data");
		}

		auto binary = std::make_unique<Binary_Format>();
		binary->data.assign(static_cast<const uint8_t*>(data),
							static_cast<const uint8_t*>(data) + size);

		detect_format(binary.get());
		if (!binary->is_valid()) {
			throw std::runtime_error("Invalid binary format");
		}

		parse_entry_point(binary.get());
		return binary;
	}

private:
	std::vector<char> payload_buffer_;
	std::vector<uint32_t> payload_string_ids_;

	[[nodiscard]] static uint32_t read_u32(const void* ptr) noexcept {
		uint32_t v = 0;
		std::memcpy(&v, ptr, sizeof(uint32_t));
		return v;
	}

	[[nodiscard]] static uint64_t read_u64(const void* ptr) noexcept {
		uint64_t v = 0;
		std::memcpy(&v, ptr, sizeof(uint64_t));
		return v;
	}

	static void detect_format(Binary_Format* fmt) noexcept {
		if (fmt == nullptr || fmt->data.size() < 4) {
			if (fmt != nullptr) {
				fmt->format_type = BinaryFormatType::ELF64;
			}
			return;
		}

		const uint8_t* bytes = fmt->data.data();
		if (bytes[0] == 0x7f && bytes[1] == 'E' && bytes[2] == 'L' && bytes[3] == 'F') {
			fmt->format_type = BinaryFormatType::ELF64;
		} else if (bytes[0] == 0xcf && bytes[1] == 0xfa && bytes[2] == 0xed && bytes[3] == 0xfe) {
			fmt->format_type = BinaryFormatType::MachO64;
		} else if (bytes[0] == 'M' && bytes[1] == 'Z') {
			fmt->format_type = BinaryFormatType::PE64;
		} else if (bytes[0] == 0x00 && bytes[1] == 0x61 && bytes[2] == 0x73 && bytes[3] == 0x6d) {
			fmt->format_type = BinaryFormatType::WASM;
		} else if (bytes[0] == 'M' && bytes[1] == 'Y' && bytes[2] == 'Z' && bytes[3] == 'N') {
			fmt->format_type = BinaryFormatType::DSL_PAYLOAD;
		} else {
			fmt->format_type = BinaryFormatType::ELF64;
		}
	}

	static void parse_entry_point(Binary_Format* fmt) noexcept {
		if (fmt == nullptr || fmt->data.size() < 64) {
			if (fmt != nullptr) {
				fmt->entry_point = 0;
			}
			return;
		}

		const uint8_t* bytes = fmt->data.data();
		switch (fmt->format_type) {
			case BinaryFormatType::ELF64:
				if (fmt->data.size() >= 32) {
					fmt->entry_point = read_u64(&bytes[24]);
				}
				break;
			case BinaryFormatType::MachO64:
				if (fmt->data.size() >= 40) {
					fmt->entry_point = read_u64(&bytes[32]);
				}
				break;
			case BinaryFormatType::PE64: {
				if (fmt->data.size() >= 0x40) {
					const uint32_t lfanew = read_u32(&bytes[0x3C]);
					const std::size_t optional_header =
						static_cast<std::size_t>(lfanew) + 4u + 20u;
					if (optional_header + 20u <= fmt->data.size()) {
						fmt->entry_point = read_u32(&bytes[optional_header + 16u]);
					} else {
						fmt->entry_point = 0;
					}
				}
				break;
			}
			case BinaryFormatType::WASM:
				fmt->entry_point = 0;
				break;
			default:
				fmt->entry_point = 0;
				break;
		}
	}
};

class Object_Loader {
public:
	Object_Loader() = default;
	~Object_Loader() = default;

	Object_Loader(const Object_Loader&) = delete;
	Object_Loader& operator=(const Object_Loader&) = delete;

	[[nodiscard]] std::unique_ptr<Object_Format> load(const std::string& filepath) {
		auto obj = std::make_unique<Object_Format>();

		std::ifstream file(filepath, std::ios::binary | std::ios::ate);
		if (!file.is_open()) {
			throw std::runtime_error("Failed to open object file: " + filepath);
		}

		const std::streamsize size = file.tellg();
		if (size < 64) {
			throw std::runtime_error("Object file too small");
		}

		file.seekg(0, std::ios::beg);
		std::vector<uint8_t> buffer(static_cast<size_t>(size));
		if (!file.read(reinterpret_cast<char*>(buffer.data()), size)) {
			throw std::runtime_error("Failed to read object file");
		}

		if (!is_elf_format(buffer)) {
			throw std::runtime_error("Unsupported object format (not ELF)");
		}

		parse_elf_object(obj.get(), buffer);
		return obj;
	}

	[[nodiscard]] std::unique_ptr<Object_Format> load_from_memory(const void* data, size_t size) {
		if (data == nullptr || size < 64) {
			throw std::runtime_error("Invalid object data");
		}

		std::vector<uint8_t> buffer(static_cast<const uint8_t*>(data),
									static_cast<const uint8_t*>(data) + size);
		if (!is_elf_format(buffer)) {
			throw std::runtime_error("Unsupported object format");
		}

		auto obj = std::make_unique<Object_Format>();
		parse_elf_object(obj.get(), buffer);
		return obj;
	}

private:
	struct ElfSectionHeaderView {
		uint32_t name_offset = 0;
		uint32_t type = 0;
		uint64_t flags = 0;
		uint64_t addr = 0;
		uint64_t offset = 0;
		uint64_t size = 0;
		uint32_t link = 0;
		uint32_t info = 0;
		uint64_t addralign = 1;
		uint64_t entsize = 0;
	};

	static constexpr uint32_t kShtNull = 0;
	static constexpr uint32_t kShtSymTab = 2;
	static constexpr uint32_t kShtStrTab = 3;
	static constexpr uint32_t kShtRela = 4;
	static constexpr uint32_t kShtNoBits = 8;
	static constexpr uint32_t kShtRel = 9;
	static constexpr uint32_t kShtDynSym = 11;
	static constexpr uint16_t kShnUndef = 0;

	[[nodiscard]] static std::string read_c_string(const uint8_t* table, size_t table_size, uint32_t offset) {
		if (table == nullptr || offset >= table_size) {
			return {};
		}
		const char* begin = reinterpret_cast<const char*>(table + offset);
		const char* end = reinterpret_cast<const char*>(table + table_size);
		const char* p = begin;
		while (p < end && *p != '\0') {
			++p;
		}
		return std::string(begin, static_cast<size_t>(p - begin));
	}

	[[nodiscard]] static std::vector<ElfSectionHeaderView> read_section_headers(
		const std::vector<uint8_t>& buffer,
		uint64_t sh_offset,
		uint16_t sh_entsize,
		uint16_t sh_num) {
		std::vector<ElfSectionHeaderView> headers;
		headers.reserve(sh_num);

		for (uint16_t i = 0; i < sh_num; ++i) {
			const uint64_t offset = sh_offset + (static_cast<uint64_t>(i) * sh_entsize);
			if (offset + 64 > buffer.size()) {
				break;
			}
			const uint8_t* sh = buffer.data() + offset;

			ElfSectionHeaderView view;
			std::memcpy(&view.name_offset, sh + 0, 4);
			std::memcpy(&view.type, sh + 4, 4);
			std::memcpy(&view.flags, sh + 8, 8);
			std::memcpy(&view.addr, sh + 16, 8);
			std::memcpy(&view.offset, sh + 24, 8);
			std::memcpy(&view.size, sh + 32, 8);
			std::memcpy(&view.link, sh + 40, 4);
			std::memcpy(&view.info, sh + 44, 4);
			std::memcpy(&view.addralign, sh + 48, 8);
			std::memcpy(&view.entsize, sh + 56, 8);
			headers.push_back(view);
		}
		return headers;
	}

	[[nodiscard]] static std::vector<std::string> build_section_name_table(
		const std::vector<uint8_t>& buffer,
		const std::vector<ElfSectionHeaderView>& headers,
		uint16_t shstrndx) {
		std::vector<std::string> names(headers.size());
		if (shstrndx >= headers.size()) {
			return names;
		}

		const auto& strtab = headers[shstrndx];
		if (strtab.type != kShtStrTab || strtab.offset + strtab.size > buffer.size()) {
			return names;
		}

		const uint8_t* table = buffer.data() + strtab.offset;
		const size_t table_size = static_cast<size_t>(strtab.size);
		for (size_t i = 0; i < headers.size(); ++i) {
			names[i] = read_c_string(table, table_size, headers[i].name_offset);
		}
		return names;
	}

	[[nodiscard]] static bool is_elf_format(const std::vector<uint8_t>& buffer) noexcept {
		return buffer.size() >= 4 && buffer[0] == 0x7f && buffer[1] == 'E' && buffer[2] == 'L' && buffer[3] == 'F';
	}

	static void parse_elf_object(Object_Format* obj, const std::vector<uint8_t>& buffer) {
		if (buffer.size() < 64) {
			throw std::runtime_error("ELF header too small");
		}

		uint64_t e_shoff = 0;
		uint16_t e_shnum = 0;
		uint16_t e_shentsize = 0;
		uint16_t e_shstrndx = 0;

		std::memcpy(&e_shoff, buffer.data() + 32, 8);
		std::memcpy(&e_shentsize, buffer.data() + 46, 2);
		std::memcpy(&e_shnum, buffer.data() + 48, 2);
		std::memcpy(&e_shstrndx, buffer.data() + 50, 2);

		if (e_shoff == 0 || e_shnum == 0 || e_shentsize < 64) {
			throw std::runtime_error("Invalid ELF section headers");
		}

		const auto section_headers = read_section_headers(buffer, e_shoff, e_shentsize, e_shnum);
		if (section_headers.empty()) {
			throw std::runtime_error("Unable to parse ELF section headers");
		}

		const auto section_names = build_section_name_table(buffer, section_headers, e_shstrndx);
		parse_sections(obj, buffer, section_headers, section_names);
		const auto symbol_tables = parse_symbols(obj, buffer, section_headers);
		parse_relocations(obj, buffer, section_headers, symbol_tables);
	}

	static void parse_sections(Object_Format* obj,
							   const std::vector<uint8_t>& buffer,
							   const std::vector<ElfSectionHeaderView>& headers,
							   const std::vector<std::string>& section_names) noexcept {
		for (size_t i = 0; i < headers.size(); ++i) {
			const auto& sh = headers[i];
			if (sh.type == kShtNull) {
				continue;
			}

			Section sec;
			sec.base_address = sh.addr;
			sec.alignment = (sh.addralign == 0) ? 1 : sh.addralign;
			sec.flags = static_cast<uint32_t>(sh.flags & 0xFFFFFFFFu);

			if (sh.type == kShtNoBits) {
				sec.data.clear();
				sec.bss_size = sh.size;
			} else {
				if (sh.offset + sh.size > buffer.size()) {
					continue;
				}
				sec.data.assign(buffer.begin() + sh.offset, buffer.begin() + sh.offset + sh.size);
			}

			const std::string_view name = (i < section_names.size()) ? std::string_view(section_names[i]) : std::string_view{};
			if (name == ".text") {
				obj->get_text_section() = std::move(sec);
			} else if (name == ".data") {
				obj->get_data_section() = std::move(sec);
			} else if (name == ".rodata") {
				obj->get_rodata_section() = std::move(sec);
			} else if (name == ".bss" || sh.type == kShtNoBits) {
				obj->get_bss_section() = std::move(sec);
			}
		}
	}

	static std::unordered_map<uint32_t, std::vector<std::string>> parse_symbols(
		Object_Format* obj,
		const std::vector<uint8_t>& buffer,
		const std::vector<ElfSectionHeaderView>& headers) noexcept {
		std::unordered_map<uint32_t, std::vector<std::string>> symbol_tables;
		symbol_tables.reserve(2);

		for (uint32_t section_index = 0; section_index < static_cast<uint32_t>(headers.size()); ++section_index) {
			const auto& sh = headers[section_index];
			if (sh.type != kShtSymTab && sh.type != kShtDynSym) {
				continue;
			}
			if (sh.entsize < 24 || sh.size == 0 || sh.offset + sh.size > buffer.size() || sh.link >= headers.size()) {
				continue;
			}

			const auto& strtab = headers[sh.link];
			if (strtab.type != kShtStrTab || strtab.offset + strtab.size > buffer.size()) {
				continue;
			}

			const uint8_t* strtab_data = buffer.data() + strtab.offset;
			const size_t strtab_size = static_cast<size_t>(strtab.size);
			const uint64_t sym_count = sh.size / sh.entsize;

			auto& symbol_names = symbol_tables[section_index];
			symbol_names.clear();
			symbol_names.resize(static_cast<size_t>(sym_count));

			for (uint64_t j = 0; j < sym_count; ++j) {
				const uint64_t sym_entry_offset = sh.offset + (j * sh.entsize);
				if (sym_entry_offset + 24 > buffer.size()) {
					break;
				}
				const uint8_t* sym = buffer.data() + sym_entry_offset;

				uint32_t st_name = 0;
				uint8_t st_info = 0;
				uint16_t st_shndx = 0;
				uint64_t st_value = 0;
				uint64_t st_size = 0;

				std::memcpy(&st_name, sym + 0, 4);
				std::memcpy(&st_info, sym + 4, 1);
				std::memcpy(&st_shndx, sym + 6, 2);
				std::memcpy(&st_value, sym + 8, 8);
				std::memcpy(&st_size, sym + 16, 8);

				std::string name = read_c_string(strtab_data, strtab_size, st_name);
				if (name.empty()) {
					name = std::string("symbol_") + std::to_string(j);
				}

				symbol_names[static_cast<size_t>(j)] = name;

				const uint8_t binding = st_info >> 4;
				const uint8_t type = st_info & 0x0F;
				if (st_shndx == kShnUndef) {
					obj->add_extern_symbol(name, binding == 0 ? 1 : binding);
				} else {
					obj->add_symbol(name, st_value, st_size, binding, type);
				}
			}
		}

		return symbol_tables;
	}

	static void parse_relocations(
		Object_Format* obj,
		const std::vector<uint8_t>& buffer,
		const std::vector<ElfSectionHeaderView>& headers,
		const std::unordered_map<uint32_t, std::vector<std::string>>& symbol_tables) noexcept {
		for (const auto& sh : headers) {
			const bool is_rela = (sh.type == kShtRela);
			if (sh.type != kShtRel && sh.type != kShtRela) {
				continue;
			}
			if (sh.entsize == 0 || sh.size == 0 || sh.offset + sh.size > buffer.size()) {
				continue;
			}

			const uint64_t rel_count = sh.size / sh.entsize;
			for (uint64_t j = 0; j < rel_count; ++j) {
				const uint64_t rel_entry_offset = sh.offset + (j * sh.entsize);
				if (rel_entry_offset + 16 > buffer.size()) {
					break;
				}

				const uint8_t* rel = buffer.data() + rel_entry_offset;
				uint64_t r_offset = 0;
				uint64_t r_info = 0;
				int64_t r_addend = 0;

				std::memcpy(&r_offset, rel + 0, 8);
				std::memcpy(&r_info, rel + 8, 8);
				if (is_rela && rel_entry_offset + 24 <= buffer.size()) {
					std::memcpy(&r_addend, rel + 16, 8);
				}

				const uint32_t rel_type = static_cast<uint32_t>(r_info & 0xFFFFFFFFu);
				const uint32_t sym_index = static_cast<uint32_t>((r_info >> 32) & 0xFFFFFFFFu);

				RelocationType reloc_type = RelocationType::Absolute;
				if (rel_type == 1u) {
					reloc_type = RelocationType::PCRelative;
				} else if (rel_type == 8u) {
					reloc_type = RelocationType::Relative;
				}

				std::string symbol_name;
				const auto table_it = symbol_tables.find(sh.link);
				if (table_it != symbol_tables.end() &&
					sym_index < table_it->second.size() &&
					!table_it->second[sym_index].empty()) {
					symbol_name = table_it->second[sym_index];
				} else {
					symbol_name = std::string("symbol_") + std::to_string(sym_index);
				}

				obj->add_relocation(r_offset, symbol_name, r_addend, reloc_type);
			}
		}
	}
};

[[nodiscard]] constexpr std::string_view target_family_name(TargetFamily family) noexcept {
    switch (family) {
        case TargetFamily::X64:    return "x64";
        case TargetFamily::ARM64:  return "arm64";
        case TargetFamily::ARM32:  return "arm32";
        case TargetFamily::WASM:   return "wasm";
        case TargetFamily::RISCV64:return "riscv64";
        case TargetFamily::CUDA:   return "cuda";
        case TargetFamily::ROCm:   return "rocm";
        case TargetFamily::FPGA:   return "fpga";
    }
    return "unknown";
}

/**
 * @brief Semantic target capability contract shared by target consumers.
 *
 * These features describe the compilation target, not the host CPU on which
 * DGTL itself is running. Host detection remains represented by
 * HardwareFeatures below. Keeping the two concepts separate prevents
 * accidental host-feature leakage into cross-compilation.
 */
enum class TargetFeature : uint8_t {
    Scalar32 = 0,
    Scalar64,
    Vector,
    VectorMemory,
    FMA,
    Atomics,
    Concurrency,
    Syscall,
    AVX2,
    AVX512,
    BMI2,
    SVE,
    NEON,
    VFP,
    RVV,
    CUDA,
    ROCm,
    Tensor,
    WasmSIMD,
    ObjectMachO,
    AMX,
    Count
};

class TargetFeatureSet {
public:
    using mask_type = uint64_t;

    constexpr TargetFeatureSet() noexcept = default;
    constexpr explicit TargetFeatureSet(mask_type mask) noexcept : mask_(mask) {}

    [[nodiscard]] constexpr bool has(TargetFeature feature) const noexcept {
        const auto bit = static_cast<unsigned>(feature);
        return bit < static_cast<unsigned>(TargetFeature::Count)
            && ((mask_ >> bit) & 1ull) != 0ull;
    }

    constexpr void add(TargetFeature feature) noexcept {
        const auto bit = static_cast<unsigned>(feature);
        if (bit < static_cast<unsigned>(TargetFeature::Count)) {
            mask_ |= (1ull << bit);
        }
    }

    constexpr void remove(TargetFeature feature) noexcept {
        const auto bit = static_cast<unsigned>(feature);
        if (bit < static_cast<unsigned>(TargetFeature::Count)) {
            mask_ &= ~(1ull << bit);
        }
    }

    [[nodiscard]] constexpr mask_type mask() const noexcept { return mask_; }
    [[nodiscard]] constexpr bool empty() const noexcept { return mask_ == 0; }

    friend constexpr bool operator==(TargetFeatureSet, TargetFeatureSet) noexcept = default;

private:
    mask_type mask_ = 0;
};

 * Mach-O is an object format, not a CPU ISA.
 */

    CPU = 0,
    GPU = 1,
    Accelerator = 2,
    VirtualMachine = 3,
    FPGA = 4
};

enum class ObjectFormat : uint8_t {

[[nodiscard]] inline SimdExecutionProfile preferred_simd_execution_profile() noexcept {
    const HardwareFeatures hw = detect_hardware();
    // AVX-512 is the wider profile; test it before AVX2.
    if (hw.has_avx512) {
        return SimdExecutionProfile::AVX512Wide;
    }
    if (hw.has_avx2) {
        return SimdExecutionProfile::AVX2DualPump;
    }
    return SimdExecutionProfile::Scalar;
}

[[nodiscard]] inline SimdTier preferred_simd_tier() noexcept {
    switch (preferred_simd_execution_profile()) {
        case SimdExecutionProfile::AVX2DualPump:
            return SimdTier::AVX2;
        case SimdExecutionProfile::AVX512Wide:
            return SimdTier::AVX512;
        case SimdExecutionProfile::Scalar:
        default:
            return SimdTier::Scalar;
    }
}

[[nodiscard]] inline ArchType select_best_x86_arch() noexcept {
    const X86SimdCaps caps = detect_x86_simd_caps();
    return caps.has_avx512 ? ArchType::X86_64_AVX512 : ArchType::X86_64_AVX2;
}

[[nodiscard]] inline ArchType select_preferred_x86_arch() noexcept {
    return preferred_simd_tier() == SimdTier::AVX512 ? ArchType::X86_64_AVX512 : ArchType::X86_64_AVX2;
}

    Vector = 1,
    Predicate = 2,
    Special = 3,

enum class CallingConventionKind : uint8_t {

    AAPCS64,

struct TargetRegisterInfo {
    uint16_t id = 0;
    [[maybe_unused]] std::string_view name{};

    uint16_t bit_width = 0;
    bool caller_saved = false;
    bool allocatable = true;
};

struct CallingConventionDesc {
    CallingConventionKind kind = CallingConventionKind::BareMetal;
    std::string_view name{};

    std::array<uint16_t, 8> vector_arg_registers{};
    uint8_t integer_arg_count = 0;
    uint8_t vector_arg_count = 0;
    uint16_t integer_return_register = 0;
    uint16_t vector_return_register = 0;
    uint16_t stack_alignment = 16;
    uint16_t shadow_space_bytes = 0;

    [[nodiscard]] std::string_view get_name() const noexcept { return name; }
    [[nodiscard]] uint16_t get_integer_arg_reg(size_t index) const noexcept {
        return (index < integer_arg_registers.size()) ? integer_arg_registers[index] : 0;
    }
    [[nodiscard]] uint16_t get_vector_arg_reg(size_t index) const noexcept {
        return (index < vector_arg_registers.size()) ? vector_arg_registers[index] : 0;
    }
};

struct InstructionDesc {
    OpCode opcode = static_cast<OpCode>(0);
    uint8_t latency_cycles = 1;
    uint8_t reciprocal_throughput = 1;
    uint16_t legal_scalar_width_mask = 0;
    bool supports_fusion = false;
};

class TargetDescription {
public:
    static constexpr uint16_t kWidth32 = 1u << 0;
    static constexpr uint16_t kWidth64 = 1u << 1;
    static constexpr uint16_t kWidth128 = 1u << 2;
    static constexpr uint16_t kWidth256 = 1u << 3;
    static constexpr uint16_t kWidth512 = 1u << 4;

    [[nodiscard]] static TargetDescription for_arch(
        ArchType arch,
        CallingConventionKind calling_convention = CallingConventionKind::SystemV_x64);

    [[nodiscard]] ArchType arch() const noexcept { return arch_; }
    [[nodiscard]] CallingConventionKind calling_convention_kind() const noexcept {
        return calling_convention_.kind;
    }
    [[nodiscard]] std::span<const TargetRegisterInfo> general_purpose_registers() const noexcept {
        return std::span<const TargetRegisterInfo>(gp_registers_.data(), gp_register_count_);
    }
    [[nodiscard]] std::span<const TargetRegisterInfo> vector_registers() const noexcept {
        return std::span<const TargetRegisterInfo>(vector_registers_.data(), vector_register_count_);
    }
    [[nodiscard]] const CallingConventionDesc& default_calling_convention() const noexcept {
        return calling_convention_;
    }
    [[nodiscard]] std::optional<InstructionDesc> instruction_desc(OpCode opcode) const noexcept {
        const auto it = std::find_if(instruction_table_.begin(), instruction_table_.begin() + instruction_count_,
                                     [opcode](const auto& desc) { return desc.opcode == opcode; });
        if (it != instruction_table_.begin() + instruction_count_) {
            return *it;
        }
        return std::nullopt;
    }
    [[nodiscard]] uint8_t instruction_latency(OpCode opcode) const noexcept {
        const auto desc = instruction_desc(opcode);
        return desc.has_value() ? desc->latency_cycles : 1;
    }
    [[nodiscard]] uint8_t instruction_reciprocal_throughput(OpCode opcode) const noexcept {
        const auto desc = instruction_desc(opcode);
        return desc.has_value() ? desc->reciprocal_throughput : 1;
    }
    [[nodiscard]] bool instruction_supports_fusion(OpCode opcode) const noexcept {
        const auto desc = instruction_desc(opcode);
        return desc.has_value() && desc->supports_fusion;
    }
    [[nodiscard]] bool supports_scalar_width(uint32_t bits) const noexcept {
        const uint16_t mask = width_to_mask(bits);
        return (supported_scalar_widths_ & mask) != 0;
    }
    [[nodiscard]] bool supports_vector_width(uint32_t bits) const noexcept {
        const uint16_t mask = width_to_mask(bits);
        return (supported_vector_widths_ & mask) != 0;
    }
    [[nodiscard]] bool supports_fma() const noexcept {
        return supports_fma_;
    }

    /**
     * @brief Returns whether the target has a direct instruction-table entry.
     *
     * This is the canonical target-local opcode legality query. Higher-level
     * legalization may still lower an opcode to another form; a false result
     * therefore means "not directly supported by this target description".
     */
    [[nodiscard]] bool supports_opcode(OpCode opcode) const noexcept {
        return instruction_desc(opcode).has_value();
    }

    /** @brief True when the vector ISA uses a scalable vector length. */
    [[nodiscard]] bool has_scalable_vectors() const noexcept {
        return scalable_vectors_;
    }
    [[nodiscard]] float average_instruction_latency() const noexcept {
        if (instruction_count_ == 0) return 1.0f;
        const uint32_t total = std::accumulate(instruction_table_.begin(), instruction_table_.begin() + instruction_count_, 0u,
                                               [](uint32_t sum, const auto& desc) { return sum + desc.latency_cycles; });
        return static_cast<float>(total) / static_cast<float>(instruction_count_);
    }
    [[nodiscard]] float average_instruction_reciprocal_throughput() const noexcept {
        if (instruction_count_ == 0) return 1.0f;
        const uint32_t total = std::accumulate(instruction_table_.begin(), instruction_table_.begin() + instruction_count_, 0u,
                                               [](uint32_t sum, const auto& desc) { return sum + desc.reciprocal_throughput; });
        return static_cast<float>(total) / static_cast<float>(instruction_count_);
    }

private:
    ArchType arch_ = ArchType::X86_64_AVX2;
    std::array<TargetRegisterInfo, 32> gp_registers_{};
    std::array<TargetRegisterInfo, 32> vector_registers_{};
    std::array<InstructionDesc, 32> instruction_table_{};
    CallingConventionDesc calling_convention_{};
    uint8_t gp_register_count_ = 0;
    uint8_t vector_register_count_ = 0;
    uint8_t instruction_count_ = 0;
    uint16_t supported_scalar_widths_ = kWidth64;
    uint16_t supported_vector_widths_ = 0;
    bool supports_fma_ = false;
    bool scalable_vectors_ = false;

    [[nodiscard]] static constexpr uint16_t width_to_mask(uint32_t bits) noexcept {
        switch (bits) {
            case 32: return kWidth32;
            case 64: return kWidth64;
            case 128: return kWidth128;
            case 256: return kWidth256;
            case 512: return kWidth512;
            default: return 0;
        }
    }

    void add_gp(uint16_t id, std::string_view name, bool caller_saved) noexcept {
        if (gp_register_count_ >= gp_registers_.size()) return;
        gp_registers_[gp_register_count_++] = TargetRegisterInfo{
            id, name, RegisterBank::GeneralPurpose, 64, caller_saved, true};
    }

    void add_vec(uint16_t id, std::string_view name, uint16_t bit_width, bool caller_saved) noexcept {
        if (vector_register_count_ >= vector_registers_.size()) return;
        vector_registers_[vector_register_count_++] = TargetRegisterInfo{
            id, name, RegisterBank::Vector, bit_width, caller_saved, true};
    }

    void add_instruction(OpCode opcode, uint8_t latency, uint8_t throughput,
                         uint16_t widths, bool fusion) noexcept {
        if (instruction_count_ >= instruction_table_.size()) return;
        instruction_table_[instruction_count_++] = InstructionDesc{
            opcode, latency, throughput, widths, fusion};
    }

    void setup_registers() noexcept;
    void setup_calling_convention(CallingConventionKind calling_convention) noexcept;
    void setup_instruction_table() noexcept;
};

class alignas(64) TargetMachine {
public:

     * @brief Target capabilities used by legalization, ISel and scheduling.
     *
     * The generic fields use the same semantic vocabulary as the generated
     * ISel target contract. Architecture-specific capability fields are retained as explicit target capability state.
     */
    struct Capabilities {
        bool scalar32 = false;
        bool scalar64 = false;
        bool vector = false;
        bool vector_memory = false;
        bool atomics = false;
        bool concurrency = false;
        bool syscall = false;
        bool tensor = false;
        bool wasm_simd = false;
        bool object_macho = false;

        bool has_avx2 = false;
        bool has_avx512 = false;
        bool has_amx = false;
        bool has_fma = false;
        bool has_bmi2 = false;
        bool has_sve = false;
        bool has_neon = false;
        bool has_vfp = false;
        bool has_rvv = false;
        bool has_cuda = false;
        bool has_rocm = false;

        uint32_t cache_line_size = 64;
        uint64_t preferred_affinity_mask = 0x1ull;
        uint32_t preferred_core_index = 0;

        [[nodiscard]] constexpr TargetFeatureSet feature_set() const noexcept {
            TargetFeatureSet result;
            if (scalar32)       result.add(TargetFeature::Scalar32);
            if (scalar64)       result.add(TargetFeature::Scalar64);
            if (vector)         result.add(TargetFeature::Vector);
            if (vector_memory)  result.add(TargetFeature::VectorMemory);
            if (has_fma)        result.add(TargetFeature::FMA);
            if (atomics)        result.add(TargetFeature::Atomics);
            if (concurrency)    result.add(TargetFeature::Concurrency);
            if (syscall)        result.add(TargetFeature::Syscall);
            if (has_avx2)       result.add(TargetFeature::AVX2);
            if (has_avx512)     result.add(TargetFeature::AVX512);
            if (has_bmi2)       result.add(TargetFeature::BMI2);
            if (has_sve)        result.add(TargetFeature::SVE);
            if (has_neon)       result.add(TargetFeature::NEON);
            if (has_vfp)        result.add(TargetFeature::VFP);
            if (has_rvv)        result.add(TargetFeature::RVV);
            if (has_cuda)       result.add(TargetFeature::CUDA);
            if (has_rocm)       result.add(TargetFeature::ROCm);
            if (tensor)         result.add(TargetFeature::Tensor);
            if (wasm_simd)      result.add(TargetFeature::WasmSIMD);
            if (object_macho)   result.add(TargetFeature::ObjectMachO);
            if (has_amx)        result.add(TargetFeature::AMX);
            return result;
        }

        [[nodiscard]] constexpr bool has(TargetFeature feature) const noexcept {
            return feature_set().has(feature);
        }
    };

    explicit TargetMachine(
        ArchType arch,
        CallingConventionKind calling_convention = CallingConventionKind::SystemV_x64)
        : TargetMachine(arch, ObjectFormat::None, calling_convention) {}

    explicit TargetMachine(
        ArchType arch,
        ObjectFormat object_format,
        CallingConventionKind calling_convention = CallingConventionKind::SystemV_x64)
        : m_arch(arch),
          object_format_(object_format),
          requested_calling_convention_(calling_convention),
          identity_(identity_for_arch(arch, object_format)) {
        initialize_target();
        finalize_capabilities();
        host_features_ = detect_hardware();
        identity_.features = m_caps.feature_set();
    }

    [[nodiscard]] ArchType get_arch() const noexcept { return m_arch; }

    [[nodiscard]] static std::optional<ArchType> arch_from_target_name(
        std::string_view name) noexcept;

    [[nodiscard]] static std::optional<ObjectFormat> object_format_from_target_name(
        std::string_view name) noexcept;
    
    [[nodiscard]] bool supports_native(OpCode op) const noexcept {
        // The target description is the single source of truth for direct
        // opcode legality. Feature-specific checks are represented there too,
        // so Legalizer and ISel do not observe contradictory target contracts.
        return description_.supports_opcode(op);
    }

    [[nodiscard]] bool host_supports_target() const noexcept {
        switch (m_arch) {
            case ArchType::X86_64_AVX512:
                return host_features_.has_avx512;
            case ArchType::X86_64_AVX2:
                return host_features_.has_avx2;
            case ArchType::AARCH64_SVE:
            case ArchType::ARM32_NEON:
            case ArchType::RISCV64_RVV:

                // must not be inferred from x86 feature detection.
                return false;

            case ArchType::CUDA_PTX:
            case ArchType::ROCM_AMDGCN:
            case ArchType::FPGA_VERILOG:

    [[nodiscard]] static constexpr TargetIdentity identity_for_arch(
        ArchType arch,
        ObjectFormat object_format = ObjectFormat::None) noexcept {
        TargetIdentity identity{};
        identity.arch = arch;
        identity.object_format = object_format;

        switch (arch) {
            case ArchType::X86_64_AVX512:
                identity.kind = TargetKind::CPU;
                identity.family = TargetFamily::X64;
                identity.canonical_name = "x64";
                identity.cpu = "x86-64";
                identity.subtarget = "avx512";
                break;
            case ArchType::X86_64_AVX2:
                identity.kind = TargetKind::CPU;
                identity.family = TargetFamily::X64;
                identity.canonical_name = "x64";
                identity.cpu = "x86-64";
                identity.subtarget = "avx2";
                break;
            case ArchType::AARCH64_SVE:
                identity.kind = TargetKind::CPU;
                identity.family = TargetFamily::ARM64;
                identity.canonical_name = "arm64";
                identity.cpu = "aarch64";
                identity.subtarget = "sve";
                break;
            case ArchType::ARM32_NEON:
                identity.kind = TargetKind::CPU;
                identity.family = TargetFamily::ARM32;
                identity.canonical_name = "arm32";
                identity.cpu = "armv7a";
                identity.subtarget = "neon-vfp";
                break;
            case ArchType::RISCV64_RVV:
                identity.kind = TargetKind::CPU;
                identity.family = TargetFamily::RISCV64;

            case machine::MachineSemanticKind::IntegerSub: if (callbacks_.sub_i32) callbacks_.sub_i32(inst, out); return;

            case machine::MachineSemanticKind::IntegerDiv: if (callbacks_.div_i32) callbacks_.div_i32(inst, out); return;

class WasmTargetLoweringContract {
public:
    using MachineOpcode = machine::MachineOpcode;

    struct Result {
        MachineOpcode opcode = machine::INVALID_MACHINE_OPCODE;
        bool assigned = false;
        const char* diagnostic = nullptr;
    };

    [[nodiscard]] static Result assign_opcode(
        const machine::MachineInstruction& instruction) noexcept {
        const auto op = dgtl::isa::wasm::opcode_for(instruction.form, instruction.semantic_kind);

            machine::INVALID_MACHINE_OPCODE,
            false,

    }

class WasmMIRBoundary {
public:
    using BlockMap = std::unordered_map<std::uint32_t, machine::BlockId>;

    [[nodiscard]] static bool build(
        const ir::IRStream& stream,
        const core::TargetMachine& target,
        dgtl::mir::isel::InstructionSelectorProgram& selector,
        machine::MachineFunction& function,
        std::string& diagnostic) {

        diagnostic.clear();

        const auto arch = target.get_arch();
        if (arch != core::ArchType::WASM_SIMD &&
            target.family() != core::TargetFamily::WASM) {
            diagnostic = "Wasm MIR boundary received a non-Wasm target";
            return false;
        }

        const auto& ir_blocks = stream.get_basic_blocks();
        const auto& ir_edges = stream.get_cfg_edges();

        BlockMap block_map;
        block_map.reserve(ir_blocks.size() * 2u + 1u);

        if (ir_blocks.empty()) {
            function.create_block(true);
        } else {
            const auto entry = stream.get_entry_block_id();

            for (const auto& src : ir_blocks) {
                const bool is_entry =
                    src.id == entry ||
                    (entry == ir::INVALID_SLOT && block_map.empty());
                block_map.emplace(
                    src.id,
                    function.create_block(is_entry).id);
            }

            for (const auto& edge : ir_edges) {
                const auto from = block_map.find(edge.from);
                const auto to = block_map.find(edge.to);
                if (from == block_map.end() ||
                    to == block_map.end() ||
                    !function.connect_blocks(from->second, to->second)) {
                    diagnostic = "failed to construct Wasm MIR CFG edge";
                    return false;
                }
            }
        }

        std::unordered_map<std::uint32_t, machine::BlockId> label_to_block;
        label_to_block.reserve(stream.size() / 4u + 4u);

        auto record_label = [&](const ir::IRNode& node,
                                machine::BlockId block) {
            if (node.opcode != ir::OpCode::LABEL) return;

            if (node.result_id != ir::INVALID_SLOT && node.result_id != 0u)
                label_to_block[node.result_id] = block;
            if (node.op_a != ir::INVALID_SLOT && node.op_a != 0u)
                label_to_block[node.op_a] = block;
            if (node.op_b != ir::INVALID_SLOT && node.op_b != 0u)
                label_to_block[node.op_b] = block;
        };

        if (!ir_blocks.empty()) {
            for (const auto& src : ir_blocks) {
                const auto it = block_map.find(src.id);
                if (it == block_map.end()) {
                    diagnostic = "Wasm MIR block mapping is incomplete";
                    return false;
                }

                const std::size_t begin =
                    std::min<std::size_t>(src.begin, stream.size());
                const std::size_t end =
                    std::min<std::size_t>(src.end, stream.size());

                for (std::size_t i = begin; i < end; ++i)
                    record_label(stream.get_node(i), it->second);
            }
        } else {
            for (std::size_t i = 0; i < stream.size(); ++i)
                record_label(stream.get_node(i), function.entry_block());
        }

        machine::MachineFunctionBuilder builder(function);

        const auto block_for_node = [&](std::size_t index) {
            for (const auto& src : ir_blocks) {
                if (index >= src.begin && index < src.end) {
                    const auto it = block_map.find(src.id);
                    if (it != block_map.end()) return it->second;
                }
            }
            return function.entry_block();
        };

        const auto classify_type = [](ir::TypeCode type) {
            switch (type) {
                case ir::TypeCode::F8:
                case ir::TypeCode::BF16:
                case ir::TypeCode::F16:
                case ir::TypeCode::F32:
                case ir::TypeCode::F64:
                case ir::TypeCode::F128:
                    return machine::RegisterClassKind::FloatingPoint;

                case ir::TypeCode::VEC3_F32:
                case ir::TypeCode::VEC4_F32:
                case ir::TypeCode::MAT4X4_F32:
                case ir::TypeCode::QUAT_F32:
                case ir::TypeCode::VECTOR_F32X16:
                case ir::TypeCode::VECTOR_I32X16:
                case ir::TypeCode::VECTOR_F16X32:
                    return machine::RegisterClassKind::Vector;

                case ir::TypeCode::PTR:
                case ir::TypeCode::ADDRESS_SPACE:
                    return machine::RegisterClassKind::Address;

                default:
                    return machine::RegisterClassKind::General;
            }
        };

        const auto width_for_type = [](ir::TypeCode type) -> std::uint32_t {
            switch (type) {
                case ir::TypeCode::I8: return 1u;
                case ir::TypeCode::I16: return 2u;
                case ir::TypeCode::I32: return 4u;
                case ir::TypeCode::I64: return 8u;
                case ir::TypeCode::F32: return 4u;
                case ir::TypeCode::F64: return 8u;
                case ir::TypeCode::VEC3_F32:
                case ir::TypeCode::VEC4_F32:
                    return 16u;
                default: return 4u;
            }
        };

        const auto apply_class =
            [](machine::MachineOperand& operand,
               machine::RegisterClassKind cls,
               std::uint32_t width) {
                if (operand.kind != machine::MachineOperandKind::Register)
                    return;
                operand.reg.reg_class = cls;
                operand.subregister = width * 8u;
            };

        for (std::size_t i = 0; i < stream.size(); ++i) {
            const ir::IRNode node = stream.get_node(i);
            const auto selected =
                selector.select_instruction(stream, i, node, target);

            if (selected.form == dgtl::mir::MachineForm::Unhandled) {
                diagnostic = "Wasm ISel produced MachineForm::Unhandled";
                return false;
            }

            machine::MachineInstruction& instruction =
                selector.append_to_machine_function(
                    builder, block_for_node(i), selected);

            const auto mapped_op = dgtl::isa::wasm::opcode_for(instruction.form, instruction.semantic_kind);
            instruction.opcode = (mapped_op != machine::GM_NOOP || instruction.form == machine::MachineForm::NoOp)
                ? mapped_op
                : static_cast<machine::MachineOpcode>(selected.semantic_opcode);

            const std::uint32_t source_ids[3] = {
                node.op_a, node.op_b, node.op_c
            };

            if (!instruction.operands.empty() &&
                instruction.operands.front().kind ==
                    machine::MachineOperandKind::Register &&
                instruction.operands.front().role ==
                    machine::OperandRole::Def) {

                const auto type =
                    stream.get_result_type_code(node.result_id);
                apply_class(
                    instruction.operands.front(),
                    classify_type(type),
                    width_for_type(type));
            }

            for (std::size_t oi = 0;
                 oi < 3u && oi < instruction.operands.size();
                 ++oi) {

                if (source_ids[oi] == ir::INVALID_SLOT)
                    continue;

                const auto type =
                    stream.get_result_type_code(source_ids[oi]);

                apply_class(
                    instruction.operands[oi],
                    classify_type(type),
                    width_for_type(type));
            }

            // Linear memory LOAD/STORE canonicalization
            if (node.opcode == ir::OpCode::LOAD ||
                node.opcode == ir::OpCode::VOLATILE_LOAD) {

                const auto type =
                    stream.get_result_type_code(node.result_id);

                if (instruction.operands.size() < 2u ||
                    instruction.operands[1].kind !=
                        machine::MachineOperandKind::Register) {
                    diagnostic = "Wasm LOAD MIR operand contract violated";
                    return false;
                }

                machine::MachineOperand memory{};
                memory.kind = machine::MachineOperandKind::Memory;
                memory.role = machine::OperandRole::Use;
                memory.memory.base = instruction.operands[1].reg;
                memory.memory.size_bytes = width_for_type(type);
                memory.memory.alignment =
                    std::min<std::uint32_t>(width_for_type(type), 16u);
                memory.memory.load = true;
                memory.memory.volatile_access =
                    node.opcode == ir::OpCode::VOLATILE_LOAD;

                instruction.operands.erase(instruction.operands.begin() + 1);
                instruction.add_operand(std::move(memory));

            } else if (node.opcode == ir::OpCode::STORE ||
                       node.opcode == ir::OpCode::VOLATILE_STORE) {

                const auto type =
                    stream.get_result_type_code(node.op_b);

                if (instruction.operands.size() < 3u ||
                    instruction.operands[1].kind !=
                        machine::MachineOperandKind::Register ||
                    instruction.operands[2].kind !=
                        machine::MachineOperandKind::Register) {
                    diagnostic = "Wasm STORE MIR operand contract violated";
                    return false;
                }

                const auto base = instruction.operands[1].reg;
                const auto value = instruction.operands[2].reg;

                machine::MachineOperand memory{};
                memory.kind = machine::MachineOperandKind::Memory;
                memory.role = machine::OperandRole::Use;
                memory.memory.base = base;
                memory.memory.size_bytes = width_for_type(type);
                memory.memory.alignment =
                    std::min<std::uint32_t>(width_for_type(type), 16u);
                memory.memory.store = true;
                memory.memory.volatile_access =
                    node.opcode == ir::OpCode::VOLATILE_STORE;

                instruction.operands.erase(
                    instruction.operands.begin() + 1,
                    instruction.operands.begin() + 3);

                auto value_operand = machine::MachineOperand::use(value);
                value_operand.subregister = width_for_type(type) * 8u;
                instruction.add_operand(std::move(value_operand));
                instruction.add_operand(std::move(memory));
            }

            // Materialize immediates
            if (selected.form == dgtl::mir::MachineForm::LoadImmediate) {
                const std::uint64_t bits =
                    (static_cast<std::uint64_t>(node.op_b) << 32u) |
                    static_cast<std::uint64_t>(node.op_a);

                instruction.operands.erase(
                    std::remove_if(
                        instruction.operands.begin(),
                        instruction.operands.end(),
                        [](const machine::MachineOperand& operand) {
                            return operand.role == machine::OperandRole::Use;
                        }),
                    instruction.operands.end());

                const auto type =
                    stream.get_result_type_code(node.result_id);

                if (node.opcode == ir::OpCode::CONST &&
                    (type == ir::TypeCode::F32 ||
                     type == ir::TypeCode::F64)) {

                    machine::MachineOperand operand{};
                    operand.kind =
                        machine::MachineOperandKind::FloatingImmediate;
                    operand.role = machine::OperandRole::Use;
                    operand.floating_immediate =
                        type == ir::TypeCode::F32
                            ? static_cast<double>(
                                  std::bit_cast<float>(
                                      static_cast<std::uint32_t>(bits)))
                            : std::bit_cast<double>(bits);
                    instruction.add_operand(std::move(operand));
                } else {
                    instruction.add_operand(
                        machine::MachineOperand::imm(
                            static_cast<std::int64_t>(bits)));
                }
            }

            // CALL/CALL_VARIADIC
            if (selected.form == dgtl::mir::MachineForm::Call &&
                (node.opcode == ir::OpCode::CALL ||
                 node.opcode == ir::OpCode::CALL_VARIADIC)) {

                if (instruction.operands.empty()) {
                    diagnostic = "Wasm CALL has no callee operand";
                    return false;
                }

                machine::MachineOperand callee{};
                callee.kind = machine::MachineOperandKind::Symbol;
                callee.role = machine::OperandRole::Use;
                callee.symbol = node.op_a;

                instruction.operands.erase(instruction.operands.begin());
                instruction.operands.insert(
                    instruction.operands.begin(), std::move(callee));

                const auto overflow = stream.get_call_overflow_args(i);
                for (const std::uint32_t value_id : overflow) {
                    if (value_id == ir::INVALID_SLOT) continue;
                    auto operand =
                        machine::MachineOperand::use(
                            builder.value_register(
                                value_id,
                                classify_type(
                                    stream.get_result_type_code(value_id))));
                    operand.subregister =
                        width_for_type(
                            stream.get_result_type_code(value_id)) * 8u;
                    instruction.add_operand(std::move(operand));
                }

                function.properties().contains_calls = true;
                function.frame_info().has_calls = true;
            }

            const auto rewrite_target =
                [&](std::size_t operand_index, std::uint32_t label) -> bool {
                    const auto it = label_to_block.find(label);
                    if (it == label_to_block.end() ||
                        operand_index >= instruction.operands.size()) {
                        return false;
                    }
                    instruction.operands[operand_index] =
                        machine::MachineOperand::block_ref(it->second);
                    return true;
                };

            switch (selected.form) {
                case dgtl::mir::MachineForm::Goto:
                    if (node.op_a != ir::INVALID_SLOT &&
                        node.op_a != 0u &&
                        !rewrite_target(0u, node.op_a)) {
                        diagnostic = "unresolved Wasm GOTO target";
                        return false;
                    }
                    break;

                case dgtl::mir::MachineForm::BranchCond: {
                    if (node.op_b != ir::INVALID_SLOT &&
                        node.op_b != 0u &&
                        !rewrite_target(1u, node.op_b)) {
                        diagnostic =
                            "unresolved Wasm conditional branch target";
                        return false;
                    }

                    if (instruction.operands.size() < 3u) {
                        const auto* current = function.block(block_for_node(i));
                        if (current != nullptr && current->successors.size() == 2u) {
                            const auto true_block =
                                instruction.operands[1].block;
                            const auto false_block =
                                current->successors[0] == true_block
                                    ? current->successors[1]
                                    : current->successors[0];
                            instruction.add_operand(
                                machine::MachineOperand::block_ref(
                                    false_block));
                        }
                    }
                    break;
                }

                case dgtl::mir::MachineForm::BranchCompare:
                    if (node.op_c != ir::INVALID_SLOT &&
                        node.op_c != 0u &&
                        !rewrite_target(2u, node.op_c)) {
                        diagnostic =
                            "unresolved Wasm compare branch target";
                        return false;
                    }
                    break;

                default:
                    break;
            }
        }

        function.set_phase(machine::MachinePhase::PreRegAlloc);

        const auto verification =
            machine::MachineVerifier::verify(function);

        if (!verification.ok()) {
            diagnostic = "MachineVerifier rejected pre-regalloc Wasm MIR";
            return false;
        }

        return true;
    }

    [[nodiscard]] static bool lower_and_allocate(
        const ir::IRStream& stream,
        const core::TargetMachine& target,
        dgtl::mir::isel::InstructionSelectorProgram& selector,
        machine::MachineFunction& function,
        std::string& diagnostic) {

        if (!build(stream, target, selector, function, diagnostic))
            return false;

        WasmTargetRegisterInfo reg_info{};
        dgtl::mir::regalloc::UniversalGraphColoringRegAlloc allocator(function, reg_info);
        if (!allocator.run()) {
            diagnostic = "Wasm universal register allocation failed";
            return false;
        }

        if (!WasmFrameCalculator::apply_machine_function(function, {})) {
            diagnostic = "Wasm linear memory frame lowering failed";
            return false;
        }

        const auto verification =
            machine::MachineVerifier::verify(function);

        if (!verification.ok()) {
            diagnostic =
                "MachineVerifier rejected frame-lowered Wasm MIR";
            return false;
        }

        return true;
    }
};

class WasmCodeBuffer {
public:
    std::vector<uint8_t> bytes;

    void emit_u8(uint8_t b) { bytes.push_back(b); }
    void emit_leb128_u32(uint32_t val) {
        do {
            uint8_t byte = val & 0x7Fu;
            val >>= 7u;
            if (val != 0) byte |= 0x80u;
            bytes.push_back(byte);
        } while (val != 0);
    }
    void emit_leb128_i32(int32_t val) {
        bool more = true;
        while (more) {
            uint8_t byte = val & 0x7F;
            val >>= 7;
            if ((val == 0 && !(byte & 0x40)) || (val == -1 && (byte & 0x40))) {
                more = false;
            } else {
                byte |= 0x80;
            }
            bytes.push_back(byte);
        }
    }
};

class WasmMachineFormLowerer {
public:
    static void emit_prologue(const WasmFramePlan& plan, WasmCodeBuffer& buf) {
        if (plan.total_frame_bytes > 0) {
            // global.get 0 (stack pointer) -> 0x23 0x00
            buf.emit_u8(0x23); buf.emit_leb128_u32(0);
            // i32.const frame_size -> 0x41
            buf.emit_u8(0x41); buf.emit_leb128_i32(static_cast<int32_t>(plan.total_frame_bytes));
            // i32.sub -> 0x6B
            buf.emit_u8(0x6B);
            // global.set 0 -> 0x24 0x00
            buf.emit_u8(0x24); buf.emit_leb128_u32(0);
        }
    }

    static void emit_epilogue(const WasmFramePlan& plan, WasmCodeBuffer& buf) {
        if (plan.total_frame_bytes > 0) {
            // global.get 0 -> 0x23 0x00
            buf.emit_u8(0x23); buf.emit_leb128_u32(0);
            // i32.const frame_size -> 0x41
            buf.emit_u8(0x41); buf.emit_leb128_i32(static_cast<int32_t>(plan.total_frame_bytes));
            // i32.add -> 0x6A
            buf.emit_u8(0x6A);
            // global.set 0 -> 0x24 0x00
            buf.emit_u8(0x24); buf.emit_leb128_u32(0);
        }
        // end -> 0x0B
        buf.emit_u8(0x0B);
    }
};

inline bool verify_frame_alignment_invariant() {
    for (uint32_t locals = 0; locals < 512; locals += 7) {
        for (uint32_t outgoing = 0; outgoing < 128; outgoing += 8) {
            const auto plan = WasmFrameCalculator::compute_plan(locals, 0, outgoing, {}, true);
            if ((plan.total_frame_bytes % 16) != 0) {
                return false;
            }
        }
    }
    return true;
}

class WasmMachineCodeEmitter final : public dgtl::machine::MachineCodeEmitter {
public:
    void emit_instruction(const dgtl::machine::MachineInstruction& inst,
                          std::vector<std::uint8_t>& out) override {
        switch (inst.form) {
            case dgtl::machine::MachineForm::Return:
                out.push_back(0x0F); // return
                break;
            case dgtl::machine::MachineForm::NoOp:
                out.push_back(0x01); // nop
                break;
            default:
                break;
        }
    }
};

[[nodiscard]] constexpr machine::GenericMachineOpcode map_to_generic_opcode(
    ir::OpCode op,
    MachineForm form,
    machine::MachineSemanticKind /*sem*/) noexcept {
    using namespace machine;

    const auto idx = static_cast<std::size_t>(op);
    if (idx < detail::kMaxOpCodeCount) {
        const auto direct = detail::kOpCodeToGenericMachineTable[idx];
        if (direct != GM_NOOP || op == ir::OpCode::NOP) {
            return direct;
        }
    }

    // Fallback based on MachineForm
    switch (form) {
        case MachineForm::NoOp:             return GM_NOOP;
        case MachineForm::Move:             return GM_COPY;
        case MachineForm::LoadImmediate:    return GM_IMMEDIATE;
        case MachineForm::LoadStack:        return GM_LOAD_STACK;
        case MachineForm::StoreStack:       return GM_STORE_STACK;
        case MachineForm::IntegerBinary32:  return GM_IADD32;
        case MachineForm::IntegerBinary64:  return GM_IADD64;
        case MachineForm::FloatBinary32:    return GM_FADD32;
        case MachineForm::FloatBinary64:    return GM_FADD64;
        case MachineForm::FloatTernaryFma32:return GM_FFMA32;
        case MachineForm::Compare32:        return GM_ICMP32;
        case MachineForm::Goto:             return GM_GOTO;
        case MachineForm::BranchCond:       return GM_BR_COND;
        case MachineForm::BranchCompare:    return GM_BR_COMPARE;
        case MachineForm::Call:             return GM_CALL;
        case MachineForm::Return:           return GM_RETURN;
        case MachineForm::Syscall:          return GM_SYSCALL;
        case MachineForm::HardwareFence:
        case MachineForm::ConcurrentSync:   return GM_FENCE;
        case MachineForm::AtomicOp:         return GM_ATOMIC_LOAD;
        case MachineForm::TensorOp:         return GM_TENSOR_OP;
        case MachineForm::VectorBinary512:  return GM_VEC_LOAD;
        case MachineForm::VectorGather512:  return GM_VEC_GATHER;
        case MachineForm::VectorStore512:   return GM_VEC_STORE;
        default:                            return GM_NOOP;
    }
}

[[nodiscard]] inline machine::MachineSelection to_machine_selection(
    const SelectedInstruction& selected,
    machine::RegisterClassKind result_class) {
    machine::MachineSelection selection{};
    selection.form = to_machine_form(selected.form);
    selection.semantic_kind = selected.semantic_kind;
    selection.opcode = map_to_generic_opcode(selected.semantic_opcode, selected.form, selected.semantic_kind);
    selection.semantic_provenance =
        dgtl::ir::schema::stable_id(selected.semantic_opcode);
    selection.debug_id = 0;
    selection.result_value = selected.node.result_id;
    selection.result_class = result_class;
    selection.intent_hash = selected.node.intent_hash;

    // Lossless transfer of IR schema traits, domains, and attributes
    if (const auto* def = dgtl::ir::gen::find_generated_op_def(selected.semantic_opcode)) {
        selection.traits = static_cast<machine::MachineTrait>(def->trait_bits);
        selection.domains = static_cast<machine::MachineDomain>(def->interface_bits);
        selection.attributes = static_cast<machine::MachineAttribute>(def->attribute_bits);
    }

    const auto default_info = machine::CommonTargetInstructionInfo::default_form_info(selection.form);
    selection.traits = selection.traits | default_info.default_traits;
    selection.domains = selection.domains | default_info.default_domains;
    selection.latency_cycles = default_info.default_latency;

    if ((selected.node.flags & dgtl::ir::Flags::IS_VOLATILE) != 0u) {
        selection.attributes = selection.attributes | machine::MachineAttribute::Volatile;
    }
    if ((selected.node.flags & dgtl::ir::Flags::IS_PREDICATED) != 0u) {
        selection.attributes = selection.attributes | machine::MachineAttribute::Predicated;
    }

    if (selected.memory_operand.has_value()) {
        selection.operands.push_back(
            machine::MachineSelectionOperand::memory_ref(
                *selected.memory_operand,
                selected.form == MachineForm::StoreStack ? machine::OperandRole::Def : machine::OperandRole::Use));
        if (selected.form == MachineForm::StoreStack) {
            const uint32_t val = (selected.node.op_b != ir::INVALID_SLOT) ? selected.node.op_b : selected.node.op_a;
            if (val != ir::INVALID_SLOT) {
                selection.operands.push_back(
                    machine::MachineSelectionOperand::value_ref(val, machine::OperandRole::Use));
            }
        }
    } else {
        const std::uint32_t operands[3] = {
            selected.node.op_a, selected.node.op_b, selected.node.op_c
        };

        for (std::size_t i = 0; i < 3; ++i) {
            if (operands[i] != ir::INVALID_SLOT) {
                selection.operands.push_back(
                    machine::MachineSelectionOperand::value_ref(
                        operands[i], machine::OperandRole::Use));
            }
        }
    }

    if (selected.node.has_side_effects()) {
        selection.flags = selection.flags | machine::MachineInstrFlag::SideEffect;
    }

    switch (selection.form) {
        case MachineForm::Goto:
        case MachineForm::BranchCond:
        case MachineForm::BranchCompare:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Terminator |
                              machine::MachineInstrFlag::Branch;
            break;
        case MachineForm::Return:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Terminator |
                              machine::MachineInstrFlag::Return;
            break;
        case MachineForm::Call:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Call |
                              machine::MachineInstrFlag::SideEffect;
            break;
        case MachineForm::LoadStack:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Load |
                              machine::MachineInstrFlag::MayLoad;
            break;
        case MachineForm::StoreStack:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Store |
                              machine::MachineInstrFlag::MayStore;
            break;
        case MachineForm::AtomicOp:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Atomic |
                              machine::MachineInstrFlag::SideEffect;
            break;
        case MachineForm::HardwareFence:
        case MachineForm::ConcurrentSync:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::Barrier |
                              machine::MachineInstrFlag::SideEffect;
            break;
        case MachineForm::Syscall:
        case MachineForm::TensorOp:
            selection.flags = selection.flags |
                              machine::MachineInstrFlag::SideEffect;
            break;
        default:
            break;
    }

    return selection;
}

namespace detail {

[[nodiscard]] inline std::string_view trim_copy(std::string_view value) noexcept {
    size_t begin = 0;
    size_t end = value.size();
    while (begin < end && std::isspace(static_cast<unsigned char>(value[begin])) != 0) {
        ++begin;
    }
    while (end > begin && std::isspace(static_cast<unsigned char>(value[end - 1])) != 0) {
        --end;
    }
    return value.substr(begin, end - begin);
}

[[nodiscard]] inline bool ascii_equal_case_insensitive(std::string_view lhs, std::string_view rhs) noexcept {
    if (lhs.size() != rhs.size()) return false;
    for (size_t i = 0; i < lhs.size(); ++i) {
        if (std::tolower(static_cast<unsigned char>(lhs[i])) != std::tolower(static_cast<unsigned char>(rhs[i]))) {
            return false;
        }
    }
    return true;
}

[[nodiscard]] inline std::vector<std::string> split_top_level(std::string_view text, char delimiter) {
    std::vector<std::string> parts;
    size_t start = 0;
    int depth = 0;
    for (size_t i = 0; i < text.size(); ++i) {
        const char ch = text[i];
        if (ch == '(') {
            ++depth;
        } else if (ch == ')') {
            --depth;
        } else if (ch == delimiter && depth == 0) {
            parts.push_back(std::string(trim_copy(text.substr(start, i - start))));
            start = i + 1;
        }
    }
    parts.push_back(std::string(trim_copy(text.substr(start))));
    return parts;
}

[[nodiscard]] inline std::optional<ir::OpCode> parse_opcode_name(
    std::string_view name) noexcept {
    return dgtl::ir::gen::find_opcode_by_name(detail::trim_copy(name));
}

struct PatternTerm {
    std::string capture;
    std::optional<ir::OpCode> opcode;
    std::vector<PatternTerm> children;
};

[[nodiscard]] inline PatternTerm parse_term(std::string_view text) {
    const std::string_view token = trim_copy(text);
    if (!token.empty() && token[0] == '$') {
        PatternTerm term{};
        term.capture = std::string(token.substr(1));
        return term;
    }

    const size_t open = token.find('(');
    if (open == std::string::npos) {
        PatternTerm term{};
        term.opcode = parse_opcode_name(token);
        return term;
    }

    const size_t close = token.rfind(')');
    PatternTerm term{};
    term.opcode = parse_opcode_name(token.substr(0, open));
    if (close != std::string::npos && close > open + 1) {
        for (const std::string& part : split_top_level(token.substr(open + 1, close - open - 1), ',')) {
            term.children.push_back(parse_term(part));
        }
    }
    return term;
}

struct TargetSelector {
    std::optional<core::TargetFamily> family;
    std::optional<core::ArchType> arch;
    std::optional<core::ObjectFormat> object_format;
};

struct SelectionRule {
    std::string name;
    PatternTerm match;
    PatternTerm replace;
    std::vector<TargetSelector> targets;
    uint32_t benefit = 1;
    bool require_fma = false;
};

[[nodiscard]] inline std::optional<TargetSelector> parse_target_name(std::string_view value) noexcept {
    const std::string_view key = trim_copy(value);

    // Exact ISA/subtarget selectors
    if (ascii_equal_case_insensitive(key, "x86_64_avx512")) {
        return TargetSelector{std::nullopt, core::ArchType::X86_64_AVX512, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "x86_64_avx2")) {
        return TargetSelector{std::nullopt, core::ArchType::X86_64_AVX2, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "aarch64_sve") ||
        ascii_equal_case_insensitive(key, "arm64_sve")) {
        return TargetSelector{std::nullopt, core::ArchType::AARCH64_SVE, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "arm32_neon")) {
        return TargetSelector{std::nullopt, core::ArchType::ARM32_NEON, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "riscv64_rvv")) {
        return TargetSelector{std::nullopt, core::ArchType::RISCV64_RVV, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "wasm_simd")) {
        return TargetSelector{std::nullopt, core::ArchType::WASM_SIMD, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "cuda_ptx")) {
        return TargetSelector{std::nullopt, core::ArchType::CUDA_PTX, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "rocm_amdgcn")) {
        return TargetSelector{std::nullopt, core::ArchType::ROCM_AMDGCN, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "fpga_verilog")) {
        return TargetSelector{std::nullopt, core::ArchType::FPGA_VERILOG, std::nullopt};
    }

    // Canonical target-family aliases
    if (ascii_equal_case_insensitive(key, "x64") ||
        ascii_equal_case_insensitive(key, "x86_64") ||
        ascii_equal_case_insensitive(key, "amd64") ||
        ascii_equal_case_insensitive(key, "x86-64")) {
        return TargetSelector{core::TargetFamily::X64, std::nullopt, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "arm64") ||
        ascii_equal_case_insensitive(key, "aarch64") ||
        ascii_equal_case_insensitive(key, "armv8") ||
        ascii_equal_case_insensitive(key, "sve")) {
        return TargetSelector{core::TargetFamily::ARM64, std::nullopt, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "arm32") ||
        ascii_equal_case_insensitive(key, "armv7a") ||
        ascii_equal_case_insensitive(key, "armv7")) {
        return TargetSelector{core::TargetFamily::ARM32, std::nullopt, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "wasm") ||
        ascii_equal_case_insensitive(key, "wasm32") ||
        ascii_equal_case_insensitive(key, "wasm64")) {
        return TargetSelector{core::TargetFamily::WASM, std::nullopt, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "riscv64") ||
        ascii_equal_case_insensitive(key, "rv64") ||
        ascii_equal_case_insensitive(key, "rv64gc")) {
        return TargetSelector{core::TargetFamily::RISCV64, std::nullopt, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "cuda") ||
        ascii_equal_case_insensitive(key, "ptx") ||
        ascii_equal_case_insensitive(key, "nvptx")) {
        return TargetSelector{core::TargetFamily::CUDA, std::nullopt, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "rocm") ||
        ascii_equal_case_insensitive(key, "amdgcn") ||
        ascii_equal_case_insensitive(key, "amd-gcn") ||
        ascii_equal_case_insensitive(key, "amdgcn-gfx") ||
        ascii_equal_case_insensitive(key, "hip")) {
        return TargetSelector{core::TargetFamily::ROCm, std::nullopt, std::nullopt};
    }
    if (ascii_equal_case_insensitive(key, "fpga") ||
        ascii_equal_case_insensitive(key, "verilog") ||
        ascii_equal_case_insensitive(key, "fpga_verilog")) {
        return TargetSelector{core::TargetFamily::FPGA, std::nullopt, std::nullopt};
    }

    if (ascii_equal_case_insensitive(key, "macho") ||
        ascii_equal_case_insensitive(key, "mach-o") ||
        ascii_equal_case_insensitive(key, "darwin")) {
        return TargetSelector{std::nullopt, std::nullopt, core::ObjectFormat::MachO};
    }

    return std::nullopt;
}

[[nodiscard]] inline std::string lower_ascii(std::string_view value) {
    std::string out(value.size(), '\0');
    std::transform(value.begin(), value.end(), out.begin(),
                   [](unsigned char ch) { return static_cast<char>(std::tolower(ch)); });
    return out;
}

[[nodiscard]] inline SelectionRule parse_rule_line(std::string_view line) {
    SelectionRule rule{};
    std::string_view text = trim_copy(line);
    if (!text.starts_with("rule ")) {
        return rule;
    }

    const size_t colon = text.find(':');
    const size_t arrow = text.find("->");
    const size_t bracket = text.find('[');
    if (colon == std::string::npos || arrow == std::string::npos || arrow <= colon) {
        return rule;
    }

    rule.name = std::string(trim_copy(text.substr(5, colon - 5)));
    const size_t replace_end = (bracket == std::string::npos) ? text.size() : bracket;
    rule.match = parse_term(text.substr(colon + 1, arrow - colon - 1));
    rule.replace = parse_term(text.substr(arrow + 2, replace_end - arrow - 2));

    if (bracket != std::string::npos) {
        const size_t end_bracket = text.rfind(']');
        if (end_bracket == std::string_view::npos || end_bracket <= bracket) {
            return rule;
        }
        const std::string_view attrs = text.substr(bracket + 1, end_bracket - bracket - 1);
        for (const std::string& item : split_top_level(attrs, ',')) {
            const size_t eq = item.find('=');
            if (eq == std::string::npos) {
                continue;
            }
            const std::string key = lower_ascii(trim_copy(item.substr(0, eq)));
            const std::string_view value = trim_copy(item.substr(eq + 1));
            if (key == "targets") {
                for (const std::string& target_name : split_top_level(value, '|')) {
                    const auto selector = parse_target_name(target_name);
                    if (selector.has_value()) {
                        rule.targets.push_back(*selector);
                    }
                }
            } else if (key == "benefit") {
                rule.benefit = static_cast<uint32_t>(std::max(1, std::stoi(std::string(value))));
            } else if (key == "require_fma") {
                rule.require_fma = ascii_equal_case_insensitive(value, "true") || value == "1" || ascii_equal_case_insensitive(value, "yes");
            }
        }
    }

    return rule;
}

struct MatchContext {
    const ir::IRStream& stream;
    std::unordered_map<uint32_t, size_t> producer_by_result;

    explicit MatchContext(const ir::IRStream& ir_stream) : stream(ir_stream) {
        producer_by_result.reserve(ir_stream.size() * 2u + 1u);

        for (size_t i = 0; i < ir_stream.size(); ++i) {
            const ir::IRNode node = ir_stream.get_node_readonly(i);
            if (node.result_id != ir::INVALID_SLOT) {
                producer_by_result.emplace(node.result_id, i);
            }
        }
    }

    [[nodiscard]] const ir::IRNode* lookup_producer(uint32_t slot) const noexcept {
        const auto it = producer_by_result.find(slot);
        if (it == producer_by_result.end()) return nullptr;
        return &stream.get_node_readonly(it->second);
    }
};

[[nodiscard]] inline bool target_selector_matches(const TargetSelector& selector,
                                                  const core::TargetMachine& target) noexcept {
    if (selector.arch.has_value() && target.get_arch() != *selector.arch) {
        return false;
    }
    if (selector.family.has_value() && target.family() != *selector.family) {
        return false;
    }
    if (selector.object_format.has_value() && target.object_format() != *selector.object_format) {
        return false;
    }
    return true;
}

[[nodiscard]] inline bool arch_matches(const SelectionRule& rule,
                                       const core::TargetMachine& target) noexcept {
    if (rule.targets.empty()) {
        return true;
    }
    return std::any_of(rule.targets.begin(), rule.targets.end(),
                       [&target](const TargetSelector& selector) {
                           return target_selector_matches(selector, target);
                       });
}

[[nodiscard]] inline bool supports_rule_requirements(const SelectionRule& rule,
                                                      const core::TargetMachine& target) noexcept {
    if (rule.require_fma && !target.has_feature(core::TargetFeature::FMA)) {
        return false;
    }
    return true;
}

[[nodiscard]] inline uint32_t operand_at(const ir::IRNode& node, size_t index) noexcept {
    switch (index) {
        case 0: return node.op_a;
        case 1: return node.op_b;
        case 2: return node.op_c;
        default: return ir::INVALID_SLOT;
    }
}

[[nodiscard]] inline bool match_term(const PatternTerm& term,
                                     const ir::IRNode& node,
                                     const MatchContext& ctx,
                                     std::unordered_map<std::string, uint32_t>& captures) {
    if (!term.capture.empty()) {
        captures[term.capture] = node.result_id;
        return true;
    }

    if (!term.opcode.has_value() || node.opcode != *term.opcode) {
        return false;
    }

    for (size_t i = 0; i < term.children.size(); ++i) {
        const PatternTerm& child = term.children[i];
        const uint32_t operand = operand_at(node, i);
        if (!child.capture.empty()) {
            captures[child.capture] = operand;
            continue;
        }
        const ir::IRNode* producer = ctx.lookup_producer(operand);
        if (producer == nullptr || !match_term(child, *producer, ctx, captures)) {
            return false;
        }
    }

    return true;
}

[[nodiscard]] inline uint32_t lookup_capture(const std::unordered_map<std::string, uint32_t>& captures,
                                             const std::string& name) noexcept {
    const auto it = captures.find(name);
    return it == captures.end() ? ir::INVALID_SLOT : it->second;
}

[[nodiscard]] inline ir::IRNode build_replacement(const SelectionRule& rule,
                                                   const ir::IRNode& original,
                                                   const std::unordered_map<std::string, uint32_t>& captures) {
    ir::IRNode node = original;
    if (rule.replace.opcode.has_value()) {
        node.opcode = *rule.replace.opcode;
    }

    node.op_a = ir::INVALID_SLOT;
    node.op_b = ir::INVALID_SLOT;
    node.op_c = ir::INVALID_SLOT;
    for (size_t i = 0; i < rule.replace.children.size() && i < 3; ++i) {
        const PatternTerm& child = rule.replace.children[i];
        const uint32_t value = child.capture.empty() ? ir::INVALID_SLOT : lookup_capture(captures, child.capture);
        switch (i) {
            case 0: node.op_a = value; break;
            case 1: node.op_b = value; break;
            case 2: node.op_c = value; break;
            default: break;
        }
    }
    node.refresh_intent_hash();
    return node;
}

[[nodiscard]] inline std::optional<int64_t> extract_constant_immediate(const ir::IRNode& node) noexcept {
    if (node.opcode == ir::OpCode::LOAD_IMM || node.opcode == ir::OpCode::IMMEDIATE || node.opcode == ir::OpCode::CONST) {
        if (node.op_a == ir::INVALID_SLOT) {
            return static_cast<int64_t>(static_cast<int32_t>(node.op_b));
        }
        const uint64_t raw = (static_cast<uint64_t>(node.op_a) << 32) | static_cast<uint64_t>(node.op_b);
        return static_cast<int64_t>(raw);
    }
    return std::nullopt;
}

[[nodiscard]] inline bool is_scale_valid_for_target(int32_t scale, const core::TargetMachine& target) noexcept {
    if (scale == 1) return true;
    if (target.family() == core::TargetFamily::RISCV64) {
        return false;
    }
    if (target.family() == core::TargetFamily::CUDA ||
        target.family() == core::TargetFamily::ROCm ||
        target.family() == core::TargetFamily::FPGA) {
        return false;
    }
    return (scale == 2 || scale == 4 || scale == 8);
}

struct SIBAddressingMode {
    uint32_t base_value = ir::INVALID_SLOT;
    uint32_t index_value = ir::INVALID_SLOT;
    int32_t scale = 1;
    int64_t displacement = 0;
    uint32_t address_space = 0;
    bool matched = false;

    [[nodiscard]] machine::MachineMemoryOperand to_memory_operand(
        uint32_t size_bytes = 4,
        uint32_t alignment = 4,
        bool is_load = false,
        bool is_store = false,
        uint32_t addr_space = 0) const noexcept {
        machine::MachineMemoryOperand mem{};
        if (base_value != ir::INVALID_SLOT) {
            mem.base = machine::MachineRegister::virtual_reg(base_value, machine::RegisterClassKind::General);
        }
        if (index_value != ir::INVALID_SLOT) {
            mem.index = machine::MachineRegister::virtual_reg(index_value, machine::RegisterClassKind::General);
        }
        mem.scale = scale;
        mem.displacement = displacement;
        mem.address_space = (addr_space != 0) ? addr_space : address_space;
        mem.size_bytes = size_bytes;
        mem.alignment = alignment;
        mem.load = is_load;
        mem.store = is_store;
        return mem;
    }
};

namespace detail {

[[nodiscard]] inline SIBAddressingMode match_addressing_mode_recursive(
    const MatchContext& ctx,
    uint32_t val,
    const core::TargetMachine& target,
    uint32_t depth = 0) noexcept {

    SIBAddressingMode mode{};
    if (val == ir::INVALID_SLOT || depth > 5) {
        mode.base_value = val;
        mode.matched = (val != ir::INVALID_SLOT);
        return mode;
    }

    const ir::IRNode* prod = ctx.lookup_producer(val);
    if (!prod) {
        mode.base_value = val;
        mode.matched = true;
        return mode;
    }

    const uint32_t node_addr_space = (prod->logic_flags >> 24) & 0xFu;
    if (node_addr_space != 0) {
        mode.address_space = node_addr_space;
    }

    if (auto imm = extract_constant_immediate(*prod); imm.has_value()) {
        mode.displacement = *imm;
        mode.matched = true;
        return mode;
    }

    if (prod->opcode == ir::OpCode::ADD_I64 || prod->opcode == ir::OpCode::ADD_I32) {
        const uint32_t lhs = prod->op_a;
        const uint32_t rhs = prod->op_b;

        const ir::IRNode* lhs_node = ctx.lookup_producer(lhs);
        const ir::IRNode* rhs_node = ctx.lookup_producer(rhs);

        auto lhs_imm = lhs_node ? extract_constant_immediate(*lhs_node) : std::nullopt;
        auto rhs_imm = rhs_node ? extract_constant_immediate(*rhs_node) : std::nullopt;

        if (rhs_imm.has_value()) {
            SIBAddressingMode base_mode = match_addressing_mode_recursive(ctx, lhs, target, depth + 1);
            base_mode.displacement += *rhs_imm;
            base_mode.matched = true;
            return base_mode;
        }

        if (lhs_imm.has_value()) {
            SIBAddressingMode base_mode = match_addressing_mode_recursive(ctx, rhs, target, depth + 1);
            base_mode.displacement += *lhs_imm;
            base_mode.matched = true;
            return base_mode;
        }

        auto try_scaled = [&](uint32_t scale_cand, uint32_t other) -> std::optional<SIBAddressingMode> {
            const ir::IRNode* scale_node = ctx.lookup_producer(scale_cand);
            if (!scale_node) return std::nullopt;

            if (scale_node->opcode == ir::OpCode::SHL_I64 || scale_node->opcode == ir::OpCode::SHL_I32) {
                const ir::IRNode* shift_node = ctx.lookup_producer(scale_node->op_b);
                auto shift_imm = shift_node ? extract_constant_immediate(*shift_node) : std::nullopt;
                if (!shift_imm && scale_node->op_b <= 3 && scale_node->op_b >= 1) {
                    shift_imm = static_cast<int64_t>(scale_node->op_b);
                }
                if (shift_imm && *shift_imm >= 1 && *shift_imm <= 3) {
                    const int32_t sc = 1 << static_cast<int32_t>(*shift_imm);
                    if (is_scale_valid_for_target(sc, target)) {
                        SIBAddressingMode m = match_addressing_mode_recursive(ctx, other, target, depth + 1);
                        if (m.index_value == ir::INVALID_SLOT) {
                            m.index_value = scale_node->op_a;
                            m.scale = sc;
                            m.matched = true;
                            return m;
                        }
                    }
                }
            }

            if (scale_node->opcode == ir::OpCode::MUL_I64 || scale_node->opcode == ir::OpCode::MUL_I32) {
                const ir::IRNode* factor_node = ctx.lookup_producer(scale_node->op_b);
                auto factor_imm = factor_node ? extract_constant_immediate(*factor_node) : std::nullopt;
                uint32_t idx_val = scale_node->op_a;
                if (!factor_imm) {
                    const ir::IRNode* lhs_f = ctx.lookup_producer(scale_node->op_a);
                    factor_imm = lhs_f ? extract_constant_immediate(*lhs_f) : std::nullopt;
                    idx_val = scale_node->op_b;
                }
                if (factor_imm && (*factor_imm == 1 || *factor_imm == 2 || *factor_imm == 4 || *factor_imm == 8)) {
                    const int32_t sc = static_cast<int32_t>(*factor_imm);
                    if (is_scale_valid_for_target(sc, target)) {
                        SIBAddressingMode m = match_addressing_mode_recursive(ctx, other, target, depth + 1);
                        if (m.index_value == ir::INVALID_SLOT) {
                            m.index_value = idx_val;
                            m.scale = sc;
                            m.matched = true;
                            return m;
                        }
                    }
                }
            }

            return std::nullopt;
        };

        if (auto m = try_scaled(rhs, lhs); m.has_value()) return *m;
        if (auto m = try_scaled(lhs, rhs); m.has_value()) return *m;

        SIBAddressingMode m = match_addressing_mode_recursive(ctx, lhs, target, depth + 1);
        if (m.index_value == ir::INVALID_SLOT && is_scale_valid_for_target(1, target)) {
            m.index_value = rhs;
            m.scale = 1;
            m.matched = true;
            return m;
        }
    }

    if (prod->opcode == ir::OpCode::GEP) {
        const uint32_t base_ptr = prod->op_a;
        const uint32_t index_or_off = prod->op_b;
        const uint32_t elem_stride = (prod->op_c != ir::INVALID_SLOT && prod->op_c > 0) ? prod->op_c : 1u;

        const ir::IRNode* off_node = ctx.lookup_producer(index_or_off);
        auto off_imm = off_node ? extract_constant_immediate(*off_node) : std::nullopt;

        if (off_imm.has_value()) {
            SIBAddressingMode base_mode = match_addressing_mode_recursive(ctx, base_ptr, target, depth + 1);
            base_mode.displacement += (*off_imm) * static_cast<int64_t>(elem_stride);
            base_mode.matched = true;
            return base_mode;
        }

        if (elem_stride == 1 || elem_stride == 2 || elem_stride == 4 || elem_stride == 8) {
            if (is_scale_valid_for_target(static_cast<int32_t>(elem_stride), target)) {
                SIBAddressingMode base_mode = match_addressing_mode_recursive(ctx, base_ptr, target, depth + 1);
                if (base_mode.index_value == ir::INVALID_SLOT) {
                    base_mode.index_value = index_or_off;
                    base_mode.scale = static_cast<int32_t>(elem_stride);
                    base_mode.matched = true;
                    return base_mode;
                }
            }
        }
    }

    if (prod->opcode == ir::OpCode::LEA) {
        return match_addressing_mode_recursive(ctx, prod->op_a, target, depth + 1);
    }

    mode.base_value = val;
    mode.matched = true;
    return mode;
}

} // namespace detail

[[nodiscard]] inline SIBAddressingMode match_addressing_mode(
    const ir::IRStream& stream,
    const detail::MatchContext& ctx,
    uint32_t address_val,
    const core::TargetMachine& target) noexcept {
    (void)stream;
    return detail::match_addressing_mode_recursive(ctx, address_val, target, 0);
}

[[nodiscard]] inline SIBAddressingMode match_addressing_mode(
    const ir::IRStream& stream,
    uint32_t address_val,
    const core::TargetMachine& target) {
    detail::MatchContext ctx(stream);
    return match_addressing_mode(stream, ctx, address_val, target);
}

[[nodiscard]] inline MachineForm classify_machine_form_with_fallback(
    const core::TargetMachine& target,
    ir::OpCode opcode) noexcept {
    const MachineForm form = dgtl::isa::classify_machine_form(target, opcode);
    if (form != MachineForm::Unhandled) [[likely]] {
        return form;
    }

    if (runtime_policy::should_route_unhandled_to_runtime_call(target, opcode)) {
        return MachineForm::Call;
    }

    return MachineForm::Unhandled;
}

class InstructionSelectorProgram {
public:
    InstructionSelectorProgram() = default;

    [[nodiscard]] static InstructionSelectorProgram from_dsl(std::string_view dsl_text) {
        InstructionSelectorProgram program;
        size_t start = 0;
        while (start < dsl_text.size()) {
            size_t end = dsl_text.find('\n', start);
            if (end == std::string_view::npos) {
                end = dsl_text.size();
            }
            const std::string_view line = detail::trim_copy(dsl_text.substr(start, end - start));
            start = end + 1;
            if (line.empty() || line[0] == '#') {
                continue;
            }
            detail::SelectionRule rule = detail::parse_rule_line(line);
            if (!rule.name.empty() && rule.match.opcode.has_value() && rule.replace.opcode.has_value()) {
                program.rules_.push_back(std::move(rule));
            }
        }
        return program;
    }

    [[nodiscard]] static InstructionSelectorProgram from_rule_lines(std::span<const std::string_view> lines) {
        InstructionSelectorProgram program;
        for (const std::string_view line_view : lines) {
            const std::string_view line = detail::trim_copy(line_view);
            if (line.empty() || line[0] == '#') {
                continue;
            }
            detail::SelectionRule rule = detail::parse_rule_line(line);
            if (!rule.name.empty() && rule.match.opcode.has_value() && rule.replace.opcode.has_value()) {
                program.rules_.push_back(std::move(rule));
            }
        }
        return program;
    }

    [[nodiscard]] ISelReplacement select_node(const ir::IRStream& stream,
                                              size_t index,
                                              const ir::IRNode& node,
                                              const core::TargetMachine& target) const {
        (void)index;
        detail::MatchContext ctx(stream);
        ISelReplacement best{};
        for (const detail::SelectionRule& rule : rules_) {
            if (!detail::arch_matches(rule, target)) {
                continue;
            }
            if (!detail::supports_rule_requirements(rule, target)) {
                continue;
            }
            if (!rule.match.opcode.has_value() || node.opcode != *rule.match.opcode) {
                continue;
            }

            std::unordered_map<std::string, uint32_t> captures;
            if (!detail::match_term(rule.match, node, ctx, captures)) {
                continue;
            }

            if (!best.matched || rule.benefit > best.benefit) {
                best.matched = true;
                best.rule_name = rule.name;
                best.benefit = rule.benefit;
                best.node = detail::build_replacement(rule, node, captures);
            }
        }
        return best;
    }

    [[nodiscard]] SelectedInstruction select_instruction(const ir::IRStream& stream,
                                                         size_t index,
                                                         const ir::IRNode& node,
                                                         const core::TargetMachine& target) const {
        const ISelReplacement replacement = select_node(stream, index, node, target);
        SelectedInstruction selected{};
        selected.node = replacement.matched ? replacement.node : node;
        selected.semantic_opcode = selected.node.opcode;
        selected.rule_name = replacement.rule_name;
        selected.matched_rule = replacement.matched;
        selected.form = classify_machine_form_with_fallback(target, selected.node.opcode);
        selected.semantic_kind = dgtl::isa::semantic_kind(target, selected.semantic_opcode);

        if (selected.form == MachineForm::LoadStack || selected.form == MachineForm::StoreStack ||
            selected.node.opcode == ir::OpCode::LOAD || selected.node.opcode == ir::OpCode::STORE ||
            selected.node.opcode == ir::OpCode::LEA) {
            detail::MatchContext ctx(stream);
            const uint32_t addr_val = selected.node.op_a;
            if (addr_val != ir::INVALID_SLOT) {
                const SIBAddressingMode sib = match_addressing_mode(stream, ctx, addr_val, target);
                if (sib.matched && (sib.index_value != ir::INVALID_SLOT || sib.displacement != 0)) {
                    const bool is_store = (selected.form == MachineForm::StoreStack || selected.node.opcode == ir::OpCode::STORE);
                    selected.memory_operand = sib.to_memory_operand(4, 4, !is_store, is_store);
                }
            }
        }
        return selected;
    }

    [[nodiscard]] machine::MachineInstruction& append_to_machine_function(
        machine::MachineFunctionBuilder& builder,
        machine::BlockId block,
        const SelectedInstruction& selected,
        machine::RegisterClassKind result_class = machine::RegisterClassKind::General) const {
        if (selected.form == MachineForm::Unhandled ||
            selected.semantic_kind == machine::MachineSemanticKind::None ||
            !machine::semantic_kind_matches_form(to_machine_form(selected.form), selected.semantic_kind)) {
            throw std::invalid_argument(
                "DGTL ISel contract violation: invalid HIR-to-MIR semantic/form pair");
        }
        return builder.append(block, to_machine_selection(selected, result_class), result_class);
    }

    [[nodiscard]] bool empty() const noexcept {
        return rules_.empty();
    }

private:
    std::vector<detail::SelectionRule> rules_;
};

[[nodiscard]] inline const InstructionSelectorProgram& make_builtin_instruction_selector_program() {
    static const InstructionSelectorProgram program = InstructionSelectorProgram::from_rule_lines({});
    return program;
}
